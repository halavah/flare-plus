```javascript
// 成功率计算器
class VerificationMetrics {
  constructor() {
    this.metrics = {
      total: 0,
      success: 0,
      failed: 0,
      expired: 0,
      skipped: 0
    };
  }

  // 记录验证结果
  record(result) {
    this.metrics.total++;
    switch (result) {
      case 'success':
        this.metrics.success++;
        break;
      case 'failed':
        this.metrics.failed++;
        break;
      case 'expired':
        this.metrics.expired++;
        break;
      case 'skipped':
        this.metrics.skipped++;
        break;
    }
  }

  // 计算成功率
  getSuccessRate() {
    if (this.metrics.total === 0) return 0;
    return (this.metrics.success / this.metrics.total * 100).toFixed(2) + '%';
  }

  // 获取详细报告
  getReport() {
    return {
      total: this.metrics.total,
      success: this.metrics.success,
      failed: this.metrics.failed,
      expired: this.metrics.expired,
      skipped: this.metrics.skipped,
      successRate: this.getSuccessRate(),
      failureRate: ((this.metrics.failed + this.metrics.expired) / this.metrics.total * 100).toFixed(2) + '%'
    };
  }
}

// 使用示例
const metrics = new VerificationMetrics();

// 在 Turnstile 回调中记录

  metrics.record('success');
  // 处理成功验证
}

function onTurnstileError(error) {
  metrics.record('failed');
  // 处理验证失败
}

// 定期生成报告
setInterval(() => {
  console.log('验证指标报告:', metrics.getReport());
```

### 2. 验证时间分析

```javascript
// 验证时间追踪器
class VerificationTimingAnalyzer {
  constructor() {
    this.sessions = [];
    this.durations = [];
  }

  // 开始追踪
  startTracking(sessionId) {
    this.sessions[sessionId] = {
      start: Date.now(),
      events: []
    };
  }

  // 记录事件
  recordEvent(sessionId, eventType) {
    const session = this.sessions[sessionId];
    if (session) {
      session.events.push({
        type: eventType,
        timestamp: Date.now(),
        duration: Date.now() - session.start
      });
    }
  }

  // 结束追踪
  endTracking(sessionId) {
    const session = this.sessions[sessionId];
    if (session) {
      const duration = Date.now() - session.start;
      this.durations.push(duration);
      delete this.sessions[sessionId];
      return duration;
    }
    return null;
  }

  // 获取统计数据
  getStatistics() {
    if (this.durations.length === 0) return null;

    const sorted = [...this.durations].sort((a, b) => a - b);
    const sum = this.durations.reduce((a, b) => a + b, 0);

    return {
      count: this.durations.length,
      average: Math.round(sum / this.durations.length),
      min: Math.min(...this.durations),
      max: Math.max(...this.durations),
      median: sorted[Math.floor(sorted.length / 2)],
      p95: sorted[Math.floor(sorted.length * 0.95)],
      p99: sorted[Math.floor(sorted.length * 0.99)]
    };
  }
}

// 使用示例
const timingAnalyzer = new VerificationTimingAnalyzer();

// 开始验证时
function startVerification() {
  const sessionId = generateSessionId();
  timingAnalyzer.startTracking(sessionId);
  return sessionId;
}

// 验证完成时
function finishVerification(sessionId, result) {
  timingAnalyzer.recordEvent(sessionId, result);
  const duration = timingAnalyzer.endTracking(sessionId);

  if (duration) {
    console.log(`验证耗时: ${duration}ms`);
  }

  // 获取统计信息
  const stats = timingAnalyzer.getStatistics();
  console.log('验证时间统计:', stats);
}
```

```javascript
// 用户行为分析器
class UserBehaviorAnalyzer {
  constructor() {
    this.behaviors = new Map();
    this.patterns = [];
  }

  // 追踪用户行为
  trackBehavior(userId, action, data) {
    if (!this.behaviors.has(userId)) {
      this.behaviors.set(userId, {
        firstSeen: Date.now(),
        actions: [],
        verificationAttempts: 0,
        successRate: 0
      });
    }

    const behavior = this.behaviors.get(userId);
    behavior.actions.push({
      action,
      timestamp: Date.now(),
      data
    });

    // 分析行为模式
    this.analyzePatterns(behavior);
  }

  // 分析行为模式
  analyzePatterns(behavior) {
    const recentActions = behavior.actions.slice(-10); // 最近10个动作

    // 检测机器人模式
    if (this.isBotPattern(recentActions)) {
      this.flagAsSuspicious(behavior, '机器人行为模式');
    }

    // 检测异常速度
    if (this.isAbnormalSpeed(recentActions)) {
      this.flagAsSuspicious(behavior, '异常操作速度');
    }
  }

  // 检测机器人模式
  isBotPattern(actions) {
    if (actions.length < 5) return false;

    // 检查是否有规律的间隔
    const intervals = [];
    for (let i = 1; i < actions.length; i++) {
      intervals.push(actions[i].timestamp - actions[i-1].timestamp);
    }

    // 计算间隔的标准差
    const avg = intervals.reduce((a, b) => a + b, 0) / intervals.length;
    const variance = intervals.reduce((a, b) => a + Math.pow(b - avg, 2), 0) / intervals.length;
    const stdDev = Math.sqrt(variance);

    // 如果标准差很小，说明间隔过于规律
    return stdDev < 100; // 100ms 以内的标准差认为过于规律
  }

  // 检测异常速度
  isAbnormalSpeed(actions) {
    if (actions.length < 3) return false;

    // 计算平均操作间隔
    const intervals = [];
    for (let i = 1; i < actions.length; i++) {
      intervals.push(actions[i].timestamp - actions[i-1].timestamp);
    }

    const avgInterval = intervals.reduce((a, b) => a + b, 0) / intervals.length;

    // 如果平均间隔小于100ms，认为速度异常
    return avgInterval < 100;
  }

  // 标记为可疑
  flagAsSuspicious(behavior, reason) {
    behavior.suspicious = true;
    behavior.suspicionReason = reason;
    behavior.flaggedAt = Date.now();
  }

  // 生成报告
  generateReport() {
    const report = {
      totalUsers: this.behaviors.size,
      suspiciousUsers: 0,
      patterns: this.patterns
    };

    for (const [userId, behavior] of this.behaviors) {
      if (behavior.suspicious) {
        report.suspiciousUsers++;
      }
    }

    report.suspicionRate = (report.suspiciousUsers / report.totalUsers * 100).toFixed(2) + '%';

    return report;
  }
}
```

```javascript
// Analytics API 客户端
class TurnstileAnalytics {
  constructor(apiToken, accountId) {
    this.apiToken = apiToken;
    this.accountId = accountId;
    this.baseUrl = 'https://api.cloudflare.com/client/v4';
  }

  // 获取验证统计
  async getVerificationStats(widgetId, startDate, endDate) {
    const url = `${this.baseUrl}/accounts/${this.accountId}/turnstile/widgets/${widgetId}/analytics`;
    const params = new URLSearchParams({
      since: startDate,
      until: endDate
    });

    try {
      const response = await fetch(`${url}?${params}`, {
        headers: {
          'Authorization': `Bearer ${this.apiToken}`,
          'Content-Type': 'application/json'
        }
      });

      const data = await response.json();
      return data.result;
    } catch (error) {
      console.error('获取验证统计失败:', error);
      throw error;
    }
  }

  // 获取挑战统计
  async getChallengeStats(widgetId) {
    const url = `${this.baseUrl}/accounts/${this.accountId}/turnstile/widgets/${widgetId}/challenges`;

    try {
      const response = await fetch(url, {
        headers: {
          'Authorization': `Bearer ${this.apiToken}`
        }
      });

      const data = await response.json();
      return data.result;
    } catch (error) {
      console.error('获取挑战统计失败:', error);
      throw error;
    }
  }

  // 获取错误分析
  async getErrorAnalysis(widgetId) {
    const url = `${this.baseUrl}/accounts/${this.accountId}/turnstile/widgets/${widgetId}/errors`;

    try {
      const response = await fetch(url, {

      return data.result;
    } catch (error) {
      console.error('获取错误分析失败:', error);
      throw error;
    }
  }
}
```

### 1. A/B 测试分析

```javascript
// A/B 测试分析器
class TurnstileABTest {
  constructor() {
    this.variants = new Map();
    this.conversions = new Map();
  }

  // 定义测试变体
  defineVariant(name, config) {
    this.variants.set(name, {
      name,
      config,
      impressions: 0,
      conversions: 0,
      value: 0
    });
  }

  // 随机分配变体
  assignVariant(userId) {
    const variants = Array.from(this.variants.keys());
    const hash = this.hashCode(userId);
    const index = Math.abs(hash) % variants.length;
    return variants[index];
  }

  // 记录展示
  recordImpression(variantName) {
    const variant = this.variants.get(variantName);
    if (variant) {
      variant.impressions++;
    }
  }

  // 记录转化
  recordConversion(variantName, value = 1) {
    const variant = this.variants.get(variantName);
    if (variant) {
      variant.conversions++;
      variant.value += value;
    }
  }

  // 计算转化率
  getConversionRate(variantName) {
    const variant = this.variants.get(variantName);
    if (!variant || variant.impressions === 0) return 0;

    return (variant.conversions / variant.impressions * 100).toFixed(2);
  }

  // 获取测试结果
  getResults() {
    const results = {};

    for (const [name, variant] of this.variants) {
      results[name] = {
        impressions: variant.impressions,
        conversions: variant.conversions,
        conversionRate: this.getConversionRate(name),
        value: variant.value,
        avgValuePerConversion: variant.conversions > 0
          ? (variant.value / variant.conversions).toFixed(2)
          : 0
      };
    }

    return results;
  }

  // 统计显著性测试
  statisticalSignificance(confidence = 0.95) {
    const variants = Array.from(this.variants.values());
    if (variants.length !== 2) return null;

    const [control, test] = variants;
    const controlRate = control.conversions / control.impressions;
    const testRate = test.conversions / test.impressions;

    // 计算标准误差
    const pooledRate = (control.conversions + test.conversions) /
                      (control.impressions + test.impressions);
    const se = Math.sqrt(pooledRate * (1 - pooledRate) *
                        (1/control.impressions + 1/test.impressions));

    // 计算 Z 分数
    const zScore = (testRate - controlRate) / se;
    const pValue = 2 * (1 - this.normalCDF(Math.abs(zScore)));

    return {
      significant: pValue < (1 - confidence),
      pValue,
      controlRate: (controlRate * 100).toFixed(2) + '%',
      testRate: (testRate * 100).toFixed(2) + '%',
      improvement: ((testRate - controlRate) / controlRate * 100).toFixed(2) + '%'
    };
  }

  // 简单的正态分布 CDF
  normalCDF(x) {
    return 0.5 * (1 + this.erf(x / Math.sqrt(2)));
  }

  // 误差函数近似
  erf(x) {
    const a1 =  0.254829592;
    const a2 = -0.284496736;
    const a3 =  1.421413741;
    const a4 = -1.453152027;
    const a5 =  1.061405429;
    const p  =  0.3275911;

    const sign = x < 0 ? -1 : 1;
    x = Math.abs(x);

    const t = 1.0 / (1.0 + p * x);
    const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-x * x);

    return sign * y;
  }

  // 哈希函数
  hashCode(str) {
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      const char = str.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash;
    }
    return hash;
  }
}
```

```javascript
// 实时监控系统
class TurnstileMonitor {
  constructor(options = {}) {
    this.options = {
      alertThresholds: {
        successRate: 95,      // 成功率低于95%告警
        avgTime: 2000,        // 平均时间超过2秒告警
        suspiciousRate: 5,    // 可疑率超过5%告警
        errorRate: 5          // 错误率超过5%告警
      },
      checkInterval: 60000,  // 检查间隔：1分钟
      alertWebhook: options.alertWebhook,
      ...options
    };

    this.metrics = {
      success: 0,
      failed: 0,
      totalTime: 0,
      suspicious: 0,
      errors: 0
    };

    this.alerts = [];
  }

  // 开始监控
  start() {
    this.monitoringInterval = setInterval(() => {
      this.checkMetrics();
    }, this.options.checkInterval);

    console.log('Turnstile 监控已启动');
  }

  // 停止监控
  stop() {
    if (this.monitoringInterval) {
      clearInterval(this.monitoringInterval);
      this.monitoringInterval = null;
    }
  }

  // 记录指标
  recordMetric(type, value = null) {
    switch (type) {
      case 'success':
        this.metrics.success++;
        if (value) this.metrics.totalTime += value;
        break;
      case 'failed':
        this.metrics.failed++;
        break;
      case 'suspicious':
        this.metrics.suspicious++;
        break;
      case 'error':
        this.metrics.errors++;
        break;
    }
  }

  // 检查指标
  checkMetrics() {
    const total = this.metrics.success + this.metrics.failed;
    if (total === 0) return;

    // 计算各项指标
    const successRate = (this.metrics.success / total * 100).toFixed(2);
    const avgTime = this.metrics.totalTime / this.metrics.success;
    const suspiciousRate = (this.metrics.suspicious / total * 100).toFixed(2);
    const errorRate = (this.metrics.errors / total * 100).toFixed(2);

    // 检查是否需要告警
    const alerts = [];

    if (successRate < this.options.alertThresholds.successRate) {
      alerts.push({
        type: 'SUCCESS_RATE_LOW',
        message: `成功率过低: ${successRate}%`,
        value: successRate
      });
    }

    if (avgTime > this.options.alertThresholds.avgTime) {
      alerts.push({
        type: 'AVG_TIME_HIGH',
        message: `平均验证时间过长: ${avgTime.toFixed(2)}ms`,
        value: avgTime
      });
    }

    if (suspiciousRate > this.options.alertThresholds.suspiciousRate) {
      alerts.push({
        type: 'SUSPICIOUS_HIGH',
        message: `可疑行为率过高: ${suspiciousRate}%`,
        value: suspiciousRate
      });
    }

    if (errorRate > this.options.alertThresholds.errorRate) {
      alerts.push({
        type: 'ERROR_RATE_HIGH',
        message: `错误率过高: ${errorRate}%`,
        value: errorRate
      });
    }

    // 发送告警
    if (alerts.length > 0) {
      this.sendAlerts(alerts);
    }

    // 记录日志
    console.log('Turnstile 指标检查:', {
      successRate,
      avgTime: avgTime.toFixed(2) + 'ms',
      suspiciousRate: suspiciousRate + '%',
      errorRate: errorRate + '%',
      alerts: alerts.length
    });
  }

  // 发送告警
  async sendAlerts(alerts) {
    const alertPayload = {
      timestamp: new Date().toISOString(),
      service: 'Turnstile',
      alerts
    };

    // 记录告警
    this.alerts.push(...alerts);

    // 发送到 webhook
    if (this.options.alertWebhook) {
      try {
        await fetch(this.options.alertWebhook, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json'
          },
          body: JSON.stringify(alertPayload)
        });
      } catch (error) {
        console.error('发送告警失败:', error);
      }
    }

    // 发送到 Slack（示例）
    if (process.env.SLACK_WEBHOOK) {
      await this.sendSlackAlert(alerts);
    }
  }

  // 发送 Slack 告警
  async sendSlackAlert(alerts) {
    const slackPayload = {
      text: '🚨 Turnstile 告警',
      attachments: alerts.map(alert => ({
        color: 'danger',
        fields: [
          {
            title: '告警类型',
            value: alert.type,
            short: true
          },
          {
            title: '详情',
            value: alert.message,
            short: true
          }
        ],
        timestamp: new Date().getTime() / 1000
      }))
    };

    try {
      await fetch(process.env.SLACK_WEBHOOK, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(slackPayload)
      });
    } catch (error) {
      console.error('发送 Slack 告警失败:', error);
    }
  }

  // 获取当前状态
  getStatus() {
    const total = this.metrics.success + this.metrics.failed;

    return {
      metrics: this.metrics,
      total,
      successRate: total > 0 ? (this.metrics.success / total * 100).toFixed(2) + '%' : '0%',
      avgTime: this.metrics.success > 0
        ? (this.metrics.totalTime / this.metrics.success).toFixed(2) + 'ms'
        : '0ms',
      activeAlerts: this.alerts.filter(a => Date.now() - a.timestamp < 3600000), // 最近1小时
      monitoringActive: !!this.monitoringInterval
    };
  }
}

// 使用示例
const monitor = new TurnstileMonitor({
  alertWebhook: 'https://your-webhook-url.com/alerts'
});

// 开始监控
monitor.start();

// 记录验证结果
function onVerificationResult(result, duration) {
  if (result.success) {
    monitor.recordMetric('success', duration);
  } else {
    monitor.recordMetric('failed');
  }

  if (result.suspicious) {
    monitor.recordMetric('suspicious');
  }

  if (result.error) {
    monitor.recordMetric('error');
  }
}
```
