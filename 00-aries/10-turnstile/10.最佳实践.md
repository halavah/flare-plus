```javascript
class RequirementAnalyzer {
  constructor(checklist) {
    this.checklist = checklist;
  }

  // 分析并推荐配置
  analyze() {
    const recommendations = {
      turnstileMode: this.recommendMode(),
      securityLevel: this.recommendSecurityLevel(),
      integrationMethod: this.recommendIntegration(),
      additionalFeatures: this.recommendAdditionalFeatures()
    };

    return recommendations;
  }

  recommendMode() {
    if (this.checklist.userExperience.priority === 'high') {
      return 'invisible';
    } else if (this.checklist.security.level === 'high') {
      return 'managed';
    }
    return 'auto';
  }

  recommendSecurityLevel() {
    if (this.checklist.security.financialTransaction) {
      return 'strict';
    } else if (this.checklist.security.level === 'high') {
      return 'balanced';
    }
    return 'low';
  }

  recommendIntegration() {
    const framework = this.checklist.technical.framework;
    const frameworks = {
      react: 'react-component',
      vue: 'vue-component',
      angular: 'angular-service',
      vanilla: 'vanilla-js'
    };
    return frameworks[framework] || 'vanilla-js';
  }

  recommendAdditionalFeatures() {
    const features = [];

    if (this.checklist.compliance.gdpr) {
      features.push('consent-management');
    }

    if (this.checklist.compliance.accessibility) {
      features.push('aria-labels', 'keyboard-navigation');
    }

    if (this.checklist.userExperience.internationalUsers) {
      features.push('multi-language');
    }

    if (this.checklist.technical.serverSide === 'python') {
      features.push('flask-integration');
    }

    return features;
  }
}
```

```javascript
class SecureKeyManager {
  constructor() {
    this.keys = new Map();
    this.rotationInterval = 30 * 24 * 60 * 60 * 1000; // 30天
  }

  // 加载密钥
  async loadKeys() {
    // 从安全存储加载
    this.keys.set('production', {
      siteKey: await this.getSecureEnv('PROD_SITE_KEY'),
      secretKey: await this.getSecureEnv('PROD_SECRET_KEY')
    });

    this.keys.set('staging', {
      siteKey: await this.getSecureEnv('STAGING_SITE_KEY'),
      secretKey: await this.getSecureEnv('STAGING_SECRET_KEY')
    });
  }

  // 获取安全的环境变量
  async getSecureEnv(key) {
    // 使用密钥管理服务
    if (process.env.USE_VAULT) {
      return await this.getFromVault(key);
    }

    // 或使用加密的环境变量
    return this.decryptEnv(process.env[key]);
  }

  // 密钥轮换
  async rotateKeys() {
    console.log('开始密钥轮换...');

    // 生成新密钥对
    const newKeys = await this.generateNewKeys();

    // 验证新密钥
    const isValid = await this.validateKeys(newKeys);

    if (isValid) {
      // 更新密钥
      await this.updateKeys(newKeys);

      // 通知相关系统
      await this.notifyRotation();

      console.log('密钥轮换成功');
    } else {
      console.error('新密钥验证失败');
    }
  }

  // 启动自动轮换
  startAutoRotation() {
    setInterval(() => {
      this.rotateKeys();
    }, this.rotationInterval);
  }

  // 获取当前密钥
  getKey(environment) {
    return this.keys.get(environment);
  }

  // 验证请求签名
  verifySignature(payload, signature) {
    // 实现 HMAC 验证
    const crypto = require('crypto');
    const secret = this.getKey('production').secretKey;
    const expectedSignature = crypto
      .createHmac('sha256', secret)
      .update(JSON.stringify(payload))
      .digest('hex');

    return crypto.timingSafeEqual(
      Buffer.from(signature),
      Buffer.from(expectedSignature)
    );
  }
}

class TurnstileJWTValidator {
  constructor(secretKey) {
    this.secretKey = secretKey;
    this.issuer = 'https://challenges.cloudflare.com';
  }

  // 验证并解析 JWT（如果使用）
  async validateToken(token) {
    try {
      // 首先验证 Turnstile
      const turnstileResult = await this.verifyTurnstile(token);
      if (!turnstileResult.success) {
        return turnstileResult;
      }

      // 如果是 JWT，进行额外验证
      if (this.isJWT(token)) {
        return await this.validateJWT(token);
      }

      return turnstileResult;
    } catch (error) {
      return {
        success: false,
        error: error.message
      };
    }
  }

  isJWT(token) {
    return token.split('.').length === 3;
  }

  async validateJWT(token) {
    const jwt = require('jsonwebtoken');

    try {
      const decoded = jwt.verify(token, this.secretKey, {
        issuer: this.issuer,
        clockTolerance: 30 // 30秒容差
      });

      return {
        success: true,
        decoded,
        source: 'jwt'
      };
    } catch (error) {
      return {
        success: false,
        error: error.message,
        source: 'jwt'
      };
    }
  }
}
```

```javascript
// 性能优化管理器
class TurnstilePerformanceOptimizer {
  constructor() {
    this.loaded = false;
    this.loadPromise = null;
    this.preloadedWidgets = new Set();
  }

  // 智能预加载策略
  async smartPreload() {
    // 检测用户意图
    const userIntent = this.detectUserIntent();

    if (userIntent.likelyToVerify) {
      this.preloadScript();
    }

    // 监听滚动事件
    this.setupIntersectionObserver();
  }

  // 检测用户意图
  detectUserIntent() {
    const signals = {
      mouseMovement: false,
      scrollDepth: false,
      formFocus: false,
      timeOnPage: false
    };

    // 监听鼠标移动
    let mouseMovements = 0;
    document.addEventListener('mousemove', () => {
      mouseMovements++;
      signals.mouseMovement = mouseMovements > 5;
    });

    // 监听滚动
    let scrollPercentage = 0;
    window.addEventListener('scroll', () => {
      const scrollTop = window.pageYOffset;
      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
      scrollPercentage = (scrollTop / docHeight) * 100;
      signals.scrollDepth = scrollPercentage > 20;
    });

    // 监听表单焦点
    document.addEventListener('focusin', (e) => {
      if (e.target.matches('input, textarea, select')) {
        signals.formFocus = true;
      }
    });

    // 监听页面停留时间
    setTimeout(() => {
      signals.timeOnPage = true;
    }, 3000);

    // 评估意图
    const score = Object.values(signals).filter(Boolean).length;

    return {
      likelyToVerify: score >= 2,
      signals
    };
  }

  // 使用 Intersection Observer 优化加载
  setupIntersectionObserver() {
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting && !this.loaded) {
          this.preloadScript();
          observer.unobserve(entry.target);
        }
      });
    }, {
      rootMargin: '100px' // 提前100px开始加载
    });

    // 观察表单元素
    document.querySelectorAll('form').forEach(form => {
      observer.observe(form);
    });
  }

  // 延迟加载脚本
  async loadScript() {
    if (this.loadPromise) {
      return this.loadPromise;
    }

    this.loadPromise = new Promise((resolve, reject) => {
      const script = document.createElement('script');
      script.src = 'https://challenges.cloudflare.com/turnstile/v0/api.js';
      script.async = true;
      script.defer = true;

      script.onload = () => {
        this.loaded = true;
        resolve();
      };

      script.onerror = () => {
        reject(new Error('Failed to load Turnstile script'));
      };

      document.head.appendChild(script);
    });

    return this.loadPromise;
  }

  // 预加载脚本
  preloadScript() {
    if (!this.loaded && !this.loadPromise) {
      this.loadScript();
    }
  }

  // 预渲染 Widget
  async prerenderWidget(containerId, options) {
    await this.loadScript();

    // 在用户可能需要之前预先渲染
    turnstile.render(containerId, {
      ...options,
      'explicit-render': true // 不立即显示
    });

    this.preloadedWidgets.add(containerId);
  }

  // 显示预渲染的 Widget
  showWidget(containerId) {
    if (this.preloadedWidgets.has(containerId)) {
      const container = document.getElementById(containerId);
      const widget = container.querySelector('.cf-turnstile');
      if (widget) {
        widget.style.display = 'block';
      }
    }
  }
}
```

### 4. 错误处理和恢复

```javascript
// 健壮的错误处理系统
class TurnstileErrorHandler {
  constructor() {
    this.errorCount = 0;
    this.lastErrorTime = 0;
    this.retryStrategy = this.createRetryStrategy();
  }

  // 创建重试策略
  createRetryStrategy() {
    return {
      maxRetries: 3,
      baseDelay: 1000,
      maxDelay: 10000,
      backoffFactor: 2,
      jitter: true
    };
  }

  // 处理错误
  async handleError(error, context = {}) {
    this.errorCount++;
    this.lastErrorTime = Date.now();

    // 分类错误
    const errorType = this.classifyError(error);

    // 记录错误
    this.logError(error, errorType, context);

    // 采取相应措施
    switch (errorType) {
      case 'NETWORK_ERROR':
        return await this.handleNetworkError(error, context);

      case 'TOKEN_EXPIRED':
        return await this.handleExpiredToken(context);

      case 'INVALID_TOKEN':
        return await this.handleInvalidToken(context);

      case 'RATE_LIMITED':
        return await this.handleRateLimit(context);

      case 'CONFIGURATION_ERROR':
        return this.handleConfigurationError(error, context);

      default:
        return this.handleUnknownError(error, context);
    }
  }

  // 错误分类
  classifyError(error) {
    const message = error.message || error.toString().toLowerCase();

    if (message.includes('network') || message.includes('fetch')) {
      return 'NETWORK_ERROR';
    }

    if (message.includes('expired') || message.includes('timeout')) {
      return 'TOKEN_EXPIRED';
    }

    if (message.includes('invalid') || message.includes('unauthorized')) {
      return 'INVALID_TOKEN';
    }

    if (message.includes('rate limit') || message.includes('too many')) {
      return 'RATE_LIMITED';
    }

    if (message.includes('sitekey') || message.includes('secret')) {
      return 'CONFIGURATION_ERROR';
    }

    return 'UNKNOWN_ERROR';
  }

  // 处理网络错误
  async handleNetworkError(error, context) {
    // 使用指数退避重试
    for (let attempt = 0; attempt < this.retryStrategy.maxRetries; attempt++) {
      try {
        const delay = this.calculateDelay(attempt);
        await this.sleep(delay);

        // 重试请求
        const result = await this.retryVerification(context);
        if (result.success) {
          return result;
        }
      } catch (retryError) {
        if (attempt === this.retryStrategy.maxRetries - 1) {
          break;
        }
      }
    }

    // 所有重试失败，使用备用方案
    return this.fallbackToAlternative();
  }

  // 计算重试延迟
  calculateDelay(attempt) {
    let delay = this.retryStrategy.baseDelay *
                Math.pow(this.retryStrategy.backoffFactor, attempt);

    if (this.retryStrategy.jitter) {
      // 添加随机抖动
      delay = delay * (0.5 + Math.random());
    }

    return Math.min(delay, this.retryStrategy.maxDelay);
  }

  // 处理过期 Token
  async handleExpiredToken(context) {
    // 自动刷新 Token
    try {
      return await this.refreshToken(context);
    } catch (error) {
      // 刷新失败，提示用户
      this.showUserMessage('验证已过期，请重试', 'warning');
      throw error;
    }
  }

  // 备用验证方案
  fallbackToAlternative() {
    // 可以实现备用验证码
    console.log('使用备用验证方案');

    return {
      success: false,
      useAlternative: true,
      message: '主验证服务不可用，使用备用方案'
    };
  }

  // 显示用户友好的错误消息
  showUserMessage(message, type = 'error') {
    // 创建消息元素
    const messageEl = document.createElement('div');
    messageEl.className = `turnstile-message turnstile-${type}`;
    messageEl.textContent = message;

    // 添加到页面
    const container = document.querySelector('.turnstile-container') || document.body;
    container.appendChild(messageEl);

    // 自动移除
    setTimeout(() => {
      messageEl.remove();
    }, 5000);
  }

  // 记录错误
  logError(error, type, context) {
    const errorLog = {
      timestamp: new Date().toISOString(),
      type,
      message: error.message,
      stack: error.stack,
      context,
      userAgent: navigator.userAgent,
      url: window.location.href
    };

    // 发送到错误追踪服务
    this.sendErrorTracking(errorLog);

    // 控制台日志（开发环境）
    if (process.env.NODE_ENV === 'development') {
      console.error('Turnstile Error:', errorLog);
    }
  }

  // 发送错误追踪
  async sendErrorTracking(errorLog) {
    // 发送到错误追踪服务
    try {
      await fetch('/api/log-error', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(errorLog)
      });
    } catch (e) {
      // 静默失败
    }
  }

  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

### 5. 监控和告警

```javascript
// 生产监控系统
class TurnstileMonitor {
  constructor(options = {}) {
    this.options = {
      alertThresholds: {
        errorRate: 0.05,      // 5%
        responseTime: 2000,   // 2秒
        failureRate: 0.02,    // 2%
        suspiciousRate: 0.1   // 10%
      },
      checkInterval: 60000,   // 1分钟
      alertWebhook: options.alertWebhook,
      slackWebhook: options.slackWebhook,
      ...options
    };

    this.metrics = {
      total: 0,
      success: 0,
      failed: 0,
      totalTime: 0,
      suspicious: 0,
      errors: []
    };

    this.alerts = [];
    this.startMonitoring();
  }

  // 开始监控
  startMonitoring() {
    setInterval(() => {
      this.checkMetrics();
    }, this.options.checkInterval);

    // 页面卸载时发送最终指标
    window.addEventListener('beforeunload', () => {
      this.sendFinalMetrics();
    });
  }

  // 记录验证指标
  recordVerification(result, duration) {
    this.metrics.total++;
    this.metrics.totalTime += duration;

    if (result.success) {
      this.metrics.success++;
    } else {
      this.metrics.failed++;
      this.metrics.errors.push({
        timestamp: Date.now(),
        error: result.error
      });
    }

    if (result.suspicious) {
      this.metrics.suspicious++;
    }
  }

  // 检查指标
  checkMetrics() {
    if (this.metrics.total === 0) return;

    const metrics = this.calculateMetrics();
    const alerts = this.detectAnomalies(metrics);

    if (alerts.length > 0) {
      this.sendAlerts(alerts, metrics);
    }
  }

  // 计算关键指标
  calculateMetrics() {
    return {
      errorRate: this.metrics.failed / this.metrics.total,
      successRate: this.metrics.success / this.metrics.total,
      avgResponseTime: this.metrics.totalTime / this.metrics.total,
      suspiciousRate: this.metrics.suspicious / this.metrics.total,
      totalVerifications: this.metrics.total,
      timeWindow: this.options.checkInterval / 1000 // 秒
    };
  }

  // 检测异常
  detectAnomalies(metrics) {
    const alerts = [];

    if (metrics.errorRate > this.options.alertThresholds.errorRate) {
      alerts.push({
        type: 'HIGH_ERROR_RATE',
        severity: 'critical',
        message: `错误率过高: ${(metrics.errorRate * 100).toFixed(2)}%`,
        value: metrics.errorRate
      });
    }

    if (metrics.avgResponseTime > this.options.alertThresholds.responseTime) {
      alerts.push({
        type: 'SLOW_RESPONSE',
        severity: 'warning',
        message: `响应时间过长: ${metrics.avgResponseTime.toFixed(2)}ms`,
        value: metrics.avgResponseTime
      });
    }

    if (metrics.suspiciousRate > this.options.alertThresholds.suspiciousRate) {
      alerts.push({
        type: 'HIGH_SUSPICIOUS_RATE',
        severity: 'warning',
        message: `可疑行为率过高: ${(metrics.suspiciousRate * 100).toFixed(2)}%`,
        value: metrics.suspiciousRate
      });
    }

    return alerts;
  }

  // 发送告警
  async sendAlerts(alerts, metrics) {
    const alertPayload = {
      timestamp: new Date().toISOString(),
      service: 'Turnstile',
      environment: process.env.NODE_ENV,
      metrics,
      alerts
    };

    // 保存告警
    this.alerts.push(...alerts);

    // 发送 Webhook
    if (this.options.alertWebhook) {
      await this.sendWebhook(this.options.alertWebhook, alertPayload);
    }

    // 发送 Slack
    if (this.options.slackWebhook) {
      await this.sendSlackAlert(alertPayload);
    }
  }

  // 发送 Webhook
  async sendWebhook(url, payload) {
    try {
      await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
    } catch (error) {
      console.error('发送 Webhook 失败:', error);
    }
  }

  // 发送 Slack 告警
  async sendSlackAlert(payload) {
    const slackMessage = {
      text: `🚨 Turnstile 监控告警`,
      attachments: [{
        color: 'danger',
        fields: [
          {
            title: '环境',
            value: payload.environment,
            short: true
          },
          {
            title: '时间',
            value: payload.timestamp,
            short: true
          },
          {
            title: '告警详情',
            value: payload.alerts.map(a => a.message).join('\n'),
            short: false
          }
        ]
      }]
    };

    try {
      await fetch(this.options.slackWebhook, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(slackMessage)
      });
    } catch (error) {
      console.error('发送 Slack 告警失败:', error);
    }
  }

  // 获取监控报告
  getReport() {
    const metrics = this.calculateMetrics();

    return {
      current: metrics,
      alerts: this.alerts.slice(-10), // 最近10个告警
      uptime: this.calculateUptime(),
      recommendations: this.generateRecommendations(metrics)
    };
  }

  // 生成建议
  generateRecommendations(metrics) {
    const recommendations = [];

    if (metrics.errorRate > 0.03) {
      recommendations.push({
        type: 'performance',
        message: '错误率偏高，建议检查网络连接和服务器状态'
      });
    }

    if (metrics.avgResponseTime > 1500) {
      recommendations.push({
        type: 'optimization',
        message: '响应时间较长，考虑启用预加载或优化脚本加载'
      });
    }

    if (metrics.suspiciousRate > 0.05) {
      recommendations.push({
        type: 'security',
        message: '可疑活动增加，建议加强安全措施'
      });
    }

    return recommendations;
  }
}
```

```javascript
// GDPR 合规管理器
class GDPRComplianceManager {
  constructor() {
    this.consent = this.loadConsent();
    this.consentBanner = null;
  }

  // 加载用户同意
  loadConsent() {
    const saved = localStorage.getItem('turnstile_consent');
    return saved ? JSON.parse(saved) : null;
  }

  // 保存用户同意
  saveConsent(consent) {
    localStorage.setItem('turnstile_consent', JSON.stringify(consent));
    this.consent = consent;

  // 检查是否需要同意
  needsConsent() {
    // 检查用户位置（基于IP或语言）
```

```javascript
// 可访问性增强
class TurnstileAccessibility {
  constructor() {
    this.announcements = [];
    this.setupARIA();
  }

  // 设置 ARIA 标签
  setupARIA() {
    const observer = new MutationObserver((mutations) => {
      mutations.forEach((mutation) => {
        mutation.addedNodes.forEach((node) => {
          if (node.classList && node.classList.contains('cf-turnstile')) {
            this.enhanceAccessibility(node);
          }
        });
      });
    });

    observer.observe(document.body, {
      childList: true,
      subtree: true
    });
  }

  // 增强可访问性
  enhanceAccessibility(element) {
    // 添加 ARIA 标签
    element.setAttribute('role', 'group');
    element.setAttribute('aria-label', '安全验证');
    element.setAttribute('aria-describedby', 'turnstile-description');

    // 添加描述
    const description = document.createElement('div');
    description.id = 'turnstile-description';
    description.className = 'sr-only';
    description.textContent = '请完成安全验证以继续操作';
    element.parentNode.insertBefore(description, element);

    // 添加键盘支持
    this.addKeyboardSupport(element);

    // 添加状态通知
    this.setupStatusAnnouncements(element);
  }

  // 添加键盘支持
  addKeyboardSupport(element) {
    element.setAttribute('tabindex', '0');

    element.addEventListener('keydown', (e) => {
      switch (e.key) {
        case 'Enter':
        case ' ':
          e.preventDefault();
          this.activateVerification(element);
          break;
        case 'Escape':
          e.preventDefault();
          this.cancelVerification(element);
          break;
      }
    });
  }

  // 设置状态通知
  setupStatusAnnouncements(element) {
    const announcer = document.createElement('div');
    announcer.setAttribute('aria-live', 'polite');
    announcer.setAttribute('aria-atomic', 'true');
    announcer.className = 'sr-only';
    element.parentNode.appendChild(announcer);

    // 监听状态变化
    const observer = new MutationObserver((mutations) => {
      mutations.forEach((mutation) => {
        if (mutation.type === 'attributes' &&
            mutation.attributeName === 'aria-expanded') {
          const isExpanded = element.getAttribute('aria-expanded') === 'true';
          this.announce(announcer, isExpanded ? '验证已展开' : '验证已关闭');
        }
      });
    });

    observer.observe(element, { attributes: true });
  }

  // 发送屏幕阅读器通知
  announce(element, message) {
    element.textContent = message;

    // 清除旧消息
    setTimeout(() => {
      element.textContent = '';
    }, 1000);
  }

  // 激活验证
  activateVerification(element) {
    element.click();
    this.announce(element, '正在激活验证');
  }

  // 取消验证
  cancelVerification(element) {
    if (window.turnstile) {
      const widgets = turnstile.getAll();
      widgets.forEach(widget => turnstile.reset(widget));
    }
    this.announce(element, '验证已取消');
  }
}
```

```javascript
class DeploymentChecklist {
  constructor() {
    this.items = [
      {
        category: '安全',
        items: [
          { id: 'prod-keys', description: '使用生产环境密钥', required: true },
          { id: 'https', description: '启用 HTTPS', required: true },
          { id: 'csp', description: '配置 CSP 头', required: false },
          { id: 'rate-limit', description: '设置速率限制', required: true }
        ]
      },
      {
        category: '性能',
        items: [
          { id: 'cdn', description: '使用 CDN', required: true },
          { id: 'compression', description: '启用 Gzip 压缩', required: true },
          { id: 'caching', description: '配置缓存策略', required: false },
          { id: 'preload', description: '启用预加载', required: false }
        ]
      },
      {
        category: '监控',
        items: [
          { id: 'analytics', description: '配置分析工具', required: true },
          { id: 'error-tracking', description: '设置错误追踪', required: true },
          { id: 'alerts', description: '配置告警通知', required: true },
          { id: 'logging', description: '启用详细日志', required: false }
        ]
      },
      {
        category: '合规',
        items: [
          { id: 'privacy-policy', description: '更新隐私政策', required: true },
          { id: 'consent', description: '实现同意管理', required: true },
          { id: 'accessibility', description: '通过可访问性测试', required: false },
          { id: 'data-retention', description: '配置数据保留策略', required: true }
        ]
      }
    ];
  }

  // 运行所有检查
  async runChecks() {
    const results = {};

    for (const category of this.items) {
      results[category.category] = {};

      for (const item of category.items) {
        try {
          const result = await this.checkItem(item);
          results[category.category][item.id] = result;
        } catch (error) {
          results[category.category][item.id] = {
            passed: false,
            error: error.message
          };
        }
      }
    }

    return results;
  }

  // 检查单个项目
  async checkItem(item) {
    switch (item.id) {
      case 'https':
        return this.checkHTTPS();

      case 'prod-keys':
        return this.checkProductionKeys();

      case 'csp':
        return this.checkCSP();

      case 'rate-limit':
        return this.checkRateLimit();

      case 'cdn':
        return this.checkCDN();

      case 'analytics':
        return this.checkAnalytics();

      case 'error-tracking':
        return this.checkErrorTracking();

      default:
        return { passed: true, message: '未实现检查' };
    }
  }

  // 检查 HTTPS
  checkHTTPS() {
    const isHTTPS = window.location.protocol === 'https:';
    return {
      passed: isHTTPS,
      message: isHTTPS ? 'HTTPS 已启用' : '未使用 HTTPS'
    };
  }

  // 检查生产密钥
  checkProductionKeys() {
    // 这里应该检查是否使用了正确的生产密钥
    const siteKey = document.querySelector('[data-sitekey]')?.getAttribute('data-sitekey');
    const isProduction = !siteKey?.includes('test') && !siteKey?.includes('dev');

    return {
      passed: isProduction,
      message: isProduction ? '使用生产密钥' : '使用测试密钥'
    };
  }

  // 生成检查报告
  generateReport(results) {
    const report = {
      timestamp: new Date().toISOString(),
      summary: {
        total: 0,
        passed: 0,
        failed: 0,
        warnings: 0
      },
      details: {}
    };

    for (const [category, items] of Object.entries(results)) {
      report.details[category] = {};

      for (const [id, result] of Object.entries(items)) {
        const item = this.findItem(id);

        report.summary.total++;

        if (result.passed) {
          report.summary.passed++;
        } else {
          if (item.required) {
            report.summary.failed++;
          } else {
            report.summary.warnings++;
          }
        }

        report.details[category][id] = {
          description: item.description,
          required: item.required,
          ...result
        };
      }
    }

    report.summary.status = report.summary.failed === 0 ? 'PASSED' : 'FAILED';

    return report;
  }

  findItem(id) {
    for (const category of this.items) {
      const item = category.items.find(i => i.id === id);
      if (item) return item;
    }
    return { description: '未知项目', required: false };
  }
}

// 部署前运行检查
async function preDeploymentCheck() {
  const checklist = new DeploymentChecklist();
  const results = await checklist.runChecks();
  const report = checklist.generateReport(results);

  console.log('部署前检查报告:');
  console.log(`状态: ${report.summary.status}`);
  console.log(`通过: ${report.summary.passed}/${report.summary.total}`);

  if (report.summary.failed > 0) {
    console.error('\n失败的检查项目:');
    for (const [category, items] of Object.entries(report.details)) {
      for (const [id, item] of Object.entries(items)) {
        if (!item.passed && item.required) {
          console.error(`- [${category}] ${item.description}: ${item.message || item.error}`);
        }
      }
    }

    return false;
  }

  return true;
}
```
