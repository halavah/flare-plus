```javascript
const { expect } = require('chai');
const sinon = require('sinon');
const TurnstileVerifier = require('../src/turnstile-verifier');

describe('TurnstileVerifier', () => {
  let verifier;
  let fetchStub;

  beforeEach(() => {
    verifier = new TurnstileVerifier({
      secretKey: 'test-secret-key'
    });

    fetchStub = sinon.stub(global, 'fetch');
  });

  afterEach(() => {
    fetchStub.restore();
  });

  describe('verifyToken', () => {
    it('should verify valid token successfully', async () => {
      // 模拟成功响应
      fetchStub.resolves({
        ok: true,
        json: () => Promise.resolve({
          success: true,
          challenge_ts: '2023-01-01T00:00:00Z',
          hostname: 'example.com'
        })
      });

      const result = await verifier.verifyToken('valid-token', '127.0.0.1');

      expect(result.success).to.be.true;
      expect(fetchStub.calledOnce).to.be.true;

      const callArgs = fetchStub.getCall(0).args[0];
      expect(callArgs.url).to.equal('https://challenges.cloudflare.com/turnstile/v0/siteverify');
      expect(callArgs.method).to.equal('POST');
    });

    it('should handle invalid token', async () => {
      // 模拟失败响应
      fetchStub.resolves({
        ok: true,
        json: () => Promise.resolve({
          success: false,
          'error-codes': ['invalid-input-response']
        })
      });

      const result = await verifier.verifyToken('invalid-token');

      expect(result.success).to.be.false;
      expect(result['error-codes']).to.include('invalid-input-response');
    });

    it('should handle network errors', async () => {
      // 模拟网络错误
      fetchStub.rejects(new Error('Network error'));

      const result = await verifier.verifyToken('test-token');

      expect(result.success).to.be.false;
      expect(result.error).to.equal('Network error');
    });

    it('should include remote IP when provided', async () => {
      fetchStub.resolves({
        ok: true,
        json: () => Promise.resolve({ success: true })
      });

      await verifier.verifyToken('token', '192.168.1.1');

      const requestBody = fetchStub.getCall(0).args[1].body;
      expect(requestBody).to.include('remoteip=192.168.1.1');
    });
  });

  describe('parseErrorCodes', () => {
    it('should parse known error codes', () => {
      const errors = ['missing-input-secret', 'invalid-input-response'];
      const result = verifier.parseErrorCodes(errors);

      expect(result).to.deep.equal({
        type: 'configuration_error',
        message: '密钥缺失或无效',
        action: '检查密钥配置'
      });
    });

    it('should handle unknown error codes', () => {
      const errors = ['unknown-error-code'];
      const result = verifier.parseErrorCodes(errors);

      expect(result.type).to.equal('unknown_error');
      expect(result.message).to.include('unknown-error-code');
    });
  });
});
```

### 3. E2E 测试框架

```javascript
// e2e-test-framework.js
class TurnstileE2ETestSuite {
  constructor() {
    this.tests = [];
    this.results = [];
  }

  // 添加测试
  addTest(name, testFunction) {
    this.tests.push({ name, testFunction });
  }

  // 运行所有测试
  async runAllTests() {
    console.log('开始运行 E2E 测试套件...\n');

    for (const test of this.tests) {
      const result = await this.runSingleTest(test);
      this.results.push(result);

      if (result.passed) {
        console.log(`✓ ${test.name}`);
      } else {
        console.log(`✗ ${test.name}`);
        console.log(`  错误: ${result.error}`);
      }
    }

    this.printSummary();
    return this.results;
  }

  // 运行单个测试
  async runSingleTest(test) {
    const startTime = Date.now();

    try {
      await test.testFunction();
      const duration = Date.now() - startTime;

      return {
        name: test.name,
        passed: true,
        duration,
        error: null
      };
    } catch (error) {
      const duration = Date.now() - startTime;

      return {
        name: test.name,
        passed: false,
        duration,
        error: error.message
      };
    }
  }

  // 打印测试摘要
  printSummary() {
    const passed = this.results.filter(r => r.passed).length;
    const failed = this.results.filter(r => !r.passed).length;
    const totalDuration = this.results.reduce((sum, r) => sum + r.duration, 0);

    console.log('\n=== 测试摘要 ===');
    console.log(`总测试数: ${this.results.length}`);
    console.log(`通过: ${passed}`);
    console.log(`失败: ${failed}`);
    console.log(`总耗时: ${totalDuration}ms`);
    console.log(`成功率: ${((passed / this.results.length) * 100).toFixed(2)}%`);
  }

  // 生成测试报告
  generateReport() {
    const report = {
      timestamp: new Date().toISOString(),
      summary: {
        total: this.results.length,
        passed: this.results.filter(r => r.passed).length,
        failed: this.results.filter(r => !r.passed).length,
        totalDuration: this.results.reduce((sum, r) => sum + r.duration, 0),
        successRate: (this.results.filter(r => r.passed).length / this.results.length * 100).toFixed(2)
      },
      tests: this.results
    };

    return report;
  }
}

// 测试辅助工具
class TurnstileTestHelper {
  constructor(browser) {
    this.browser = browser;
  }

  // 创建测试页面
  async createTestPage(options = {}) {
    const page = await this.browser.newPage();

    // 设置默认超时
    page.setDefaultTimeout(30000);

    // 拦截网络请求
    await page.setRequestInterception(true);
    page.on('request', (request) => {
      // 允许必要请求
      if (request.url().includes('challenges.cloudflare.com') ||
          request.url().includes('analytics') ||
          request.resourceType() === 'document' ||
          request.resourceType() === 'script' ||
          request.resourceType() === 'stylesheet') {
        request.continue();
      } else {
        request.abort();
      }
    });

    return page;
  }

  // 等待验证完成
  async waitForVerification(page, timeout = 30000) {
    return await page.waitForFunction(() => {
      return window.turnstileResponse;
    }, { timeout });
  }

  // 模拟用户行为
  async simulateHumanBehavior(page) {
    // 随机移动鼠标
    await page.mouse.move(
      Math.random() * 800,
      Math.random() * 600
    );

    // 随机滚动
    await page.evaluate(() => {
      window.scrollBy(0, Math.random() * 200);
    });

    // 等待随机时间
    await new Promise(resolve => setTimeout(resolve, Math.random() * 1000));
  }

  // 检查验证结果
  async getVerificationResult(page) {
    return await page.evaluate(() => {
      return window.turnstileResponse || null;
    });
  }
}

// 使用示例
async function runE2ETests() {
  const browser = await puppeteer.launch({ headless: false });
  const testSuite = new TurnstileE2ETestSuite();
  const helper = new TurnstileTestHelper(browser);

  // 添加测试用例
  testSuite.addTest('基本验证流程', async () => {
    const page = await helper.createTestPage();
    await page.goto('http://localhost:3000/test');

    // 等待 Turnstile 加载
    await page.waitForSelector('.cf-turnstile');

    // 模拟人类行为
    await helper.simulateHumanBehavior(page);

    // 完成验证
    await page.click('.cf-turnstile');

    // 等待验证完成
    await helper.waitForVerification(page);

    const result = await helper.getVerificationResult(page);
    if (!result) {
      throw new Error('验证未完成');
    }

    await page.close();
  });

  testSuite.addTest('表单提交测试', async () => {
    const page = await helper.createTestPage();
```

```javascript
const browsers = [

async function runBrowserCompatibilityTests() {
  const results = [];

  for (const browser of browsers) {
    console.log(`测试 ${browser.name} ${browser.version} (${browser.platform})`);

    const result = await testBrowser(browser);
    results.push(result);

    console.log(`${browser.name}: ${result.passed ? '✓' : '✗'}`);
  }

  return results;
}

async function testBrowser(browserConfig) {
  const capabilities = {
    browserName: browserConfig.name.toLowerCase(),
    browserVersion: browserConfig.version,
    platformName: browserConfig.platform === 'mobile' ? 'Android' : 'Windows',
  };

  try {
    // 使用 WebDriver 或 Puppeteer 启动浏览器
    const driver = await createDriver(capabilities);

    // 运行基本测试
    const testResult = await runBasicTests(driver);

    await driver.quit();

    return {
      browser: browserConfig.name,
      platform: browserConfig.platform,
      passed: testResult.passed,
      issues: testResult.issues || []
    };
  } catch (error) {
    return {
      browser: browserConfig.name,
      platform: browserConfig.platform,
      passed: false,
      issues: [error.message]
    };
  }
}
```
