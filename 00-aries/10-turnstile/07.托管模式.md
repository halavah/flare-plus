# Cloudflare Turnstile 托管模式

## 托管模式概述

Cloudflare Turnstile 提供了三种主要的托管模式，每种模式都有其特定的使用场景和优势。

## 模式类型

### 1. 托管 Widget (Managed Widget)

#### 特点
- Cloudflare 完全托管所有资源
- 最简单的集成方式
- 自动处理所有验证逻辑
- 最高的可靠性

#### 使用场景
- 大多数网站的标准使用
- 快速集成需求
- 不想管理验证逻辑的网站
- 需要最高可靠性的场景

#### 实现示例
```html
<!DOCTYPE html>
<html>
<head>
  <title>托管 Widget 示例</title>
  <script src="https://challenges.cloudflare.com/turnstile/v0/api.js" async defer></script>
</head>
<body>
  <form action="/submit" method="POST">
    <input type="email" name="email" placeholder="邮箱" required>
    <input type="password" name="password" placeholder="密码" required>

    <!-- 托管 Widget -->
    <div class="cf-turnstile"
         data-sitekey="YOUR_SITE_KEY"
         data-callback="onSuccess"
         data-error-callback="onError">
    </div>

    <button type="submit">登录</button>
  </form>

  <script>
  function onSuccess(token) {
    console.log('验证成功:', token);
    // 可以自动提交表单
    document.forms[0].submit();
  }

  function onError(error) {
    console.error('验证失败:', error);
    alert('验证失败，请重试');
  }
  </script>
</body>
</html>
```

### 2. 非托管模式 (Non-Managed)

#### 特点
- 自托管验证资源
- 更多的控制权
- 需要额外配置
- 适合特殊需求

#### 使用场景
- 需要完全控制验证流程
- 离线环境或特殊网络需求
- 需要自定义验证逻辑
- 符合特定合规要求

#### 实现示例
```html
<!DOCTYPE html>
<html>
<head>
  <title>非托管模式示例</title>
</head>
<body>
  <form id="secureForm">
    <input type="text" name="username" placeholder="用户名" required>

    <!-- 自定义验证触发器 -->
    <button type="button" onclick="startVerification()">
      开始安全验证
    </button>

    <div id="status"></div>
    <button type="submit" disabled>提交</button>
  </form>

  <script>
  let verificationToken = null;

  // 自定义验证流程
  async function startVerification() {
    const statusEl = document.getElementById('status');
    const submitBtn = document.querySelector('button[type="submit"]');

    try {
      statusEl.textContent = '正在验证...';

      // 自定义验证逻辑
      const challenge = await generateChallenge();
      const response = await solveChallenge(challenge);

      // 验证响应
      const result = await verifyResponse(response);

      if (result.success) {
        verificationToken = result.token;
        statusEl.textContent = '验证成功 ✓';
        submitBtn.disabled = false;
      } else {
        throw new Error('验证失败');
      }

    } catch (error) {
      statusEl.textContent = '验证失败: ' + error.message;
      verificationToken = null;
      submitBtn.disabled = true;
    }
  }

  // 生成挑战
  async function generateChallenge() {
    // 自定义挑战生成逻辑
    const response = await fetch('/api/generate-challenge', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        sitekey: 'YOUR_SITE_KEY',
        action: 'custom_verification'
      })
    });

    return response.json();
  }

  // 解决挑战
  async function solveChallenge(challenge) {
    // 实现挑战解决逻辑
    // 可以是密码学计算、机器学习推理等
    return {
      solution: computeSolution(challenge),
      timestamp: Date.now()
    };
  }

  // 验证响应
  async function verifyResponse(response) {
    const verifyResponse = await fetch('https://challenges.cloudflare.com/turnstile/v0/siteverify', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/x-www-form-urlencoded'
      },
      body: `secret=${YOUR_SECRET_KEY}&response=${response.token}`
    });

    return verifyResponse.json();
  }

  // 表单提交
  document.getElementById('secureForm').addEventListener('submit', async (e) => {
    e.preventDefault();

    if (!verificationToken) {
      alert('请先完成验证');
      return;
    }

    // 提交表单
    const formData = new FormData(e.target);
    formData.append('turnstile-token', verificationToken);

    const response = await fetch('/api/submit', {
      method: 'POST',
      body: formData
    });

    const result = await response.json();
    if (result.success) {
      alert('提交成功');
    } else {
      alert('提交失败');
    }
  });
  </script>
</body>
</html>
```

### 3. 不可见模式 (Invisible)

#### 特点
- 用户完全无感知
- 自动在后台进行验证
- 最佳的用户体验
- 智能风险检测

#### 使用场景
- 需要无缝用户体验
- 高转化率要求的页面
- 移动应用和SPA
- 不想中断用户流程

#### 实现示例
```html
<!DOCTYPE html>
<html>
<head>
  <title>不可见验证示例</title>
  <script src="https://challenges.cloudflare.com/turnstile/v0/api.js" async defer></script>
</head>
<body>
  <form id="invisibleForm" action="/submit" method="POST">
    <input type="email" name="email" placeholder="邮箱" required>
    <input type="password" name="password" placeholder="密码" required>

    <!-- 不可见验证容器 -->
    <div id="turnstile-invisible" style="display: none;">
      <div class="cf-turnstile"
           data-sitekey="YOUR_SITE_KEY"
           data-size="invisible"
           data-callback="onInvisibleSuccess">
      </div>
    </div>

    <button type="submit">登录</button>
  </form>

  <script>
  let formSubmitted = false;

  // 表单提交处理
  document.getElementById('invisibleForm').addEventListener('submit', async (e) => {
    e.preventDefault();

    if (formSubmitted) {
      return; // 防止重复提交
    }

    formSubmitted = true;
    const submitBtn = e.target.querySelector('button[type="submit"]');
    submitBtn.disabled = true;
    submitBtn.textContent = '验证中...';

    // 触发不可见验证
    turnstile.render('#turnstile-invisible');
  });

  // 验证成功回调
  async function onInvisibleSuccess(token) {
    const form = document.getElementById('invisibleForm');
    const submitBtn = form.querySelector('button[type="submit"]');

    // 添加 token 到表单
    const hiddenInput = document.createElement('input');
    hiddenInput.type = 'hidden';
    hiddenInput.name = 'turnstile-token';
    hiddenInput.value = token;
    form.appendChild(hiddenInput);

    // 提交表单
    try {
      const response = await fetch(form.action, {
        method: form.method,
        body: new FormData(form)
      });

      const result = await response.json();

      if (result.success) {
        // 跳转到成功页面
        window.location.href = '/dashboard';
      } else {
        throw new Error(result.error || '登录失败');
      }
    } catch (error) {
      alert(error.message);
      submitBtn.disabled = false;
      submitBtn.textContent = '登录';
      formSubmitted = false;
    }
  }

  // 页面加载时预初始化
  window.onload = function() {
    turnstile.ready(function() {
      // 预渲染但不显示
      turnstile.render('#turnstile-invisible', {
        sitekey: 'YOUR_SITE_KEY',
        size: 'invisible',
        'callback': onInvisibleSuccess
      });
    });
  };
  </script>
</body>
</html>
```

## 模式选择指南

### 决策矩阵

| 需求 | 托管 Widget | 非托管 | 不可见 |
|------|-------------|--------|--------|
| 快速集成 | ✓✓✓ | ✓ | ✓✓ |
| 完全控制 | ✓ | ✓✓✓ | ✓ |
| 无感知体验 | ✓ | ✓ | ✓✓✓ |
| 离线支持 | ✗ | ✓✓✓ | ✗ |
| 自定义UI | ✓ | ✓✓✓ | ✓ |
| 维护成本 | 低 | 高 | 中 |
| 可靠性 | 高 | 中 | 高 |

### 选择建议

#### 选择托管 Widget 当：
- 你是初次使用 Turnstile
- 需要快速集成
- 不想管理验证逻辑
- 需要最高的可靠性

#### 选择非托管当：
- 需要完全控制验证流程
- 有特殊的合规要求
- 需要离线或内网部署
- 需要自定义验证算法

#### 选择不可见当：
- 用户体验是首要考虑
- 不想中断用户流程
- 页面转化率至关重要
- 可以接受稍高的复杂度

### 1. 动态模式切换

```javascript
class TurnstileModeManager {
  constructor() {
    this.currentMode = 'managed';
    this.modeConfigs = {
      managed: {
        container: '#managed-turnstile',
        sitekey: 'MANAGED_SITE_KEY',
        options: {
          callback: this.onSuccess.bind(this),
          'error-callback': this.onError.bind(this)
        }
      },
      invisible: {
        container: '#invisible-turnstile',
        sitekey: 'INVISIBLE_SITE_KEY',
        options: {
          size: 'invisible',
          callback: this.onSuccess.bind(this),
          'error-callback': this.onError.bind(this)
        }
      },
      non_managed: {
        sitekey: 'NON_MANAGED_SITE_KEY',
        customHandler: this.customHandler.bind(this)
      }
    };
  }

  // 根据风险选择模式
  selectMode(riskScore) {
    if (riskScore < 0.3) {
      // 低风险：不可见模式
      return 'invisible';
    } else if (riskScore < 0.7) {
      // 中风险：托管 Widget
      return 'managed';
    } else {
      // 高风险：非托管模式
      return 'non_managed';
    }
  }

  // 动态切换模式
  async switchMode(mode) {
    if (mode === this.currentMode) return;

    // 清理当前模式
    await this.cleanupCurrentMode();

    // 初始化新模式
    this.currentMode = mode;
    await this.initializeMode(mode);
  }

  async initializeMode(mode) {
    const config = this.modeConfigs[mode];

    switch (mode) {
      case 'managed':
      case 'invisible':
        const container = document.querySelector(config.container);
        container.style.display = 'block';
        turnstile.render(container, config.options);
        break;

      case 'non_managed':
        // 初始化自定义验证处理器
        this.setupCustomVerification(config);
        break;
    }
  }

  async cleanupCurrentMode() {
    // 清理当前 Widget
    const widgets = turnstile.getAll();
    widgets.forEach(widget => turnstile.remove(widget));

    // 隐藏容器
    document.querySelectorAll('[id$="-turnstile"]').forEach(
      container => container.style.display = 'none'
    );
  }

  // 自定义验证处理器
  setupCustomVerification(config) {
    // 实现自定义验证逻辑
    console.log('设置自定义验证');
  }

  // 成功回调
  onSuccess(token) {
    console.log('验证成功:', token);
    this.handleToken(token);
  }

  // 错误回调
  onError(error) {
    console.error('验证失败:', error);
    this.handleError(error);
  }

  // 处理 token
  handleToken(token) {
    // 统一的 token 处理逻辑
  }

  // 处理错误
  handleError(error) {
    // 统一的错误处理逻辑
  }
}

// 使用示例
const modeManager = new TurnstileModeManager();

// 根据风险评估选择模式
async function initTurnstile() {
  const riskScore = await assessRisk();
  const mode = modeManager.selectMode(riskScore);
  await modeManager.switchMode(mode);
}
```

### 2. 混合模式策略

```javascript
// 混合验证策略
class HybridTurnstileStrategy {
  constructor(options = {}) {
    this.options = {
      primaryMode: options.primaryMode || 'invisible',
      fallbackMode: options.fallbackMode || 'managed',
      retryAttempts: options.retryAttempts || 3,
      riskThreshold: options.riskThreshold || 0.5
    };
  }

  // 执行验证
  async execute(context = {}) {
    let lastError = null;

    // 尝试主要模式
    for (let attempt = 0; attempt < this.options.retryAttempts; attempt++) {
      try {
        const result = await this.attemptVerification(
          this.options.primaryMode,
          context
        );

        if (result.success) {
          return result;
        }

        lastError = result.error;
      } catch (error) {
        lastError = error;
      }
    }

    // 主要模式失败，使用备用模式
    console.warn('主模式验证失败，切换到备用模式', lastError);

    try {
      return await this.attemptVerification(
        this.options.fallbackMode,
        context
      );
    } catch (error) {
      throw new Error(`所有验证模式失败: ${error.message}`);
    }
  }

  // 尝试特定模式的验证
  async attemptVerification(mode, context) {
    switch (mode) {
      case 'invisible':
        return await this.invisibleVerification(context);
      case 'managed':
        return await this.managedVerification(context);
      case 'non_managed':
        return await this.nonManagedVerification(context);
      default:
        throw new Error(`未知的验证模式: ${mode}`);
    }
  }

  // 不可见验证
  async invisibleVerification(context) {
    return new Promise((resolve, reject) => {
      turnstile.ready(async () => {
        try {
          const widgetId = turnstile.render('#turnstile-container', {
            sitekey: context.sitekey,
            size: 'invisible',
            callback: (token) => resolve({ success: true, token }),
            'error-callback': (error) => reject(new Error(error))
          });

          // 立即触发验证
          turnstile.execute(widgetId);
        } catch (error) {
          reject(error);
        }
      });
    });
  }

  // 托管验证
  async managedVerification(context) {
    return new Promise((resolve, reject) => {
      turnstile.ready(async () => {
        try {
          turnstile.render('#turnstile-container', {
            sitekey: context.sitekey,
            callback: (token) => resolve({ success: true, token }),
            'error-callback': (error) => reject(new Error(error))
          });
        } catch (error) {
          reject(error);
        }
      });
    });
  }

  // 非托管验证
  async nonManagedVerification(context) {
    // 实现自定义验证逻辑
    const response = await fetch('/api/custom-verification', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(context)
    });

    const result = await response.json();

    if (result.success) {
      return { success: true, token: result.token };
    } else {
      throw new Error(result.error);
    }
  }
}
```

### 1. 延迟加载

```javascript
// 基于用户行为的延迟加载
class LazyTurnstileLoader {
  constructor() {
    this.loaded = false;
    this.initiated = false;
  }

  // 监听用户交互
  observeUserInteraction() {
    const interactions = [
      'mousemove',
      'scroll',
      'keydown',
      'touchstart',
      'click'
    ];

    const loadHandler = () => {
      if (!this.initiated) {
        this.initiated = true;
        this.loadScript();
      }
    };

    interactions.forEach(event => {
      document.addEventListener(event, loadHandler, { once: true });
    });

    // 5秒后自动加载
    setTimeout(loadHandler, 5000);
  }

  // 加载脚本
  loadScript() {
    if (this.loaded) return;

    const script = document.createElement('script');
    script.src = 'https://challenges.cloudflare.com/turnstile/v0/api.js';
    script.async = true;
    script.defer = true;
    script.onload = () => {
      this.loaded = true;
      this.onScriptLoaded();
    };

    document.head.appendChild(script);
  }

  // 脚本加载完成
  onScriptLoaded() {
    console.log('Turnstile 脚本已加载');
    // 可以预先渲染 Widget
    this.preRenderWidgets();
  }

  // 预渲染 Widget
  preRenderWidgets() {
    const containers = document.querySelectorAll('[data-turnstile-lazy]');
    containers.forEach(container => {
      const config = {
        sitekey: container.dataset.sitekey,
        callback: container.dataset.callback,
        'error-callback': container.dataset.errorCallback
      };

      turnstile.render(container, config);
    });
  }
}

// 使用示例
const lazyLoader = new LazyTurnstileLoader();
lazyLoader.observeUserInteraction();
```

### 2. 缓存策略

```javascript
// 验证结果缓存
class TurnstileCache {
  constructor() {
    this.cache = new Map();
    this.ttl = 5 * 60 * 1000; // 5分钟
  }

  // 缓存验证结果
  set(key, value) {
    this.cache.set(key, {
      value,
      timestamp: Date.now()
    });
  }

  // 获取缓存的验证结果
  get(key) {
    const item = this.cache.get(key);
    if (!item) return null;

    if (Date.now() - item.timestamp > this.ttl) {
      this.cache.delete(key);
      return null;
    }

    return item.value;
  }

  // 生成缓存键
  generateKey(context) {
    const parts = [
      context.sitekey,
      context.action || '',
      context.userId || '',
      Math.floor(Date.now() / (60 * 1000)) // 分钟级缓存
    ];
    return parts.join(':');
  }
}
```
