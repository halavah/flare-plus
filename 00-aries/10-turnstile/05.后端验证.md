# Cloudflare Turnstile 后端验证

## 验证流程概述

Turnstile 验证采用客户端-服务器模式：
1. 客户端获取验证 token
2. 将 token 提交到服务器
3. 服务器向 Cloudflare 验证 token
4. 根据验证结果处理请求

## 验证 API

### 端点信息
```
URL: https://challenges.cloudflare.com/turnstile/v0/siteverify
方法: POST
Content-Type: application/x-www-form-urlencoded
```

### 请求参数
```javascript
const params = {
  secret: 'YOUR_SECRET_KEY',        // 必需：密钥
  response: 'TURNSTILE_TOKEN',      // 必需：客户端 token
  remoteip: 'USER_IP_ADDRESS',      // 可选：用户 IP
  id: 'ACTION_ID'                   // 可选：操作 ID
};
```

### 响应格式
```json
{
  "success": true,                  // 验证是否成功
  "error-codes": [],                // 错误代码列表
  "challenge_ts": "2023-01-01T12:00:00Z",  // 验证时间
  "hostname": "example.com",        // 验证的域名
  "action": "login",               // 操作类型
  "cdata": "custom_data"           // 自定义数据
}
```

### 1. Node.js (Express)

```javascript
const express = require('express');
const fetch = require('node-fetch');
const app = express();

// 配置
const TURNSTILE_SECRET_KEY = process.env.TURNSTILE_SECRET_KEY;

// 中间件
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// 验证函数
async function verifyTurnstileToken(token, remoteIp) {
  const formData = new URLSearchParams();
  formData.append('secret', TURNSTILE_SECRET_KEY);
  formData.append('response', token);

  if (remoteIp) {
    formData.append('remoteip', remoteIp);
  }

  try {
    const response = await fetch('https://challenges.cloudflare.com/turnstile/v0/siteverify', {
      method: 'POST',
      body: formData
    });

    const result = await response.json();
    return result;
  } catch (error) {
    console.error('Turnstile 验证错误:', error);
    return { success: false, error: 'network_error' };
  }
}

// 登录路由
app.post('/api/login', async (req, res) => {
  try {
    const { username, password, 'turnstile-token': token } = req.body;
    const userIp = req.ip || req.connection.remoteAddress;

    // 验证 Turnstile token
    if (!token) {
      return res.status(400).json({
        success: false,
        error: 'missing_turnstile_token'
      });
    }

    const verification = await verifyTurnstileToken(token, userIp);

    if (!verification.success) {
      console.error('Turnstile 验证失败:', verification['error-codes']);
      return res.status(400).json({
        success: false,
        error: 'turnstile_verification_failed',
        details: verification['error-codes']
      });
    }

    // 验证用户凭据
    const user = await authenticateUser(username, password);

    if (!user) {
      return res.status(401).json({
        success: false,
        error: 'invalid_credentials'
      });
    }

    // 创建会话
    const session = await createSession(user);

    res.json({
      success: true,
      session,
      message: '登录成功'
    });

  } catch (error) {
    console.error('登录错误:', error);
    res.status(500).json({
      success: false,
      error: 'server_error'
    });
  }
});

// 错误处理中间件
app.use((error, req, res, next) => {
  console.error('未处理的错误:', error);
  res.status(500).json({
    success: false,
    error: 'internal_server_error'
  });
});

// 启动服务器
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`服务器运行在端口 ${PORT}`);
});
```

```python
def verify_turnstile(f):
    """Turnstile 验证装饰器"""
    @wraps(f)
    async def decorated_function(*args, **kwargs):
        token = request.form.get('turnstile-token') or request.json.get('turnstile-token')

        if not token:
            return jsonify({
                'success': False,
                'error': 'missing_turnstile_token'
            }), 400

        # 准备验证数据
        data = {
            'secret': TURNSTILE_SECRET_KEY,
            'response': token
        }

        # 添加 IP（可选）
        if request.remote_addr:
            data['remoteip'] = request.remote_addr

        try:
            # 发送验证请求
            response = requests.post(
                'https://challenges.cloudflare.com/turnstile/v0/siteverify',
                data=data,
                timeout=10
            )

            result = response.json()

            if not result.get('success'):
                app.logger.error(f"Turnstile 验证失败: {result.get('error-codes')}")
                return jsonify({
                    'success': False,
                    'error': 'turnstile_verification_failed',
                    'details': result.get('error-codes')
                }), 400

        except requests.RequestException as e:
            app.logger.error(f"Turnstile API 错误: {str(e)}")
            return jsonify({
                'success': False,
                'error': 'verification_service_error'
            }), 500

        return await f(*args, **kwargs)

    return decorated_function
```

```java
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;
import org.springframework.http.*;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.client.RestTemplate;
import org.springframework.util.LinkedMultiValueMap;
import org.springframework.util.MultiValueMap;

import java.util.Collections;
import java.util.Map;

@SpringBootApplication
@RestController
public class TurnstileDemoApplication {

    private final String TURNSTILE_SECRET_KEY = System.getenv("TURNSTILE_SECRET_KEY");
    private final String TURNSTILE_VERIFY_URL = "https://challenges.cloudflare.com/turnstile/v0/siteverify";
    private final RestTemplate restTemplate = new RestTemplate();

    // Turnstile 验证结果类
    public static class TurnstileResponse {
        private boolean success;
        private String[] errorCodes;
        private String challengeTs;
        private String hostname;

        // Getters and Setters
        public boolean isSuccess() { return success; }
        public void setSuccess(boolean success) { this.success = success; }
        public String[] getErrorCodes() { return errorCodes; }
        public void setErrorCodes(String[] errorCodes) { this.errorCodes = errorCodes; }
        public String getChallengeTs() { return challengeTs; }
        public void setChallengeTs(String challengeTs) { this.challengeTs = challengeTs; }
        public String getHostname() { return hostname; }
        public void setHostname(String hostname) { this.hostname = hostname; }
    }

    // 验证 Turnstile Token
    public boolean verifyTurnstileToken(String token, String remoteIp) {
        MultiValueMap<String, String> formData = new LinkedMultiValueMap<>();
        formData.add("secret", TURNSTILE_SECRET_KEY);
        formData.add("response", token);

        if (remoteIp != null) {
            formData.add("remoteip", remoteIp);
        }

        HttpHeaders headers = new HttpHeaders();
        headers.setContentType(MediaType.APPLICATION_FORM_URLENCODED);

        HttpEntity<MultiValueMap<String, String>> entity = new HttpEntity<>(formData, headers);

        try {
            ResponseEntity<TurnstileResponse> response = restTemplate.postForEntity(
                TURNSTILE_VERIFY_URL,
                entity,
                TurnstileResponse.class
            );

            TurnstileResponse result = response.getBody();
            return result != null && result.isSuccess();

        } catch (Exception e) {
            System.err.println("Turnstile 验证错误: " + e.getMessage());
            return false;
        }
    }

    // 评论提交接口
```

### 4. PHP (Laravel)

```php
<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Validator;

class CommentController extends Controller
{
    private $turnstileSecret;

    public function __construct()
    {
        $this->turnstileSecret = env('TURNSTILE_SECRET_KEY');
    }

    /**
     * 验证 Turnstile Token
     */
    private function verifyTurnstileToken($token, $remoteIp = null)
    {
        $response = Http::asForm()->post('https://challenges.cloudflare.com/turnstile/v0/siteverify', [
            'secret' => $this->turnstileSecret,
            'response' => $token,
            'remoteip' => $remoteIp
        ]);

        return $response->json();
    }

    /**
     * 提交评论
     */
    public function submit(Request $request)
    {
        // 验证输入
        $validator = Validator::make($request->all(), [
            'content' => 'required|string|max:1000',
            'author' => 'required|string|max:100',
            'turnstile-token' => 'required|string'
        ]);

        if ($validator->fails()) {
            return response()->json([
                'success' => false,
                'error' => 'validation_failed',
                'details' => $validator->errors()
            ], 400);
        }

        // 验证 Turnstile
        $token = $request->input('turnstile-token');
        $remoteIp = $request->ip();

        $verification = $this->verifyTurnstileToken($token, $remoteIp);

        if (!$verification['success']) {
            \Log::error('Turnstile 验证失败', $verification['error-codes'] ?? []);

            return response()->json([
                'success' => false,
                'error' => 'turnstile_verification_failed',
                'details' => $verification['error-codes'] ?? []
            ], 400);
        }

        try {
            // 创建评论
            $comment = new Comment();
            $comment->author = $request->input('author');
            $comment->content = $request->input('content');
            $comment->ip_address = $remoteIp;
            $comment->user_agent = $request->userAgent();
            $comment->verified_at = now();
            $comment->save();

            return response()->json([
                'success' => true,
                'comment' => [
                    'id' => $comment->id,
                    'author' => $comment->author,
                    'content' => $comment->content,
                    'created_at' => $comment->created_at->toISOString()
                ]
            ]);

        } catch (\Exception $e) {
            \Log::error('创建评论错误: ' . $e->getMessage());

            return response()->json([
                'success' => false,
                'error' => 'server_error'
            ], 500);
        }
    }
}
```

## 错误处理

### 常见错误代码

| 错误代码 | 说明 | 处理建议 |
|---------|------|----------|
| `missing-input-secret` | 缺少密钥 | 检查服务端配置 |
| `invalid-input-secret` | 无效密钥 | 验证密钥是否正确 |
| `missing-input-response` | 缺少响应 | 确保客户端提交了 token |
| `invalid-input-response` | 无效响应 | token 可能已过期或无效 |
| `timeout-or-duplicate` | 超时或重复 | 重新获取 token |
| `bad-request` | 错误请求 | 检查请求格式 |
| `internal-error` | 内部错误 | 重试或联系支持 |

### 错误处理示例

```javascript
// 错误处理工具
class TurnstileErrorHandler {
  static handleVerificationError(errorCodes, context = {}) {
    const errorMap = {
      'missing-input-secret': {
        message: '服务器配置错误',
        statusCode: 500,
        shouldRetry: false
      },
      'invalid-input-secret': {
        message: '验证密钥无效',
        statusCode: 500,
        shouldRetry: false
      },
      'missing-input-response': {
        message: '请完成验证',
        statusCode: 400,
        shouldRetry: false
      },
      'invalid-input-response': {
        message: '验证已过期，请重新验证',
        statusCode: 400,
        shouldRetry: false
      },
      'timeout-or-duplicate': {
        message: '验证超时，请重试',
        statusCode: 408,
        shouldRetry: true
      },
      'bad-request': {
        message: '请求格式错误',
        statusCode: 400,
        shouldRetry: false
      },
      'internal-error': {
        message: '验证服务暂时不可用',
        statusCode: 503,
        shouldRetry: true
      }
    };

    // 取第一个错误代码
    const errorCode = errorCodes?.[0];
    const error = errorMap[errorCode] || {
      message: '未知验证错误',
      statusCode: 500,
      shouldRetry: false
    };

    // 记录错误日志
    console.error('Turnstile 验证错误:', {
      errorCodes,
      errorCode,
      message: error.message,
      context
    });

    return {
      ...error,
      errorCode,
      allErrorCodes: errorCodes
    };
  }

  static async retryVerification(token, remoteIp, maxRetries = 3) {
    for (let i = 0; i < maxRetries; i++) {
      const result = await verifyTurnstileToken(token, remoteIp);

      if (result.success) {
        return result;
      }

      // 如果不是可重试的错误，直接返回
      const error = this.handleVerificationError(result['error-codes']);
      if (!error.shouldRetry || i === maxRetries - 1) {
        return result;
      }

      // 指数退避
      await new Promise(resolve => setTimeout(resolve, Math.pow(2, i) * 1000));
    }

    return { success: false, 'error-codes': ['max_retries_exceeded'] };
  }
}

// 使用示例
app.post('/api/submit', async (req, res) => {
  const { token } = req.body;
  const remoteIp = req.ip;

  const result = await verifyTurnstileToken(token, remoteIp);

  if (!result.success) {
    const error = TurnstileErrorHandler.handleVerificationError(
      result['error-codes'],
      { endpoint: '/api/submit' }
    );

    return res.status(error.statusCode).json({
      success: false,
      error: error.message,
      code: error.errorCode,
      shouldRetry: error.shouldRetry
    });
  }

  // 处理成功逻辑
  res.json({ success: true });
});
```

## 安全增强

### 1. Token 缓存

```javascript
// Redis 缓存实现
const Redis = require('redis');
const redis = Redis.createClient();

class TurnstileCache {
  static async cacheToken(token, verification, ttl = 300) {
    await redis.setex(`turnstile:${token}`, ttl, JSON.stringify(verification));
  }

  static async getCachedToken(token) {
    const cached = await redis.get(`turnstile:${token}`);
    return cached ? JSON.parse(cached) : null;
  }

  static async invalidateToken(token) {
    await redis.del(`turnstile:${token}`);
  }
}

// 使用缓存的验证函数
async function verifyTokenWithCache(token, remoteIp) {
  // 先检查缓存
  const cached = await TurnstileCache.getCachedToken(token);
  if (cached) {
    return cached;
  }

  // 执行验证
  const result = await verifyTurnstileToken(token, remoteIp);

  // 缓存结果
  if (result.success) {
    await TurnstileCache.cacheToken(token, result, 300); // 5分钟缓存
  }

  return result;
}
```

```javascript
// IP 白名单配置
const TRUSTED_IPS = new Set([
  '127.0.0.1',
  '::1',
  // 添加更多可信 IP
]);

function isTrustedIp(ip) {
  return TRUSTED_IPS.has(ip);
}

// 修改验证函数
async function verifyTurnstileTokenWithTrust(token, remoteIp) {
  // 可信 IP 跳过验证（仅限内部测试）
  if (isTrustedIp(remoteIp)) {
    return { success: true };
  }

  return await verifyTurnstileToken(token, remoteIp);
}
```
