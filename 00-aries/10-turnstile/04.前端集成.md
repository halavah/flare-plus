# Cloudflare Turnstile 前端集成

## 基本集成步骤

### 1. 加载 Turnstile 脚本

```html
<!-- 方法1：直接引入 -->
<script src="https://challenges.cloudflare.com/turnstile/v0/api.js" async defer></script>

<!-- 方法2：ES6 模块 -->
<script type="module">
  import Turnstile from 'https://challenges.cloudflare.com/turnstile/v0/api.js';
  // 使用 Turnstile API
</script>

<!-- 方法3：动态加载 -->
<script>
function loadTurnstile() {
  const script = document.createElement('script');
  script.src = 'https://challenges.cloudflare.com/turnstile/v0/api.js';
  script.async = true;
  script.defer = true;
  document.head.appendChild(script);
}
</script>
```

### 2. 创建 Widget 容器

```html
<!-- 基本容器 -->
<div class="cf-turnstile" data-sitekey="YOUR_SITE_KEY"></div>

<!-- 带回调的容器 -->
<div class="cf-turnstile"
     data-sitekey="YOUR_SITE_KEY"
     data-callback="javascriptCallback"
     data-error-callback="javascriptErrorCallback">
</div>
```

```html
    function onTurnstileSuccess(token) {
      turnstileToken = token;

      document.getElementById('error-message').style.display = 'none';
    }

    // Turnstile 错误回调
    function onTurnstileError(error) {
      console.error('验证失败:', error);
      document.getElementById('error-message').textContent = '验证失败，请重试';
      document.getElementById('error-message').style.display = 'block';
      turnstileToken = null;
    }

    // Turnstile 过期回调
    function onTurnstileExpired() {
      console.warn('验证已过期');
      document.getElementById('error-message').textContent = '验证已过期，请重新验证';
      document.getElementById('error-message').style.display = 'block';
      turnstileToken = null;
    }

    // 表单提交处理
    async function handleSubmit(event) {
      event.preventDefault();

      if (!turnstileToken) {
        document.getElementById('error-message').textContent = '请先完成验证';
        document.getElementById('error-message').style.display = 'block';
        return;
      }

      const formData = new FormData(event.target);
      formData.append('turnstile-token', turnstileToken);

      try {
        const response = await fetch('/api/login', {
          method: 'POST',
          body: formData
        });

        const result = await response.json();

        if (result.success) {
          // 登录成功
          window.location.href = '/dashboard';
        } else {
          // 登录失败
          document.getElementById('error-message').textContent = result.error || '登录失败';
          document.getElementById('error-message').style.display = 'block';

          // 重置 Turnstile
          turnstile.reset();
        }
      } catch (error) {
        console.error('提交错误:', error);
        document.getElementById('error-message').textContent = '网络错误，请重试';
        document.getElementById('error-message').style.display = 'block';
      }
    }
  </script>
</body>
</html>
```

## 高级集成技巧

### 1. 动态渲染

```javascript
class TurnstileManager {
  constructor() {
    this.instances = new Map();
    this.siteKey = 'YOUR_SITE_KEY';
  }

  // 动态创建 Widget
  createWidget(containerId, options = {}) {
    const container = document.getElementById(containerId);
    if (!container) {
      throw new Error(`Container ${containerId} not found`);
    }

    // 默认配置
    const defaultOptions = {
      sitekey: this.siteKey,
      callback: this.onSuccess.bind(this),
      'error-callback': this.onError.bind(this),
      'expired-callback': this.onExpired.bind(this),
      theme: 'auto',
      language: 'zh-CN'
    };

    // 合并配置
    const widgetOptions = { ...defaultOptions, ...options };

    // 清空容器
    container.innerHTML = '';

    // 创建 Widget
    const widgetId = turnstile.render(container, widgetOptions);
    this.instances.set(containerId, widgetId);

    return widgetId;
  }

  // 重置 Widget
  reset(containerId) {
    const widgetId = this.instances.get(containerId);
    if (widgetId) {
      turnstile.reset(widgetId);
    }
  }

  // 获取 Token
  getToken(containerId) {
    const widgetId = this.instances.get(containerId);
    if (widgetId) {
      return turnstile.getResponse(widgetId);
    }
    return null;
  }

  // 销毁 Widget
  destroy(containerId) {
    const widgetId = this.instances.get(containerId);
    if (widgetId) {
      turnstile.remove(widgetId);
      this.instances.delete(containerId);
    }
  }

  // 成功回调
  onSuccess(token) {
    console.log('验证成功:', token);
  }

  // 错误回调
  onError(error) {
    console.error('验证失败:', error);
  }

  // 过期回调
  onExpired() {
    console.warn('验证已过期');
  }
}

// 使用示例
const turnstileManager = new TurnstileManager();

// 页面加载后创建
document.addEventListener('DOMContentLoaded', () => {
  turnstileManager.createWidget('turnstile-container', {
    theme: 'dark',
    action: 'login'
  });
});
```

### 3. 多语言支持

```javascript
// 语言配置
const translations = {
  'zh-CN': {
    error: '验证失败，请重试',
    expired: '验证已过期，请重新验证',
    required: '请完成验证'
  },
  'en': {
    error: 'Verification failed, please try again',
    expired: 'Verification expired, please verify again',
    required: 'Please complete verification'
  },
  'ja': {
    error: '認証に失敗しました。もう一度お試しください',
    expired: '認証の有効期限が切れました。再度認証してください',
    required: '認証を完了してください'
  }
};

// 多语言 Turnstile 管理器
class MultiLanguageTurnstile {
  constructor() {
    this.currentLang = this.detectLanguage();
    this.translations = translations[this.currentLang] || translations['en'];
  }

  // 检测语言
  detectLanguage() {
    const browserLang = navigator.language || navigator.userLanguage;
    const supportedLangs = Object.keys(translations);

    // 直接匹配
    if (supportedLangs.includes(browserLang)) {
      return browserLang;
    }

    // 匹配语言前缀
    const langPrefix = browserLang.split('-')[0];
    for (const lang of supportedLangs) {
      if (lang.startsWith(langPrefix)) {
        return lang;
      }
    }

    // 默认英文
    return 'en';
  }

  // 创建多语言 Widget
  createWidget(containerId, options = {}) {
    const container = document.getElementById(containerId);

    // 错误消息容器
    const errorDiv = document.createElement('div');
    errorDiv.id = 'turnstile-error';
    errorDiv.className = 'turnstile-error';
    errorDiv.style.display = 'none';
    container.parentNode.insertBefore(errorDiv, container.nextSibling);

    // 创建 Widget
    const widgetId = turnstile.render(container, {
      sitekey: options.sitekey,
      language: this.currentLang,
      callback: (token) => this.onSuccess(token, errorDiv),
      'error-callback': (error) => this.onError(error, errorDiv),
      'expired-callback': () => this.onExpired(errorDiv),
      ...options
    });

    return widgetId;
  }

  // 显示错误消息
  showError(message, errorDiv) {
    errorDiv.textContent = message;
    errorDiv.style.display = 'block';
  }

  // 隐藏错误消息
  hideError(errorDiv) {
    errorDiv.style.display = 'none';
  }

  // 成功回调
  onSuccess(token, errorDiv) {
    this.hideError(errorDiv);
    console.log('验证成功');
  }

  // 错误回调
  onError(error, errorDiv) {
    this.showError(this.translations.error, errorDiv);
  }

  // 过期回调
  onExpired(errorDiv) {
    this.showError(this.translations.expired, errorDiv);
  }
}

// 使用示例
const multiLangTurnstile = new MultiLanguageTurnstile();
document.addEventListener('DOMContentLoaded', () => {
  multiLangTurnstile.createWidget('turnstile-container', {
    sitekey: 'YOUR_SITE_KEY'
  });
});
```

```html
// 主题管理器
class TurnstileTheme {
  constructor() {
    this.currentTheme = this.detectTheme();
    this.applyTheme();
  }

  // 检测系统主题
  detectTheme() {
    if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
      return 'dark';
    }
    return 'light';
  }

  // 应用主题
  applyTheme() {
    const wrapper = document.querySelector('.turnstile-wrapper');
    if (wrapper) {
      wrapper.className = `turnstile-wrapper turnstile-${this.currentTheme} turnstile-animated`;
    }
  }

  // 切换主题
  toggle() {
    this.currentTheme = this.currentTheme === 'dark' ? 'light' : 'dark';
    this.applyTheme();
  }
}

// 监听系统主题变化
window.matchMedia('(prefers-color-scheme: dark)').addEventListener('change', (e) => {
  const theme = new TurnstileTheme();
});

// 创建带主题的 Turnstile
function createThemedTurnstile() {

  wrapper.className = 'turnstile-wrapper';

  const turnstileDiv = document.createElement('div');
  turnstileDiv.className = 'cf-turnstile';
  turnstileDiv.setAttribute('data-sitekey', 'YOUR_SITE_KEY');
  turnstileDiv.setAttribute('data-theme', 'auto');

  wrapper.appendChild(turnstileDiv);
  return wrapper;
}
</script>
```

## 框架集成

### 1. React 集成

```jsx
// Turnstile 组件
import React, { useEffect, useRef, useState } from 'react';

const Turnstile = ({ sitekey, onVerify, theme = 'auto', language = 'auto' }) => {
  const containerRef = useRef(null);
  const [widgetId, setWidgetId] = useState(null);
  const [isLoaded, setIsLoaded] = useState(false);

  useEffect(() => {
    // 加载 Turnstile 脚本
    const script = document.createElement('script');
    script.src = 'https://challenges.cloudflare.com/turnstile/v0/api.js';
    script.async = true;
    script.defer = true;
    script.onload = () => setIsLoaded(true);

    document.head.appendChild(script);

    return () => {
      // 清理
      if (widgetId) {
        window.turnstile?.remove(widgetId);
      }
    };
  }, []);

  useEffect(() => {
    if (isLoaded && containerRef.current && !widgetId) {
      const id = window.turnstile.render(containerRef.current, {
        sitekey,
        theme,
        language,
        callback: onVerify,
        'error-callback': (error) => console.error('Turnstile error:', error),
        'expired-callback': () => console.log('Turnstile expired')
      });
      setWidgetId(id);
    }
  }, [isLoaded, sitekey, theme, language, onVerify]);

  const reset = () => {
    if (widgetId) {
      window.turnstile.reset(widgetId);
    }
  };

  return {
    reset,
    render: <div ref={containerRef} />
  };
};

// 使用示例
const LoginForm = () => {
  const [token, setToken] = useState(null);
  const turnstileRef = useRef();

  const handleSubmit = async (e) => {
    e.preventDefault();

    if (!token) {
      alert('请先完成验证');
      return;
    }

    // 提交表单...
  };

  const turnstile = Turnstile({
    sitekey: 'YOUR_SITE_KEY',
    onVerify: setToken,
    theme: 'auto'
  });

  turnstileRef.current = turnstile;

  return (
    <form onSubmit={handleSubmit}>
      <input type="email" placeholder="邮箱" required />
      <input type="password" placeholder="密码" required />

      {turnstile.render}

      <button type="submit">登录</button>

      <button type="button" onClick={() => turnstileRef.current?.reset()}>
        重置验证
      </button>

  );
};
```

```vue
<!-- Turnstile.vue -->
<template>
  <div ref="container" class="turnstile-container"></div>

<script>
export default {
  name: 'Turnstile',
  props: {
    sitekey: {
      type: String,
      required: true
    },
    theme: {
      type: String,
      default: 'auto'
    },

      type: String,
      default: 'auto'

        this.isLoaded = true;
        this.renderWidget();

      script.src = 'https://challenges.cloudflare.com/turnstile/v0/api.js';
```

### 3. Angular 集成

```typescript
// turnstile.component.ts
import { Component, Input, Output, EventEmitter, OnInit, OnDestroy } from '@angular/core';

declare global {
  interface Window {
    turnstile: any;
  }
}

@Component({
  selector: 'app-turnstile',
  template: '<div #container></div>',
  styles: [':host { display: block; }']
})
export class TurnstileComponent implements OnInit, OnDestroy {
  @Input() sitekey: string;
  @Input() theme: string = 'auto';
  @Input() language: string = 'auto';

  @Output() verify = new EventEmitter<string>();
  @Output() error = new EventEmitter<string>();
  @Output() expired = new EventEmitter<void>();

  @ViewChild('container', { static: true }) container: ElementRef;

  private widgetId: string | null = null;

  ngOnInit(): void {
    this.loadTurnstileScript();
  }

  ngOnDestroy(): void {
    if (this.widgetId && window.turnstile) {
      window.turnstile.remove(this.widgetId);
    }
  }

  private loadTurnstileScript(): void {
    if (window.turnstile) {
      this.renderWidget();
      return;
    }

    const script = document.createElement('script');
    script.src = 'https://challenges.cloudflare.com/turnstile/v0/api.js';
    script.async = true;
    script.defer = true;
    script.onload = () => this.renderWidget();

    document.head.appendChild(script);
  }

  private renderWidget(): void {
    if (!window.turnstile || !this.container.nativeElement) return;

    this.widgetId = window.turnstile.render(this.container.nativeElement, {
      sitekey: this.sitekey,
      theme: this.theme,
      language: this.language,
      callback: (token: string) => this.verify.emit(token),
      'error-callback': (error: string) => this.error.emit(error),
      'expired-callback': () => this.expired.emit()
    });
  }

  public reset(): void {
    if (this.widgetId && window.turnstile) {
      window.turnstile.reset(this.widgetId);
    }
  }

  public getResponse(): string | null {
    if (this.widgetId && window.turnstile) {
      return window.turnstile.getResponse(this.widgetId);
    }
    return null;
  }
}
```

## 最佳实践

1. **延迟加载**：在需要时才加载 Turnstile 脚本
2. **错误处理**：提供清晰的错误提示
3. **可访问性**：添加适当的 ARIA 标签
4. **性能优化**：避免重复加载脚本
5. **用户体验**：提供重置选项
6. **安全性**：不要在客户端存储密钥
