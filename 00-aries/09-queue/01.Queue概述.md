# Queue 概述

## 🚀 什么是 Cloudflare Queue？

Cloudflare Queue 是一个强大、完全托管的消息队列服务，专为构建可扩展的分布式应用程序而设计。它提供了可靠的消息传递、精确的顺序保证和高吞吐量的消息处理能力，同时保持了 Cloudflare 的全球分布式架构优势。

### 核心特性

#### 🎯 关键功能
- **高可靠性**：消息持久化，确保不会丢失
- **全球分布式**：利用 Cloudflare 的全球网络实现低延迟
- **自动扩缩**：根据负载自动调整资源
- **简单易用**：RESTful API，易于集成
- **成本效益**：按需付费，无需前期投入
- **安全可靠**：内置安全机制，符合企业级要求

#### 🏗️ 架构优势
```
生产者 → Cloudflare Queue → 消费者
    ↓
全球分布式网络 → 智能路由 → 负载均衡
    ↓
自动扩缩 → 故障转移 → 持久化存储
```

## 📊 核心概念

### 1. 队列 (Queue)

队列是消息的容器，负责存储和传递消息。每个队列都有唯一的标识符和特定的配置参数。

## 🔄 消息处理流程

### 典型处理模式

#### 1. 生产者-消费者模式

```javascript
// 消息流程示例
class MessageFlow {
  async produce(queueId, message) {
    // 1. 发送消息到队列
    const response = await fetch(`https://api.cloudflare.com/client/v4/accounts/${account_id}/queues/${queueId}/messages`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${API_TOKEN}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        content: message,
        metadata: {
          source: 'order_service',
          version: '1.0'
        }
      })
    });

    const result = await response.json();
    console.log(`Message sent: ${result.result.id}`);
    return result;
  }

  async consume(queueId, handler) {
    // 2. 接收消息
    while (true) {
      const response = await fetch(`https://api.cloudflare.com/client/v4/accounts/${account_id}/queues/${queueId}/messages/consume`, {
      method: 'POST',
      headers: {
        'Authorization': `essageId: ${API_TOKEN}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        visibility_timeout: 30000,
        batch_size: 10
      })
    });

      const result = await response.json();
      const messages = result.result.messages;

      // 3. 处理消息
      for (const message of messages) {
        try {
          await handler(message);
          // 自动确认成功
          await this.acknowledge(queueId, message.id);
        } catch (error) {
          console.error('Message processing failed:', error);
          // 处理失败，消息将重新进入队列
        }
      }
    }
  }

  // 消息确认
  async acknowledge(queueId, messageId) {
    await fetch(`https://api.cloudflare.com/client/v4/accounts/${account_id}/queues/${queueId}/messages/${messageId}/acknowledge`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${API_TOKEN}`,
        'Content-Type': 'application/json'
      }
    });
  }
}
```

```javascript
// 发布订阅示例
class PubSubQueue {
  constructor() {
    this.topics = new Map();
    this.subscriptions = new Map();
  }

  // 创建主题
  async createTopic(topicConfig) {
    const response = await fetch('https://api.cloudflare.com/client/v4/accounts/${account_id}/topics', {

      headers: {
        'Authorization': `Bearer ${API_TOKEN}`,

        description: topicConfig.description,

      })
    });

    const topic = await response.json();
    this.topics.set(topicConfig.name, topic.result);
    return topic;
  }

  // 发布消息
  async publish(topicName, message) {
    const topic = this.topics.get(topicName);
    if (!topic) {
      throw new Error(`Topic not found: ${topicName}`);
    }

    const response = await fetch(`https://api.cloudflare.com/client/v4/accounts/${account_id}/topics/${topic.id}/publish`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${API_TOKEN}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        content: message,
        metadata: {
          timestamp: new Date().toISOString()
        }
      })
    });

    return await response.json();
  }

  // 订阅主题
  async subscribe(topicName, handler) {
    const topic = this.topics.get(topicName);
    if (!topic) {
      throw new Error(`Topic not found: ${topicName}`);
    }

    // 创建订阅
    const response = await fetch(`https://api.cloudflare.com/client/v4/accounts/${account_id}/subscriptions`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${API_TOKEN}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        topic_id: topic.id,
        name: `subscription_${Date.now()}`,
        filter: {
          pattern: '*',
          attributes: {}
        }
      })
    });

    const subscription = await response.json();
    this.subscriptions.set(subscription.result.id, {
      subscription,
      handler
    });

    return subscription;
  }
}
```

```javascript
// 任务队列示例
class TaskQueue {
  constructor(queueId, options = {}) {
    this.queueId = queueId;
    this.workers = [];
    this.taskHandlers = new Map();
    this.priorityLevels = options.priorityLevels || ['low', 'medium', 'high', 'critical'];
    this.setupWorkers(options.workerCount || 5);
  }

  // 设置工作进程
  setupWorkers(count) {
    for (let i = 0; i < count; i++) {
      this.workers.push({
        id: i,
        busy: false,
        processTasks: async () => {
          while (true) {
            const task = await this.getTask();
            if (!task) {
              await new Promise(resolve => setTimeout(resolve, 1000));
              continue;
            }

            this.workers[i].busy = true;
            await this.processTask(task);
            this.workers[i].busy = false;
          }
        }
      });

      // 启动工作进程
      this.workers[i].processTasks();
    }
  }

  // 添加任务
  async addTask(task, priority = 'medium') {
    const taskObj = {
      id: crypto.randomUUID(),
      task,
      priority: this.getPriorityValue(priority),
      created_at: Date.now(),
      attempts: 0,
      status: 'pending'
    };

    const response = await fetch(`https://api.cloudflare.com/client/v4/accounts/${account_id}/queues/${this.queueId}/messages`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${API_TOKEN}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        content: taskObj,
        metadata: {
          priority,
          task_id: taskObj.id
        },
        delay: priority === 'critical' ? 0 : null
      })
    });

    return await response.json();
  }

  // 获取任务
  async getTask() {
    const response = await fetch(`https://api.cloudflare.com/client-v4/accounts/${account_id}/queues/${this.queueId}/messages/consume`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${API_TOKEN}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        visibility_timeout: 30000,
        batch_size: 1,
        filter: {
          metadata: {
            status: 'pending'
          }
        }
      })
    });

    const result = await response.json();
    if (result.result.messages.length > 0) {
      return result.result.messages[0];
    }
    return null;
  }

  // 处理任务
  async processTask(message) {
    const task = JSON.parse(message.content);
    const handler = this.taskHandlers.get(task.type);

    if (!handler) {
      console.log(`No handler for task type: ${task.type}`);
      return;
    }

    try {
      const result = await handler(task);

      // 成功，确认消息
      await this.acknowledge(message.id);

      // 记录成功指标
      this.recordMetric('task_success', task.type);

    } catch (error) {
      console.error(`Task processing failed:`, error);

      // 失败，处理重试逻辑
      await this.handleTaskFailure(message, task, error);
    }
  }

  // 获取优先级值
  getPriorityValue(priority) {
    return this.priorityLevels.indexOf(priority) || 0;
  }

  // 记录指标
  recordMetric(type, value) {
    console.log(`${type}: ${value}`);
    // 这里可以发送到监控系统
  }
}
```

### 1. 全球消息路由

Cloudflare Queue 利用 Cloudflare 的全球网络，将消息智能路由到最近的边缘节点，确保低延迟和高可用性。

```javascript
// 智于位置的队列
export class LocationAwareQueue {
  constructor(options = {}) {
    this.regions = ['us-east', 'us-west', 'eu-west', 'asia-pacific', 'australia'];
    this.localQueue = options.localQueue;
    this.backupQueues = options.backupQueues;
    this.failoverPolicy = options.failover || 'nearest';
    this.geolocation = options.geolocation;
  }

  // 智于地理位置选择队列
  async selectOptimalQueue(event) {
    const userLocation = await this.getUserLocation(event.user_id);
    const optimalRegion = await this.findNearestRegion(userLocation);

    // 优先使用本地队列
    if (optimalRegion === this.localQueue.region) {
      return this.localQueue;
    }

    // 查找该区域的队列
    for (const queue of this.backupQueues) {
      if (queue.region === optimalRegion) {
        return queue;
      }
    }

    // 回退到默认队列
    return this.defaultQueue;
  }

  // 获取用户位置
  async getUserLocation(userId) {
    // 从数据库或缓存中获取用户位置
    return { latitude: 40.7128, longitude: -74.0060, region: 'us-east' };
  }

  // 查找最近的区域
  async findNearestRegion(userLocation) {
    let minDistance = Infinity;
    let nearestRegion = this.regions[0];

    for (const region of this.regions) {
      const distance = this.calculateDistance(userLocation, region.center);
      if (distance < minDistance) {
        minDistance = distance;
        nearestRegion = region;
      }
    }

    return nearestRegion;
  }
}
```

### 2. 边缘消息处理

```javascript
// 边缘消息处理器
class EdgeMessageProcessor {
  constructor() {
    this.edgeNodes = new Map();
    this.messageCache = new Map();
    this.replicationFactor = 3;
  }

  // 处理边缘消息
  async processEdgeEvent(event) {
    // 选择最近的边缘节点
    const edgeNode = await this.selectEdgeNode(event);

    // 在边缘节点处理消息
    const result = await this.processAtEdge(edgeNode, event);

    // 复制到其他边缘节点（高可用性）
    await this.replicateEvent(event, edgeNode);

    return result;
  }

  // 在边缘处理消息
  async processAtEdge(node, event) {
    // 利用边缘缓存
    const cachedResult = await this.getFromCache(node, event.id);
    if (cachedResult) {
      return cachedResult;
    }

    // 在边缘节点处理
    const result = await node.processMessage(event);

    // 缓存结果
    await this.cacheResult(node, event.id, result);

    return result;
  }

  // 复制事件到其他边缘节点
  async replicateEvent(event, sourceNode) {
    const otherNodes = Array.from(this.edgeNodes.values())
      .filter(node => node.id !== sourceNode.id)
      .slice(0, this.replicationFactor - 1);

    const promises = otherNodes.map(node =>
      node.replicateEvent(event).catch(error => {
        console.error(`Replication failed for node ${node.id}:`, error);
      })
    );

    await Promise.allSettled(promises);
  }
}
```

```javascript
// 智内容的消息路由
class MessageRouter {
  constructor() {
    this.routes = new Map();
    this.defaultRoute = null;
  }

  // 添加路由规则
  addRoute(condition, handler) {
    this.routes.set(condition, handler);
  }

  // 设置默认路由
  setDefaultRoute(handler) {
    this.defaultRoute = handler;
  }

  // 路由消息
  async routeMessage(message) {
    const content = message.content;

    // 检查路由规则
    for (const [condition, handler] of this.routes) {
      if (this.matchCondition(condition, content)) {
        return await handler(message);
      }
    }

    // 使用默认路由
    if (this.defaultRoute) {
      return await this.defaultRoute(message);
    }

    throw new Error('No matching route for message');
  }

  // 条件匹配
  matchCondition(condition, content) {
    switch (condition.type) {
      case 'field_equals':
        return content[condition.field] === condition.value;
      case 'field_contains':
        return content[condition.field].includes(condition.value);
      case 'regex':
        return new RegExp(condition.pattern).test(JSON.stringify(content));
      case 'function':
        return condition.fn(content);
      default:
        return false;
    }
  }
}
```

```javascript
// 工作流队列
class WorkflowQueue {
  constructor(options = {}) {
    this.workflows = new Map();
    this.stateMachine = new StateMachine();
  }

  // 创建工作流
  async createWorkflow(workflowConfig) {
    const workflow = {
      id: workflowConfig.id,
      name: workflowConfig.name,
      states: workflowConfig.states,
      transitions: workflowConfig.transitions,
      current_state: workflowConfig.initial_state,
      variables: {},
      history: []
    };

    this.workflows.set(workflow.id, workflow);

    // 设置状态机
    this.stateMachine.setup(workflow);

    return workflow;
  }

  // 处理工作流消息
  async processWorkflowMessage(message) {
    const workflow = this.workflows.get(message.workflow_id);
    if (!workflow) {
      throw new Error(`Workflow not found: ${message.workflow_id}`);
    }

    const transition = this.stateMachine.processEvent(
      workflow,
      message.event,
      message.data
    );

    if (transition) {
      // 更新工作流状态
      workflow.current_state = transition.to_state;
      workflow.variables = { ...workflow.variables, ...transition.variables };
      workflow.history.push(transition);

      // 记录转换
      await this.recordTransition(transition);

      // 处理进入新状态的动作
      await this.handleStateChange(workflow, transition);
    }

    return workflow;
  }
}
```
