# Cloudflare Queue 消费者管理

## 消费者管理概述

消费者是消息队列中处理消息的组件。有效的消费者管理对于确保消息可靠处理、优化系统性能和实现可扩展性至关重要。

## 1. 消费者组概念

### 消费者组特点
- **负载均衡**：消息在组内消费者之间分配
- **容错性**：单个消费者失败不影响整体
- **可扩展性**：动态增减消费者
- **消息独占**：每条消息只被组内一个消费者处理

### 动态消费者组实现
```javascript
class ConsumerGroup {
  constructor(name, queue, handler) {
    this.name = name;
    this.queue = queue;
    this.handler = handler;
    this.consumers = new Map();
    this.assignedPartitions = new Set();
  }

  async addConsumer(consumerId) {
    if (this.consumers.has(consumerId)) {
      throw new Error(`Consumer ${consumerId} already exists`);
    }

    const consumer = {
      id: consumerId,
      status: 'active',
      lastHeartbeat: Date.now(),
      processedCount: 0,
      errorCount: 0
    };

    this.consumers.set(consumerId, consumer);
    await this.rebalance();

    return consumer;
  }

  async removeConsumer(consumerId) {
    const consumer = this.consumers.get(consumerId);
    if (!consumer) return;

    consumer.status = 'offline';
    this.consumers.delete(consumerId);
    await this.rebalance();
  }

  async rebalance() {
    // 重新分配分区给活跃消费者
    const activeConsumers = Array.from(this.consumers.values())
      .filter(c => c.status === 'active');

    // 简单的轮询分配策略
    const partitionCount = await this.getPartitionCount();
    const partitionsPerConsumer = Math.floor(partitionCount / activeConsumers.length);

    activeConsumers.forEach((consumer, index) => {
      const startPartition = index * partitionsPerConsumer;
      const endPartition = index === activeConsumers.length - 1
        ? partitionCount
        : (index + 1) * partitionsPerConsumer;

      consumer.assignedPartitions = Array.from(
        { length: endPartition - startPartition },
        (_, i) => startPartition + i
      );
    });
  }
}
```

class ConsumerLifecycle {
  constructor(consumerId, options = {}) {
    this.id = consumerId;
    this.state = 'stopped';
    this.startTime = null;
    this.lastActivity = null;
    this.healthCheckInterval = options.healthCheckInterval || 30000;
    this.maxIdleTime = options.maxIdleTime || 300000; // 5分钟
    this.healthTimer = null;
  }

  async start() {
    if (this.state === 'running') {
      throw new Error('Consumer is already running');
    }

    this.state = 'starting';
    this.startTime = Date.now();
    this.lastActivity = Date.now();

    try {
      await this.initialize();
      this.state = 'running';
      this.startHealthCheck();
      console.log(`Consumer ${this.id} started`);
    } catch (error) {
      this.state = 'failed';
      throw error;
    }
  }

  async stop() {
    if (this.state !== 'running') return;

    this.state = 'stopping';
    this.stopHealthCheck();

    try {
      await this.shutdown();
      this.state = 'stopped';
      console.log(`Consumer ${this.id} stopped`);
    } catch (error) {
      this.state = 'failed';
      console.error(`Error stopping consumer ${this.id}:`, error);
    }
  }

  startHealthCheck() {
    this.healthTimer = setInterval(() => {
      this.checkHealth();
    }, this.healthCheckInterval);
  }

  stopHealthCheck() {
    if (this.healthTimer) {
      clearInterval(this.healthTimer);
      this.healthTimer = null;
    }
  }

  checkHealth() {
    const idleTime = Date.now() - this.lastActivity;

    if (idleTime > this.maxIdleTime && this.state === 'running') {
      console.warn(`Consumer ${this.id} is idle for ${idleTime}ms`);
      // 可以触发重启或其他恢复操作
    }
  }

  updateActivity() {
    this.lastActivity = Date.now();
  }

  async initialize() {
    // 初始化资源
  }

  async shutdown() {
    // 清理资源
  }
}

### 优雅关闭机制
```javascript
class GracefulShutdown {
  constructor(consumer) {
    this.consumer = consumer;
    this.isShuttingDown = false;
    this.pendingMessages = new Set();
    this.shutdownTimeout = 30000; // 30秒超时
  }

  async shutdown() {
    if (this.isShuttingDown) return;
    this.isShuttingDown = true;

    console.log('Starting graceful shutdown...');

    // 停止接收新消息
    await this.stopReceiving();

    // 等待当前消息处理完成
    await this.waitForPendingMessages();

    // 清理资源
    await this.cleanup();

    console.log('Graceful shutdown completed');
  }

  async stopReceiving() {
    // 停止从队列接收新消息
    this.consumer.stopPolling();
  }

  async waitForPendingMessages() {
    const startTime = Date.now();

    while (this.pendingMessages.size > 0) {
      const elapsed = Date.now() - startTime;
      if (elapsed > this.shutdownTimeout) {
        console.warn(`Shutdown timeout, ${this.pendingMessages.size} messages pending`);
        break;
      }

      await new Promise(resolve => setTimeout(resolve, 100));
    }
  }

  trackMessage(message) {
    if (!this.isShuttingDown) {
      this.pendingMessages.add(message.id);
    }
  }

  untrackMessage(messageId) {
    this.pendingMessages.delete(messageId);
  }

  async cleanup() {
    // 清理资源
    await this.consumer.closeConnections();
  }
}

// 在消息处理中使用
export default {
  async queue(batch, env) {
    const gracefulShutdown = new GracefulShutdown(this);

    // 监听关闭信号
    process.on('SIGTERM', () => gracefulShutdown.shutdown());
    process.on('SIGINT', () => gracefulShutdown.shutdown());

    for (const message of batch.messages) {
      if (gracefulShutdown.isShuttingDown) {
        // 重新排队未处理的消息
        message.retry();
        break;
      }

      gracefulShutdown.trackMessage(message);

      try {
        await processMessage(message);
        message.ack();
      } catch (error) {
        console.error('Message processing failed:', error);
        message.retry();
      } finally {
        gracefulShutdown.untrackMessage(message.id);
      }
    }
  }
};
```

## 3. 消费者负载均衡

### 负载均衡策略
```javascript
class LoadBalancer {
  constructor(strategy = 'round-robin') {
    this.strategy = strategy;
    this.currentIndex = 0;
    this.consumerStats = new Map();
  }

  selectConsumer(consumers, message = null) {
    const activeConsumers = consumers.filter(c => c.status === 'active');

    if (activeConsumers.length === 0) {
      throw new Error('No active consumers available');
    }

    switch (this.strategy) {
      case 'round-robin':
        return this.roundRobinSelect(activeConsumers);

      case 'least-connections':
        return this.leastConnectionsSelect(activeConsumers);

      case 'weighted':
        return this.weightedSelect(activeConsumers);

      case 'hash-based':
        return this.hashBasedSelect(activeConsumers, message);

      default:
        return this.roundRobinSelect(activeConsumers);
    }
  }

  roundRobinSelect(consumers) {
    const consumer = consumers[this.currentIndex % consumers.length];
    this.currentIndex++;
    return consumer;
  }

  leastConnectionsSelect(consumers) {
    return consumers.reduce((min, current) => {
      const minStats = this.consumerStats.get(min.id) || { connections: 0 };
      const currentStats = this.consumerStats.get(current.id) || { connections: 0 };

      return currentStats.connections < minStats.connections ? current : min;
    });
  }

  weightedSelect(consumers) {
    const totalWeight = consumers.reduce((sum, c) => sum + (c.weight || 1), 0);
    let random = Math.random() * totalWeight;

    for (const consumer of consumers) {
      random -= consumer.weight || 1;
      if (random <= 0) {
        return consumer;
      }
    }

    return consumers[consumers.length - 1];
  }

  hashBasedSelect(consumers, message) {
    if (!message) return this.roundRobinSelect(consumers);

    const hash = this.hash(message.id || message.body);
    const index = hash % consumers.length;
    return consumers[index];
  }

  hash(str) {
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      const char = str.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // 转换为32位整数
    }
    return Math.abs(hash);
  }

  updateConsumerStats(consumerId, stats) {
    this.consumerStats.set(consumerId, stats);
  }
}
```

### 动态负载调整
```javascript
class DynamicLoadManager {
  constructor() {
    this.consumers = new Map();
    this.metrics = new Map();
    this.scaleUpThreshold = 0.8; // 80% 负载
    this.scaleDownThreshold = 0.2; // 20% 负载
    this.checkInterval = 60000; // 1分钟检查一次
    this.maxConsumers = 10;
    this.minConsumers = 2;
  }

  async startAutoScaling(consumerFactory) {
    setInterval(async () => {
      await this.checkAndScale(consumerFactory);
    }, this.checkInterval);
  }

  async checkAndScale(consumerFactory) {
    const avgLoad = this.calculateAverageLoad();
    const currentCount = this.consumers.size;

    console.log(`Average load: ${avgLoad}, Consumers: ${currentCount}`);

    if (avgLoad > this.scaleUpThreshold && currentCount < this.maxConsumers) {
      await this.scaleUp(consumerFactory);
    } else if (avgLoad < this.scaleDownThreshold && currentCount > this.minConsumers) {
      await this.scaleDown();
    }
  }

  calculateAverageLoad() {
    const loads = Array.from(this.metrics.values());
    if (loads.length === 0) return 0;

    return loads.reduce((sum, load) => sum + load, 0) / loads.length;
  }

  async scaleUp(consumerFactory) {
    const newConsumerId = `consumer-${Date.now()}`;
    console.log(`Scaling up: adding ${newConsumerId}`);

    const consumer = await consumerFactory(newConsumerId);
    this.consumers.set(newConsumerId, consumer);
    this.metrics.set(newConsumerId, 0);
  }

  async scaleDown() {
    // 找到负载最低的消费者
    let lowestLoad = Infinity;
    let consumerToRemove = null;

    for (const [id, load] of this.metrics) {
      if (load < lowestLoad) {
        lowestLoad = load;
        consumerToRemove = id;
      }
    }

    if (consumerToRemove) {
      console.log(`Scaling down: removing ${consumerToRemove}`);
      const consumer = this.consumers.get(consumerToRemove);
      await consumer.shutdown();
      this.consumers.delete(consumerToRemove);
      this.metrics.delete(consumerToRemove);
    }
  }

  updateConsumerLoad(consumerId, load) {
    if (this.metrics.has(consumerId)) {
      this.metrics.set(consumerId, load);
    }
  }
}
```

## 4. 消费者监控

### 消费者指标收集
```javascript
class ConsumerMetrics {
  constructor(consumerId) {
    this.consumerId = consumerId;
    this.startTime = Date.now();
    this.metrics = {
      messagesProcessed: 0,
      messagesFailed: 0,
      totalProcessingTime: 0,
      lastProcessedTime: null,
      errors: []
    };
  }

  recordMessage(processingTime, success = true, error = null) {
    if (success) {
      this.metrics.messagesProcessed++;
      this.metrics.totalProcessingTime += processingTime;
    } else {
      this.metrics.messagesFailed++;
      if (error) {
        this.metrics.errors.push({
          timestamp: Date.now(),
          message: error.message,
          stack: error.stack
        });
      }
    }

    this.metrics.lastProcessedTime = Date.now();
  }

  getMetrics() {
    const avgProcessingTime = this.metrics.messagesProcessed > 0
      ? this.metrics.totalProcessingTime / this.metrics.messagesProcessed
      : 0;

    const uptime = Date.now() - this.startTime;
    const throughput = this.metrics.messagesProcessed / (uptime / 1000);

    const errorRate = (this.metrics.messagesProcessed + this.metrics.messagesFailed) > 0
      ? this.metrics.messagesFailed / (this.metrics.messagesProcessed + this.metrics.messagesFailed)
      : 0;

    return {
      consumerId: this.consumerId,
      uptime,
      messagesProcessed: this.metrics.messagesProcessed,
      messagesFailed: this.metrics.messagesFailed,
      avgProcessingTime,
      throughput,
      errorRate,
      recentErrors: this.metrics.errors.slice(-10) // 最近10个错误
    };
  }
}
```

### 健康检查实现
```javascript
class ConsumerHealthChecker {
  constructor(consumer, options = {}) {
    this.consumer = consumer;
    this.checkInterval = options.checkInterval || 30000;
    this.unhealthyThreshold = options.unhealthyThreshold || 3;
    this.healthStatus = 'healthy';
    this.consecutiveFailures = 0;
    this.lastCheck = null;
  }

  async start() {
    this.healthTimer = setInterval(async () => {
      await this.performHealthCheck();
    }, this.checkInterval);
  }

  async stop() {
    if (this.healthTimer) {
      clearInterval(this.healthTimer);
      this.healthTimer = null;
    }
  }

  async performHealthCheck() {
    try {
      const metrics = await this.consumer.getMetrics();
      const isHealthy = this.evaluateHealth(metrics);

      this.lastCheck = {
        timestamp: Date.now(),
        status: isHealthy ? 'healthy' : 'unhealthy',
        metrics
      };

      if (isHealthy) {
        this.consecutiveFailures = 0;
        if (this.healthStatus !== 'healthy') {
          this.healthStatus = 'healthy';
          console.log(`Consumer ${this.consumer.id} is healthy`);
        }
      } else {
        this.consecutiveFailures++;
        if (this.consecutiveFailures >= this.unhealthyThreshold) {
          this.healthStatus = 'unhealthy';
          console.error(`Consumer ${this.consumer.id} is unhealthy`);
          await this.handleUnhealthy();
        }
      }
    } catch (error) {
      console.error('Health check failed:', error);
      this.consecutiveFailures++;
    }
  }

  evaluateHealth(metrics) {
    // 检查错误率
    if (metrics.errorRate > 0.1) { // 10%
      return false;
    }

    // 检查是否长时间没有处理消息
    const timeSinceLastMessage = Date.now() - metrics.lastProcessedTime;
    if (timeSinceLastMessage > 300000) { // 5分钟
      return false;
    }

    // 检查平均处理时间
    if (metrics.avgProcessingTime > 30000) { // 30秒
      return false;
    }

    return true;
  }

  async handleUnhealthy() {
    // 可以实现自动恢复逻辑
    // 例如：重启消费者、发送告警等
    console.log('Attempting to restart unhealthy consumer...');
    await this.consumer.restart();
  }

  getHealthStatus() {
    return {
      status: this.healthStatus,
      consecutiveFailures: this.consecutiveFailures,
      lastCheck: this.lastCheck
    };
  }
}
```

## 5. 消费者组协调

### 分布式协调器
```javascript
class ConsumerGroupCoordinator {
  constructor(groupId, storage) {
    this.groupId = groupId;
    this.storage = storage; // KV Store 或 Durable Object
    this.heartbeatInterval = 30000;
    this.sessionTimeout = 90000; // 3次心跳超时
    this.members = new Map();
  }

  async join(memberId, metadata = {}) {
    const member = {
      id: memberId,
      metadata,
      lastHeartbeat: Date.now(),
      status: 'active'
    };

    await this.storage.put(
      `${this.groupId}:${memberId}`,
      JSON.stringify(member)
    );

    this.members.set(memberId, member);
    await this.rebalance();

    return member;
  }

  async leave(memberId) {
    await this.storage.delete(`${this.groupId}:${memberId}`);
    this.members.delete(memberId);
    await this.rebalance();
  }

  async heartbeat(memberId) {
    const key = `${this.groupId}:${memberId}`;
    const member = await this.storage.get(key);

    if (member) {
      const memberData = JSON.parse(member);
      memberData.lastHeartbeat = Date.now();

      await this.storage.put(key, JSON.stringify(memberData));
      this.members.set(memberId, memberData);
    }
  }

  async checkMembers() {
    const now = Date.now();
    const expiredMembers = [];

    for (const [memberId, member] of this.members) {
      if (now - member.lastHeartbeat > this.sessionTimeout) {
        expiredMembers.push(memberId);
      }
    }

    for (const memberId of expiredMembers) {
      await this.storage.delete(`${this.groupId}:${memberId}`);
      this.members.delete(memberId);
    }

    if (expiredMembers.length > 0) {
      await this.rebalance();
    }
  }

  async rebalance() {
    const activeMembers = Array.from(this.members.values())
      .filter(m => m.status === 'active');

    // 实现重平衡逻辑
    const assignment = this.calculateAssignment(activeMembers);

    // 通知所有成员新的分配
    for (const member of activeMembers) {
      await this.notifyAssignment(member.id, assignment[member.id]);
    }
  }

  calculateAssignment(members) {
    // 简单的轮询分配算法
    const assignment = {};
    const partitions = Array.from({ length: 10 }, (_, i) => i); // 假设10个分区

    members.forEach((member, index) => {
      const start = Math.floor(index * partitions.length / members.length);
      const end = Math.floor((index + 1) * partitions.length / members.length);

      assignment[member.id] = partitions.slice(start, end);
    });

    return assignment;
  }

  async notifyAssignment(memberId, partitions) {
    // 通知消费者新的分区分配
    const notification = {
      type: 'assignment',
      memberId,
      partitions,
      timestamp: Date.now()
    };

    await this.storage.put(
      `${this.groupId}:${memberId}:assignment`,
      JSON.stringify(notification)
    );
  }
}
```
