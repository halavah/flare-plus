### 错误处理原则
```javascript
export default {
  async queue(batch, env) {
    for (const message of batch.messages) {
      try {
        await processMessage(message);
        message.ack();
      } catch (error) {
        // 记录详细错误信息
        console.error({
          messageId: message.id,
          error: error.message,
          stack: error.stack,
          timestamp: new Date().toISOString()

          message.retry();
        } else {
          // 发送到死信队列
          await env.DLQ.send({
            body: JSON.stringify({
              originalMessage: message.body,
              error: error.message,
              timestamp: new Date().toISOString()
            })
          });
          message.ack();
        }
      }
    }
  }
};

function isTransientError(error) {
  const transientErrors = [
    'NetworkError',
    'TimeoutError',
    'TemporaryServiceUnavailable'
  ];
  return transientErrors.includes(error.name);
}
```

```javascript
  constructor() {
    this.buffer = [];
    this.bufferSize = 100;
  }

  async process(message) {
    this.buffer.push(message);

    if (this.buffer.length >= this.bufferSize) {
      await this.flush();
    }
  }

  async flush() {
    if (this.buffer.length === 0) return;

    // 批量处理
    await this.processBatch(this.buffer);
    this.buffer = [];
  }
}
```

## 3. 可靠性保证

### 幂等性实现
```javascript
class IdempotentProcessor {
  constructor(storage) {
    this.storage = storage;
    this.processedIds = new Set();
  }

  async process(message) {
    const messageId = message.id;
    const idempotencyKey = message.metadata?.idempotencyKey || messageId;

    // 检查是否已处理
    if (await this.isProcessed(idempotencyKey)) {
      console.log(`Message ${idempotencyKey} already processed`);
      return;
    }

    // 标记为处理中（防止并发处理）
    const processing = await this.markProcessing(idempotencyKey);
    if (!processing) {
      console.log(`Message ${idempotencyKey} is being processed by another consumer`);
      return;
    }

    try {
      // 处理消息
      await this.doProcess(message);

      // 标记为已完成
      await this.markCompleted(idempotencyKey);
    } catch (error) {
      // 清除处理中标记
      await this.clearProcessing(idempotencyKey);
      throw error;
    }
  }

  async isProcessed(idempotencyKey) {
    const key = `processed:${idempotencyKey}`;
    return await this.storage.get(key) !== null;
  }

  async markProcessing(idempotencyKey) {
    const key = `processing:${idempotencyKey}`;
    const existing = await this.storage.get(key);

    if (existing) {
      const data = JSON.parse(existing);
      // 检查是否超时
      if (Date.now() - data.timestamp < 300000) { // 5分钟
        return false;
      }
    }

    await this.storage.put(key, JSON.stringify({
      timestamp: Date.now()
    }), {
      expirationTtl: 600 // 10分钟过期
    });

    return true;
  }

  async markCompleted(idempotencyKey) {
    const processedKey = `processed:${idempotencyKey}`;
    const processingKey = `processing:${idempotencyKey}`;

    await this.storage.put(processedKey, 'true', {
      expirationTtl: 7 * 24 * 3600 // 7天
    });

    await this.storage.delete(processingKey);
  }
}
```

### 重试策略
```javascript
class RetryStrategy {
  constructor(options = {}) {
    this.maxRetries = options.maxRetries || 3;
    this.baseDelay = options.baseDelay || 1000;
    this.maxDelay = options.maxDelay || 30000;
    this.backoffFactor = options.backoffFactor || 2;
  }

  async executeWithRetry(fn, context) {
    let lastError;

    for (let attempt = 0; attempt <= this.maxRetries; attempt++) {
      try {
        return await fn();
      } catch (error) {
        lastError = error;

        if (attempt === this.maxRetries) {
          break;
        }

        const delay = this.calculateDelay(attempt);
        console.warn(`Attempt ${attempt + 1} failed, retrying in ${delay}ms`, error.message);

        await this.sleep(delay);
      }
    }

    throw lastError;
  }

  calculateDelay(attempt) {
    const delay = this.baseDelay * Math.pow(this.backoffFactor, attempt);
    const jitter = delay * 0.1 * Math.random(); // 10% 抖动
    return Math.min(delay + jitter, this.maxDelay);
  }

  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

// 使用示例
const retry = new RetryStrategy({
  maxRetries: 5,
  baseDelay: 2000,
  maxDelay: 60000
});

await retry.executeWithRetry(async () => {
  await processExternalAPI(message);
});
```

### 关键指标监控
```javascript
class QueueMonitor {
  constructor() {
    this.metrics = {
      messagesProcessed: 0,
      messagesFailed: 0,
      processingTime: [],
      queueDepth: 0,
      consumerLag: 0
    };
  }

  recordProcessed(processingTime) {
    this.metrics.messagesProcessed++;
    this.metrics.processingTime.push(processingTime);

    // 只保留最近1000个样本
    if (this.metrics.processingTime.length > 1000) {
      this.metrics.processingTime = this.metrics.processingTime.slice(-1000);
    }
  }

  recordFailed() {
    this.metrics.messagesFailed++;
  }

  getStats() {
    const times = this.metrics.processingTime;
    const avgTime = times.length > 0
      ? times.reduce((a, b) => a + b, 0) / times.length
      : 0;

    const sorted = [...times].sort((a, b) => a - b);
    const p95 = sorted[Math.floor(sorted.length * 0.95)] || 0;
    const p99 = sorted[Math.floor(sorted.length * 0.99)] || 0;

    const total = this.metrics.messagesProcessed + this.metrics.messagesFailed;
    const errorRate = total > 0 ? (this.metrics.messagesFailed / total) * 100 : 0;

    return {
      processed: this.metrics.messagesProcessed,
      failed: this.metrics.messagesFailed,
      errorRate: errorRate.toFixed(2) + '%',
      avgProcessingTime: avgTime.toFixed(2) + 'ms',
      p95ProcessingTime: p95 + 'ms',
      p99ProcessingTime: p99 + 'ms',
      queueDepth: this.metrics.queueDepth,
      consumerLag: this.metrics.consumerLag
    };
  }
}
```

```javascript
class BatchSender {
  constructor(queue, options = {}) {
    this.queue = queue;
    this.batchSize = options.batchSize || 100;
    this.flushInterval = options.flushInterval || 5000;
    this.buffer = [];
    this.timer = null;
  }

  async send(message) {
    this.buffer.push(message);

    if (this.buffer.length >= this.batchSize) {
      await this.flush();
    } else if (!this.timer) {
      this.timer = setTimeout(() => this.flush(), this.flushInterval);
    }
  }

  async flush() {
    if (this.buffer.length === 0) return;

    const batch = this.buffer.splice(0);
    if (this.timer) {
      clearTimeout(this.timer);
      this.timer = null;
    }

    try {
      await this.queue.sendBatch(batch);
      console.log(`Sent batch of ${batch.length} messages`);
    } catch (error) {
      console.error('Batch send failed:', error);
      // 重新加入缓冲区或发送到死信队列
      this.buffer.unshift(...batch);
    }
  }
}
```
