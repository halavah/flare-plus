# Cloudflare Queue 死信队列（DLQ）

## 死信队列概述

死信队列（Dead Letter Queue，DLQ）是处理无法正常处理的消息的特殊队列。当消息在正常队列中处理失败超过最大重试次数后，会被转移到死信队列，以便后续分析和处理。

## 1. 为什么需要死信队列

### 主要用途
- **错误隔离**：防止失败消息阻塞正常处理
- **问题诊断**：分析失败原因
- **数据恢复**：修复问题后重新处理
- **监控告警**：及时发现问题
- **审计跟踪**：记录处理失败的事件

### 常见失败场景
- 消息格式错误
- 业务逻辑异常
- 依赖服务不可用
- 资源不足
- 数据校验失败
- 超时异常

## 2. 配置死信队列

### 在 wrangler.toml 中配置
```toml
# 主队列
[[queues.producers]]
  binding = "MY_QUEUE"
  queue = "my-queue"

# 主队列的消费者
[[queues.consumers]]
  queue = "my-queue"
  max_batch_size = 10
  max_wait_time_ms = 5000
  max_retries = 3
  dead_letter_queue = "my-dlq"

# 死信队列的消费者
[[queues.consumers]]
  queue = "my-dlq"
  max_batch_size = 10
  max_wait_time_ms = 5000
```

### 使用 CLI 创建
```bash
# 创建主队列
wrangler queues create my-queue

# 创建死信队列
wrangler queues create my-dlq

# 配置死信队列
wrangler queues update my-queue --dead-letter-queue my-dlq
```

## 3. 死信队列的实现

### 消息处理与重试
```javascript
export default {
  async queue(batch, env) {
    const results = [];

    // 处理每条消息
    for (const message of batch.messages) {
      try {
        // 解析消息
        const data = JSON.parse(message.body);

        // 处理消息
        await processMessage(data);

        // 成功，确认消息
        message.ack();
        results.push({ id: message.id, status: 'success' });

      } catch (error) {
        console.error(`Failed to process ${message.id}:`, error);

        // 检查重试次数
        const retryCount = message.attempts || 0;

        if (retryCount < 3) {
          // 重试消息
          message.retry();
          results.push({ id: message.id, status: 'retry' });
        } else {
          // 超过重试次数，发送到死信队列
          await sendToDLQ(env.MY_DLQ, message, error);
          message.ack(); // 从原队列删除
          results.push({ id: message.id, status: 'dead_letter' });
        }
      }
    }

    // 记录处理结果
    logProcessingResults(results);
  }
};

async function sendToDLQ(dlq, originalMessage, error) {
  const dlqMessage = {
    body: JSON.stringify({
      originalMessage: originalMessage.body,
      messageId: originalMessage.id,
      originalQueue: originalMessage.queueName,
      failureReason: error.message,
      stackTrace: error.stack,
      failedAt: new Date().toISOString(),
      retryCount: originalMessage.attempts
    }),
    metadata: {
      originalMessageId: originalMessage.id,
      failureType: error.constructor.name,
      errorCode: error.code
    }
  };

  await dlq.send(dlqMessage);
}
```

### 死信队列处理器
```javascript
export default {
  async queue(batch, env) {
    for (const message of batch.messages) {
      const dlqData = JSON.parse(message.body);

      // 分析失败原因
      const analysis = analyzeFailure(dlqData);

      // 根据分析结果采取行动
      switch (analysis.action) {
        case 'RETRY':
          await retryMessage(env.MY_QUEUE, dlqData);
          message.ack();
          break;

        case 'MANUAL_REVIEW':
          await notifyTeam(dlqData);
          message.ack();
          break;

        case 'DISCARD':
          logAndDiscard(dlqData);
          message.ack();
          break;

        case 'REPAIR':
          const repaired = await repairMessage(dlqData);
          if (repaired) {
            await env.MY_QUEUE.send(repaired);
          }
          message.ack();
          break;

        default:
          // 保留在死信队列中供进一步调查
          console.warn('Unknown action for DLQ message:', dlqData);
      }
    }
  }
};

function analyzeFailure(dlqData) {
  const error = dlqData.failureReason;

  // 系统错误，可以重试
  if (error.includes('timeout') || error.includes('network')) {
    return { action: 'RETRY', confidence: 0.8 };
  }

  // 数据格式错误，可能需要修复
  if (error.includes('JSON') || error.includes('format')) {
    return { action: 'REPAIR', confidence: 0.6 };
  }

  // 业务逻辑错误，需要人工审查
  if (error.includes('validation') || error.includes('business')) {
    return { action: 'MANUAL_REVIEW', confidence: 0.9 };
  }

  // 未知错误，默认保留
  return { action: 'UNKNOWN', confidence: 0.1 };
}
```

## 4. 死信队列管理

### 监控死信队列
```javascript
class DLQMonitor {
  constructor(dlq) {
    this.dlq = dlq;
    this.alertThreshold = 100; // 死信消息数量阈值
    this.checkInterval = 60000; // 检查间隔：1分钟
  }

  async start() {
    setInterval(async () => {
      await this.checkDLQ();
    }, this.checkInterval);
  }

  async checkDLQ() {
    const stats = await this.dlq.getStats();
    const messageCount = stats.approximateMessageCount;

    if (messageCount > this.alertThreshold) {
      await this.sendAlert({
        type: 'DLQ_OVERFLOW',
        messageCount,
        threshold: this.alertThreshold,
        timestamp: new Date().toISOString()
      });
    }

    // 记录指标
    this.recordMetrics({
      messageCount,
      oldestMessage: stats.approximateAgeOfOldestMessage
    });
  }

  async sendAlert(alert) {
    // 发送到告警系统
    await fetch(process.env.ALERT_WEBHOOK, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(alert)
    });
  }

  recordMetrics(metrics) {
    // 发送到监控系统
    console.log('DLQ Metrics:', JSON.stringify(metrics));
  }
}
```

### 死信消息分析工具
```javascript
class DLQAnalyzer {
  async analyze(dlq, limit = 100) {
    const batch = await dlq.receive({
      batchSize: limit,
      waitTimeMs: 1000
    });

    const analysis = {
      totalMessages: batch.messages.length,
      errorTypes: {},
      failureReasons: {},
      timeDistribution: {},
      retryDistribution: {}
    };

    for (const message of batch.messages) {
      const data = JSON.parse(message.body);

      // 统计错误类型
      const errorType = data.failureReason.split(':')[0];
      analysis.errorTypes[errorType] = (analysis.errorTypes[errorType] || 0) + 1;

      // 统计失败原因
      analysis.failureReasons[data.failureReason] =
        (analysis.failureReasons[data.failureReason] || 0) + 1;

      // 统计时间分布
      const hour = new Date(data.failedAt).getHours();
      analysis.timeDistribution[hour] = (analysis.timeDistribution[hour] || 0) + 1;

      // 统计重试次数
      const retryCount = data.retryCount || 0;
      analysis.retryDistribution[retryCount] =
        (analysis.retryDistribution[retryCount] || 0) + 1;

      // 不要确认消息，保留在死信队列
    }

    return analysis;
  }

  generateReport(analysis) {
    return `
死信队列分析报告
================
总消息数: ${analysis.totalMessages}

错误类型分布:
${Object.entries(analysis.errorTypes)
  .sort((a, b) => b[1] - a[1])
  .map(([type, count]) => `  ${type}: ${count}`)
  .join('\n')}

主要失败原因:
${Object.entries(analysis.failureReason)
  .sort((a, b) => b[1] - a[1])
  .slice(0, 5)
  .map(([reason, count]) => `  ${reason}: ${count}`)
  .join('\n')}

时间分布:
${Object.entries(analysis.timeDistribution)
  .sort((a, b) => a[0] - b[0])
  .map(([hour, count]) => `  ${hour}:00 - ${count}`)
  .join('\n')}
    `;
  }
}
```

## 5. 死信队列恢复策略

### 自动重试策略
```javascript
class DLQRetryManager {
  constructor(dlq, targetQueue) {
    this.dlq = dlq;
    this.targetQueue = targetQueue;
    this.retryRules = new Map();
    this.setupDefaultRules();
  }

  setupDefaultRules() {
    // 网络错误：5分钟后重试
    this.retryRules.set('NetworkError', { delay: 300000, maxRetries: 3 });

    // 超时错误：10分钟后重试
    this.retryRules.set('TimeoutError', { delay: 600000, maxRetries: 2 });

    // 临时错误：1分钟后重试
    this.retryRules.set('TemporaryError', { delay: 60000, maxRetries: 5 });
  }

  async processDLQ() {
    const batch = await this.dlq.receive({
      batchSize: 10,
      waitTimeMs: 5000
    });

    for (const message of batch.messages) {
      const data = JSON.parse(message.body);
      const shouldRetry = await this.evaluateRetry(data);

      if (shouldRetry) {
        await this.retryMessage(data);
        message.ack();
      } else {
        // 保留消息等待人工处理
        console.log('Message not eligible for retry:', data.originalMessageId);
      }
    }
  }

  async evaluateRetry(data) {
    const rule = this.retryRules.get(data.failureType);
    if (!rule) return false;

    // 检查重试次数
    if (data.retryCount >= rule.maxRetries) {
      return false;
    }

    // 检查时间间隔
    const timeSinceFailure = Date.now() - new Date(data.failedAt).getTime();
    if (timeSinceFailure < rule.delay) {
      return false;
    }

    return true;
  }

  async retryMessage(data) {
    // 创建新消息，更新元数据
    const retryMessage = {
      body: data.originalMessage,
      metadata: {
        ...data.originalMetadata,
        dlqRetry: true,
        originalFailure: data.failureReason,
        retriedAt: new Date().toISOString()
      }
    };

    await this.targetQueue.send(retryMessage);
  }
}
```

### 批量重试工具
```javascript
// 管理员工具：批量重试死信消息
async function bulkRetryDLQ(dlq, targetQueue, filter = null) {
  const retried = [];
  const failed = [];

  let hasMore = true;
  while (hasMore) {
    const batch = await dlq.receive({
      batchSize: 25,
      waitTimeMs: 1000
    });

    for (const message of batch.messages) {
      try {
        const data = JSON.parse(message.body);

        // 应用过滤器
        if (filter && !filter(data)) {
          continue;
        }

        // 重试消息
        await targetQueue.send({
          body: data.originalMessage,
          metadata: { dlqRetry: true }
        });

        retried.push(message.id);
        message.ack();
      } catch (error) {
        failed.push({ id: message.id, error });
      }
    }

    hasMore = batch.messages.length > 0;
  }

  return { retried, failed };
}

// 使用示例
const result = await bulkRetryDLQ(
  env.MY_DLQ,
  env.MY_QUEUE,
  (data) => data.failureType === 'NetworkError' // 只重试网络错误
);
```

## 6. 最佳实践

### 1. 死信队列设计原则
- 不要让死信队列无限增长
- 定期分析和清理
- 实现适当的监控和告警
- 保持死信消息的完整上下文

### 3. 死信队列清理策略
```javascript
// 定期清理策略
async function cleanupDLQ(dlq, olderThanDays = 30) {
  const cutoffDate = new Date();
  cutoffDate.setDate(cutoffDate.getDate() - olderThanDays);

  let deletedCount = 0;
  let hasMore = true;

  while (hasMore) {
    const batch = await dlq.receive({
      batchSize: 25,
      waitTimeMs: 1000
    });

    for (const message of batch.messages) {
      const data = JSON.parse(message.body);
      const failedDate = new Date(data.failedAt);

      if (failedDate < cutoffDate) {
        // 归档到存储系统
        await archiveMessage(data);
        message.ack();
        deletedCount++;
      }
    }

    hasMore = batch.messages.length > 0;
  }

  return deletedCount;
}
```
