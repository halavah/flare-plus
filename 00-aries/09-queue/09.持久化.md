# Cloudflare Queue 消息持久化

## 持久化概述

Cloudflare Queue 提供可靠的消息持久化机制，确保消息在系统故障、网络中断或其他异常情况下不会丢失。持久化是消息队列系统的核心特性之一。

## 1. 持久化级别

### 至少一次（At-Least-Once）投递
```javascript
// Cloudflare Queue 默认保证至少一次投递
export default {
  async queue(batch, env) {
    for (const message of batch.messages) {
      // 处理消息前检查是否已处理（实现幂等性）
      const messageId = message.id;

      if (await isAlreadyProcessed(messageId)) {
        console.log(`Message ${messageId} already processed, skipping`);
        message.ack();
        continue;
      }

      try {
        // 处理消息
        await processMessage(message);

        // 标记为已处理
        await markAsProcessed(messageId);

        // 确认消息
        message.ack();
      } catch (error) {
        console.error(`Failed to process message ${messageId}:`, error);

        // 不确认消息，让它重新投递
        // 或者使用 message.retry() 主动重试
      }
    }
  }
};

// 实现幂等性的辅助函数
async function isAlreadyProcessed(messageId) {
  const key = `processed:${messageId}`;
  const value = await env.KV.get(key);
  return value !== null;
}

async function markAsProcessed(messageId) {
  const key = `processed:${messageId}`;
  const value = {
    processedAt: new Date().toISOString(),
    messageId
  };
  await env.KV.put(key, JSON.stringify(value), {
    expirationTtl: 7 * 24 * 3600 // 7天过期
  });
}
```

### 消息去重实现
```javascript
class MessageDeduplicator {
  constructor(storage, windowSize = 3600000) { // 默认1小时窗口
    this.storage = storage;
    this.windowSize = windowSize;
  }

  async checkDuplicate(messageId, contentHash = null) {
    const key = `msg:${messageId}`;
    const existing = await this.storage.get(key);

    if (existing) {
      const data = JSON.parse(existing);
      if (contentHash && data.contentHash !== contentHash) {
        // 相同ID但内容不同，可能是重新发送
        await this.updateRecord(key, messageId, contentHash);
        return false;
      }
      return true; // 重复消息
    }

    // 记录新消息
    await this.updateRecord(key, messageId, contentHash);
    return false;
  }

  async updateRecord(key, messageId, contentHash) {
    const data = {
      messageId,
      contentHash,
      firstSeen: new Date().toISOString(),
      lastSeen: new Date().toISOString()
    };

    await this.storage.put(key, JSON.stringify(data), {
      expirationTtl: Math.ceil(this.windowSize / 1000)
    });
  }
}
```

## 2. 消息存储机制

### 存储架构
```javascript
// Cloudflare Queue 使用多层存储架构
class QueueStorageArchitecture {
  constructor() {
    this.hotStorage = new Map(); // 热数据，内存缓存
    this.warmStorage = new DurableObjectStorage(); // 温数据，快速访问
    this.coldStorage = new R2Bucket(); // 冷数据，大容量存储
  }

  // 写入消息时的存储策略
  async storeMessage(message) {
    const messageData = {
      id: message.id,
      body: message.body,
      metadata: message.metadata,
      timestamp: Date.now(),
      attempts: 0
    };

    // 1. 写入热存储（内存）
    this.hotStorage.set(message.id, messageData);

    // 2. 异步写入温存储
    this.warmStorage.put(message.id, JSON.stringify(messageData));

    // 3. 如果消息很大，存储到冷存储
    if (message.size > 1024 * 1024) { // 1MB
      const key = `large:${message.id}`;
      await this.coldStorage.put(key, message.body);
      messageData.body = `ref:${key}`;
    }

    return messageData;
  }

  // 读取消息时的存储策略
  async retrieveMessage(messageId) {
    // 1. 先查热存储
    let message = this.hotStorage.get(messageId);
    if (message) return message;

    // 2. 查温存储
    const stored = await this.warmStorage.get(messageId);
    if (stored) {
      message = JSON.parse(stored);

      // 如果是冷存储引用，加载实际内容
      if (message.body.startsWith('ref:')) {
        const key = message.body.substring(4);
        message.body = await this.coldStorage.get(key);
      }

      // 加载到热存储
      this.hotStorage.set(messageId, message);
      return message;
    }

    return null;
  }
}
```

### 分区存储策略
```javascript
class PartitionedStorage {
  constructor(partitionCount = 16) {
    this.partitionCount = partitionCount;
    this.partitions = Array.from({ length: partitionCount }, () => ({
      messages: new Map(),
      size: 0,
      lastAccess: Date.now()
    }));
  }

  getPartition(messageId) {
    const hash = this.hash(messageId);
    return this.partitions[hash % this.partitionCount];
  }

  hash(str) {
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      const char = str.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash;
    }
    return Math.abs(hash);
  }

  async store(message) {
    const partition = this.getPartition(message.id);

    partition.messages.set(message.id, {
      ...message,
      storedAt: Date.now()
    });

    partition.size += JSON.stringify(message).length;
    partition.lastAccess = Date.now();

    // 如果分区过大，触发清理
    if (partition.size > 100 * 1024 * 1024) { // 100MB
      await this.compactPartition(partition);
    }
  }

  async retrieve(messageId) {
    const partition = this.getPartition(messageId);
    const message = partition.messages.get(messageId);

    if (message) {
      partition.lastAccess = Date.now();
      return message;
    }

    return null;
  }

  async compactPartition(partition) {
    const messages = Array.from(partition.messages.entries());
    const now = Date.now();

    // 移除过期消息
    for (const [id, message] of messages) {
      const age = now - message.storedAt;
      const maxAge = 4 * 24 * 3600 * 1000; // 4天

      if (age > maxAge) {
        partition.messages.delete(id);
        partition.size -= JSON.stringify(message).length;
      }
    }
  }
}
```

## 3. 事务和一致性

### 消息事务实现
```javascript
class MessageTransaction {
  constructor(queue, storage) {
    this.queue = queue;
    this.storage = storage;
    this.operations = [];
    this.status = 'active';
    this.transactionId = this.generateId();
  }

  async begin() {
    const transaction = {
      id: this.transactionId,
      status: 'active',
      operations: [],
      createdAt: Date.now()
    };

    await this.storage.put(
      `txn:${this.transactionId}`,
      JSON.stringify(transaction)
    );

    return this.transactionId;
  }

  async send(message) {
    if (this.status !== 'active') {
      throw new Error('Transaction is not active');
    }

    const operation = {
      type: 'send',
      message,
      timestamp: Date.now()
    };

    this.operations.push(operation);

    // 记录操作到事务日志
    await this.appendOperation(operation);
  }

  async commit() {
    if (this.status !== 'active') {
      throw new Error('Transaction is not active');
    }

    try {
      // 执行所有操作
      const sendOperations = this.operations.filter(op => op.type === 'send');

      if (sendOperations.length > 0) {
        const messages = sendOperations.map(op => ({
          body: op.message.body,
          metadata: {
            ...op.message.metadata,
            transactionId: this.transactionId
          }
        }));

        await this.queue.sendBatch(messages);
      }

      // 更新事务状态
      this.status = 'committed';
      await this.updateTransactionStatus('committed');

    } catch (error) {
      await this.rollback();
      throw error;
    }
  }

  async rollback() {
    this.status = 'rolled_back';
    await this.updateTransactionStatus('rolled_back');
  }

  async appendOperation(operation) {
    const key = `txn:${this.transactionId}:ops`;
    const existing = await this.storage.get(key);
    const operations = existing ? JSON.parse(existing) : [];

    operations.push(operation);

    await this.storage.put(key, JSON.stringify(operations));
  }

  async updateTransactionStatus(status) {
    const key = `txn:${this.transactionId}`;
    const transaction = JSON.parse(await this.storage.get(key));

    transaction.status = status;
    transaction.updatedAt = Date.now();

    await this.storage.put(key, JSON.stringify(transaction));
  }

  generateId() {
    return `txn_${Date.now()}_${Math.random().toString(36).substring(2, 15)}`;
  }
}
```

### 两阶段提交实现
```javascript
class TwoPhaseCommit {
  constructor(participants) {
    this.participants = participants;
    this.coordinatorId = `coord_${Date.now()}`;
  }

  async executeTransaction(operations) {
    const transactionId = this.generateTransactionId();

    try {
      // Phase 1: Prepare
      const prepared = await this.preparePhase(transactionId, operations);

      if (!prepared) {
        await this.abortPhase(transactionId);
        throw new Error('Transaction prepare phase failed');
      }

      // Phase 2: Commit
      await this.commitPhase(transactionId);

      return { success: true, transactionId };
    } catch (error) {
      await this.abortPhase(transactionId);
      throw error;
    }
  }

  async preparePhase(transactionId, operations) {
    const promises = this.participants.map(participant =>
      this.sendPrepare(participant, transactionId, operations)
    );

    const results = await Promise.allSettled(promises);

    // 检查是否所有参与者都准备好
    return results.every(result =>
      result.status === 'fulfilled' && result.value === 'ready'
    );
  }

  async sendPrepare(participant, transactionId, operations) {
    const prepareMessage = {
      type: 'prepare',
      transactionId,
      operations
    };

    try {
      const response = await participant.prepare(prepareMessage);
      return response.status === 'ready';
    } catch (error) {
      console.error(`Prepare failed for ${participant.id}:`, error);
      return false;
    }
  }

  async commitPhase(transactionId) {
    const promises = this.participants.map(participant =>
      this.sendCommit(participant, transactionId)
    );

    const results = await Promise.allSettled(promises);

    // 记录提交结果
    const failures = results.filter(r => r.status === 'rejected');
    if (failures.length > 0) {
      console.error(`${failures.length} participants failed to commit`);
    }
  }

  async sendCommit(participant, transactionId) {
    const commitMessage = {
      type: 'commit',
      transactionId
    };

    await participant.commit(commitMessage);
  }

  async abortPhase(transactionId) {
    const promises = this.participants.map(participant =>
      this.sendAbort(participant, transactionId)
    );

    await Promise.allSettled(promises);
  }

  async sendAbort(participant, transactionId) {
    const abortMessage = {
      type: 'abort',
      transactionId
    };

    await participant.abort(abortMessage);
  }

  generateTransactionId() {
    return `2pc_${Date.now()}_${Math.random().toString(36).substring(2, 15)}`;
  }
}
```

## 4. 备份和恢复

### 消息备份策略
```javascript
class MessageBackup {
  constructor(sourceQueue, backupStorage) {
    this.sourceQueue = sourceQueue;
    this.backupStorage = backupStorage;
    this.backupInterval = 3600000; // 1小时备份一次
    this.compressionEnabled = true;
  }

  async startBackup() {
    setInterval(async () => {
      await this.performBackup();
    }, this.backupInterval);
  }

  async performBackup() {
    const backupId = `backup_${Date.now()}`;
    const backup = {
      id: backupId,
      timestamp: new Date().toISOString(),
      messages: []
    };

    let hasMore = true;
    let processedCount = 0;

    while (hasMore) {
      const batch = await this.sourceQueue.receive({
        batchSize: 100,
        waitTimeMs: 1000
      });

      if (batch.messages.length > 0) {
        for (const message of batch.messages) {
          backup.messages.push({
            id: message.id,
            body: message.body,
            metadata: message.metadata,
            timestamp: message.timestamp
          });

          // 确认消息（不移除，只是备份）
          processedCount++;
        }

        // 释放消息回队列
        for (const message of batch.messages) {
          message.retry();
        }
      } else {
        hasMore = false;
      }

      // 避免长时间处理
      if (processedCount > 10000) {
        hasMore = false;
      }
    }

    // 压缩备份
    const backupData = this.compressionEnabled
      ? await this.compress(JSON.stringify(backup))
      : JSON.stringify(backup);

    // 存储备份
    const backupKey = `backups/${backupId}.json${this.compressionEnabled ? '.gz' : ''}`;
    await this.backupStorage.put(backupKey, backupData);

    console.log(`Backup completed: ${backupId}, Messages: ${backup.messages.length}`);
  }

  async compress(data) {
    // 实现压缩逻辑
    return data; // 简化示例
  }

  async decompress(data) {
    // 实现解压缩逻辑
    return data; // 简化示例
  }

  async restore(backupId) {
    const backupKey = `backups/${backupId}.json${this.compressionEnabled ? '.gz' : ''}`;
    const backupData = await this.backupStorage.get(backupKey);

    if (!backupData) {
      throw new Error(`Backup ${backupId} not found`);
    }

    const decompressed = this.compressionEnabled
      ? await this.decompress(backupData)
      : backupData;

    const backup = JSON.parse(decompressed);

    // 恢复消息到队列
    const messages = backup.messages.map(msg => ({
      body: msg.body,
      metadata: {
        ...msg.metadata,
        restoredFrom: backupId,
        restoredAt: new Date().toISOString()
      }
    }));

    // 批量发送
    const batchSize = 100;
    for (let i = 0; i < messages.length; i += batchSize) {
      const batch = messages.slice(i, i + batchSize);
      await this.sourceQueue.sendBatch(batch);
    }

    console.log(`Restored ${messages.length} messages from backup ${backupId}`);
    return messages.length;
  }
}
```

## 5. 持久化性能优化

### 写入优化
```javascript
class WriteOptimizedPersistence {
  constructor(storage) {
    this.storage = storage;
    this.writeBuffer = [];
    this.bufferSize = 100;
    this.flushInterval = 5000; // 5秒
    this.compressionEnabled = true;
  }

  async write(message) {
    this.writeBuffer.push({
      data: message,
      timestamp: Date.now()
    });

    if (this.writeBuffer.length >= this.bufferSize) {
      await this.flush();
    }
  }

  async flush() {
    if (this.writeBuffer.length === 0) return;

    const batch = [...this.writeBuffer];
    this.writeBuffer = [];

    const batchData = {
      messages: batch.map(item => item.data),
      timestamp: Date.now()
    };

    let serialized = JSON.stringify(batchData);

    if (this.compressionEnabled) {
      serialized = await this.compress(serialized);
    }

    const batchId = `batch_${Date.now()}_${Math.random().toString(36).substring(2, 15)}`;
    await this.storage.put(batchId, serialized);

    return batchId;
  }

  async startPeriodicFlush() {
    setInterval(async () => {
      if (this.writeBuffer.length > 0) {
        await this.flush();
      }
    }, this.flushInterval);
  }
}
```

### 读取优化
```javascript
class ReadOptimizedPersistence {
  constructor(storage) {
    this.storage = storage;
    this.readCache = new Map();
    this.cacheSize = 1000;
    this.hitCount = 0;
    this.missCount = 0;
  }

  async read(messageId) {
    // 1. 检查缓存
    if (this.readCache.has(messageId)) {
      this.hitCount++;
      return this.readCache.get(messageId);
    }

    // 2. 从存储读取
    this.missCount++;
    const message = await this.storage.get(messageId);

    if (message) {
      // 3. 更新缓存
      this.updateCache(messageId, message);
    }

    return message;
  }

  updateCache(messageId, message) {
    // LRU 缓存策略
    if (this.readCache.size >= this.cacheSize) {
      const oldestKey = this.readCache.keys().next().value;
      this.readCache.delete(oldestKey);
    }

    this.readCache.set(messageId, message);
  }

  getCacheStats() {
    const total = this.hitCount + this.missCount;
    return {
      hits: this.hitCount,
      misses: this.missCount,
      hitRate: total > 0 ? (this.hitCount / total * 100).toFixed(2) + '%' : '0%',
      cacheSize: this.readCache.size
    };
  }

  preloadCache(messageIds) {
    // 预加载常用消息到缓存
    messageIds.forEach(async id => {
      if (!this.readCache.has(id)) {
        const message = await this.storage.get(id);
        if (message) {
          this.updateCache(id, message);
        }
      }
    });
  }
}
```

## 最佳实践

1. **实现幂等性**：防止重复处理
2. **合理设置TTL**：平衡存储成本和消息保留时间
