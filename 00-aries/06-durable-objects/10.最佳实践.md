# Durable Objects 最佳实践

## 概述

Durable Objects 是一个强大的有状态计算平台，但要充分发挥其潜力，需要遵循一些最佳实践。本文档总结了在实际开发中积累的经验和技巧，帮助开发者构建高性能、可扩展、可靠的应用。

```javascript
// 集中式状态管理
export class StateManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.stateSchema = {
      version: '1.0.0',
      metadata: {},
      data: {},
      indexes: {}
    };
  }

  // 统一的状态访问接口
  async getState() {
    const state = await this.state.storage.get('state');
    return state || this.createInitialState();
  }

  async updateState(updates) {
    return await this.state.storage.transaction(async (txn) => {
      const currentState = await txn.get('state') || this.createInitialState();

      // 应用更新
      Object.assign(currentState.data, updates);
      currentState.metadata.lastUpdated = Date.now();

      // 更新索引
      await this.updateIndexes(txn, currentState);

      await txn.put('state', currentState);
      return currentState;
    });
  }

  async createInitialState() {
    return {
      version: this.stateSchema.version,
      metadata: {
        created: Date.now(),
        lastUpdated: Date.now()
      },
      data: {},
      indexes: {}
    };
  }

  async updateIndexes(txn, state) {
    // 更新搜索索引
    for (const [key, value] of Object.entries(state.data)) {
      if (typeof value === 'object' && value !== null) {
        const indexValue = this.extractIndexValue(value);
        if (indexValue) {
          await txn.put(`index:${indexValue}:${key}`, true);
        }
      }
    }
  }

  extractIndexValue(obj) {
    // 提取可索引的值
    return obj.id || obj.name || obj.type || null;
  }
}
```

### 1. 批量操作优化

```javascript
// 智能批量处理器
export class SmartBatchProcessor {
  constructor(state, options = {}) {
    this.state = state;
    this.batchSize = options.batchSize || 100;
    this.flushInterval = options.flushInterval || 5000;
    this.priorityQueue = new PriorityQueue();
    this.batchQueue = [];
    this.lastFlush = Date.now();
  }

  async add(operation, priority = 'normal') {
    operation.addedAt = Date.now();
    operation.priority = priority;

    if (priority === 'high') {
      // 高优先级立即处理
      await this.processOperation(operation);
    } else {
      this.priorityQueue.enqueue(operation);
      await this.checkFlush();
    }
  }

  async checkFlush() {
    const now = Date.now();
    const shouldFlush =
      this.priorityQueue.size() >= this.batchSize ||
      (now - this.lastFlush) >= this.flushInterval;

    if (shouldFlush) {
      await this.flush();
    }
  }

  async flush() {
    if (this.priorityQueue.isEmpty()) return;

    // 收集批次
    const batch = [];
    while (!this.priorityQueue.isEmpty() && batch.length < this.batchSize) {
      batch.push(this.priorityQueue.dequeue());
    }

    this.lastFlush = Date.now();

    // 按类型分组
    const groups = this.groupByType(batch);
    const results = [];

    // 并行执行不同类型的操作
    const promises = Array.from(groups.entries()).map(async ([type, operations]) => {
      return await this.processBatch(type, operations);
    });

    const batchResults = await Promise.all(promises);
    results.push(...batchResults.flat());

    return results;
  }

  groupByType(operations) {
    const groups = new Map();
    for (const op of operations) {
      if (!groups.has(op.type)) {
        groups.set(op.type, []);
      }
      groups.get(op.type).push(op);
    }
    return groups;
  }

  async processBatch(type, operations) {
    switch (type) {
      case 'write':
        return await this.processWriteBatch(operations);
      case 'read':
        return await this.processReadBatch(operations);
      case 'delete':
        return await this.processDeleteBatch(operations);
      default:
        return await this.processGenericBatch(operations);
    }
  }

  async processWriteBatch(operations) {
    return await this.state.storage.transaction(async (txn) => {
      const results = [];
      for (const op of operations) {
        try {
          await txn.put(op.key, op.value, op.options);
          results.push({ id: op.id, success: true });
        } catch (error) {
          results.push({ id: op.id, success: false, error: error.message });
        }
      }
      return results;
    });
  }
}

// 优先级队列实现
class PriorityQueue {
  constructor() {
    this.items = [];
  }

  enqueue(item) {
    const priority = this.getPriority(item.priority);
    let added = false;

    for (let i = 0; i < this.items.length; i++) {
      if (this.getPriority(this.items[i].priority) < priority) {
        this.items.splice(i, 0, item);
        added = true;
        break;
      }
    }

    if (!added) {
      this.items.push(item);
    }
  }

  dequeue() {
    return this.items.shift();
  }

  size() {
    return this.items.length;
  }

  isEmpty() {
    return this.items.length === 0;
  }

  getPriority(priority) {
    const priorities = { low: 1, normal: 2, high: 3, critical: 4 };
    return priorities[priority] || 2;
  }
}
```

```javascript
// 多层缓存系统
export class MultiLayerCache {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.l1Cache = new Map(); // 内存缓存
    this.l2Cache = env.CACHE_KV; // KV 缓存
    this.l1MaxSize = 1000;
    this.l1TTL = 30000; // 30秒
    this.l2TTL = 300; // 5分钟
  }

  async get(key) {
    // L1 缓存检查
    const l1Value = this.l1Cache.get(key);
    if (l1Value && Date.now() - l1Value.timestamp < this.l1TTL) {
      return l1Value.value;
    }

    // L2 缓存检查
    const l2Value = await this.l2Cache.get(key);
    if (l2Value) {
      const parsed = JSON.parse(l2Value);
      // 回填 L1
      this.setL1(key, parsed.value);
      return parsed.value;
    }

    return null;
  }

  async set(key, value, customTTL = null) {
    // 设置 L1 缓存
    this.setL1(key, value);

    // 设置 L2 缓存
    const ttl = customTTL || this.l2TTL;
    await this.l2Cache.put(
      key,
      JSON.stringify({ value: value, timestamp: Date.now() }),
      { expirationTtl: ttl }
    );
  }

  setL1(key, value) {
    // LRU 淘汰
    if (this.l1Cache.size >= this.l1MaxSize) {
      const firstKey = this.l1Cache.keys().next().value;
      this.l1Cache.delete(firstKey);
    }

    this.l1Cache.set(key, {
      value: value,
      timestamp: Date.now()
    });
  }

  // 智能预热
  async warmup(pattern) {
    const keys = await this.getKeysByPattern(pattern);
    const promises = keys.map(key => this.get(key));
    await Promise.all(promises);
  }

  async getKeysByPattern(pattern) {
    // 实现键模式匹配
    const allKeys = await this.l2Cache.list({ prefix: pattern });
    return allKeys.keys.map(key => key.name);
  }
}
```

### 1. 错误处理模式
```javascript
// 全面错误处理
export class RobustErrorHandler {
  constructor(logger, metrics) {
    this.logger = logger;
    this.metrics = metrics;
    this.retryStrategies = new Map();
    this.circuitBreakers = new Map();
  }

  async executeWithRetry(operation, options = {}) {
    const {
      maxRetries = 3,
      baseDelay = 1000,
      maxDelay = 30000,
      backoff = 'exponential'
    } = options;

    let lastError;
    let attempt = 0;

    while (attempt <= maxRetries) {
      try {
        const result = await operation();

        // 成功时记录指标
        this.metrics.recordSuccess(operation.name, attempt);

        return result;
      } catch (error) {
        lastError = error;
        attempt++;

        // 检查是否应该重试
        if (!this.shouldRetry(error, attempt, maxRetries)) {
          break;
        }

        // 计算延迟
        const delay = this.calculateDelay(
          baseDelay,
          attempt,
          maxDelay,
          backoff
        );

        this.logger.warn(`Attempt ${attempt} failed, retrying in ${delay}ms`, {
          error: error.message,
          attempt: attempt
        });

        await this.sleep(delay);
      }
    }

    // 失败时记录指标
    this.metrics.recordFailure(operation.name, attempt, lastError);

    throw lastError;
  }

  shouldRetry(error, attempt, maxRetries) {
    // 不重试的错误类型
    const noRetryErrors = [
      'ValidationError',
      'AuthenticationError',
      'PermissionError'
    ];

    if (noRetryErrors.includes(error.name)) {
      return false;
    }

    // 达到最大重试次数
    if (attempt > maxRetries) {
      return false;
    }

    return true;
  }

  calculateDelay(baseDelay, attempt, maxDelay, backoff) {
    let delay;

    switch (backoff) {
      case 'exponential':
        delay = baseDelay * Math.pow(2, attempt - 1);
        break;
      case 'linear':
        delay = baseDelay * attempt;
        break;
      case 'fixed':
        delay = baseDelay;
        break;
      default:
        delay = baseDelay;
    }

    // 添加抖动
    const jitter = delay * 0.1 * Math.random();
    delay += jitter;

    return Math.min(delay, maxDelay);
  }

  async sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

// 断路器模式
export class CircuitBreaker {
  constructor(options = {}) {
    this.failureThreshold = options.failureThreshold || 5;
    this.recoveryTimeout = options.recoveryTimeout || 60000;
    this.monitoringPeriod = options.monitoringPeriod || 10000;

    this.state = 'CLOSED'; // CLOSED, OPEN, HALF_OPEN
    this.failureCount = 0;
    this.lastFailureTime = 0;
    this.successCount = 0;
  }

  async execute(operation) {
    if (this.state === 'OPEN') {
      if (Date.now() - this.lastFailureTime < this.recoveryTimeout) {
        throw new Error('Circuit breaker is OPEN');
      } else {
        this.state = 'HALF_OPEN';
      }
    }

    try {
      const result = await operation();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      throw error;
    }
  }

  onSuccess() {
    this.failureCount = 0;

    if (this.state === 'HALF_OPEN') {
      this.state = 'CLOSED';
      this.successCount++;
    }
  }

  onFailure() {
    this.failureCount++;
    this.lastFailureTime = Date.now();

    if (this.failureCount >= this.failureThreshold) {
      this.state = 'OPEN';
    }
  }

  getState() {
    return {
      state: this.state,
      failureCount: this.failureCount,
      successCount: this.successCount,
      lastFailureTime: this.lastFailureTime
    };
  }
}
```

```javascript
// 事务管理器
export class TransactionManager {
  constructor(state) {
    this.state = state;
    this.transactions = new Map();
  }

  async executeTransaction(operations) {
    const transactionId = crypto.randomUUID();
    const startTime = Date.now();

    try {
      // 验证操作
      this.validateOperations(operations);

      // 执行事务
      const results = await this.state.storage.transaction(async (txn) => {
        const transactionResults = [];

        for (const operation of operations) {
          const result = await this.executeOperation(txn, operation);
          transactionResults.push(result);
        }

        return transactionResults;
      });

      // 记录成功事务
      await this.recordTransaction(transactionId, operations, results, true, Date.now() - startTime);

      return results;

    } catch (error) {
      // 记录失败事务
      await this.recordTransaction(transactionId, operations, null, false, Date.now() - startTime, error);
      throw error;
    }
  }

  validateOperations(operations) {
    // 检查操作完整性
    if (!operations || operations.length === 0) {
      throw new Error('No operations provided');
    }

    // 检查操作有效性
    for (const op of operations) {
      if (!op.type || !op.key) {
        throw new Error(`Invalid operation: ${JSON.stringify(op)}`);
      }
    }
  }

  async executeOperation(txn, operation) {
    switch (operation.type) {
      case 'put':
        await txn.put(operation.key, operation.value, operation.options);
        return { type: 'put', key: operation.key, success: true };

      case 'delete':
        await txn.delete(operation.key);
        return { type: 'delete', key: operation.key, success: true };

      case 'check':
        const exists = await txn.get(operation.key) !== null;
        if (!exists && operation.required) {
          throw new Error(`Required key not found: ${operation.key}`);
        }
        return { type: 'check', key: operation.key, exists: exists };

      default:
        throw new Error(`Unknown operation type: ${operation.type}`);
    }
  }

  async recordTransaction(id, operations, results, success, duration, error = null) {
    const record = {
      id: id,
      operations: operations,
      results: results,
      success: success,
      duration: duration,
      timestamp: Date.now(),
      error: error ? error.message : null
    };

    const key = `transaction:${id}`;
    await this.state.storage.put(key, record, {
      expirationTtl: 7 * 24 * 60 * 60 // 7天
    });
  }
}
```

```javascript
// 性能监控器
export class PerformanceMonitor {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.metrics = new Map();
    this.alertThresholds = new Map();
  }

  startOperation(name) {
    const operation = {
      name: name,
      startTime: performance.now(),
      startCPUTime: 0,
      endCPUTime: 0
    };

    // 设置性能监控
    const observer = new PerformanceObserver((list) => {
      const entries = list.getEntries();
      for (const entry of entries) {
        if (entry.name === name) {
          operation.endCPUTime = entry.duration;
        }
      }
    });

    observer.observe({ entryTypes: ['measure'] });

    // 标记操作开始
    performance.mark(`${name}-start`);

    return {
      operation: operation,
      observer: observer,
      end: () => this.endOperation(operation)
    };
  }

  async endOperation(operation) {
    const endTime = performance.now();
    const duration = endTime - operation.startTime;

    // 标记操作结束
    performance.mark(`${operation.name}-end`);
    performance.measure(
      operation.name,
      `${operation.name}-start`,
      `${operation.name}-end`
    );

    // 计算指标
    const metrics = {
      duration: duration,
      cpuTime: operation.endCPUTime || 0,
      timestamp: Date.now()
    };

    // 更新统计数据
    await this.updateMetrics(operation.name, metrics);

    // 检查告警阈值
    await this.checkAlerts(operation.name, metrics);

    // 清理
    if (operation.observer) {
      operation.observer.disconnect();
    }

    return metrics;
  }

  async updateMetrics(name, metrics) {
    if (!this.metrics.has(name)) {
      this.metrics.set(name, {
        count: 0,
        totalDuration: 0,
        totalCPUTime: 0,
        minDuration: Infinity,
        maxDuration: 0,
        lastUpdate: Date.now()
      });
    }

    const existing = this.metrics.get(name);
    existing.count++;
    existing.totalDuration += metrics.duration;
    existing.totalCPUTime += metrics.cpuTime;
    existing.minDuration = Math.min(existing.minDuration, metrics.duration);
    existing.maxDuration = Math.max(existing.maxDuration, metrics.duration);
    existing.lastUpdate = Date.now();

    // 计算平均值
    existing.avgDuration = existing.totalDuration / existing.count;
    existing.avgCPUTime = existing.totalCPUTime / existing.count;

    // 定期持久化
    if (existing.count % 100 === 0) {
      await this.persistMetrics(name, existing);
    }
  }

  async persistMetrics(name, metrics) {
    const key = `metrics:${name}:${Date.now()}`;
    await this.state.storage.put(key, metrics, {
      expirationTtl: 30 * 24 * 60 * 60 // 30天
    });
  }

  setAlertThreshold(name, threshold) {
    this.alertThresholds.set(name, threshold);
  }

  async checkAlerts(name, metrics) {
    const threshold = this.alertThresholds.get(name);
    if (!threshold) return;

    // 检查各种阈值
    if (threshold.maxDuration && metrics.duration > threshold.maxDuration) {
      await this.sendAlert({
        type: 'performance',
        severity: 'warning',
        message: `Operation ${name} exceeded max duration`,
        value: metrics.duration,
        threshold: threshold.maxDuration
      });
    }

    if (threshold.maxCPUTime && metrics.cpuTime > threshold.maxCPUTime) {
      await this.sendAlert({
        type: 'performance',
        severity: 'warning',
        message: `Operation ${name} exceeded max CPU time`,
        value: metrics.cpuTime,
        threshold: threshold.maxCPUTime
      });
    }
  }

  async sendAlert(alert) {
    console.warn('ALERT:', alert);

    // 持久化告警
    const alertKey = `alert:${Date.now()}:${Math.random()}`;
    await this.state.storage.put(alertKey, alert, {
      expirationTtl: 7 * 24 * 60 * 60 // 7天
    });
  }

  async getMetricsSummary(name) {
    const metrics = this.metrics.get(name);
    if (!metrics) {
      return null;
    }

    return {
      name: name,
      count: metrics.count,
      avgDuration: metrics.avgDuration,
      minDuration: metrics.minDuration,
      maxDuration: metrics.maxDuration,
      avgCPUTime: metrics.avgCPUTime,
      lastUpdate: metrics.lastUpdate
    };
  }
}
```

```javascript
// 调试助手
export class DebugHelper {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.debugMode = env.DEBUG_MODE === 'true';
    this.traces = [];
    this.maxTraces = 1000;
  }

  trace(operation, data = {}) {
    if (!this.debugMode) return;

    const trace = {
      id: crypto.randomUUID(),
      operation: operation,
      data: data,
      timestamp: Date.now(),
      stackTrace: new Error().stack
    };

    this.traces.push(trace);

    // 限制跟踪数量
    if (this.traces.length > this.maxTraces) {
      this.traces.shift();
    }

    console.trace(`[TRACE] ${operation}:`, data);
  }

  async dumpState(label = null) {
    if (!this.debugMode) return;

    const state = await this.captureState();
    const dump = {
      id: crypto.randomUUID(),
      label: label || `dump-${Date.now()}`,
      state: state,
      timestamp: Date.now()
    };

    // 持久化转储
    const key = `dump:${dump.id}`;
    await this.state.storage.put(key, dump, {
      expirationTtl: 7 * 24 * 60 * 60 // 7天
    });

    console.log(`[DUMP] State dump saved: ${dump.id}`);
    return dump.id;
  }

  async captureState() {
    const allKeys = await this.state.storage.list();
    const state = {};

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value) {
        state[key.name] = value;
      }
    }

    return state;
  }

  // 性能分析
  async analyzePerformance() {
    const traces = this.getTracesByType();
    const analysis = {};

    for (const [type, typeTraces] of traces) {
      const times = typeTraces.map(t => t.timestamp);
      const intervals = [];

      for (let i = 1; i < times.length; i++) {
        intervals.push(times[i] - times[i - 1]);
      }

      analysis[type] = {
        count: typeTraces.length,
        frequency: intervals.length > 0 ? intervals.reduce((a, b) => a + b) / intervals.length : 0,
        firstSeen: times[0],
        lastSeen: times[times.length - 1]
      };
    }

    return analysis;
  }

  getTracesByType() {
    const grouped = new Map();

    for (const trace of this.traces) {
      if (!grouped.has(trace.operation)) {
        grouped.set(trace.operation, []);
      }
      grouped.get(trace.operation).push(trace);
    }

    return grouped;
  }

  // 生成调试报告
  async generateDebugReport() {
    const report = {
      timestamp: Date.now(),
      traces: this.traces,
      performance: await this.analyzePerformance(),
      state: await this.captureState(),
      memory: this.getMemoryUsage(),
      environment: this.getEnvironmentInfo()
    };

    // 持久化报告
    const reportKey = `debug_report:${Date.now()}`;
    await this.state.storage.put(reportKey, report, {
      expirationTtl: 24 * 60 * 60 // 24小时
    });

    return report;
  }

  getMemoryUsage() {
    if (typeof process !== 'undefined' && process.memoryUsage) {
      return process.memoryUsage();
    }
    return null;
  }

  getEnvironmentInfo() {
    return {
      debugMode: this.debugMode,
      traceCount: this.traces.length,
      maxTraces: this.maxTraces
    };
  }

  // 启用/禁用调试模式
  setDebugMode(enabled) {
    this.debugMode = enabled;
    if (!enabled) {
      this.traces = [];
    }
  }
}
```

```javascript
// 动态配置管理
export class ConfigurationManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.config = new Map();
    this.watchers = new Map();
    this.version = 0;
  }

  async initialize() {
    // 加载初始配置
    await this.loadConfiguration();

    // 设置配置更新监听
    this.setupConfigWatcher();
  }

  async loadConfiguration() {
    const configData = await this.state.storage.get('config');
    if (configData) {
      this.config = new Map(Object.entries(configData));
      this.version = configData._version || 0;
    }
  }

  get(key, defaultValue = null) {
    return this.config.get(key) || defaultValue;
  }

  async set(key, value, options = {}) {
    const oldValue = this.config.get(key);
    this.config.set(key, value);
    this.version++;

    // 持久化配置
    await this.persistConfiguration();

    // 通知观察者
    if (oldValue !== value) {
      await this.notifyWatchers(key, value, oldValue);
    }

    // 设置 TTL（如果指定）
    if (options.ttl) {
      setTimeout(() => {
        this.config.delete(key);
        this.notifyWatchers(key, null, value);
      }, options.ttl);
    }
  }

  async watch(key, callback) {
    if (!this.watchers.has(key)) {
      this.watchers.set(key, new Set());
    }

    this.watchers.get(key).add(callback);

    // 返回取消监听的函数
    return () => {
      const callbacks = this.watchers.get(key);
      if (callbacks) {
        callbacks.delete(callback);
        if (callbacks.size === 0) {
          this.watchers.delete(key);
        }
      }
    };
  }

  async notifyWatchers(key, newValue, oldValue) {
    const callbacks = this.watchers.get(key);
    if (!callbacks) return;

    const changeEvent = {
      key: key,
      newValue: newValue,
      oldValue: oldValue,
      version: this.version,
      timestamp: Date.now()
    };

    for (const callback of callbacks) {
      try {
        await callback(changeEvent);
      } catch (error) {
        console.error(`Config watcher error for ${key}:`, error);
      }
    }
  }

  async persistConfiguration() {
    const configData = {
      ...Object.fromEntries(this.config),
      _version: this.version,
      _updatedAt: Date.now()
    };

    await this.state.storage.put('config', configData);
  }

  setupConfigWatcher() {
    // 定期检查外部配置更新
    setInterval(async () => {
      await this.checkForExternalUpdates();
    }, 60000); // 每分钟检查一次
  }

  async checkForExternalUpdates() {
    // 实现外部配置源检查
    // 例如：从 KV、R2 或外部 API 获取配置
  }

  // 配置验证
  validateConfiguration(schema) {
    const errors = [];

    for (const [key, rules] of Object.entries(schema)) {
      const value = this.get(key);
      const error = this.validateValue(key, value, rules);
      if (error) {
        errors.push(error);
      }
    }

    return errors;
  }

  validateValue(key, value, rules) {
    if (rules.required && value === null) {
      return `${key} is required`;
    }

    if (rules.type && typeof value !== rules.type) {
      return `${key} must be of type ${rules.type}`;
    }

    if (rules.validator && !rules.validator(value)) {
      return `${key} failed validation`;
    }

    return null;
  }

  async reloadConfiguration() {
    const oldVersion = this.version;
    await this.loadConfiguration();

    if (this.version !== oldVersion) {
      console.log(`Configuration reloaded: ${oldVersion} -> ${this.version}`);
      await this.notifyWatchers('*', this.config, null);
    }
```
