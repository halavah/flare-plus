# Durable Objects 概述

## 什么是 Durable Objects

Durable Objects 是 Cloudflare 提供的分布式强一致性存储对象，专门用于构建需要实时协作和状态共享的应用。与无状态的 Workers 不同，Durable Objects 提供了有状态的计算能力，每个对象都有唯一的标识符和持久化状态。

## 核心特性

### 1. 强一致性
```javascript
// Durable Objects 提供强一致性保证
export class ChatRoom {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.sessions = new Map(); // 存储所有 WebSocket 连接
  }

  // 所有操作都在同一个对象中串行执行
  async handleMessage(message, sender) {
    // 串行处理确保一致性
    const messageData = {
      id: crypto.randomUUID(),
      text: message.text,
      sender: message.sender,
      timestamp: Date.now()
    };

    // 存储到持久化存储
    await this.state.storage.put(`message:${messageData.id}`, messageData);

    // 广播给所有连接的客户端
    this.broadcast(messageData);

    return messageData;
  }
}
```

```javascript
    // 执行定期维护任务
    await this.performMaintenance();

    // 设置下一个定时器
    this.setAlarm();
  }

  // 定期维护
  async performMaintenance() {
    const now = Date.now();
    const uptime = now - this.startTime;

    // 记录统计信息
    await this.state.storage.put('lastMaintenance', now);
    await this.state.storage.put('uptime', uptime);

    console.log(`Uptime: ${uptime}ms`);
  }

  // 设置定时器（每小时触发）
  setAlarm() {
    const currentHour = Math.floor(Date.now() / (1000 * 60 * 60));
    const nextHour = (currentHour + 1) * (1000 * 60 * 60);

    this.state.storage.setAlarm(nextHour);
  }

  // 关闭前清理
  async shutdown() {
    console.log('Shutting down object...');
    await this.state.storage.put('shutdown', Date.now());
  }
}
```

```javascript
// 分布式锁实现
export class DistributedLock {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.locks = new Map();
  }

  // 获取锁
  async acquireLock(resource, timeout = 30000) {
    const lockKey = `lock:${resource}`;
    const lockId = crypto.randomUUID();
    const expiresAt = Date.now() + timeout;

    // 尝试获取锁
    const existingLock = await this.state.storage.get(lockKey);

    if (!existingLock || existingLock.expiresAt < Date.now()) {
      // 锁可用或已过期
      const newLock = {
        id: lockId,
        expiresAt: expiresAt,
        acquiredAt: Date.now()
      };

      await this.state.storage.put(lockKey, newLock, {
        expirationTtl: Math.ceil(timeout / 1000) + 1
      });

      this.locks.set(resource, lockId);
      return lockId;
    }

    // 锁被占用
    throw new Error(`Lock for ${resource} is currently held`);
  }

  // 释放锁
  async releaseLock(resource, lockId) {
    const lockKey = `lock:${resource}`;
    const lock = await this.state.storage.get(lockKey);

    if (lock && lock.id === lockId) {
      await this.state.storage.delete(lockKey);
      this.locks.delete(resource);
      return true;
    }

    return false;
  }

  // 自动释放过期锁
  async cleanupExpiredLocks() {
    const locks = await this.state.storage.list({ prefix: 'lock:' });
    const now = Date.now();

    for (const [key, lock] of locks.entries()) {
      if (lock.expiresAt < now) {
        await this.state.storage.delete(key);
      }
    }
  }
}

// 事件协调器
export class EventCoordinator {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.eventQueue = [];
    this.subscribers = new Map();
  }

  // 发布事件
  async publish(event, data) {
    const eventData = {
      id: crypto.randomUUID(),
      type: event,
      data: data,
      timestamp: Date.now(),
      processed: false
    };

    // 存储事件
    await this.state.storage.put(`event:${eventData.id}`, eventData);

    // 立即处理事件
    await this.processEvent(eventData);

    return eventData.id;
  }

  // 处理事件
  async processEvent(eventData) {
    // 标记为已处理
    eventData.processed = true;
    await this.state.storage.put(`event:${eventData.id}`, eventData);

    // 通知订阅者
    const subscribers = this.subscribers.get(eventData.type) || [];
    for (const subscriber of subscribers) {
      try {
        await subscriber(eventData);
      } catch (error) {
        console.error('Event processing error:', error);
      }
    }
  }

  // 订阅事件
  subscribe(eventType, callback) {
    if (!this.subscribers.has(eventType)) {
      this.subscribers.set(eventType, new Set());
    }
    this.subscribers.get(eventType).add(callback);
  }

  // 取消订阅
  unsubscribe(eventType, callback) {
    const subscribers = this.subscribers.get(eventType);
    if (subscribers) {
      subscribers.delete(callback);
    }
  }
}
```

```javascript
// 事务操作示例
export class TransactionManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
  }

  // 原子转账操作
  async transfer(fromAccount, toAccount, amount) {
    // 开始事务
    const transaction = await this.state.storage.transaction();

    try {
      // 获取账户余额
      const fromBalance = await transaction.get(`balance:${fromAccount}`) || 0;
      const toBalance = await transaction.get(`balance:${toAccount}`) || 0;

      // 检查余额
      if (fromBalance < amount) {
        throw new Error('Insufficient balance');
      }

      // 更新余额
      await transaction.put(`balance:${fromAccount}`, fromBalance - amount);
      await transaction.put(`balance:${toAccount}`, toBalance + amount);

      // 记录交易
      const transactionId = crypto.randomUUID();
      await transaction.put(`transaction:${transactionId}`, {
        id: transactionId,
        from: fromAccount,
        to: toAccount,
        amount: amount,
        timestamp: Date.now()
      });

      // 提交事务
      await transaction.commit();

      return { transactionId, success: true };

    } catch (error) {
      // 回滚事务
      await transaction.rollback();
      throw error;
    }
  }

  // 批量更新
  async batchUpdate(updates) {
    const transaction = await this.state.storage.transaction();

    try {
      for (const update of updates) {
        await transaction.put(update.key, update.value);
      }

      await transaction.commit();
      return { success: true, updated: updates.length };

    } catch (error) {
      await transaction.rollback();
      throw error;
    }
  }
}
```

```javascript
// 状态同步管理器
export class StateSyncManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.lastSyncTime = 0;
    this.syncInterval = 5000; // 5秒
  }

  // 初始化状态同步
  async initializeSync() {
    const lastSync = await this.state.storage.get('lastSyncTime');
    this.lastSyncTime = lastSync || Date.now();

    // 启动定期同步
    this.startPeriodicSync();
  }

  // 获取状态变更
  async getStateChanges(since) {
    const changes = {
      adds: new Map(),
      updates: new Map(),
      deletes: new Set()
    };

    const allEntries = await this.state.storage.list();
    for (const [key, value] of allEntries.entries()) {
      if (value.timestamp > since) {
        if (value.deleted) {
          changes.deletes.add(key);
        } else if (value.created > since) {
          changes.adds.set(key, value);
        } else {
          changes.updates.set(key, value);
        }
      }
    }

    return changes;
  }

  // 同步状态到客户端
  async syncToClient(client) {
    const changes = await this.getStateChanges(this.lastSyncTime);

    // 发送增量更新
    for (const [key, value] of changes.adds) {
      client.send({ type: 'add', key, value });
    }

    for (const [key, value] of changes.updates) {
      client.send({ type: 'update', key, value });
    }

    for (const key of changes.deletes) {
      client.send({ type: 'delete', key });
    }

    // 更新同步时间
    this.lastSyncTime = Date.now();
    await this.state.storage.put('lastSyncTime', this.lastSyncTime);
  }

  // 启动定期同步
  startPeriodicSync() {
    this.state.storage.setAlarm(Date.now() + this.syncInterval);
  }

  // 定时同步回调
  async alarm() {
    // 执行同步操作
    await this.performFullSync();

    // 设置下一次同步
    this.startPeriodicSync();
  }

  // 执行完整同步
  async performFullSync() {
    console.log('Performing full state sync...');

    // 这里可以添加具体的同步逻辑
    // 例如：备份到外部存储、清理过期数据等

    const syncTime = Date.now();
    await this.state.storage.put('lastFullSync', syncTime);
  }
}
```

### 1. WebSocket 管理
```javascript
// WebSocket 连接管理器
export class WebSocketManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.connections = new Map();
    this.rooms = new Map();
  }

  // 处理 WebSocket 连接
  async handleWebSocket(ws, request) {
    const clientId = this.generateClientId();
    const clientInfo = {
      id: clientId,
      ws: ws,
      joined: Date.now(),
      lastPing: Date.now(),
      metadata: this.parseClientMetadata(request)
    };

    this.connections.set(clientId, clientInfo);

    // 设置 WebSocket 事件处理
    ws.accept();
    this.setupWebSocketHandlers(clientInfo);

    // 发送欢迎消息
    ws.send(JSON.stringify({
      type: 'welcome',
      clientId: clientId,
      timestamp: Date.now()
    }));

    // 通知其他客户端
    this.broadcast({
      type: 'user_joined',
      clientId: clientId,
      userCount: this.connections.size
    }, clientId);

    return clientId;
  }

  // 设置 WebSocket 事件处理
  setupWebSocketHandlers(clientInfo) {
    const { ws, id: clientId } = clientInfo;

    ws.addEventListener('message', async (event) => {
      try {
        const message = JSON.parse(event.data);
        await this.handleMessage(clientInfo, message);
      } catch (error) {
        console.error('Message handling error:', error);
        ws.send(JSON.stringify({
          type: 'error',
          message: 'Invalid message format'
        }));
      }
    });

    ws.addEventListener('close', async () => {
      await this.handleDisconnect(clientInfo);
    });

    ws.addEventListener('error', (error) => {
      console.error('WebSocket error:', error);
    });

    // 定期心跳
    this.startHeartbeat(clientInfo);
  }

  // 处理客户端消息
  async handleMessage(clientInfo, message) {
    const { type, data } = message;

    switch (type) {
      case 'ping':
        clientInfo.ws.send(JSON.stringify({ type: 'pong' }));
        clientInfo.lastPing = Date.now();
        break;

      case 'join_room':
        await this.joinRoom(clientInfo.id, data.roomId);
        break;

      case 'leave_room':
        await this.leaveRoom(clientInfo.id, data.roomId);
        break;

      case 'chat_message':
        await this.handleChatMessage(clientInfo, data);
        break;

      default:
        console.warn('Unknown message type:', type);
    }
  }

  // 加入房间
  async joinRoom(clientId, roomId) {
    if (!this.rooms.has(roomId)) {
      this.rooms.set(roomId, new Set());
    }

    this.rooms.get(roomId).add(clientId);

    // 通知房间内其他用户
    const roomClients = this.rooms.get(roomId);
    for (const id of roomClients) {
      if (id !== clientId) {
        const client = this.connections.get(id);
        if (client && client.ws.readyState === WebSocket.OPEN) {
          client.ws.send(JSON.stringify({
            type: 'user_joined_room',
            roomId: roomId,
            clientId: clientId
          }));
        }
      }
    }
  }

  // 处理聊天消息
  async handleChatMessage(clientInfo, data) {
    const message = {
      id: crypto.randomUUID(),
      senderId: clientInfo.id,
      senderName: clientInfo.metadata.name || 'Anonymous',
      text: data.text,
      timestamp: Date.now(),
      roomId: data.roomId
    };

    // 存储消息
    await this.state.storage.put(`message:${message.id}`, message);

    // 广播消息
    if (data.roomId) {
      this.broadcastToRoom(data.roomId, {
        type: 'chat_message',
        message: message
      });
    } else {
      this.broadcast({
        type: 'chat_message',
        message: message
      });
    }
  }

  // 广播消息到所有客户端
  broadcast(message, excludeClientId = null) {
    const messageString = JSON.stringify(message);

    for (const [clientId, client] of this.connections) {
      if (clientId !== excludeClientId &&
          client.ws.readyState === WebSocket.OPEN) {
        try {
          client.ws.send(messageString);
        } catch (error) {
          console.error('Broadcast error:', error);
        }
      }
    }
  }

  // 广播消息到房间
  broadcastToRoom(roomId, message) {
    const roomClients = this.rooms.get(roomId);
    if (!roomClients) return;

    const messageString = JSON.stringify(message);

    for (const clientId of roomClients) {
      const client = this.connections.get(clientId);
      if (client && client.ws.readyState === WebSocket.OPEN) {
        try {
          client.ws.send(messageString);
        } catch (error) {
          console.error('Room broadcast error:', error);
        }
      }
    }
  }

  // 处理客户端断开
  async handleDisconnect(clientInfo) {
    const { id: clientId } = clientInfo;

    // 从所有房间移除
    for (const [roomId, roomClients] of this.rooms) {
      roomClients.delete(clientId);

      // 通知房间内其他用户
      for (const id of roomClients) {
        const client = this.connections.get(id);
        if (client && client.ws.readyState === WebSocket.OPEN) {
          client.ws.send(JSON.stringify({
            type: 'user_left_room',
            roomId: roomId,
            clientId: clientId
          }));
        }
      }
    }

    // 从连接列表移除
    this.connections.delete(clientId);

    // 通知其他客户端
    this.broadcast({
      type: 'user_left',
      clientId: clientId,
      userCount: this.connections.size
    });
  }

  // 生成客户端 ID
  generateClientId() {
    return crypto.randomUUID();
  }

  // 解析客户端元数据
  parseClientMetadata(request) {
    const url = new URL(request.url);
    return {
      userAgent: request.headers.get('User-Agent'),
      ip: request.headers.get('CF-Connecting-IP'),
      country: request.cf?.country,
      name: url.searchParams.get('name')
    };
  }

  // 启动心跳检测
  startHeartbeat(clientInfo) {
    const heartbeatInterval = setInterval(() => {
      if (clientInfo.ws.readyState === WebSocket.OPEN) {
        if (Date.now() - clientInfo.lastPing > 30000) {
          // 超时，关闭连接
          clientInfo.ws.close(1000, 'Heartbeat timeout');
          clearInterval(heartbeatInterval);
        } else {
          // 发送心跳
          clientInfo.ws.send(JSON.stringify({ type: 'heartbeat' }));
        }
      } else {
        clearInterval(heartbeatInterval);
      }
    }, 15000); // 15秒心跳间隔
  }
}
```

### 2. 事件广播
```javascript
// 事件广播系统
export class EventBroadcaster {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.subscribers = new Map();
    this.eventHistory = [];
    this.maxHistorySize = 1000;
  }

  // 发布事件
  async publish(event, data, options = {}) {
    const {
      persist = true,
      broadcast = true,
      target = 'all'
    } = options;

    const eventData = {
      id: crypto.randomUUID(),
      type: event,
      data: data,
      timestamp: Date.now(),
      target: target
    };

    // 持久化事件
    if (persist) {
      await this.state.storage.put(`event:${eventData.id}`, eventData);
    }

    // 添加到历史记录
    this.eventHistory.push(eventData);
    if (this.eventHistory.length > this.maxHistorySize) {
      this.eventHistory.shift();
    }

    // 广播事件
    if (broadcast) {
      await this.broadcastEvent(eventData);
    }

    return eventData.id;
  }

  // 广播事件
  async broadcastEvent(eventData) {
    const message = JSON.stringify({
      type: 'event',
      event: eventData
    });

    // 广播给所有订阅者
    for (const [clientId, subscriber] of this.subscribers) {
      if (this.shouldSendEvent(subscriber, eventData)) {
        try {
          subscriber.ws.send(message);
        } catch (error) {
          console.error('Broadcast failed:', error);
          this.removeSubscriber(clientId);
        }
      }
    }
  }

  // 添加订阅者
  addSubscriber(clientId, ws, filters = {}) {
    this.subscribers.set(clientId, {
      id: clientId,
      ws: ws,
      filters: filters,
      subscribedAt: Date.now()
    });
  }

  // 移除订阅者
  removeSubscriber(clientId) {
    this.subscribers.delete(clientId);
  }

  // 判断是否应该发送事件
  shouldSendEvent(subscriber, event) {
    const { filters } = subscriber;

    // 检查事件类型过滤
    if (filters.eventTypes && !filters.eventTypes.includes(event.type)) {
      return false;
    }

    // 检查目标过滤
    if (event.target !== 'all' && filters.targets && !filters.targets.includes(event.target)) {
      return false;
    }

    // 检查时间范围过滤
    if (filters.since && event.timestamp < filters.since) {
      return false;
    }

    return true;
  }

  // 获取事件历史
  async getEventHistory(filters = {}) {
    const {
      limit = 100,
      eventTypes,
      since,
      until
    } = filters;

    let filtered = [...this.eventHistory];

    // 应用过滤条件
    if (eventTypes) {
      filtered = filtered.filter(event => eventTypes.includes(event.type));
    }

    if (since) {
      filtered = filtered.filter(event => event.timestamp >= since);
    }

    if (until) {
      filtered = filtered.filter(event => event.timestamp <= until);
    }

    // 按时间倒序排列
    filtered.sort((a, b) => b.timestamp - a.timestamp);

    // 限制数量
    return filtered.slice(0, limit);
  }
}
```
