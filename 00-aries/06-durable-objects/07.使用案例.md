# Durable Objects 使用案例

## 概述

Durable Objects 在各种实际场景中都有出色的表现，特别是在需要状态管理、实时协作和高并发处理的应用中。本章将介绍多个具体的使用案例，展示如何利用 Durable Objects 解决实际问题。

## 实时聊天应用

### 1. 多人聊天室
```javascript
// 聊天室 Durable Object
export class ChatRoom {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.participants = new Map();
    this.messages = [];
    this.metadata = {
      name: '',
      created: Date.now(),
      lastActivity: Date.now()
    };
  }

  // 初始化聊天室
  async initialize(roomId, options = {}) {
    this.roomId = roomId;
    this.metadata.name = options.name || roomId;
    this.metadata.maxParticipants = options.maxParticipants || 100;
    this.metadata.isPrivate = options.private || false;
    this.metadata.password = options.password || null;

    // 加载历史消息
    await this.loadMessages();
  }

  // 处理 WebSocket 连接
  async handleWebSocket(ws, request, userInfo) {
    const participantId = userInfo.id || crypto.randomUUID();
    const participant = {
      id: participantId,
      name: userInfo.name || 'Anonymous',
      avatar: userInfo.avatar || null,
      role: userInfo.role || 'member',
      joinedAt: Date.now(),
      lastPing: Date.now(),
      ws: ws,
      typing: false,
      lastTypingTime: null
    };

    // 检查聊天室容量
    if (this.participants.size >= this.metadata.maxParticipants) {
      ws.close(1013, 'Chat room is full');
      return;
    }

    // 添加参与者
    this.participants.set(participantId, participant);

    // 接受 WebSocket 连接
    ws.accept();

    // 设置事件处理
    this.setupWebSocketHandlers(participant);

    // 发送欢迎消息
    this.sendToParticipant(participantId, {
      type: 'welcome',
      roomInfo: this.metadata,
      participant: this.sanitizeParticipant(participant),
      recentMessages: this.messages.slice(-50),
      participantCount: this.participants.size
    });

    // 广播用户加入
    await this.broadcast({
      type: 'user_joined',
      participant: this.sanitizeParticipant(participant),
      participantCount: this.participants.size
    }, participantId);

    // 启动心跳
    this.startHeartbeat(participantId);

    return participantId;
  }

  // 设置 WebSocket 事件处理
  setupWebSocketHandlers(participant) {
    const ws = participant.ws;
    const participantId = participant.id;

    ws.addEventListener('message', async (event) => {
      try {
        const message = JSON.parse(event.data);
        await this.handleMessage(participantId, message);
      } catch (error) {
        console.error('Message handling error:', error);
        this.sendToParticipant(participantId, {
          type: 'error',
          message: 'Invalid message format'
        });
      }
    });

    ws.addEventListener('close', async () => {
      await this.handleDisconnect(participantId);
    });

    ws.addEventListener('error', (error) => {
      console.error('WebSocket error:', error);
    });
  }

  // 处理消息
  async handleMessage(participantId, message) {
    const participant = this.participants.get(participantId);
    if (!participant) return;

    switch (message.type) {
      case 'chat_message':
        await this.handleChatMessage(participantId, message);
        break;

      case 'typing':
        await this.handleTyping(participantId, message);
        break;

      case 'private_message':
        await this.handlePrivateMessage(participantId, message);
        break;

      case 'user_info_update':
        await this.handleUserInfoUpdate(participantId, message);
        break;

      case 'reaction':
        await this.handleReaction(participantId, message);
        break;
    }
  }

  // 处理聊天消息
  async handleChatMessage(participantId, message) {
    const participant = this.participants.get(participantId);
    if (!participant) return;

    const chatMessage = {
      id: crypto.randomUUID(),
      type: message.messageType || 'text',
      content: message.content,
      sender: {
        id: participantId,
        name: participant.name,
        avatar: participant.avatar
      },
      timestamp: Date.now(),
      replyTo: message.replyTo || null,
      attachments: message.attachments || [],
      reactions: {},
      edited: false,
      deleted: false
    };

    // 添加到消息列表
    this.messages.push(chatMessage);

    // 限制消息历史
    if (this.messages.length > 1000) {
      this.messages = this.messages.slice(-500);
    }

    // 持久化消息
    await this.persistMessage(chatMessage);

    // 广播消息
    await this.broadcast({
      type: 'new_message',
      message: chatMessage
    });

    // 更新最后活动时间
    this.metadata.lastActivity = Date.now();
    await this.updateMetadata();
  }

  // 处理输入状态
  async handleTyping(participantId, message) {
    const participant = this.participants.get(participantId);
    if (!participant) return;

    participant.typing = message.isTyping;
    participant.lastTypingTime = Date.now();

    // 广播输入状态（不发送给发送者）
    await this.broadcast({
      type: 'typing_status',
      participantId: participantId,
      isTyping: message.isTyping
    }, participantId);

    // 自动停止输入状态
    if (message.isTyping) {
      setTimeout(() => {
        const p = this.participants.get(participantId);
        if (p && p.typing && Date.now() - p.lastTypingTime > 3000) {
          p.typing = false;
          this.broadcast({
            type: 'typing_status',
            participantId: participantId,
            isTyping: false
          });
        }
      }, 5000);
    }
  }

  // 处理私聊消息
  async handlePrivateMessage(participantId, message) {
    const targetParticipant = this.participants.get(message.targetUserId);
    if (!targetParticipant) {
      this.sendToParticipant(participantId, {
        type: 'error',
        message: 'User not found'
      });
      return;
    }

    const sender = this.participants.get(participantId);
    const privateMessage = {
      id: crypto.randomUUID(),
      content: message.content,
      sender: {
        id: participantId,
        name: sender.name
      },
      receiver: {
        id: message.targetUserId,
        name: targetParticipant.name
      },
      timestamp: Date.now()
    };

    // 发送给接收者
    this.sendToParticipant(message.targetUserId, {
      type: 'private_message',
      message: privateMessage
    });

    // 发送回执给发送者
    this.sendToParticipant(participantId, {
      type: 'private_message_sent',
      messageId: privateMessage.id
    });
  }

  // 处理用户信息更新
  async handleUserInfoUpdate(participantId, message) {
    const participant = this.participants.get(participantId);
    if (!participant) return;

    // 更新信息
    if (message.name) participant.name = message.name;
    if (message.avatar) participant.avatar = message.avatar;
    if (message.status) participant.status = message.status;

    // 广播更新
    await this.broadcast({
      type: 'user_info_updated',
      participantId: participantId,
      updates: message
    });
  }

  // 处理表情回应
  async handleReaction(participantId, message) {
    const messageId = message.messageId;
    const reaction = message.reaction;

    // 找到消息
    const messageIndex = this.messages.findIndex(m => m.id === messageId);
    if (messageIndex === -1) return;

    const chatMessage = this.messages[messageIndex];

    // 添加或移除反应
    if (!chatMessage.reactions[reaction]) {
      chatMessage.reactions[reaction] = new Set();
    }

    if (chatMessage.reactions[reaction].has(participantId)) {
      chatMessage.reactions[reaction].delete(participantId);
    } else {
      chatMessage.reactions[reaction].add(participantId);
    }

    // 转换为数组以便序列化
    const reactionsArray = {};
    for (const [emoji, users] of Object.entries(chatMessage.reactions)) {
      reactionsArray[emoji] = Array.from(users);
    }

    // 广播反应更新
    await this.broadcast({
      type: 'reaction_updated',
      messageId: messageId,
      reactions: reactionsArray
    });
  }

  // 处理断开连接
  async handleDisconnect(participantId) {
    const participant = this.participants.get(participantId);
    if (!participant) return;

    // 移除参与者
    this.participants.delete(participantId);

    // 广播用户离开
    await this.broadcast({
      type: 'user_left',
      participantId: participantId,
      participantCount: this.participants.size
    });

    // 如果聊天室为空，设置定时清理
    if (this.participants.size === 0) {
      this.scheduleCleanup();
    }
  }

  // 广播消息
  async broadcast(message, excludeParticipantId = null) {
    const messageString = JSON.stringify(message);

    for (const [id, participant] of this.participants) {
      if (id !== excludeParticipantId && participant.ws.readyState === WebSocket.OPEN) {
        try {
          participant.ws.send(messageString);
        } catch (error) {
          console.error(`Broadcast failed for ${id}:`, error);
        }
      }
    }
  }

  // 发送消息给特定参与者
  sendToParticipant(participantId, message) {
    const participant = this.participants.get(participantId);
    if (!participant || participant.ws.readyState !== WebSocket.OPEN) {
      return false;
    }

    try {
      participant.ws.send(JSON.stringify(message));
      return true;
    } catch (error) {
      console.error(`Send failed for ${participantId}:`, error);
      return false;
    }
  }

  // 启动心跳
  startHeartbeat(participantId) {
    const participant = this.participants.get(participantId);
    if (!participant) return;

    const interval = setInterval(() => {
      const p = this.participants.get(participantId);
      if (!p || p.ws.readyState !== WebSocket.OPEN) {
        clearInterval(interval);
        this.handleDisconnect(participantId);
        return;
      }

      if (Date.now() - p.lastPing > 30000) {
        p.ws.close(1000, 'No response to ping');
        clearInterval(interval);
      } else {
        p.ws.send(JSON.stringify({ type: 'ping' }));
      }
    }, 15000);
  }

  // 持久化消息
  async persistMessage(message) {
    const key = `message:${message.id}`;
    await this.state.storage.put(key, message, {
      expirationTtl: 7 * 24 * 60 * 60 // 7天
    });

    // 更新消息索引
    const messageIds = await this.state.storage.get('message_ids') || [];
    messageIds.push(message.id);

    // 限制索引大小
    if (messageIds.length > 1000) {
      const toRemove = messageIds.splice(0, 500);
      for (const id of toRemove) {
        await this.state.storage.delete(`message:${id}`);
      }
    }

    await this.state.storage.put('message_ids', messageIds);
  }

  // 加载历史消息
  async loadMessages() {
    const messageIds = await this.state.storage.get('message_ids') || [];
    this.messages = [];

    for (const id of messageIds) {
      const message = await this.state.storage.get(`message:${id}`);
      if (message) {
        this.messages.push(message);
      }
    }

    // 按时间排序
    this.messages.sort((a, b) => a.timestamp - b.timestamp);
  }

  // 更新元数据
  async updateMetadata() {
    await this.state.storage.put('metadata', this.metadata);
  }

  // 定时清理
  scheduleCleanup() {
    // 1小时后清理空聊天室
    this.state.storage.setAlarm(Date.now() + 60 * 60 * 1000);
  }

  // 定时器回调
  async alarm() {
    if (this.participants.size === 0) {
      // 清理聊天室数据
      await this.cleanup();
    }
  }

  // 清理聊天室
  async cleanup() {
    console.log(`Cleaning up chat room ${this.roomId}`);

    // 保留重要数据，清理临时数据
    await this.state.storage.delete('participants');
    // 保留消息历史以供查询
  }

  // 清理参与者信息
  sanitizeParticipant(participant) {
    const { ws, ...sanitized } = participant;
    return sanitized;
  }
}

// 聊天应用入口
export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname;

    if (path.startsWith('/chat/')) {
      const roomId = path.split('/')[2];
      const id = env.CHAT_ROOM.idFromName(roomId);
      const stub = env.CHAT_ROOM.get(id);

      return stub.fetch(request);
    }

    return new Response('Not Found', { status: 404 });
  }
};
```

## 在线游戏

### 1. 多人游戏房间
```javascript
// 游戏房间 Durable Object
export class GameRoom {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.players = new Map();
    this.gameState = 'waiting'; // waiting, playing, finished
    this.gameData = {};
    this.tickRate = 60; // 60 FPS
  }

  // 初始化游戏
  async initialize(gameId, gameType, options = {}) {
    this.gameId = gameId;
    this.gameType = gameType;
    this.options = {
      maxPlayers: options.maxPlayers || 4,
      minPlayers: options.minPlayers || 2,
      autoStart: options.autoStart || false,
      ...options
    };

    // 加载游戏状态
    await this.loadGameState();
  }

  // 处理玩家加入
  async addPlayer(playerInfo) {
    if (this.players.size >= this.options.maxPlayers) {
      throw new Error('Game room is full');
    }

    const playerId = playerInfo.id || crypto.randomUUID();
    const player = {
      id: playerId,
      name: playerInfo.name || `Player ${this.players.size + 1}`,
      score: 0,
      position: { x: 0, y: 0 },
      velocity: { x: 0, y: 0 },
      status: 'ready',
      joinedAt: Date.now(),
      lastUpdate: Date.now(),
      ws: null
    };

    this.players.set(playerId, player);

    // 发送游戏状态给新玩家
    if (playerInfo.ws) {
      player.ws = playerInfo.ws;
      this.sendToPlayer(playerId, {
        type: 'game_state',
        gameState: this.gameState,
        players: this.getAllPlayerData(),
        gameData: this.gameData
      });
    }

    // 广播玩家加入
    await this.broadcast({
      type: 'player_joined',
      player: this.sanitizePlayer(player),
      playerCount: this.players.size
    });

    // 自动开始游戏
    if (this.options.autoStart && this.players.size >= this.options.minPlayers) {
      await this.startGame();
    }

    return playerId;
  }

  // 处理玩家操作
  async handlePlayerAction(playerId, action) {
    const player = this.players.get(playerId);
    if (!player || player.status !== 'active') {
      return;
    }

    switch (action.type) {
      case 'move':
        await this.handlePlayerMove(playerId, action);
        break;

      case 'attack':
        await this.handlePlayerAttack(playerId, action);
        break;

      case 'interact':
        await this.handlePlayerInteract(playerId, action);
        break;

      case 'chat':
        await this.handlePlayerChat(playerId, action);
        break;
    }

    player.lastUpdate = Date.now();
  }

  // 开始游戏
  async startGame() {
    if (this.gameState !== 'waiting') {
      return;
    }

    if (this.players.size < this.options.minPlayers) {
      return;
    }

    this.gameState = 'playing';
    this.gameData = this.initializeGameData();

    // 激活所有玩家
    for (const player of this.players.values()) {
      player.status = 'active';
      player.score = 0;
      this.resetPlayerPosition(player);
    }

    // 广播游戏开始
    await this.broadcast({
      type: 'game_started',
      gameData: this.gameData,
      players: this.getAllPlayerData()
    });

    // 启动游戏循环
    this.startGameLoop();
  }

  // 游戏循环
  startGameLoop() {
    const gameLoop = async () => {
      if (this.gameState !== 'playing') {
        return;
      }

      // 更新游戏逻辑
      await this.updateGame();

      // 发送游戏状态更新
      await this.broadcast({
        type: 'game_update',
        gameData: this.gameData,
        players: this.getAllPlayerData(),
        timestamp: Date.now()
      });

      // 检查游戏结束
      if (this.checkGameEnd()) {
        await this.endGame();
      } else {
        // 继续循环
        setTimeout(gameLoop, 1000 / this.tickRate);
      }
    };

    gameLoop();
  }

  // 更新游戏逻辑
  async updateGame() {
    // 更新所有玩家
    for (const player of this.players.values()) {
      if (player.status === 'active') {
        this.updatePlayer(player);
      }
    }

    // 检查碰撞
    await this.checkCollisions();

    // 更新游戏对象
    await this.updateGameObjects();

    // 检查特殊规则
    await this.checkGameRules();
  }

  // 更新玩家
  updatePlayer(player) {
    // 更新位置
    player.position.x += player.velocity.x;
    player.position.y += player.velocity.y;

    // 应用摩擦力
    player.velocity.x *= 0.95;
    player.velocity.y *= 0.95;

    // 边界检查
    const bounds = this.gameData.bounds;
    player.position.x = Math.max(bounds.minX, Math.min(bounds.maxX, player.position.x));
    player.position.y = Math.max(bounds.minY, Math.min(bounds.maxY, player.position.y));
  }

  // 检查碰撞
  async checkCollisions() {
    const players = Array.from(this.players.values()).filter(p => p.status === 'active');

    for (let i = 0; i < players.length; i++) {
      for (let j = i + 1; j < players.length; j++) {
        if (this.checkPlayerCollision(players[i], players[j])) {
          await this.handlePlayerCollision(players[i], players[j]);
        }
      }
    }
  }

  // 检查玩家碰撞
  checkPlayerCollision(player1, player2) {
    const distance = Math.sqrt(
      Math.pow(player1.position.x - player2.position.x, 2) +
      Math.pow(player1.position.y - player2.position.y, 2)
    );

    return distance < (player1.radius + player2.radius);
  }

  // 处理玩家碰撞
  async handlePlayerCollision(player1, player2) {
    // 弹性碰撞
    const dx = player2.position.x - player1.position.x;
    const dy = player2.position.y - player1.position.y;
    const distance = Math.sqrt(dx * dx + dy * dy);

    // 避免除零
    if (distance === 0) return;

    // 单位方向向量
    const nx = dx / distance;
    const ny = dy / distance;

    // 相对速度
    const dvx = player2.velocity.x - player1.velocity.x;
    const dvy = player2.velocity.y - player1.velocity.y;

    // 相对速度在碰撞方向上的分量
    const dvn = dvx * nx + dvy * ny;

    // 如果物体正在分离，不需要处理
    if (dvn > 0) return;

    // 弹性碰撞
    const impulse = 2 * dvn / 2; // 假设质量相等

    player1.velocity.x += impulse * nx;
    player1.velocity.y += impulse * ny;
    player2.velocity.x -= impulse * nx;
    player2.velocity.y -= impulse * ny;

    // 分离物体避免重叠
    const overlap = (player1.radius + player2.radius) - distance;
    const separationX = nx * overlap / 2;
    const separationY = ny * overlap / 2;

    player1.position.x -= separationX;
    player1.position.y -= separationY;
    player2.position.x += separationX;
    player2.position.y += separationY;

    // 广播碰撞事件
    await this.broadcast({
      type: 'collision',
      players: [player1.id, player2.id],
      positions: [player1.position, player2.position]
    });
  }

  // 结束游戏
  async endGame() {
    this.gameState = 'finished';

    // 计算最终得分
    const rankings = this.calculateRankings();

    // 广播游戏结束
    await this.broadcast({
      type: 'game_ended',
      rankings: rankings,
      finalScores: this.getAllPlayerScores()
    });

    // 设置重置定时器
    setTimeout(() => {
      this.resetGame();
    }, 30000); // 30秒后重置
  }

  // 计算排名
  calculateRankings() {
    const players = Array.from(this.players.values());
    return players
      .sort((a, b) => b.score - a.score)
      .map((player, index) => ({
        playerId: player.id,
        name: player.name,
        score: player.score,
        rank: index + 1
      }));
  }

  // 重置游戏
  resetGame() {
    this.gameState = 'waiting';
    this.gameData = {};

    for (const player of this.players.values()) {
      player.status = 'ready';
      player.score = 0;
      this.resetPlayerPosition(player);
    }
  }

  // 初始化游戏数据
  initializeGameData() {
    return {
      bounds: {
        minX: -400,
        maxX: 400,
        minY: -300,
        maxY: 300
      },
      objects: [],
      powerUps: [],
      startTime: Date.now(),
      duration: 0
    };
  }

  // 重置玩家位置
  resetPlayerPosition(player) {
    // 在游戏区域内随机位置
    player.position = {
      x: (Math.random() - 0.5) * 600,
      y: (Math.random() - 0.5) * 400
    };
    player.velocity = { x: 0, y: 0 };
    player.radius = 20;
  }

  // 获取所有玩家数据
  getAllPlayerData() {
    const playerData = {};
    for (const [id, player] of this.players) {
      playerData[id] = this.sanitizePlayer(player);
    }
    return playerData;
  }

  // 获取所有玩家得分
  getAllPlayerScores() {
    const scores = {};
    for (const [id, player] of this.players) {
      scores[id] = player.score;
    }
    return scores;
  }

  // 发送消息给玩家
  sendToPlayer(playerId, message) {
    const player = this.players.get(playerId);
    if (!player || !player.ws || player.ws.readyState !== WebSocket.OPEN) {
      return false;
    }

    try {
      player.ws.send(JSON.stringify(message));
      return true;
    } catch (error) {
      console.error(`Send failed for ${playerId}:`, error);
      return false;
    }
  }

  // 广播消息
  async broadcast(message, excludePlayerId = null) {
    const messageString = JSON.stringify(message);

    for (const [id, player] of this.players) {
      if (id !== excludePlayerId && player.ws && player.ws.readyState === WebSocket.OPEN) {
        try {
          player.ws.send(messageString);
        } catch (error) {
          console.error(`Broadcast failed for ${id}:`, error);
        }
      }
    }
  }

  // 清理玩家信息
  sanitizePlayer(player) {
    const { ws, ...sanitized } = player;
    return sanitized;
  }

  // 加载游戏状态
  async loadGameState() {
```

```javascript
// 白板 Durable Object
export class Whiteboard {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.elements = new Map();
    this.version = 0;
    this.collaborators = new Map();
  }

  // 初始化白板
  async initialize(whiteboardId, options = {}) {
    this.whiteboardId = whiteboardId;
    this.options = {
      width: options.width || 1920,
      height: options.height || 1080,
      backgroundColor: options.backgroundColor || '#ffffff',
      ...options
    };

    await this.loadWhiteboard();
  }

  // 添加协作者
  async addCollaborator(collaboratorInfo) {
    const collaboratorId = collaboratorInfo.id || crypto.randomUUID();
    const collaborator = {
      id: collaboratorId,
      name: collaboratorInfo.name || 'Anonymous',
      color: collaboratorInfo.color || this.generateColor(),
      cursor: { x: 0, y: 0 },
      selection: null,
      tool: collaboratorInfo.tool || 'pen',
      joinedAt: Date.now(),
      lastActivity: Date.now(),
      ws: null
    };

    this.collaborators.set(collaboratorId, collaborator);

    // 发送白板状态
    if (collaboratorInfo.ws) {
      collaborator.ws = collaboratorInfo.ws;
      this.sendToCollaborator(collaboratorId, {
        type: 'whiteboard_state',
        elements: Array.from(this.elements.values()),
        version: this.version,
        options: this.options,
        collaborators: this.getAllCollaboratorData()
      });
    }

    // 广播新协作者加入
    await this.broadcast({
      type: 'collaborator_joined',
      collaborator: this.sanitizeCollaborator(collaborator)
    }, collaboratorId);

    return collaboratorId;
  }

  // 绘制元素
  async drawElement(elementData, authorId) {
    const element = {
      id: crypto.randomUUID(),
      type: elementData.type,
      points: elementData.points || [],
      style: {
        color: elementData.color || '#000000',
        width: elementData.width || 2,
        fill: elementData.fill || null,
        ...elementData.style
      },
      position: elementData.position || { x: 0, y: 0 },
      size: elementData.size || { width: 0, height: 0 },
      transform: elementData.transform || null,
      authorId: authorId,
      createdAt: Date.now(),
      lastModified: Date.now(),
      version: 1
    };

    this.elements.set(element.id, element);
    this.version++;

    await this.persistElement(element);
    await this.broadcast({
      type: 'element_added',
      element: element,
      version: this.version
    });

    return element.id;
  }

  // 更新元素
  async updateElement(elementId, updates, authorId) {
    const element = this.elements.get(elementId);
    if (!element) {
      throw new Error(`Element ${elementId} not found`);
    }

    // 应用更新
    Object.assign(element, updates);
    element.lastModified = Date.now();
    element.version++;
    this.version++;

    await this.persistElement(element);
    await this.broadcast({
      type: 'element_updated',
      element: element,
      version: this.version
    });

    return element;
  }

  // 删除元素
  async deleteElement(elementId, authorId) {
    const element = this.elements.get(elementId);
    if (!element) {
      return false;
    }

    this.elements.delete(elementId);
    this.version++;

    await this.state.storage.delete(`element:${elementId}`);
    await this.broadcast({
      type: 'element_deleted',
      elementId: elementId,
      version: this.version
    });

    return true;
  }

  // 更新光标
  async updateCursor(collaboratorId, cursor) {
    const collaborator = this.collaborators.get(collaboratorId);
    if (!collaborator) return;

    collaborator.cursor = cursor;
    collaborator.lastActivity = Date.now();

    await this.broadcast({
      type: 'cursor_updated',
      collaboratorId: collaboratorId,
      cursor: cursor
    }, collaboratorId);
  }

  // 更新选择
  async updateSelection(collaboratorId, selection) {
    const collaborator = this.collaborators.get(collaboratorId);
    if (!collaborator) return;

    collaborator.selection = selection;
    collaborator.lastActivity = Date.now();

    await this.broadcast({
      type: 'selection_updated',
      collaboratorId: collaboratorId,
      selection: selection
    }, collaboratorId);
  }

  // 持久化元素
  async persistElement(element) {
    await this.state.storage.put(`element:${element.id}`, element);
  }

  // 加载白板
  async loadWhiteboard() {
    const options = await this.state.storage.get('whiteboard_options');
    if (options) {
      this.options = options;
    }

    const elementKeys = await this.state.storage.list({
      prefix: 'element:'
    });

    this.elements.clear();
    for (const key of elementKeys.keys) {
      const element = await this.state.storage.get(key.name);
      if (element) {
        this.elements.set(element.id, element);
      }
    }

    // 获取版本
    this.version = await this.state.storage.get('whiteboard_version') || 0;
  }

  // 保存白板
  async saveWhiteboard() {
    await this.state.storage.put('whiteboard_options', this.options);
    await this.state.storage.put('whiteboard_version', this.version);
  }

  // 广播消息
  async broadcast(message, excludeCollaboratorId = null) {
    const messageString = JSON.stringify(message);

    for (const [id, collaborator] of this.collaborators) {
      if (id !== excludeCollaboratorId &&
          collaborator.ws &&
          collaborator.ws.readyState === WebSocket.OPEN) {
        try {
          collaborator.ws.send(messageString);
        } catch (error) {
          console.error(`Broadcast failed for ${id}:`, error);
        }
      }
    }
  }

  // 发送消息给协作者
  sendToCollaborator(collaboratorId, message) {
    const collaborator = this.collaborators.get(collaboratorId);
    if (!collaborator || !collaborator.ws || collaborator.ws.readyState !== WebSocket.OPEN) {
      return false;
    }

    try {
      collaborator.ws.send(JSON.stringify(message));
      return true;
    } catch (error) {
      console.error(`Send failed for ${collaboratorId}:`, error);
      return false;
    }
  }

  // 获取所有协作者数据
  getAllCollaboratorData() {
    const collaboratorData = {};
    for (const [id, collaborator] of this.collaborators) {
      collaboratorData[id] = this.sanitizeCollaborator(collaborator);
    }
    return collaboratorData;
  }

  // 清理协作者信息
  sanitizeCollaborator(collaborator) {
    const { ws, ...sanitized } = collaborator;
    return sanitized;
  }

  // 生成颜色
  generateColor() {
    const colors = [
      '#FF6B6B', '#4ECDC4', '#45B7D1', '#96CEB4',
      '#FFEAA7', '#DDA0DD', '#98D8C8', '#F7DC6F'
    ];
    const usedColors = Array.from(this.collaborators.values()).map(c => c.color);
    const availableColors = colors.filter(c => !usedColors.includes(c));
    return availableColors.length > 0 ? availableColors[0] : colors[Math.floor(Math.random() * colors.length)];
  }
}
```

```javascript
// 设备管理 Durable Object
export class DeviceManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.devices = new Map();
    this.alerts = [];
    this.heartbeatInterval = 30000; // 30秒
  }

  // 注册设备
  async registerDevice(deviceInfo) {
    const deviceId = deviceInfo.id || crypto.randomUUID();
    const device = {
      id: deviceId,
      type: deviceInfo.type,
      name: deviceInfo.name,
      location: deviceInfo.location || {},
      metadata: deviceInfo.metadata || {},
      status: 'offline',
      lastSeen: null,
      registeredAt: Date.now(),
      data: {},
      alerts: []
    };

    this.devices.set(deviceId, device);
    await this.persistDevice(device);

    return deviceId;
  }

  // 处理设备数据上报
  async handleDeviceData(deviceId, data) {
    const device = this.devices.get(deviceId);
    if (!device) {
      throw new Error(`Device ${deviceId} not registered`);
    }

    // 更新设备状态
    device.status = 'online';
    device.lastSeen = Date.now();
    device.data = { ...device.data, ...data };

    // 处理数据
    await this.processDeviceData(deviceId, data);

    // 检查告警
    await this.checkAlerts(deviceId, data);

    // 持久化
    await this.persistDevice(device);

    return { success: true };
  }

  // 处理设备心跳
  async handleHeartbeat(deviceId) {
    const device = this.devices.get(deviceId);
    if (!device) {
      return false;
    }

    device.status = 'online';
    device.lastSeen = Date.now();

    await this.persistDevice(device);
    return true;
  }

  // 发送命令到设备
  async sendCommand(deviceId, command) {
    const device = this.devices.get(deviceId);
    if (!device) {
      throw new Error(`Device ${deviceId} not found`);
    }

    const commandData = {
      id: crypto.randomUUID(),
      deviceId: deviceId,
      command: command,
      sentAt: Date.now(),
      status: 'sent'
    };

    // 这里应该实现实际的命令发送逻辑
    // 例如通过 WebSocket、MQTT 或其他协议
    await this.sendToDevice(deviceId, commandData);

    // 记录命令历史
    await this.recordCommand(commandData);

    return commandData;
  }

  // 处理设备数据
  async processDeviceData(deviceId, data) {
    const device = this.devices.get(deviceId);
    if (!device) return;

    // 根据设备类型处理数据
    switch (device.type) {
      case 'sensor':
        await this.processSensorData(deviceId, data);
        break;
      case 'actuator':
        await this.processActuatorData(deviceId, data);
        break;
      case 'camera':
        await this.processCameraData(deviceId, data);
        break;
    }

    // 存储历史数据
    await this.storeDataHistory(deviceId, data);
  }

  // 处理传感器数据
  async processSensorData(deviceId, data) {
    // 温度告警检查
    if (data.temperature) {
      if (data.temperature > 40) {
        await this.createAlert({
          deviceId: deviceId,
          type: 'high_temperature',
          severity: 'warning',
          message: `Temperature too high: ${data.temperature}°C`,
          value: data.temperature
        });
      }
    }

    // 湿度告警检查
    if (data.humidity) {
      if (data.humidity > 80) {
        await this.createAlert({
          deviceId: deviceId,
          type: 'high_humidity',
          severity: 'warning',
          message: `Humidity too high: ${data.humidity}%`,
          value: data.humidity
        });
      }
    }
  }

  // 创建告警
  async createAlert(alertData) {
    const alert = {
      id: crypto.randomUUID(),
      ...alertData,
      createdAt: Date.now(),
      acknowledged: false,
      resolved: false
    };

    this.alerts.push(alert);
    await this.persistAlert(alert);

    // 发送告警通知
    await this.sendAlertNotification(alert);

    // 限制告警历史
    if (this.alerts.length > 1000) {
      this.alerts = this.alerts.slice(-500);
    }
  }

  // 获取设备状态
  async getDeviceStatus(deviceId) {
    const device = this.devices.get(deviceId);
    if (!device) {
      return null;
    }

    return {
      id: device.id,
      name: device.name,
      type: device.type,
      status: device.status,
      lastSeen: device.lastSeen,
      location: device.location,
      data: device.data
    };
  }

  // 获取所有设备
  async getAllDevices() {
    return Array.from(this.devices.values());
  }

  // 获取设备历史数据
  async getDeviceHistory(deviceId, options = {}) {
    const {
      startTime = Date.now() - 24 * 60 * 60 * 1000, // 默认24小时
      endTime = Date.now(),
      limit = 1000
    } = options;

    const historyKey = `history:${deviceId}`;
    const history = await this.state.storage.get(historyKey) || [];

    return history
      .filter(entry => entry.timestamp >= startTime && entry.timestamp <= endTime)
      .slice(-limit);
  }

  // 存储数据历史
  async storeDataHistory(deviceId, data) {
    const historyKey = `history:${deviceId}`;
    const history = await this.state.storage.get(historyKey) || [];

    history.push({
      timestamp: Date.now(),
      data: data
    });

    // 限制历史大小
    if (history.length > 10000) {
      history.splice(0, 5000);
    }

    await this.state.storage.put(historyKey, history, {
      expirationTtl: 30 * 24 * 60 * 60 // 30天过期
    });
  }

  // 检查设备离线
  async checkOfflineDevices() {
    const now = Date.now();
    const offlineThreshold = this.heartbeatInterval * 2;

    for (const [deviceId, device] of this.devices) {
      if (device.status === 'online' && device.lastSeen && (now - device.lastSeen > offlineThreshold)) {
        device.status = 'offline';
        await this.createAlert({
          deviceId: deviceId,
          type: 'device_offline',
          severity: 'error',
          message: `Device ${device.name} is offline`,
          lastSeen: device.lastSeen
        });

        await this.persistDevice(device);
      }
    }
  }

  // 持久化设备
  async persistDevice(device) {
    await this.state.storage.put(`device:${device.id}`, device);
  }

  // 持久化告警
  async persistAlert(alert) {
    await this.state.storage.put(`alert:${alert.id}`, alert, {
      expirationTtl: 7 * 24 * 60 * 60 // 7天过期
    });
  }

  // 记录命令
  async recordCommand(command) {
    const commandKey = `command:${command.id}`;
    await this.state.storage.put(commandKey, command, {
      expirationTtl: 24 * 60 * 60 // 24小时过期
    });
  }

  // 发送告警通知
  async sendAlertNotification(alert) {
    // 实现告警通知逻辑
    console.log('Alert:', alert);
  }

  // 发送命令到设备
  async sendToDevice(deviceId, command) {
    // 实现设备命令发送逻辑
    console.log(`Sending command to ${deviceId}:`, command);
  }
}
```
