# Durable Objects API 使用指南

## API 概览

Durable Objects 提供了完整的 API 接口，包括对象管理、状态存储、定时器和 WebSocket 支持等。掌握这些 API 是构建实时应用的基础。

## 核心 API

### 1. Durable Object 定义
```javascript
// 定义 Durable Object 类
export class MyDurableObject {
  constructor(state, env) {
    // state - DurableObjectState 对象，提供存储和同步能力
    // env - 环境变量，包含绑定的其他服务
    this.state = state;
    this.env = env;

    // 初始化实例变量
    this.sessions = new Set();
    this.startTime = Date.now();
  }

  // 必须实现的 fetch 方法
  async fetch(request) {
    const url = new URL(request.url);
    const path = url.pathname;

    switch (path) {
      case '/connect':
        return this.handleWebSocket(request);
      case '/message':
        return this.handleMessage(request);
      case '/status':
        return this.handleStatus(request);
      default:
        return new Response('Not Found', { status: 404 });
    }
  }

  // WebSocket 连接处理
  async handleWebSocket(request) {
    const pair = new WebSocketPair();
    const [client, server] = Object.values(pair);

    await this.handleSession(server, request);

    return new Response(null, {
      status: 101,
      webSocket: client
    });
  }
}

// 在 Worker 中绑定 Durable Object
```

### 3. 定时器 API
```javascript
export class TimerManagement {
  constructor(state, env) {
    this.state = state;
    this.env = env;
  }

  // 设置定时器
  async setTimers() {
    // 设置单次定时器
    const nextHour = Math.ceil(Date.now() / 3600000) * 3600000;
    await this.state.storage.setAlarm(nextHour);

    // 设置重复定时器（需要在 alarm 回调中重新设置）
    await this.scheduleRepeatingTask('daily_cleanup', 24 * 60 * 60 * 1000); // 每天

    // 设置延迟任务
    await this.scheduleDelayedTask('send_reminder', 5 * 60 * 1000); // 5分钟后
  }

  // 定时器回调
  async alarm() {
    console.log('Alarm triggered at:', new Date().toISOString());

    // 获取触发的定时器信息
    const alarms = await this.state.storage.list({
      prefix: 'alarm:'
    });

    for (const [key, alarm] of alarms.entries()) {
      await this.handleAlarm(alarm);
      await this.state.storage.delete(key);
    }

    // 重新设置重复任务
    await this.scheduleRepeatingTask('daily_cleanup', 24 * 60 * 60 * 1000);
  }

  // 处理定时器事件
  async handleAlarm(alarm) {
    switch (alarm.type) {
      case 'daily_cleanup':
        await this.performDailyCleanup();
        break;
      case 'send_reminder':
        await this.sendReminder(alarm.data);
        break;
      case 'generate_report':
        await this.generateReport(alarm.data);
        break;
    }
  }

  // 调度重复任务
  async scheduleRepeatingTask(type, interval) {
    const nextExecution = Date.now() + interval;

    await this.state.storage.put(`alarm:${nextExecution}`, {
      type: type,
      scheduledAt: nextExecution,
      interval: interval
    });

    await this.state.storage.setAlarm(nextExecution);
  }

  // 调度延迟任务
  async scheduleDelayedTask(type, delay, data = null) {
    const executionTime = Date.now() + delay;

    await this.state.storage.put(`alarm:${executionTime}`, {
      type: type,
      data: data,
      scheduledAt: executionTime
    });

    // 更新最早的定时器
    await this.updateEarliestAlarm();
  }

  // 更新最早的定时器
  async updateEarliestAlarm() {
    const alarms = await this.state.storage.list({
      prefix: 'alarm:'
    });

    if (alarms.keys.length > 0) {
      const earliest = Math.min(...alarms.keys.map(k =>
        parseInt(k.name.split(':')[1])
      ));
      await this.state.storage.setAlarm(earliest);
    }
  }

  // 日常清理任务
  async performDailyCleanup() {
    console.log('Performing daily cleanup...');

    // 清理过期数据
    const cutoffTime = Date.now() - 7 * 24 * 60 * 60 * 1000; // 7天前
    const allKeys = await this.state.storage.list();

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value && value.timestamp && value.timestamp < cutoffTime) {
        await this.state.storage.delete(key.name);
      }
    }

    // 压缩存储
    await this.optimizeStorage();

    console.log('Daily cleanup completed');
  }

  // 存储优化
  async optimizeStorage() {
    // 实施存储优化逻辑
    const allData = await this.state.storage.list();

    // 合并相关数据
    const groupedData = {};
    for (const key of allData.keys) {
      const [group, ...rest] = key.name.split(':');
      if (!groupedData[group]) {
        groupedData[group] = [];
      }
      groupedData[group].push({
        key: key.name,
        value: await this.state.storage.get(key.name)
      });
    }

    // 处理分组数据
    for (const [group, items] of Object.entries(groupedData)) {
      if (items.length > 100) { // 如果分组数据过多
        await this.compressGroup(group, items);
      }
    }
  }

  // 压缩数据分组
  async compressGroup(group, items) {
    // 实施数据压缩逻辑
    const compressedData = {
      group: group,
      items: items.map(item => ({
        key: item.key,
        value: item.value
      })),
      compressed: true,
      compressedAt: Date.now()
    };

    // 存储压缩后的数据
    await this.state.storage.put(`compressed:${group}`, compressedData);

    // 删除原始数据
    for (const item of items) {
      await this.state.storage.delete(item.key);
    }
  }
}
```

## WebSocket API

### 1. WebSocket 连接管理
```javascript
export class WebSocketHandler {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.connections = new Map();
  }

  // 处理 WebSocket 连接
  async handleWebSocket(ws, request) {
    const clientId = this.generateClientId();
    const url = new URL(request.url);
    const roomId = url.searchParams.get('room') || 'default';

    // 创建连接对象
    const connection = {
      id: clientId,
      ws: ws,
      roomId: roomId,
      connectedAt: Date.now(),
      lastPing: Date.now(),
      metadata: this.extractMetadata(request)
    };

    // 注册连接
    this.connections.set(clientId, connection);

    // 加入房间
    await this.joinRoom(clientId, roomId);

    // 设置 WebSocket 事件处理
    this.setupWebSocketHandlers(clientId, ws);

    // 发送欢迎消息
    ws.send(JSON.stringify({
      type: 'connected',
      clientId: clientId,
      roomId: roomId,
      timestamp: Date.now()
    }));

    // 通知其他用户
    await this.broadcastToRoom(roomId, {
      type: 'user_joined',
      clientId: clientId,
      userCount: this.getConnectionsInRoom(roomId).length
    }, clientId);

    return clientId;
  }

  // 设置 WebSocket 事件处理
  setupWebSocketHandlers(clientId, ws) {
    // 消息处理
    ws.addEventListener('message', async (event) => {
      try {
        const message = JSON.parse(event.data);
        await this.handleMessage(clientId, message);
      } catch (error) {
        console.error('Message parse error:', error);
        this.sendError(clientId, 'Invalid message format');
      }
    });

    // 关闭处理
    ws.addEventListener('close', async () => {
      await this.handleDisconnect(clientId);
    });

    // 错误处理
    ws.addEventListener('error', (error) => {
      console.error('WebSocket error:', error);
    });

    // 启动心跳
    this.startHeartbeat(clientId);
  }

  // 处理消息
  async handleMessage(clientId, message) {
    const connection = this.connections.get(clientId);
    if (!connection) return;

    const { type, data } = message;

    switch (type) {
      case 'ping':
        this.handlePing(clientId);
        break;

      case 'chat_message':
        await this.handleChatMessage(clientId, data);
        break;

      case 'room_action':
        await this.handleRoomAction(clientId, data);
        break;

      case 'private_message':
        await this.handlePrivateMessage(clientId, data);
        break;

      case 'typing':
        await this.handleTyping(clientId, data);
        break;

      default:
        console.warn('Unknown message type:', type);
    }
  }

  // 处理聊天消息
  async handleChatMessage(clientId, data) {
    const connection = this.connections.get(clientId);
    const message = {
      id: crypto.randomUUID(),
      senderId: clientId,
      senderName: connection.metadata.name || 'Anonymous',
      text: data.text,
      roomId: connection.roomId,
      timestamp: Date.now(),
      type: data.type || 'text'
    };

    // 存储消息
    await this.state.storage.put(`message:${message.id}`, message);

    // 广播到房间
    await this.broadcastToRoom(connection.roomId, {
      type: 'chat_message',
      message: message
    });

    // 触发消息事件
    await this.triggerMessageEvent(message);
  }

  // 处理房间操作
  async handleRoomAction(clientId, data) {
    const connection = this.connections.get(clientId);

    switch (data.action) {
      case 'join':
        await this.switchRoom(clientId, data.roomId);
        break;

      case 'leave':
        await this.leaveRoom(clientId);
        break;

      case 'kick':
        await this.kickUser(clientId, data.targetId);
        break;

      case 'ban':
        await this.banUser(clientId, data.targetId);
        break;
    }
  }

  // 处理私聊消息
  async handlePrivateMessage(clientId, data) {
    const targetConnection = this.connections.get(data.targetId);
    if (!targetConnection) {
      this.sendError(clientId, 'User not found');
      return;
    }

    const message = {
      id: crypto.randomUUID(),
      senderId: clientId,
      receiverId: data.targetId,
      text: data.text,
      timestamp: Date.now()
    };

    // 发送给接收者
    targetConnection.ws.send(JSON.stringify({
      type: 'private_message',
      message: message
    }));

    // 发送回执给发送者
    const senderConnection = this.connections.get(clientId);
    senderConnection.ws.send(JSON.stringify({
      type: 'message_sent',
      messageId: message.id,
      timestamp: message.timestamp
    }));
  }

  // 处理打字状态
  async handleTyping(clientId, data) {
    const connection = this.connections.get(clientId);

    // 广播打字状态（不发送给发送者）
    await this.broadcastToRoom(connection.roomId, {
      type: 'user_typing',
      clientId: clientId,
      isTyping: data.isTyping
    }, clientId);
  }

  // 广播到房间
  async broadcastToRoom(roomId, message, excludeClientId = null) {
    const roomConnections = this.getConnectionsInRoom(roomId);
    const messageString = JSON.stringify(message);

    for (const [clientId, connection] of roomConnections) {
      if (clientId !== excludeClientId && connection.ws.readyState === WebSocket.OPEN) {
        try {
          connection.ws.send(messageString);
        } catch (error) {
          console.error('Broadcast error:', error);
        }
      }
    }
  }

  // 获取房间内连接
  getConnectionsInRoom(roomId) {
    const roomConnections = new Map();
    for (const [clientId, connection] of this.connections) {
      if (connection.roomId === roomId) {
        roomConnections.set(clientId, connection);
      }
    }
    return roomConnections;
  }

  // 处理连接断开
  async handleDisconnect(clientId) {
    const connection = this.connections.get(clientId);
    if (!connection) return;

    // 从房间移除
    await this.leaveRoom(clientId);

    // 从连接列表移除
    this.connections.delete(clientId);

    // 通知其他用户
    await this.broadcastToRoom(connection.roomId, {
      type: 'user_left',
      clientId: clientId,
      userCount: this.getConnectionsInRoom(connection.roomId).length
    });

    // 记录断开事件
    await this.recordDisconnection(clientId, connection);
  }

  // 心跳处理
  startHeartbeat(clientId) {
    const connection = this.connections.get(clientId);
    if (!connection) return;

    const interval = setInterval(() => {
      const conn = this.connections.get(clientId);
      if (!conn || conn.ws.readyState !== WebSocket.OPEN) {
        clearInterval(interval);
        return;
      }

      // 检查最后活动时间
      if (Date.now() - conn.lastPing > 30000) { // 30秒无响应
        conn.ws.close(1000, 'No response to ping');
        clearInterval(interval);
        return;
      }

      // 发送心跳
      conn.ws.send(JSON.stringify({ type: 'ping' }));
    }, 15000); // 每15秒一次心跳
  }

  // 发送错误消息
  sendError(clientId, message) {
    const connection = this.connections.get(clientId);
    if (connection && connection.ws.readyState === WebSocket.OPEN) {
      connection.ws.send(JSON.stringify({
        type: 'error',
        message: message
      }));
    }
  }

  // 生成客户端 ID
  generateClientId() {
    return crypto.randomUUID();
  }

  // 提取元数据
  extractMetadata(request) {
    const url = new URL(request.url);
    return {
      userAgent: request.headers.get('User-Agent'),
      ip: request.headers.get('CF-Connecting-IP'),
      country: request.cf?.country,
      name: url.searchParams.get('name') || 'Anonymous'
    };
  }

  // 记录断开事件
  async recordDisconnection(clientId, connection) {
    const disconnectEvent = {
      clientId: clientId,
      roomId: connection.roomId,
      connectedAt: connection.connectedAt,
      disconnectedAt: Date.now(),
      duration: Date.now() - connection.connectedAt,
      messagesSent: connection.messagesSent || 0
    };

    await this.state.storage.put(
      `disconnect:${clientId}:${Date.now()}`,
      disconnectEvent
    );
  }
}
```

### 2. 高级 WebSocket 功能
```javascript
export class AdvancedWebSocketFeatures {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.rooms = new Map();
    this.messageQueues = new Map();
  }

  // 房间管理
  async createRoom(roomId, options = {}) {
    const room = {
      id: roomId,
      name: options.name || roomId,
      maxUsers: options.maxUsers || 100,
      isPrivate: options.private || false,
      password: options.password || null,
      created: Date.now(),
      owner: options.owner || null,
      metadata: options.metadata || {},
      users: new Set()
    };

    this.rooms.set(roomId, room);
    await this.state.storage.put(`room:${roomId}`, room);

    return room;
  }

  // 加入房间
  async joinRoom(clientId, roomId, password = null) {
    const room = this.rooms.get(roomId);
    if (!room) {
      throw new Error('Room not found');
    }

    // 检查密码
    if (room.isPrivate && room.password !== password) {
      throw new Error('Invalid password');
    }

    // 检查容量
    if (room.users.size >= room.maxUsers) {
      throw new Error('Room is full');
    }

    // 添加用户到房间
    room.users.add(clientId);

    // 更新房间信息
    await this.state.storage.put(`room:${roomId}`, room);

    return room;
  }

  // 离开房间
  async leaveRoom(clientId) {
    for (const [roomId, room] of this.rooms) {
      if (room.users.has(clientId)) {
        room.users.delete(clientId);

        // 如果房间为空，删除房间
        if (room.users.size === 0) {
          this.rooms.delete(roomId);
          await this.state.storage.delete(`room:${roomId}`);
        } else {
          await this.state.storage.put(`room:${roomId}`, room);
        }
      }
    }
  }

  // 消息队列
  async enqueueMessage(roomId, message) {
    if (!this.messageQueues.has(roomId)) {
      this.messageQueues.set(roomId, []);
    }

    const queue = this.messageQueues.get(roomId);
    queue.push({
      id: crypto.randomUUID(),
      message: message,
      timestamp: Date.now()
    });

    // 限制队列大小
    if (queue.length > 1000) {
      queue.shift(); // 移除最旧的消息
    }

    // 持久化队列
    await this.state.storage.put(`queue:${roomId}`, queue);
  }

  // 获取离线消息
  async getOfflineMessages(clientId, roomId) {
    const queue = await this.state.storage.get(`queue:${roomId}`) || [];
    const lastSeen = await this.state.storage.get(`last_seen:${clientId}:${roomId}`);

    const offlineMessages = queue.filter(item =>
      !lastSeen || item.timestamp > lastSeen
    );

    // 更新最后看到的时间
    await this.state.storage.put(`last_seen:${clientId}:${roomId}`, Date.now());

    return offlineMessages;
  }

  // 消息广播优化
  async optimizedBroadcast(roomId, message, options = {}) {
    const {
      exclude = [],
      delay = 0,
      priority = 'normal'
    } = options;

    // 根据优先级调整广播策略
    if (priority === 'high') {
      await this.immediateBroadcast(roomId, message, exclude);
    } else if (priority === 'low') {
      await this.delayedBroadcast(roomId, message, delay, exclude);
    } else {
      await this.normalBroadcast(roomId, message, exclude);
    }
  }

  // 立即广播
  async immediateBroadcast(roomId, message, exclude = []) {
    const messageString = JSON.stringify(message);
    const room = this.rooms.get(roomId);

    if (!room) return;

    const promises = [];
    for (const clientId of room.users) {
      if (!exclude.includes(clientId)) {
        promises.push(this.sendToClient(clientId, messageString));
      }
    }

    await Promise.allSettled(promises);
  }

  // 延迟广播
  async delayedBroadcast(roomId, message, delay, exclude = []) {
    setTimeout(async () => {
      await this.normalBroadcast(roomId, message, exclude);
    }, delay);
  }

  // 普通广播
  async normalBroadcast(roomId, message, exclude = []) {
    const messageString = JSON.stringify(message);
    const room = this.rooms.get(roomId);

    if (!room) return;

    // 批量处理，避免阻塞
    const batchSize = 10;
    const users = Array.from(room.users).filter(id => !exclude.includes(id));

    for (let i = 0; i < users.length; i += batchSize) {
      const batch = users.slice(i, i + batchSize);
      const promises = batch.map(clientId =>
        this.sendToClient(clientId, messageString)
      );

      await Promise.allSettled(promises);

      // 小延迟避免过载
      if (i + batchSize < users.length) {
        await new Promise(resolve => setTimeout(resolve, 1));
      }
    }
  }

  // 发送给特定客户端
  async sendToClient(clientId, messageString) {
    const connection = this.connections.get(clientId);
    if (connection && connection.ws.readyState === WebSocket.OPEN) {
      try {
        connection.ws.send(messageString);
        return true;
      } catch (error) {
        console.error('Send error:', error);
        return false;
      }
    }
    return false;
  }

  // 文件传输支持
  async handleFileTransfer(clientId, data) {
    const {
      fileId,
      fileName,
      fileSize,
      chunkIndex,
      totalChunks,
      chunkData
    } = data;

    // 存储文件块
    const chunkKey = `file:${fileId}:chunk:${chunkIndex}`;
    await this.state.storage.put(chunkKey, {
      fileId: fileId,
      fileName: fileName,
      fileSize: fileSize,
      chunkIndex: chunkIndex,
      totalChunks: totalChunks,
      data: chunkData,
      receivedAt: Date.now()
    });

    // 检查是否接收完成
    const allChunks = await this.state.storage.list({
      prefix: `file:${fileId}:chunk:`
    });

    if (allChunks.keys.length === totalChunks) {
      // 组装完整文件
      const fileData = await this.assembleFile(fileId, totalChunks);

      // 通知文件接收完成
      await this.broadcastToRoom(
        this.connections.get(clientId).roomId,
        {
          type: 'file_received',
          fileId: fileId,
          fileName: fileName,
          size: fileSize
        }
      );

      // 清理临时块
      await this.cleanupFileChunks(fileId);
    }
  }

  // 组装文件
  async assembleFile(fileId, totalChunks) {
    const chunks = [];

    for (let i = 0; i < totalChunks; i++) {
      const chunkKey = `file:${fileId}:chunk:${i}`;
      const chunk = await this.state.storage.get(chunkKey);
      if (chunk) {
        chunks.push(chunk);
      }
    }

    // 按索引排序
    chunks.sort((a, b) => a.chunkIndex - b.chunkIndex);

    // 组装文件数据
    const fileData = {
      id: fileId,
      chunks: chunks,
      assembledAt: Date.now()
    };

    await this.state.storage.put(`file:${fileId}`, fileData);
    return fileData;
  }

  // 清理文件块
  async cleanupFileChunks(fileId) {
    const chunks = await this.state.storage.list({
      prefix: `file:${fileId}:chunk:`
    });

    for (const chunk of chunks.keys) {
      await this.state.storage.delete(chunk.name);
    }
  }
}
```

```javascript
        if (options.backgroundTasks) {

        }

        // 设置定期清理

        }
      }

      async alarm() {
        // 执行清理任务
        if (this.cleanup) {
          await this.cleanup();
        }

        // 重新设置定时器
        this.state.storage.setAlarm(Date.now() + this.config.cleanupInterval);
      }
    };
  }

  static withLogging(BaseClass) {
    return class extends BaseClass {
      async fetch(request) {
        const startTime = Date.now();
        const url = new URL(request.url);

        console.log(`[${this.constructor.name}] Request: ${request.method} ${url.pathname}`);

        try {
          const response = await super.fetch(request);
          const duration = Date.now() - startTime;

          console.log(`[${this.constructor.name}] Response: ${response.status} (${duration}ms)`);
          return response;
        } catch (error) {
          const duration = Date.now() - startTime;
          console.error(`[${this.constructor.name}] Error: ${error.message} (${duration}ms)`);
          throw error;
        }
      }
    };
  }

  static withMetrics(BaseClass) {
    return class extends BaseClass {
      async fetch(request) {
        const startTime = performance.now();
        let response;

        try {
          response = await super.fetch(request);

          // 记录成功的请求指标
          await this.recordMetrics({
            success: true,
            duration: performance.now() - startTime,
            status: response.status,
            path: new URL(request.url).pathname
          });

          return response;
        } catch (error) {
          // 记录失败的请求指标
          await this.recordMetrics({
            success: false,
            duration: performance.now() - startTime,
            error: error.message,
            path: new URL(request.url).pathname
          });

          throw error;
        }
      }

      async recordMetrics(metrics) {
        const today = new Date().toISOString().split('T')[0];
        const key = `metrics:${today}`;

        const existing = await this.state.storage.get(key) || {
          requests: 0,
          errors: 0,
          totalDuration: 0,
          paths: {}
        };

        existing.requests++;
        if (!metrics.success) {
          existing.errors++;
        }
        existing.totalDuration += metrics.duration;

        if (!existing.paths[metrics.path]) {
          existing.paths[metrics.path] = { count: 0, duration: 0 };
        }
        existing.paths[metrics.path].count++;
        existing.paths[metrics.path].duration += metrics.duration;

        await this.state.storage.put(key, existing, {
          expirationTtl: 30 * 24 * 60 * 60 // 30天
        });
      }
    };
  }
```

```javascript
// 装饰器用于扩展 Durable Object 功能
export const DurableObjectDecorators = {
  // 缓存装饰器
  cache(ttl = 300) {
    return function(target, propertyKey, descriptor) {
      const originalMethod = descriptor.value;
      const cache = new Map();

      descriptor.value = async function(...args) {
        const cacheKey = JSON.stringify(args);
        const cached = cache.get(cacheKey);

        if (cached && Date.now() - cached.timestamp < ttl * 1000) {
          return cached.value;
        }

        const result = await originalMethod.apply(this, args);
        cache.set(cacheKey, {
          value: result,
          timestamp: Date.now()
        });

        return result;
      };
    };
  },

  // 重试装饰器
  retry(maxRetries = 3, delay = 1000) {
    return function(target, propertyKey, descriptor) {
      const originalMethod = descriptor.value;

      descriptor.value = async function(...args) {
        let lastError;

        for (let i = 0; i <= maxRetries; i++) {
          try {
            return await originalMethod.apply(this, args);
          } catch (error) {
            lastError = error;

            if (i < maxRetries) {
              console.warn(`Retry ${i + 1}/${maxRetries} for ${propertyKey}:`, error.message);
              await new Promise(resolve => setTimeout(resolve, delay * Math.pow(2, i)));
            }
          }
        }

        throw lastError;
      };
    };
  },

  // 限流装饰器
  rateLimit(maxCalls = 100, windowMs = 60000) {
    return function(target, propertyKey, descriptor) {
      const originalMethod = descriptor.value;
      const calls = [];

      descriptor.value = async function(...args) {
        const now = Date.now();

        // 清理过期调用记录
        while (calls.length > 0 && calls[0] < now - windowMs) {
          calls.shift();
        }

        if (calls.length >= maxCalls) {
          throw new Error(`Rate limit exceeded for ${propertyKey}`);
        }

        calls.push(now);
        return await originalMethod.apply(this, args);
      };
    };
  }
};

// 使用示例
export class ExampleDurableObject {
  constructor(state, env) {
    this.state = state;
    this.env = env;
  }

  @DurableObjectDecorators.cache(60) // 缓存60秒
  @DurableObjectDecorators.retry(3)  // 重试3次
  async expensiveOperation(param) {
    // 耗时操作
    await new Promise(resolve => setTimeout(resolve, 100));
    return `Result for ${param}`;
  }

  @DurableObjectDecorators.rateLimit(10, 60000) // 每分钟最多10次
  async limitedOperation(data) {
    // 受限的操作
    return `Processed: ${data}`;
  }
}
```
