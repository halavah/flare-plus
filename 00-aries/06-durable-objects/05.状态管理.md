# Durable Objects 状态管理

## 状态管理概述

Durable Objects 提供了强大的状态管理能力，通过事务性存储保证数据的一致性和持久性。合理的状态管理策略是构建可靠应用的关键。

## 状态存储基础

### 1. 状态对象初始化
```javascript
// 状态管理基础类
export class StateManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.localCache = new Map();
    this.cacheExpiry = new Map();
    this.cacheTimeout = 30000; // 30秒缓存超时
  }

  // 初始化状态
  async initialize() {
    // 检查是否首次初始化
    const initialized = await this.state.storage.get('initialized');
    if (!initialized) {
      await this.setupInitialState();
      await this.state.storage.put('initialized', true);
      console.log('State initialized for the first time');
    }

    // 加载状态到缓存
    await this.loadStateToCache();

    // 设置定期同步
    this.schedulePeriodicSync();
  }

  // 设置初始状态
  async setupInitialState() {
    const initialState = {
      version: '1.0.0',
      created: Date.now(),
      counters: {
        requests: 0,
        connections: 0,
        messages: 0
      },
      settings: {
        maxConnections: 1000,
        messageHistorySize: 100,
        cleanupInterval: 3600000 // 1小时
      },
      metadata: {
        lastUpdate: Date.now(),
        lastSync: Date.now()
      }
    };

    await this.state.storage.put('state', initialState);
    await this.state.storage.put('version', initialState.version);
  }

  // 加载状态到内存缓存
  async loadStateToCache() {
    try {
      const state = await this.state.storage.get('state');
      if (state) {
        this.localCache.set('state', state);
        this.cacheExpiry.set('state', Date.now() + this.cacheTimeout);
      }
    } catch (error) {
      console.error('Failed to load state to cache:', error);
    }
  }

  // 获取状态（优先从缓存）
  async getState() {
    const cached = this.getCached('state');
    if (cached) {
      return cached;
    }

    // 从存储加载
    const state = await this.state.storage.get('state') || this.getDefaultState();
    this.setCached('state', state);

    return state;
  }

  // 更新状态
  async updateState(updates) {
    return await this.state.storage.transaction(async (txn) => {
      const currentState = await txn.get('state') || this.getDefaultState();

      // 应用更新
      const newState = this.mergeState(currentState, updates);

      // 更新元数据
      newState.metadata.lastUpdate = Date.now();
      newState.metadata.lastSync = Date.now();

      // 保存新状态
      await txn.put('state', newState);

      // 更新本地缓存
      this.setCached('state', newState);

      // 记录变更历史
      await this.recordStateChange(txn, currentState, newState);

      return newState;
    });
  }

  // 合并状态
  mergeState(current, updates) {
    const merged = { ...current };

    for (const [key, value] of Object.entries(updates)) {
      if (typeof value === 'object' && !Array.isArray(value) && value !== null) {
        merged[key] = { ...merged[key], ...value };
      } else {
        merged[key] = value;
      }
    }

    return merged;
  }

  // 记录状态变更
  async recordStateChange(txn, oldState, newState) {
    const change = {
      timestamp: Date.now(),
      oldState: this.sanitizeState(oldState),
      newState: this.sanitizeState(newState),
      diff: this.calculateStateDiff(oldState, newState)
    };

    const changeHistory = await txn.get('state_changes') || [];
    changeHistory.push(change);

    // 限制历史记录数量
    if (changeHistory.length > 100) {
      changeHistory.shift();
    }

    await txn.put('state_changes', changeHistory);
  }

  // 计算状态差异
  calculateStateDiff(oldState, newState) {
    const diff = {};

    for (const [key, newValue] of Object.entries(newState)) {
      const oldValue = oldState[key];
      if (JSON.stringify(oldValue) !== JSON.stringify(newValue)) {
        diff[key] = {
          old: oldValue,
          new: newValue,
          changed: true
        };
      }
    }

    return diff;
  }

  // 清理状态数据
  sanitizeState(state) {
    // 移除敏感或大型数据，只保留必要信息
    const sanitized = { ...state };

    // 移除大型缓存
    if (sanitized.cache) {
      delete sanitized.cache;
    }

    // 移除临时数据
    if (sanitized.temp) {
      delete sanitized.temp;
    }

    return sanitized;
  }

  // 缓存操作
  getCached(key) {
    const expiry = this.cacheExpiry.get(key);
    if (expiry && Date.now() > expiry) {
      this.localCache.delete(key);
      this.cacheExpiry.delete(key);
      return null;
    }

    return this.localCache.get(key);
  }

  setCached(key, value) {
    this.localCache.set(key, value);
    this.cacheExpiry.set(key, Date.now() + this.cacheTimeout);
  }

  // 获取默认状态
  getDefaultState() {
    return {
      version: '1.0.0',
      created: Date.now(),
      counters: { requests: 0, connections: 0, messages: 0 },
      settings: {
        maxConnections: 1000,
        messageHistorySize: 100,
        cleanupInterval: 3600000
      },
      metadata: {
        lastUpdate: Date.now(),
        lastSync: Date.now()
      }
    };
  }
}
```

```javascript
// 计数器管理器
export class CounterManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.counters = new Map();
  }

  // 初始化计数器
  async initializeCounter(name, initialValue = 0) {
    const key = `counter:${name}`;
    const exists = await this.state.storage.get(key);

    if (!exists) {
      const counter = {
        name: name,
        value: initialValue,
        created: Date.now(),
        lastUpdated: Date.now(),
        history: []
      };

      await this.state.storage.put(key, counter);
      this.counters.set(name, counter);
    }

    return await this.getCounter(name);
  }

  // 增加计数器
  async incrementCounter(name, amount = 1, metadata = {}) {
    const key = `counter:${name}`;

    return await this.state.storage.transaction(async (txn) => {
      const counter = await txn.get(key) || await this.createDefaultCounter(name);

      // 增加值
      counter.value += amount;
      counter.lastUpdated = Date.now();

      // 记录历史
      const historyEntry = {
        timestamp: Date.now(),
        delta: amount,
        value: counter.value,
        metadata: metadata
      };

      counter.history.push(historyEntry);

      // 限制历史记录
      if (counter.history.length > 1000) {
        counter.history = counter.history.slice(-500);
      }

      await txn.put(key, counter);
      this.counters.set(name, counter);

      return {
        name: name,
        value: counter.value,
        delta: amount,
        timestamp: counter.lastUpdated
      };
    });
  }

  // 减少计数器
  async decrementCounter(name, amount = 1, metadata = {}) {
    return await this.incrementCounter(name, -amount, metadata);
  }

  // 设置计数器值
  async setCounter(name, value, metadata = {}) {
    const key = `counter:${name}`;

    return await this.state.storage.transaction(async (txn) => {
      const counter = await txn.get(key) || await this.createDefaultCounter(name);
      const oldValue = counter.value;

      counter.value = value;
      counter.lastUpdated = Date.now();

      // 记录历史
      const historyEntry = {
        timestamp: Date.now(),
        delta: value - oldValue,
        value: value,
        metadata: metadata
      };

      counter.history.push(historyEntry);

      await txn.put(key, counter);
      this.counters.set(name, counter);

      return {
        name: name,
        oldValue: oldValue,
        newValue: value,
        timestamp: counter.lastUpdated
      };
    });
  }

  // 获取计数器
  async getCounter(name) {
    const key = `counter:${name}`;
    let counter = this.counters.get(name);

    if (!counter) {
      counter = await this.state.storage.get(key);
      if (counter) {
        this.counters.set(name, counter);
      }
    }

    return counter;
  }

  // 获取计数器值
  async getCounterValue(name) {
    const counter = await this.getCounter(name);
    return counter ? counter.value : 0;
  }

  // 获取计数器历史
  async getCounterHistory(name, options = {}) {
    const {
      limit = 100,
      since = null,
      until = null
    } = options;

    const counter = await this.getCounter(name);
    if (!counter || !counter.history) {
      return [];
    }

    let history = [...counter.history];

    // 时间过滤
    if (since) {
      history = history.filter(entry => entry.timestamp >= since);
    }

    if (until) {
      history = history.filter(entry => entry.timestamp <= until);
    }

    // 排序和限制
    history.sort((a, b) => b.timestamp - a.timestamp);
    return history.slice(0, limit);
  }

  // 重置计数器
  async resetCounter(name, metadata = {}) {
    return await this.setCounter(name, 0, { ...metadata, action: 'reset' });
  }

  // 批量操作
  async batchIncrement(operations) {
    const results = [];

    await this.state.storage.transaction(async (txn) => {
      for (const operation of operations) {
        const { name, amount = 1, metadata = {} } = operation;
        const key = `counter:${name}`;

        const counter = await txn.get(key) || await this.createDefaultCounter(name);
        counter.value += amount;
        counter.lastUpdated = Date.now();

        // 记录历史
        counter.history.push({
          timestamp: Date.now(),
          delta: amount,
          value: counter.value,
          metadata: metadata
        });

        await txn.put(key, counter);
        this.counters.set(name, counter);

        results.push({
          name: name,
          value: counter.value,
          delta: amount
        });
      }
    });

    return results;
  }

  // 获取所有计数器
  async getAllCounters() {
    const counters = await this.state.storage.list({
      prefix: 'counter:'
    });

    const results = {};
    for (const key of counters.keys) {
      const name = key.name.replace('counter:', '');
      const counter = await this.getCounter(name);
      if (counter) {
        results[name] = {
          value: counter.value,
          lastUpdated: counter.lastUpdated,
          created: counter.created
        };
      }
    }

    return results;
  }

  // 创建默认计数器
  async createDefaultCounter(name) {
    return {
      name: name,
      value: 0,
      created: Date.now(),
      lastUpdated: Date.now(),
      history: []
    };
  }
}
```

### 3. 配置管理
```javascript
// 配置管理器
export class ConfigManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.config = new Map();
    this.watchers = new Map();
  }

  // 加载配置
  async loadConfig(key = null) {
    if (key) {
      const configValue = await this.state.storage.get(`config:${key}`);
      if (configValue) {
        this.config.set(key, configValue);
      }
      return configValue;
    } else {
      // 加载所有配置
      const configs = await this.state.storage.list({
        prefix: 'config:'
      });

      for (const configKey of configs.keys) {
        const configKeyShort = configKey.name.replace('config:', '');
        const value = await this.state.storage.get(configKey.name);
        this.config.set(configKeyShort, value);
      }

      return Object.fromEntries(this.config);
    }
  }

  // 设置配置
  async setConfig(key, value, options = {}) {
    const {
      persist = true,
      notify = true,
      metadata = {}
    } = options;

    const oldValue = this.config.get(key);

    // 更新内存中的配置
    this.config.set(key, value);

    // 持久化
    if (persist) {
      const configData = {
        key: key,
        value: value,
        metadata: {
          ...metadata,
          updated: Date.now(),
          version: (oldValue?.metadata?.version || 0) + 1
        }
      };

      await this.state.storage.put(`config:${key}`, configData);

      // 记录配置变更历史
      await this.recordConfigChange(key, oldValue, value, metadata);
    }

    // 通知观察者
    if (notify && oldValue !== value) {
      await this.notifyWatchers(key, value, oldValue);
    }

    return {
      key: key,
      value: value,
      oldValue: oldValue,
      changed: oldValue !== value
    };
  }

  // 获取配置
  async getConfig(key, defaultValue = null) {
    let value = this.config.get(key);

    if (value === undefined) {
      // 从存储加载
      const configData = await this.state.storage.get(`config:${key}`);
      if (configData) {
        value = configData.value;
        this.config.set(key, value);
      }
    }

    return value !== undefined ? value : defaultValue;
  }

  // 删除配置
  async deleteConfig(key) {
    const oldValue = this.config.get(key);

    // 从内存删除
    this.config.delete(key);

    // 从存储删除
    await this.state.storage.delete(`config:${key}`);

    // 通知观察者
    await this.notifyWatchers(key, null, oldValue);

    return {
      key: key,
      deleted: true,
      oldValue: oldValue
    };
  }

  // 监听配置变更
  watchConfig(key, callback) {
    if (!this.watchers.has(key)) {
      this.watchers.set(key, new Set());
    }

    this.watchers.get(key).add(callback);

    // 返回取消监听的函数
    return () => {
      const keyWatchers = this.watchers.get(key);
      if (keyWatchers) {
        keyWatchers.delete(callback);
        if (keyWatchers.size === 0) {
          this.watchers.delete(key);
        }
      }
    };
  }

  // 通知观察者
  async notifyWatchers(key, newValue, oldValue) {
    const keyWatchers = this.watchers.get(key);
    if (!keyWatchers) return;

    const changeEvent = {
      key: key,
      newValue: newValue,
      oldValue: oldValue,
      timestamp: Date.now()
    };

    for (const callback of keyWatchers) {
      try {
        await callback(changeEvent);
      } catch (error) {
        console.error(`Config watcher error for key ${key}:`, error);
      }
    }
  }

  // 批量设置配置
  async batchSetConfig(configs, options = {}) {
    const results = [];

    await this.state.storage.transaction(async (txn) => {
      for (const [key, value] of Object.entries(configs)) {
        const oldValue = this.config.get(key);

        this.config.set(key, value);

        const configData = {
          key: key,
          value: value,
          metadata: {
            ...options.metadata,
            updated: Date.now(),
            batch: true
          }
        };

        await txn.put(`config:${key}`, configData);

        results.push({
          key: key,
          value: value,
          oldValue: oldValue,
          changed: oldValue !== value
        });
      }
    });

    // 批量通知
    for (const change of results) {
      if (change.changed) {
        await this.notifyWatchers(change.key, change.value, change.oldValue);
      }
    }

    return results;
  }

  // 记录配置变更历史
  async recordConfigChange(key, oldValue, newValue, metadata) {
    const historyKey = `config_history:${key}`;
    let history = await this.state.storage.get(historyKey) || [];

    const changeRecord = {
      timestamp: Date.now(),
      key: key,
      oldValue: oldValue,
      newValue: newValue,
      metadata: metadata
    };

    history.push(changeRecord);

    // 限制历史记录
    if (history.length > 100) {
      history = history.slice(-50);
    }

    await this.state.storage.put(historyKey, history);
  }

  // 获取配置变更历史
  async getConfigHistory(key, limit = 50) {
    const historyKey = `config_history:${key}`;
    const history = await this.state.storage.get(historyKey) || [];

    return history
      .sort((a, b) => b.timestamp - a.timestamp)
      .slice(0, limit);
  }

  // 验证配置
  validateConfig(key, value) {
    const validators = {
      'maxConnections': (v) => typeof v === 'number' && v > 0 && v <= 10000,
      'timeout': (v) => typeof v === 'number' && v > 0 && v <= 300000,
      'enableFeature': (v) => typeof v === 'boolean',
      'retryCount': (v) => typeof v === 'number' && v >= 0 && v <= 10
    };

    const validator = validators[key];
    return validator ? validator(value) : true;
  }

  // 导出配置
  async exportConfig(keys = null) {
    let configsToExport = this.config;

    if (keys) {
      configsToExport = new Map();
      for (const key of keys) {
        const value = await this.getConfig(key);
        if (value !== null) {
          configsToExport.set(key, value);
        }
      }
    }

    return {
      timestamp: Date.now(),
      version: '1.0.0',
      configs: Object.fromEntries(configsToExport)
    };
  }

  // 导入配置
  async importConfig(configData, options = {}) {
    const {
      overwrite = false,
      validate = true
    } = options;

    const results = {
      imported: [],
      skipped: [],
      errors: []
    };

    for (const [key, value] of Object.entries(configData.configs)) {
      try {
        // 验证
        if (validate && !this.validateConfig(key, value)) {
          results.errors.push({
            key: key,
            error: 'Validation failed'
          });
          continue;
        }

        // 检查是否需要覆盖
        const existingValue = await this.getConfig(key);
        if (!overwrite && existingValue !== null) {
          results.skipped.push({
            key: key,
            reason: 'Already exists'
          });
          continue;
        }

        // 导入
        await this.setConfig(key, value, {
          persist: true,
          notify: false,
          metadata: {
            imported: true,
            source: 'import'
          }
        });

        results.imported.push({
          key: key,
          value: value
        });

      } catch (error) {
        results.errors.push({
          key: key,
          error: error.message
        });
      }
    }

    return results;
  }
}
```

### 1. 自动同步机制
```javascript
// 状态同步管理器
export class StateSyncManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.syncInterval = 60000; // 1分钟
    this.lastSync = 0;
    this.pendingChanges = new Map();
    this.conflictResolver = new ConflictResolver();
  }

  // 初始化同步
  async initializeSync() {
    // 获取最后同步时间
    this.lastSync = await this.state.storage.get('last_sync_time') || 0;

    // 启动定期同步
    this.startPeriodicSync();

    // 设置定时器
    this.state.storage.setAlarm(Date.now() + this.syncInterval);
  }

  // 启动定期同步
  startPeriodicSync() {
    this.state.storage.setAlarm(Date.now() + this.syncInterval);
  }

  // 定时器回调
  async alarm() {
    await this.performSync();
    this.startPeriodicSync();
  }

  // 执行同步
  async performSync() {
    console.log('Performing state sync...');

    try {
      // 收集所有变更
      const changes = await this.collectChanges();

      if (changes.length > 0) {
        // 处理变更
        await this.processChanges(changes);

        // 更新同步时间
        this.lastSync = Date.now();
        await this.state.storage.put('last_sync_time', this.lastSync);

        console.log(`Sync completed: ${changes.length} changes processed`);
      } else {
        console.log('No changes to sync');
      }

    } catch (error) {
      console.error('Sync failed:', error);
    }
  }

  // 收集变更
  async collectChanges() {
    const changes = [];

    // 获取所有带时间戳的数据
    const allData = await this.state.storage.list();

    for (const key of allData.keys) {
      const value = await this.state.storage.get(key.name);

      if (value && value.lastModified && value.lastModified > this.lastSync) {
        changes.push({
          key: key.name,
          value: value,
          timestamp: value.lastModified,
          operation: value.operation || 'update'
        });
      }
    }

    // 按时间排序
    return changes.sort((a, b) => a.timestamp - b.timestamp);
  }

  // 处理变更
  async processChanges(changes) {
    for (const change of changes) {
      await this.applyChange(change);
    }

    // 清理已处理的变更标记
    await this.cleanupProcessedChanges();
  }

  // 应用变更
  async applyChange(change) {
    switch (change.operation) {
      case 'create':
      case 'update':
        await this.applyUpdate(change);
        break;
      case 'delete':
        await this.applyDelete(change);
        break;
      case 'merge':
        await this.applyMerge(change);
        break;
    }
  }

  // 应用更新
  async applyUpdate(change) {
    const currentValue = await this.state.storage.get(change.key);

    // 检查冲突
    if (currentValue && currentValue.lastModified !== change.timestamp) {
      const resolution = await this.conflictResolver.resolve(
        change.key,
        currentValue,
        change.value
      );

      if (resolution.conflict) {
        console.warn(`Conflict detected for ${change.key}, using resolution:`, resolution.action);
      }

      if (resolution.action === 'skip') {
        return;
      }

      change.value = resolution.value;
    }

    // 应用更新
    const updatedValue = {
      ...change.value,
      lastSynced: Date.now(),
      synced: true
    };

    await this.state.storage.put(change.key, updatedValue);
  }

  // 应用删除
  async applyDelete(change) {
    await this.state.storage.delete(change.key);
  }

  // 应用合并
  async applyMerge(change) {
    const currentValue = await this.state.storage.get(change.key);

    if (currentValue) {
      const mergedValue = this.mergeValues(currentValue, change.value);
      await this.state.storage.put(change.key, mergedValue);
    }
  }

  // 合并值
  mergeValues(current, incoming) {
    const merged = { ...current };

    // 简单的合并策略
    for (const [key, value] of Object.entries(incoming)) {
      if (key === 'lastModified' || key === 'operation') {
        continue; // 跳过元数据
      }

      if (Array.isArray(value) && Array.isArray(merged[key])) {
        // 数组合并
        merged[key] = [...new Set([...merged[key], ...value])];
      } else if (typeof value === 'object' && typeof merged[key] === 'object') {
        // 对象合并
        merged[key] = { ...merged[key], ...value };
      } else {
        // 直接覆盖
        merged[key] = value;
      }
    }

    merged.lastModified = Date.now();
    merged.merged = true;

    return merged;
  }

  // 清理已处理的变更
  async cleanupProcessedChanges() {
    // 清理临时数据
    const tempData = await this.state.storage.list({
      prefix: 'temp:'
    });

    for (const key of tempData.keys) {
      const value = await this.state.storage.get(key.name);
      if (value && value.expiresAt && value.expiresAt < Date.now()) {
        await this.state.storage.delete(key.name);
      }
    }
  }

  // 手动触发同步
  async forceSync() {
    console.log('Force sync triggered');
    await this.performSync();
  }

  // 获取同步状态
  async getSyncStatus() {
    const pendingChanges = await this.collectChanges();

    return {
      lastSync: this.lastSync,
      nextSync: this.lastSync + this.syncInterval,
      pendingChanges: pendingChanges.length,
      syncInterval: this.syncInterval
    };
  }
}
```

```javascript
// 冲突解决器
export class ConflictResolver {
  constructor() {
    this.strategies = new Map();
    this.setupDefaultStrategies();
  }

  // 设置默认策略
  setupDefaultStrategies() {
    // 时间戳策略：使用最新的数据
    this.strategies.set('timestamp', (current, incoming) => {
      if (current.lastModified > incoming.lastModified) {
        return { action: 'keep_current', value: current, conflict: true };
      } else {
        return { action: 'use_incoming', value: incoming, conflict: true };
      }
    });

    // 计数器策略：累加
    this.strategies.set('counter', (current, incoming) => {
      const merged = {
        ...current,
        value: current.value + incoming.value,
        lastModified: Date.now()
      };
      return { action: 'merge', value: merged, conflict: true };
    });

    // 集合策略：合并集合
    this.strategies.set('set', (current, incoming) => {
      const merged = {
        ...current,
        items: [...new Set([...current.items, ...incoming.items])],
        lastModified: Date.now()
      };
      return { action: 'merge', value: merged, conflict: true };
    });

    // 版本策略：使用版本号
    this.strategies.set('version', (current, incoming) => {
      const currentVersion = current.version || 0;
      const incomingVersion = incoming.version || 0;

      if (currentVersion > incomingVersion) {
        return { action: 'keep_current', value: current, conflict: true };
      } else if (incomingVersion > currentVersion) {
        return { action: 'use_incoming', value: incoming, conflict: true };
      } else {
        // 版本相同，使用时间戳
        return this.strategies.get('timestamp')(current, incoming);
      }
    });

    // 优先级策略：使用优先级字段
    this.strategies.set('priority', (current, incoming) => {
      const currentPriority = current.priority || 0;
      const incomingPriority = incoming.priority || 0;

      if (currentPriority > incomingPriority) {
        return { action: 'keep_current', value: current, conflict: true };
      } else if (incomingPriority > currentPriority) {
        return { action: 'use_incoming', value: incoming, conflict: true };
      } else {
        // 优先级相同，使用时间戳
        return this.strategies.get('timestamp')(current, incoming);
      }
    });
  }

  // 解决冲突
  async resolve(key, current, incoming) {
    // 根据键名或数据类型选择策略
    const strategy = this.selectStrategy(key, current, incoming);

    if (strategy) {
      return strategy(current, incoming);
    }

    // 默认策略：使用传入的数据
    return { action: 'use_incoming', value: incoming, conflict: true };
  }

  // 选择解决策略
  selectStrategy(key, current, incoming) {
    // 基于键名的策略选择
    if (key.includes('counter') || key.includes('count')) {
      return this.strategies.get('counter');
    }

    if (key.includes('config') || key.includes('settings')) {
      return this.strategies.get('version');
    }

    if (key.includes('list') || key.includes('items')) {
      return this.strategies.get('set');
    }

    // 基于数据类型的策略选择
    if (current.type === 'counter' && incoming.type === 'counter') {
      return this.strategies.get('counter');
    }

    if (current.items && incoming.items) {
      return this.strategies.get('set');
    }

    // 检查是否有优先级字段
    if ((current.priority !== undefined) || (incoming.priority !== undefined)) {
      return this.strategies.get('priority');
    }

    // 检查是否有版本字段
    if ((current.version !== undefined) || (incoming.version !== undefined)) {
      return this.strategies.get('version');
    }

    // 默认使用时间戳策略
    return this.strategies.get('timestamp');
  }

  // 注册自定义策略
  registerStrategy(name, strategy) {
    this.strategies.set(name, strategy);
  }

  // 移除策略
  removeStrategy(name) {
    return this.strategies.delete(name);
  }

  // 获取所有策略
  getStrategies() {
    return Array.from(this.strategies.keys());
  }
}
```

```javascript
// 数据迁移管理器
export class MigrationManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.migrations = new Map();
    this.completedMigrations = new Set();
  }

  // 注册迁移
  registerMigration(version, migration) {
    this.migrations.set(version, {
      version: version,
      up: migration.up,
      down: migration.down,
      description: migration.description,
      dependencies: migration.dependencies || []
    });
  }

  // 执行迁移
  async migrate(targetVersion = null) {
    console.log('Starting data migration...');

    // 获取当前版本
    const currentVersion = await this.getCurrentVersion();
    console.log(`Current version: ${currentVersion}`);

    // 获取已完成的迁移
    await this.loadCompletedMigrations();

    // 确定需要执行的迁移
    const migrationsToRun = this.getMigrationsToRun(currentVersion, targetVersion);

    if (migrationsToRun.length === 0) {
      console.log('No migrations to run');
      return { success: true, migrations: [] };
    }

    console.log(`Running ${migrationsToRun.length} migrations...`);

    // 执行迁移
    const results = [];
    for (const migration of migrationsToRun) {
      try {
        console.log(`Running migration ${migration.version}: ${migration.description}`);

        await this.runMigration(migration);

        results.push({
          version: migration.version,
          status: 'success',
          description: migration.description
        });

        console.log(`Migration ${migration.version} completed`);

      } catch (error) {
        console.error(`Migration ${migration.version} failed:`, error);

        results.push({
          version: migration.version,
          status: 'failed',
          error: error.message,
          description: migration.description
        });

        // 迁移失败，停止执行
        break;
      }
    }

    // 更新版本
    if (results.every(r => r.status === 'success')) {
      await this.updateVersion(targetVersion || migrationsToRun[migrationsToRun.length - 1].version);
    }

    return {
      success: results.every(r => r.status === 'success'),
      migrations: results
    };
  }

  // 运行单个迁移
  async runMigration(migration) {
    // 检查依赖
    for (const dependency of migration.dependencies) {
      if (!this.completedMigrations.has(dependency)) {
        throw new Error(`Missing dependency: ${dependency}`);
      }
    }

    // 执行迁移
    await migration.up(this.state);

    // 记录完成的迁移
    this.completedMigrations.add(migration.version);
    await this.recordCompletedMigration(migration.version);
  }

  // 获取当前版本
  async getCurrentVersion() {
    return await this.state.storage.get('schema_version') || '0.0.0';
  }

  // 更新版本
  async updateVersion(version) {
    await this.state.storage.put('schema_version', version);
  }

  // 加载已完成的迁移
  async loadCompletedMigrations() {
    const completed = await this.state.storage.get('completed_migrations') || [];
    this.completedMigrations = new Set(completed);
  }

  // 记录完成的迁移
  async recordCompletedMigration(version) {
    this.completedMigrations.add(version);
    const completed = Array.from(this.completedMigrations);
    await this.state.storage.put('completed_migrations', completed);
  }

  // 获取需要执行的迁移
  getMigrationsToRun(currentVersion, targetVersion) {
    const versions = Array.from(this.migrations.keys()).sort(this.compareVersions);
    const migrationsToRun = [];

    for (const version of versions) {
      const comparison = this.compareVersions(version, currentVersion);

      if (comparison > 0) {
        // 版本大于当前版本
        if (!targetVersion || this.compareVersions(version, targetVersion) <= 0) {
          migrationsToRun.push(this.migrations.get(version));
        }
      }
    }

    return migrationsToRun;
  }

  // 版本比较
  compareVersions(a, b) {
    const aParts = a.split('.').map(Number);
    const bParts = b.split('.').map(Number);
    const maxLength = Math.max(aParts.length, bParts.length);

    for (let i = 0; i < maxLength; i++) {
      const aPart = aParts[i] || 0;
      const bPart = bParts[i] || 0;

      if (aPart > bPart) return 1;
      if (aPart < bPart) return -1;
    }

    return 0;
  }

  // 回滚迁移
  async rollback(targetVersion) {
    console.log(`Rolling back to version ${targetVersion}...`);

    const currentVersion = await this.getCurrentVersion();
    await this.loadCompletedMigrations();

    // 获取需要回滚的迁移
    const migrationsToRollback = [];
    const versions = Array.from(this.migrations.keys()).sort(this.compareVersions).reverse();

    for (const version of versions) {
      if (this.compareVersions(version, targetVersion) > 0 &&
          this.compareVersions(version, currentVersion) <= 0) {
        migrationsToRollback.push(this.migrations.get(version));
      }
    }

    console.log(`Rolling back ${migrationsToRollback.length} migrations...`);

    // 执行回滚
    const results = [];
    for (const migration of migrationsToRollback) {
      try {
        console.log(`Rolling back migration ${migration.version}`);

        await migration.down(this.state);

        results.push({
          version: migration.version,
          status: 'success'
        });

        // 移除已完成的迁移记录
        this.completedMigrations.delete(migration.version);

      } catch (error) {
        console.error(`Rollback failed for ${migration.version}:`, error);

        results.push({
          version: migration.version,
          status: 'failed',
          error: error.message
        });
      }
    }

    // 更新版本和记录
    if (results.every(r => r.status === 'success')) {
      await this.updateVersion(targetVersion);
      const completed = Array.from(this.completedMigrations);
      await this.state.storage.put('completed_migrations', completed);
    }

    return {
      success: results.every(r => r.status === 'success'),
      rollbacks: results
    };
  }
}
```
