# Durable Objects 性能测试

## 测试概述

Durable Objects 的性能测试对于了解其在实际应用中的表现至关重要。本章节将介绍如何进行全面的性能测试，包括延迟测试、吞吐量测试、并发连接测试和压力测试。

## 测试框架

### 1. 性能测试基础框架
```javascript
// 性能测试框架
export class PerformanceTestFramework {
  constructor(env) {
    this.env = env;
    this.testResults = [];
    this.testConfig = {
      iterations: 1000,
      concurrency: 10,
      warmupIterations: 100,
      timeoutMs: 30000
    };
  }

  // 运行性能测试
  async runTest(testName, testFunction, options = {}) {
    const config = { ...this.testConfig, ...options };
    const results = {
      testName: testName,
      timestamp: Date.now(),
      config: config,
      iterations: [],
      summary: {}
    };

    console.log(`Starting performance test: ${testName}`);

    // 预热阶段
    console.log(`Warming up with ${config.warmupIterations} iterations...`);
    for (let i = 0; i < config.warmupIterations; i++) {
      try {
        await testFunction();
      } catch (error) {
        console.warn('Warmup iteration failed:', error.message);
      }
    }

    // 实际测试阶段
    console.log(`Running ${config.iterations} iterations with ${config.concurrency} concurrency...`);
    const promises = [];

    for (let batch = 0; batch < Math.ceil(config.iterations / config.concurrency); batch++) {
      const batchPromises = [];

      for (let i = 0; i < config.concurrency && (batch * config.concurrency + i) < config.iterations; i++) {
        const iterationId = batch * config.concurrency + i;
        const promise = this.runSingleIteration(testFunction, iterationId);
        batchPromises.push(promise);
      }

      promises.push(Promise.allSettled(batchPromises));
    }

    const batchResults = await Promise.all(promises);

    // 收集结果
    for (const batch of batchResults) {
      for (const result of batch) {
        if (result.status === 'fulfilled') {
          results.iterations.push(result.value);
        } else {
          console.error('Test iteration failed:', result.reason);
          results.iterations.push({
            success: false,
            error: result.reason.message,
            duration: config.timeoutMs
          });
        }
      }
    }

    // 计算统计信息
    results.summary = this.calculateSummary(results.iterations);

    // 保存结果
    await this.saveTestResults(results);

    console.log(`Test completed: ${testName}`);
    console.log(`Average latency: ${results.summary.avgLatency.toFixed(2)}ms`);
    console.log(`Throughput: ${results.summary.throughput.toFixed(2)} ops/s`);

    return results;
  }

  // 运行单个测试迭代
  async runSingleIteration(testFunction, iterationId) {
    const startTime = performance.now();
    let success = true;
    let error = null;

    try {
      await Promise.race([
        testFunction(),
        new Promise((_, reject) =>
          setTimeout(() => reject(new Error('Timeout')), this.testConfig.timeoutMs)
        )
      ]);
    } catch (err) {
      success = false;
      error = err.message;
    }

    const endTime = performance.now();
    const duration = endTime - startTime;

    return {
      iterationId: iterationId,
      success: success,
      duration: duration,
      error: error,
      timestamp: Date.now()
    };
  }

  // 计算统计信息
  calculateSummary(iterations) {
    const successful = iterations.filter(i => i.success);
    const failed = iterations.filter(i => !i.success);
    const durations = successful.map(i => i.duration).sort((a, b) => a - b);

    const summary = {
      totalIterations: iterations.length,
      successful: successful.length,
      failed: failed.length,
      successRate: (successful.length / iterations.length) * 100,
      minLatency: durations.length > 0 ? Math.min(...durations) : 0,
      maxLatency: durations.length > 0 ? Math.max(...durations) : 0,
      avgLatency: durations.length > 0 ? durations.reduce((sum, d) => sum + d, 0) / durations.length : 0,
      p50: durations.length > 0 ? this.percentile(durations, 50) : 0,
      p95: durations.length > 0 ? this.percentile(durations, 95) : 0,
      p99: durations.length > 0 ? this.percentile(durations, 99) : 0,
      throughput: 0
    };

    // 计算吞吐量
    if (successful.length > 0) {
      const totalTime = Math.max(...iterations.map(i => i.timestamp)) - Math.min(...iterations.map(i => i.timestamp));
      summary.throughput = totalTime > 0 ? (successful.length / totalTime) * 1000 : 0;
    }

    // 错误分析
    summary.errors = {};
    for (const fail of failed) {
      summary.errors[fail.error] = (summary.errors[fail.error] || 0) + 1;
    }

    return summary;
  }

  // 计算百分位数
  percentile(sortedNumbers, p) {
    if (sortedNumbers.length === 0) return 0;
    const index = Math.ceil((p / 100) * sortedNumbers.length) - 1;
    return sortedNumbers[Math.max(0, index)];
  }

  // 保存测试结果
  async saveTestResults(results) {
    const key = `perf_test:${results.testName}:${results.timestamp}`;
    await this.env.TEST_RESULTS.put(
      key,
      JSON.stringify(results),
      { expirationTtl: 7 * 24 * 60 * 60 } // 7天
    );

    // 更新测试索引
    await this.updateTestIndex(results);
  }

  // 更新测试索引
  async updateTestIndex(results) {
    const indexKey = 'perf_test_index';
    let index = [];

    try {
      const indexValue = await this.env.TEST_RESULTS.get(indexKey);
      if (indexValue) {
        index = JSON.parse(indexValue);
      }
    } catch (error) {
      console.error('Failed to parse test index:', error);
    }

    index.push({
      testName: results.testName,
      timestamp: results.timestamp,
      summary: results.summary
    });

    // 保留最近100次测试
    index = index.slice(-100);

    await this.env.TEST_RESULTS.put(
      indexKey,
      JSON.stringify(index),
      { expirationTtl: 30 * 24 * 60 * 60 } // 30天
    );
  }

  // 获取测试历史
  async getTestHistory(testName = null, limit = 50) {
    const indexKey = 'perf_test_index';
    let history = [];

    try {
      const indexValue = await this.env.TEST_RESULTS.get(indexKey);
      if (indexValue) {
        const index = JSON.parse(indexValue);

        if (testName) {
          history = index.filter(test => test.testName === testName);
        } else {
          history = index;
        }
      }
    } catch (error) {
      console.error('Failed to get test history:', error);
    }

    return history
      .sort((a, b) => b.timestamp - a.timestamp)
      .slice(0, limit);
  }
}
```

```javascript
export class LatencyTestSuite {

      const value = { data: 'test', timestamp: Date.now() };

      // 写入测试
      const writeStart = performance.now();
      await testObject.put(key, value);
      const writeLatency = performance.now() - writeStart;

      // 读取测试
      const readStart = performance.now();
      const retrieved = await testObject.get(key);
      const readLatency = performance.now() - readStart;

      // 删除测试
      const deleteStart = performance.now();
      await testObject.delete(key);
      const deleteLatency = performance.now() - deleteStart;

      return {
        writeLatency: writeLatency,
        readLatency: readLatency,
        deleteLatency: deleteLatency,
        totalLatency: writeLatency + readLatency + deleteLatency
      };
    };

    return await this.framework.runTest('storage_latency', testFunction, {
      iterations: 500,
      concurrency: 5
    });
  }

  // WebSocket 延迟测试
  async testWebSocketLatency() {
    const testObject = this.env.TEST_OBJECT;

    return await this.framework.runTest('websocket_latency', async () => {
      const startTime = performance.now();

      // 创建 WebSocket 连接
      const response = await fetch(`https://example.com/test-connection`, {
        headers: {
          'Upgrade': 'websocket'
        }
      });

      if (response.status !== 101) {
        throw new Error('WebSocket upgrade failed');
      }

      const ws = response.webSocket;
      ws.accept();

      // 发送消息并等待回复
      const messageReceived = new Promise((resolve) => {
        ws.addEventListener('message', (event) => {
          resolve(event.data);
        });
      });

      ws.send(JSON.stringify({ type: 'ping', timestamp: Date.now() }));

      // 等待响应
      await messageReceived;
      const latency = performance.now() - startTime;

      ws.close();

      return { websocketLatency: latency };
    }, {
      iterations: 100,
      concurrency: 2
    });
  }

  // 计算密集型操作延迟测试
  async testComputationLatency() {
    const testObject = this.env.TEST_OBJECT;

    return await this.framework.runTest('computation_latency', async () => {
      const startTime = performance.now();

      // 执行一些计算密集型操作
      const result = await testObject.compute({
        type: 'complex_calculation',
        size: 1000
      });

      const latency = performance.now() - startTime;

      return {
        computationLatency: latency,
        resultSize: JSON.stringify(result).length
      };
    }, {
      iterations: 200,
      concurrency: 3
    });
  }

  // 事务操作延迟测试
  async testTransactionLatency() {
    const testObject = this.env.TEST_OBJECT;

    return await this.framework.runTest('transaction_latency', async () => {
      const startTime = performance.now();

      // 执行事务操作
      await testObject.transaction(async (txn) => {
        // 多个操作
        await txn.put('tx_key1', 'value1');
        await txn.put('tx_key2', 'value2');
        await txn.put('tx_key3', 'value3');

        // 读取验证
        const value = await txn.get('tx_key1');
        if (!value) {
          throw new Error('Transaction failed');
        }
      });

      const latency = performance.now() - startTime;

      return { transactionLatency: latency };
    }, {
      iterations: 100,
      concurrency: 2
    });
  }
}
```

```javascript
export class ThroughputTestSuite {
  constructor(env) {
    this.env = env;
    this.framework = new PerformanceTestFramework(env);
  }

  // 存储吞吐量测试
  async testStorageThroughput() {
    const testObject = this.env.TEST_OBJECT;
    const batchSize = 10;
    const totalOperations = 1000;

    return await this.framework.runTest('storage_throughput', async () => {
      const startTime = performance.now();
      const operations = [];

      // 创建批量操作
      for (let i = 0; i < batchSize; i++) {
        const key = `throughput_test_${Date.now()}_${Math.random()}`;
        const value = {
          data: 'x'.repeat(1024), // 1KB 数据
          index: i,
          timestamp: Date.now()
        };

        operations.push(testObject.put(key, value));
      }

      // 并发执行
      await Promise.all(operations);

      const duration = performance.now() - startTime;
      const throughput = (batchSize / duration) * 1000;

      return {
        batchSize: batchSize,
        duration: duration,
        throughput: throughput
      };
    }, {
      iterations: Math.floor(totalOperations / batchSize),
      concurrency: 20
    });
  }

  // WebSocket 并发连接测试
  async testWebSocketConcurrency() {
    const testObject = this.env.TEST_OBJECT;
    const targetConnections = 100;
    const connections = [];

    return await this.framework.runTest('websocket_concurrency', async () => {
      const connectionPromises = [];

      // 创建多个并发连接
      for (let i = 0; i < targetConnections; i++) {
        const promise = this.createWebSocketConnection(i);
        connectionPromises.push(promise);
      }

      // 等待所有连接建立
      const results = await Promise.allSettled(connectionPromises);
      const successful = results.filter(r => r.status === 'fulfilled').length;

      // 清理连接
      for (const ws of connections) {
        try {
          ws.close();
        } catch (error) {
          console.error('Error closing connection:', error);
        }
      }

      return {
        targetConnections: targetConnections,
        successfulConnections: successful,
        connectionRate: (successful / targetConnections) * 100
      };
    }, {
      iterations: 10,
      concurrency: 1
    });
  }

  // 消息广播吞吐量测试
  async testMessageBroadcastThroughput() {
    const testObject = this.env.TEST_OBJECT;
    const messageSize = 512; // bytes
    const recipientCount = 50;

    return await this.framework.runTest('broadcast_throughput', async () => {
      const startTime = performance.now();

      // 创建接收者
      const receivers = [];
      for (let i = 0; i < recipientCount; i++) {
        const receiver = await testObject.createReceiver(`receiver_${i}`);
        receivers.push(receiver);
      }

      // 广播消息
      const message = {
        type: 'broadcast_test',
        data: 'x'.repeat(messageSize),
        timestamp: Date.now()
      };

      await testObject.broadcast(message, receivers.map(r => r.id));

      const duration = performance.now() - startTime;
      const totalMessages = recipientCount;
      const throughput = (totalMessages / duration) * 1000;

      // 清理接收者
      for (const receiver of receivers) {
        await testObject.removeReceiver(receiver.id);
      }

      return {
        recipientCount: recipientCount,
        messageSize: messageSize,
        totalMessages: totalMessages,
        duration: duration,
        throughput: throughput
      };
    }, {
      iterations: 100,
      concurrency: 5
    });
  }

  // 并发读写测试
  async testConcurrentReadWrite() {
    const testObject = this.env.TEST_OBJECT;
    const concurrentOperations = 20;
    const operationsPerThread = 10;

    return await this.framework.runTest('concurrent_read_write', async () => {
      const promises = [];

      // 创建读写操作
      for (let i = 0; i < concurrentOperations; i++) {
        if (i % 2 === 0) {
          // 写操作
          promises.push(this.performWriteOperations(testObject, operationsPerThread, i));
        } else {
          // 读操作
          promises.push(this.performReadOperations(testObject, operationsPerThread));
        }
      }

      const results = await Promise.allSettled(promises);
      const successful = results.filter(r => r.status === 'fulfilled').length;

      return {
        concurrentOperations: concurrentOperations,
        successful: successful,
        successRate: (successful / concurrentOperations) * 100
      };
    }, {
      iterations: 50,
      concurrency: 1
    });
  }

  // 执行写操作
  async performWriteOperations(testObject, count, threadId) {
    for (let i = 0; i < count; i++) {
      const key = `concurrent_write_${threadId}_${i}`;
      const value = {
        threadId: threadId,
        index: i,
        data: 'test data',
        timestamp: Date.now()
      };

      await testObject.put(key, value);
    }
  }

  // 执行读操作
  async performReadOperations(testObject, count) {
    for (let i = 0; i < count; i++) {
      const key = `concurrent_write_0_${i}`; // 读取第一个线程写入的数据
      await testObject.get(key);
    }
  }
}
```

```javascript
// 压力测试套件
export class StressTestSuite {
  constructor(env) {
    this.env = env;
    this.framework = new PerformanceTestFramework(env);
  }

  // 高负载存储测试
  async testHighLoadStorage() {
    const testObject = this.env.TEST_OBJECT;
    const duration = 60000; // 60秒
    const targetRPS = 100; // 每秒100个请求

    return await this.framework.runTest('high_load_storage', async () => {
      const startTime = Date.now();
      const endTime = startTime + duration;
      let requestCount = 0;
      let errors = 0;

      while (Date.now() < endTime) {
        const batchStart = Date.now();
        const batchPromises = [];

        // 创建批量请求
        for (let i = 0; i < targetRPS; i++) {
          const promise = testObject.put(
            `stress_test_${requestCount + i}`,
            {
              data: 'x'.repeat(1024),
              index: requestCount + i,
              timestamp: Date.now()
            }
          ).catch(error => {
            errors++;
            return null;
          });

          batchPromises.push(promise);
        }

        // 执行批量请求
        await Promise.allSettled(batchPromises);
        requestCount += targetRPS;

        // 控制请求速率
        const batchDuration = Date.now() - batchStart;
        if (batchDuration < 1000) {
          await new Promise(resolve => setTimeout(resolve, 1000 - batchDuration));
        }
      }

      const actualDuration = Date.now() - startTime;
      const actualRPS = requestCount / (actualDuration / 1000);

      return {
        duration: actualDuration,
        targetRPS: targetRPS,
        actualRPS: actualRPS,
        totalRequests: requestCount,
        errors: errors,
        errorRate: (errors / requestCount) * 100
      };
    }, {
      iterations: 1,
      concurrency: 1
    });
  }

  // 内存压力测试
  async testMemoryPressure() {
    const testObject = this.env.TEST_OBJECT;

    return await this.framework.runTest('memory_pressure', async () => {
      const largeDataSize = 1024 * 1024; // 1MB
      const dataCount = 50; // 50个1MB数据块

      // 创建大量数据
      const promises = [];
      for (let i = 0; i < dataCount; i++) {
        const key = `memory_test_${i}`;
        const value = {
          data: 'x'.repeat(largeDataSize),
          index: i,
          timestamp: Date.now()
        };

        promises.push(testObject.put(key, value));
      }

      await Promise.all(promises);

      // 读取所有数据
      const readPromises = [];
      for (let i = 0; i < dataCount; i++) {
        readPromises.push(testObject.get(`memory_test_${i}`));
      }

      const results = await Promise.all(readPromises);
      const successful = results.filter(r => r !== null).length;

      // 清理数据
      for (let i = 0; i < dataCount; i++) {
        await testObject.delete(`memory_test_${i}`);
      }

      return {
        dataSize: largeDataSize,
        dataCount: dataCount,
        totalDataSize: largeDataSize * dataCount,
        successfulReads: successful,
        readSuccessRate: (successful / dataCount) * 100
      };
    }, {
      iterations: 10,
      concurrency: 2
    });
  }

  // 连接风暴测试
  async testConnectionStorm() {
    const testObject = this.env.TEST_OBJECT;
    const connectionBurst = 200;
    const burstCount = 5;

    return await this.framework.runTest('connection_storm', async () => {
      const results = [];

      for (let burst = 0; burst < burstCount; burst++) {
        const burstStart = performance.now();
        const connections = [];

        // 创建连接突发
        for (let i = 0; i < connectionBurst; i++) {
          try {
            const connection = await testObject.createConnection(`storm_${burst}_${i}`);
            connections.push(connection);
          } catch (error) {
            console.error(`Connection ${i} failed:`, error.message);
          }
        }

        const burstDuration = performance.now() - burstStart;

        // 保持连接一段时间
        await new Promise(resolve => setTimeout(resolve, 1000));

        // 关闭所有连接
        for (const conn of connections) {
          try {
            await testObject.closeConnection(conn.id);
          } catch (error) {
            console.error('Error closing connection:', error);
          }
        }

        results.push({
          burst: burst,
          targetConnections: connectionBurst,
          actualConnections: connections.length,
          duration: burstDuration,
          connectionRate: connections.length / (burstDuration / 1000)
        });

        // 突发间隔
        await new Promise(resolve => setTimeout(resolve, 2000));
      }

      return results;
    }, {
      iterations: 1,
      concurrency: 1
    });
  }

  // 长时间运行测试
  async testEndurance() {
    const testObject = this.env.TEST_OBJECT;
    const duration = 10 * 60 * 1000; // 10分钟
    const operationsPerSecond = 10;

    return await this.framework.runTest('endurance', async () => {
      const startTime = Date.now();
      const endTime = startTime + duration;
      let operationCount = 0;
      let errors = 0;
      const metrics = {
        latencies: [],
        memoryUsage: [],
        errorTypes: {}
      };

      while (Date.now() < endTime) {
        const operationStart = performance.now();

        try {
          // 混合操作
          const operationType = operationCount % 4;
          switch (operationType) {
            case 0: // 写入
              await testObject.put(`endurance_${operationCount}`, {
                type: 'write',
                index: operationCount,
                timestamp: Date.now()
              });
              break;
            case 1: // 读取
              await testObject.get(`endurance_${operationCount - 1}`);
              break;
            case 2: // 列表
              await testObject.list({ limit: 100 });
              break;
            case 3: // 删除
              if (operationCount > 10) {
                await testObject.delete(`endurance_${operationCount - 10}`);
              }
              break;
          }

          const latency = performance.now() - operationStart;
          metrics.latencies.push(latency);

        } catch (error) {
          errors++;
          const errorType = error.name || 'UnknownError';
          metrics.errorTypes[errorType] = (metrics.errorTypes[errorType] || 0) + 1;
        }

        operationCount++;

        // 控制操作频率
        const operationDuration = performance.now() - operationStart;
        const targetInterval = 1000 / operationsPerSecond;
        if (operationDuration < targetInterval) {
          await new Promise(resolve => setTimeout(resolve, targetInterval - operationDuration));
        }
      }

      const actualDuration = Date.now() - startTime;
      const avgLatency = metrics.latencies.reduce((sum, lat) => sum + lat, 0) / metrics.latencies.length;

      return {
        duration: actualDuration,
        totalOperations: operationCount,
        actualOpsPerSecond: operationCount / (actualDuration / 1000),
        errors: errors,
        errorRate: (errors / operationCount) * 100,
        avgLatency: avgLatency,
        maxLatency: Math.max(...metrics.latencies),
        minLatency: Math.min(...metrics.latencies),
        errorTypes: metrics.errorTypes
      };
    }, {
      iterations: 1,
      concurrency: 1
    });
  }
}
```

### 1. 性能报告生成器

```javascript
// 性能报告生成器
export class PerformanceReportGenerator {
  constructor(env) {
    this.env = env;
  }

  // 生成综合性能报告
  async generateComprehensiveReport() {
    const report = {
      timestamp: Date.now(),
      summary: {},
      testResults: {},
      trends: {},
      recommendations: []
    };

    // 获取所有测试结果
    const testHistory = await this.getAllTestHistory();

    // 分析各个测试类型
    const testTypes = ['storage_latency', 'websocket_latency', 'throughput', 'stress'];
    for (const testType of testTypes) {
      const typeResults = testHistory.filter(test => test.testName.includes(testType));
      report.testResults[testType] = await this.analyzeTestType(testType, typeResults);
    }

    // 生成趋势分析
    report.trends = await this.analyzeTrends(testHistory);

    // 生成建议
    report.recommendations = this.generateRecommendations(report);

    // 保存报告
    await this.saveReport(report);

    return report;
  }

  // 分析特定测试类型
  async analyzeTestType(testType, results) {
    if (results.length === 0) {
      return { status: 'no_data' };
    }

    const latest = results[0];
    const previous = results[1] || latest;

    const analysis = {
      latest: latest.summary,
      previous: previous.summary,
      trend: this.calculateTrend(previous.summary, latest.summary),
      performance: this.assessPerformance(testType, latest.summary)
    };

    return analysis;
  }

  // 计算趋势
  calculateTrend(previous, current) {
    const trends = {};

    for (const [key, value] of Object.entries(current)) {
      if (typeof value === 'number' && previous[key] !== undefined) {
        const previousValue = previous[key];
        const change = ((value - previousValue) / previousValue) * 100;

        trends[key] = {
          previous: previousValue,
          current: value,
          change: change,
          direction: change > 5 ? 'improving' : change < -5 ? 'degrading' : 'stable'
        };
      }
    }

    return trends;
  }

  // 评估性能
  assessPerformance(testType, summary) {
    const thresholds = {
      storage_latency: {
        avgLatency: { good: 10, acceptable: 50 },
        p95: { good: 25, acceptable: 100 }
      },
      websocket_latency: {
        avgLatency: { good: 20, acceptable: 100 },
        p95: { good: 50, acceptable: 200 }
      },
      throughput: {
        throughput: { good: 1000, acceptable: 500 }
      }
    };

    const threshold = thresholds[testType];
    if (!threshold) return { status: 'unknown' };

    let score = 0;
    let totalChecks = 0;

    for (const [metric, limits] of Object.entries(threshold)) {
      const value = summary[metric];
      if (value !== undefined) {
        totalChecks++;

        if (value <= limits.good) {
          score += 2;
        } else if (value <= limits.acceptable) {
          score += 1;
        }
      }
    }

    const performanceRatio = totalChecks > 0 ? score / (totalChecks * 2) : 0;

    return {
      score: performanceRatio,
      status: performanceRatio >= 0.8 ? 'excellent' :
              performanceRatio >= 0.6 ? 'good' :
              performanceRatio >= 0.4 ? 'acceptable' : 'poor',
      details: threshold
    };
  }

  // 分析趋势
  async analyzeTrends(testHistory) {
    const trends = {};
    const timeWindows = [7, 30]; // 7天和30天

    for (const window of timeWindows) {
      const cutoffTime = Date.now() - (window * 24 * 60 * 60 * 1000);
      const windowTests = testHistory.filter(test => test.timestamp > cutoffTime);

      if (windowTests.length > 1) {
        const first = windowTests[windowTests.length - 1];
        const last = windowTests[0];

        trends[`${window}days`] = {
          improvement: this.calculateImprovement(first.summary, last.summary),
          stability: this.calculateStability(windowTests)
        };
      }
    }

    return trends;
  }

  // 计算改进程度
  calculateImprovement(first, last) {
    const improvements = {};

    for (const [key, value] of Object.entries(last)) {
      if (typeof value === 'number' && first[key] !== undefined) {
        const change = ((value - first[key]) / first[key]) * 100;
        improvements[key] = change;
      }
    }

    return improvements;
  }

  // 计算稳定性
  calculateStability(tests) {
    if (tests.length < 3) return 'insufficient_data';

    const metric = 'avgLatency'; // 使用平均延迟作为稳定性指标
    const values = tests.map(test => test.summary[metric]).filter(v => v !== undefined);

    if (values.length < 3) return 'insufficient_data';

    const mean = values.reduce((sum, v) => sum + v, 0) / values.length;
    const variance = values.reduce((sum, v) => sum + Math.pow(v - mean, 2), 0) / values.length;
    const standardDeviation = Math.sqrt(variance);
    const coefficientOfVariation = standardDeviation / mean;

    if (coefficientOfVariation < 0.1) return 'very_stable';
    if (coefficientOfVariation < 0.2) return 'stable';
    if (coefficientOfVariation < 0.3) return 'moderately_stable';
    return 'unstable';
  }

  // 生成建议
  generateRecommendations(report) {
    const recommendations = [];

    // 分析延迟性能
    const latencyTest = report.testResults.storage_latency;
    if (latencyTest.performance && latencyTest.performance.status === 'poor') {
      recommendations.push({
        category: 'performance',
        priority: 'high',
        title: 'High storage latency detected',
        description: 'Average storage latency is above acceptable thresholds',
        actions: [
          'Consider implementing caching strategies',
          'Review data access patterns',
          'Optimize object size and structure'
        ]
      });
    }

    // 分析错误率
    for (const [testType, analysis] of Object.entries(report.testResults)) {
      if (analysis.latest && analysis.latest.errorRate > 5) {
        recommendations.push({
          category: 'reliability',
          priority: 'high',
          title: `High error rate in ${testType}`,
          description: `Error rate is ${analysis.latest.errorRate.toFixed(1)}%`,
          actions: [
            'Implement proper error handling',
            'Add retry mechanisms',
            'Review resource limits'
          ]
        });
      }
    }

    // 分析趋势
    if (report.trends['7days'] && report.trends['7days'].improvement.avgLatency > 10) {
      recommendations.push({
        category: 'trend',
        priority: 'medium',
        title: 'Degrading performance trend',
        description: 'Latency has increased by over 10% in the last 7 days',
        actions: [
          'Investigate recent changes',
          'Monitor system resources',
          'Consider scaling strategies'
        ]
      });
    }

    return recommendations;
  }

  // 获取所有测试历史
  async getAllTestHistory() {
    const indexKey = 'perf_test_index';
    let history = [];

    try {
      const indexValue = await this.env.TEST_RESULTS.get(indexKey);
      if (indexValue) {
        history = JSON.parse(indexValue);
      }
    } catch (error) {
      console.error('Failed to get test history:', error);
    }

    return history.sort((a, b) => b.timestamp - a.timestamp);
  }

  // 保存报告
  async saveReport(report) {
    const reportKey = `performance_report:${report.timestamp}`;
    await this.env.TEST_RESULTS.put(
      reportKey,
      JSON.stringify(report),
      { expirationTtl: 90 * 24 * 60 * 60 } // 90天
    );
  }
}
```

### 1. 自动化测试调度器
```javascript
// 自动化测试调度器
export class TestScheduler {
  constructor(env) {
    this.env = env;
    this.testSuites = new Map();
  }

  // 注册测试套件
  registerTestSuite(name, suite) {
    this.testSuites.set(name, suite);
  }

  // 运行所有测试
  async runAllTests() {
    const results = {};

    for (const [name, suite] of this.testSuites) {
      console.log(`Running test suite: ${name}`);
      try {
        results[name] = await this.runTestSuite(suite);
      } catch (error) {
        console.error(`Test suite ${name} failed:`, error);
        results[name] = { error: error.message };
      }
    }

    return results;
  }

  // 运行测试套件
  async runTestSuite(suite) {
    const suiteResults = {};

    for (const [testName, testFunction] of Object.entries(suite)) {
      if (typeof testFunction === 'function') {
        console.log(`  Running test: ${testName}`);
        try {
          const result = await testFunction.call(suite);
          suiteResults[testName] = result;
        } catch (error) {
          console.error(`    Test ${testName} failed:`, error);
          suiteResults[testName] = { error: error.message };
        }
      }
    }

    return suiteResults;
  }

  // 定期运行测试
  async schedulePeriodicTests(intervalHours = 24) {
    const nextRun = Date.now() + (intervalHours * 60 * 60 * 1000);

    // 设置定时器
    await this.env.SCHEDULER.put('next_test_run', nextRun);

    // 注册告警
    console.log(`Next performance test scheduled for: ${new Date(nextRun).toISOString()}`);
  }

  // 检查并运行定期测试
  async checkAndRunPeriodicTests() {
    const nextRun = await this.env.SCHEDULER.get('next_test_run');

    if (nextRun && Date.now() >= nextRun) {
      console.log('Running scheduled performance tests...');
      const results = await this.runAllTests();

      // 保存结果
      await this.env.SCHEDULER.put(
        `scheduled_test_results:${Date.now()}`,
        JSON.stringify(results),
        { expirationTtl: 30 * 24 * 60 * 60 }
      );

      // 调度下次测试
      await this.schedulePeriodicTests();

      return results;
    }

    return null;
  }
}
```

## 测试最佳实践

### 1. 测试设计原则
- **真实性**: 模拟真实的使用场景
- **可重复性**: 确保测试结果可重复
- **渐进性**: 从小规模开始逐步增加负载
- **多维度**: 测试延迟、吞吐量、并发等多个指标

### 2. 测试执行建议
- 在不同的时间段运行测试
- 使用多个地理位置进行测试
- 监控系统资源使用情况
- 记录详细的测试日志

### 3. 结果分析要点
- 关注百分位数而不仅仅是平均值
- 分析错误类型和原因
- 比较不同版本的性能差异
- 识别性能瓶颈
