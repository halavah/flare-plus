# Durable Objects 扩展模式

## 概述

Durable Objects 可以通过各种扩展模式来增强功能、提高性能和改善用户体验。本章将介绍几种常见的扩展模式，包括分片、缓存、批处理和组合模式。

## 分片模式

### 1. 水平分片
```javascript
// 分片管理器
export class ShardManager {
  constructor(env) {
    this.env = env;
    this.shardCount = 10;
    this.shards = new Map();
  }

  // 获取分片 ID
  getShardId(key) {
    // 使用一致性哈希
    const hash = this.hashKey(key);
    return hash % this.shardCount;
  }

  // 获取分片对象
  async getShard(shardId) {
    if (!this.shards.has(shardId)) {
      const id = this.env.SHARDED_OBJECT.idFromName(`shard-${shardId}`);
      const stub = this.env.SHARDED_OBJECT.get(id);
      this.shards.set(shardId, stub);
    }

    return this.shards.get(shardId);
  }

  // 路由到分片
  async routeToShard(key, operation, ...args) {
    const shardId = this.getShardId(key);
    const shard = await this.getShard(shardId);

    return await shard.fetch(new Request('https://internal/api', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        operation: operation,
        key: key,
        args: args
      })
    }));
  }

  // 哈希键
  hashKey(key) {
    let hash = 0;
    for (let i = 0; i < key.length; i++) {
      const char = key.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // 转换为32位整数
    }
    return Math.abs(hash);
  }

  // 重新平衡分片
  async rebalanceShards() {
    const stats = await this.getShardStatistics();
    const avgLoad = stats.reduce((sum, stat) => sum + stat.load, 0) / stats.length;

    // 识别负载不均衡的分片
    const unbalancedShards = stats.filter(stat =>
      Math.abs(stat.load - avgLoad) > avgLoad * 0.3
    );

    for (const shard of unbalancedShards) {
      if (shard.load > avgLoad * 1.3) {
        // 需要拆分
        await this.splitShard(shard.id);
      } else if (shard.load < avgLoad * 0.7) {
        // 需要合并
        await this.mergeShard(shard.id);
      }
    }
  }

  // 获取分片统计
  async getShardStatistics() {
    const stats = [];

    for (let i = 0; i < this.shardCount; i++) {
      const shard = await this.getShard(i);
      const response = await shard.fetch(new Request('https://internal/api/stats'));
      const stat = await response.json();

      stats.push({
        id: i,
        load: stat.load,
        keyCount: stat.keyCount,
        memoryUsage: stat.memoryUsage
      });
    }

    return stats;
  }

  // 拆分分片
  async splitShard(shardId) {
    console.log(`Splitting shard ${shardId}`);

    // 创建新的分片
    const newShardId = this.shardCount;
    this.shardCount++;

    // 迁移部分数据到新分片
    const shard = await this.getShard(shardId);
    const migrationData = await this.getMigrationData(shardId);

    // 分配数据到新旧分片
    for (const [key, value] of migrationData) {
      const targetShardId = this.getShardId(key);
      if (targetShardId === newShardId) {
        await this.moveToShard(newShardId, key, value);
        await shard.fetch(new Request('https://internal/api/delete', {
          method: 'POST',
          body: JSON.stringify({ key: key })
        }));
      }
    }

    console.log(`Shard ${shardId} split, new shard ${newShardId} created`);
  }

  // 合并分片
  async mergeShard(shardId) {
    console.log(`Merging shard ${shardId}`);

    // 找到目标分片
    const targetShardId = this.findMergeTarget(shardId);
    if (targetShardId === null) return;

    // 迁移数据
    const shard = await this.getShard(shardId);
    const migrationData = await this.getMigrationData(shardId);

    for (const [key, value] of migrationData) {
      await this.moveToShard(targetShardId, key, value);
    }

    // 清理原分片
    await shard.fetch(new Request('https://internal/api/clear'));

    console.log(`Shard ${shardId} merged into ${targetShardId}`);
  }

  // 获取迁移数据
  async getMigrationData(shardId) {
    const shard = await this.getShard(shardId);
    const response = await shard.fetch(new Request('https://internal/api/migration-data'));
    return await response.json();
  }

  // 移动数据到分片
  async moveToShard(shardId, key, value) {
    const shard = await this.getShard(shardId);
    await shard.fetch(new Request('https://internal/api/move', {
      method: 'POST',
      body: JSON.stringify({ key: key, value: value })
    }));
  }

  // 查找合并目标
  findMergeTarget(shardId) {
    // 简化实现：找到负载最低的分片
    let targetShardId = null;
    let minLoad = Infinity;

    for (let i = 0; i < this.shardCount; i++) {
      if (i !== shardId) {
        const shard = this.shards.get(i);
        if (shard && shard.load < minLoad) {
          minLoad = shard.load;
          targetShardId = i;
        }
      }
    }

    return targetShardId;
  }
}

// 分片对象
export class ShardedObject {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.load = 0;
  }

  async fetch(request) {
    const url = new URL(request.url);
    const operation = url.pathname.split('/').pop();

    switch (operation) {
      case 'api':
        return this.handleAPI(request);
      case 'stats':
        return this.handleStats();
      case 'migration-data':
        return this.handleMigrationData();
      case 'delete':
        return this.handleDelete(request);
      case 'clear':
        return this.handleClear();
      case 'move':
        return this.handleMove(request);
      default:
        return new Response('Not Found', { status: 404 });
    }
  }

  async handleAPI(request) {
    const { operation, key, args } = await request.json();

    switch (operation) {
      case 'get':
        return this.get(key);
      case 'set':
        return this.set(key, ...args);
      case 'delete':
        return this.delete(key);
      case 'exists':
        return this.exists(key);
      default:
        return new Response('Invalid operation', { status: 400 });
    }
  }

  async handleStats() {
    const allKeys = await this.state.storage.list();
    const keyCount = allKeys.keys.length;

    let totalSize = 0;
    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value) {
        totalSize += JSON.stringify(value).length;
      }
    }

    return Response.json({
      load: this.load,
      keyCount: keyCount,
      memoryUsage: totalSize
    });
  }

  async handleMigrationData() {
    const allKeys = await this.state.storage.list();
    const data = new Map();

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value) {
        data.set(key.name, value);
      }
    }

    return Response.json(Array.from(data.entries()));
  }

  async handleDelete(request) {
    const { key } = await request.json();
    await this.state.storage.delete(key);
    this.load = Math.max(0, this.load - 1);
    return Response.json({ success: true });
  }

  async handleClear() {
    await this.state.storage.list().then(keys =>
      Promise.all(keys.keys.map(key => this.state.storage.delete(key.name)))
    );
    this.load = 0;
    return Response.json({ success: true });
  }

  async handleMove(request) {
    const { key, value } = await request.json();
    await this.state.storage.put(key, value);
    this.load++;
    return Response.json({ success: true });
  }

  async get(key) {
    const value = await this.state.storage.get(key);
    return Response.json({ value });
  }

  async set(key, value) {
    await this.state.storage.put(key, value);
    this.load++;
    return Response.json({ success: true });
  }

  async delete(key) {
    await this.state.storage.delete(key);
    this.load = Math.max(0, this.load - 1);
    return Response.json({ success: true });
  }

  async exists(key) {
    const exists = await this.state.storage.get(key) !== null;
    return Response.json({ exists });
  }
}
```

```javascript
// 垂直分片管理器
export class VerticalShardManager {
  constructor(env) {
    this.env = env;
    this.shards = {
      user: new UserShard(env),
      content: new ContentShard(env),
      analytics: new AnalyticsShard(env),
      config: new ConfigShard(env)
    };
  }

  // 获取用户分片
  getUserShard() {
    return this.shards.user;
  }

  // 获取内容分片
  getContentShard() {
    return this.shards.content;
  }

  // 获取分析分片
  getAnalyticsShard() {
    return this.shards.analytics;
  }

  // 获取配置分片
  getConfigShard() {
    return this.shards.config;
  }

  // 跨分片事务
  async executeTransaction(operations) {
    const results = [];

    try {
      // 按分片分组操作
      const shardOperations = this.groupOperationsByShard(operations);

      // 执行各分片操作
      for (const [shardName, ops] of shardOperations) {
        const shard = this.shards[shardName];
        const shardResults = await shard.executeBatch(ops);
        results.push(...shardResults);
      }

      return { success: true, results: results };

    } catch (error) {
      // 简单的回滚策略
      console.error('Transaction failed:', error);
      return { success: false, error: error.message };
    }
  }

  // 按分片分组操作
  groupOperationsByShard(operations) {
    const groups = {};

    for (const op of operations) {
      const shardName = this.determineShard(op);
      if (!groups[shardName]) {
        groups[shardName] = [];
      }
      groups[shardName].push(op);
    }

    return groups;
  }

  // 确定分片
  determineShard(operation) {
    switch (operation.type) {
      case 'user':
      case 'profile':
      case 'preferences':
        return 'user';
      case 'content':
      case 'post':
      case 'comment':
        return 'content';
      case 'analytics':
      case 'metric':
      case 'stat':
        return 'analytics';
      case 'config':
      case 'setting':
        return 'config';
      default:
        return 'user'; // 默认分片
    }
  }
}
```

## 缓存模式

### 1. 多级缓存
```javascript
// 多级缓存管理器
export class MultiLevelCacheManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.l1Cache = new Map(); // 内存缓存
    this.l2Cache = env.CACHE_KV; // KV 缓存
    this.l3Cache = env.CACHE_R2; // R2 缓存
  }

  // 获取缓存值
  async get(key, options = {}) {
    const {
      useL1 = true,
      useL2 = true,
      useL3 = true,
      refreshThreshold = 0.8
    } = options;

    // L1 缓存
    if (useL1) {
      const l1Value = this.getL1(key);
      if (l1Value && this.isFresh(l1Value, refreshThreshold)) {
        return l1Value.value;
      }
    }

    // L2 缓存
    if (useL2) {
      const l2Value = await this.getL2(key);
      if (l2Value && this.isFresh(l2Value, refreshThreshold)) {
        // 回填 L1
        if (useL1) {
          this.setL1(key, l2Value);
        }
        return l2Value.value;
      }
    }

    // L3 缓存
    if (useL3) {
      const l3Value = await this.getL3(key);
      if (l3Value && this.isFresh(l3Value, refreshThreshold)) {
        // 回填 L2 和 L1
        if (useL2) {
          await this.setL2(key, l3Value);
        }
        if (useL1) {
          this.setL1(key, l3Value);
        }
        return l3Value.value;
      }
    }

    return null;
  }

  // 设置缓存值
  async set(key, value, options = {}) {
    const {
      useL1 = true,
      useL2 = true,
      useL3 = false,
      ttl = 3600 // 1小时
    } = options;

    const cacheEntry = {
      value: value,
      timestamp: Date.now(),
      ttl: ttl * 1000
    };

    // 设置到所有指定级别
    if (useL1) {
      this.setL1(key, cacheEntry);
    }

    if (useL2) {
      await this.setL2(key, cacheEntry, ttl);
    }

    if (useL3) {
      await this.setL3(key, cacheEntry, ttl);
    }
  }

  // L1 缓存操作
  getL1(key) {
    return this.l1Cache.get(key);
  }

  setL1(key, value) {
    // LRU 淘汰
    if (this.l1Cache.size >= 1000) {
      const firstKey = this.l1Cache.keys().next().value;
      this.l1Cache.delete(firstKey);
    }

    this.l1Cache.set(key, value);
  }

  deleteL1(key) {
    return this.l1Cache.delete(key);
  }

  // L2 缓存操作
  async getL2(key) {
    try {
      const value = await this.l2Cache.get(key);
      return value ? JSON.parse(value) : null;
    } catch (error) {
      console.error('L2 cache get error:', error);
      return null;
    }
  }

  async setL2(key, value, ttl) {
    try {
      await this.l2Cache.put(
        key,
        JSON.stringify(value),
        { expirationTtl: ttl }
      );
    } catch (error) {
      console.error('L2 cache set error:', error);
    }
  }

  async deleteL2(key) {
    try {
      await this.l2Cache.delete(key);
    } catch (error) {
      console.error('L2 cache delete error:', error);
    }
  }

  // L3 缓存操作
  async getL3(key) {
    try {
      const object = await this.l3Cache.get(`cache/${key}`);
      if (object) {
        const text = await object.text();
        return JSON.parse(text);
      }
      return null;
    } catch (error) {
      console.error('L3 cache get error:', error);
      return null;
    }
  }

  async setL3(key, value, ttl) {
    try {
      const json = JSON.stringify(value);
      await this.l3Cache.put(`cache/${key}`, json, {
        customMetadata: { ttl: ttl.toString() }
      });
    } catch (error) {
      console.error('L3 cache set error:', error);
    }
  }

  async deleteL3(key) {
    try {
      await this.l3Cache.delete(`cache/${key}`);
    } catch (error) {
      console.error('L3 cache delete error:', error);
    }
  }

  // 检查是否新鲜
  isFresh(cacheEntry, threshold = 0.8) {
    const age = Date.now() - cacheEntry.timestamp;
    const maxAge = cacheEntry.ttl;
    return (age / maxAge) < threshold;
  }

  // 预热缓存
  async warmUp(keys) {
    const promises = keys.map(key => this.get(key));
    await Promise.all(promises);
  }

  // 清理缓存
  async cleanup() {
    // 清理过期的 L1 缓存
    for (const [key, value] of this.l1Cache) {
      if (!this.isFresh(value, 1)) {
        this.l1Cache.delete(key);
      }
    }

    // L2 和 L3 缓存会自动过期
  }

  // 获取缓存统计
  async getStats() {
    const l1Size = this.l1Cache.size;
    const l2Size = await this.getL2Size();
    const l3Size = await this.getL3Size();

    return {
      l1: { size: l1Size, max: 1000 },
      l2: { size: l2Size },
      l3: { size: l3Size }
    };
  }

  async getL2Size() {
    const list = await this.l2Cache.list();
    return list.keys.length;
  }

  async getL3Size() {
    const list = await this.l3Cache.list({ prefix: 'cache/' });
    return list.keys.length;
  }
}
```

## 批处理模式

### 1. 批量操作处理器
```javascript
// 批量操作处理器
export class BatchProcessor {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.batchQueue = [];
    this.batchSize = 100;
    this.flushInterval = 5000; // 5秒
    this.maxWaitTime = 30000; // 30秒
  }

  // 添加操作到批次
  async add(operation) {
    operation.id = crypto.randomUUID();
    operation.addedAt = Date.now();

    this.batchQueue.push(operation);

    // 立即检查是否需要刷新
    if (this.batchQueue.length >= this.batchSize) {
      await this.flush();
    } else if (!this.flushTimer) {
      // 设置延迟刷新
      this.flushTimer = setTimeout(() => this.flush(), this.flushInterval);
    }

    return operation.id;
  }

  // 刷新批次
  async flush() {
    if (this.batchQueue.length === 0) {
      return;
    }

    // 清除定时器
    if (this.flushTimer) {
      clearTimeout(this.flushTimer);
      this.flushTimer = null;
    }

    // 获取批次
    const batch = this.batchQueue.splice(0, this.batchSize);

    try {
      // 按类型分组
      const groups = this.groupByType(batch);

      // 执行各类型批次
      const results = [];
      for (const [type, operations] of groups) {
        const typeResults = await this.executeBatch(type, operations);
        results.push(...typeResults);
      }

      return results;

    } catch (error) {
      // 失败的操作重新加入队列
      this.batchQueue.unshift(...batch);
      throw error;
    }
  }

  // 按类型分组
  groupByType(operations) {
    const groups = new Map();

    for (const op of operations) {
      if (!groups.has(op.type)) {
        groups.set(op.type, []);
      }
      groups.get(op.type).push(op);
    }

    return groups;
  }

  // 执行批次
  async executeBatch(type, operations) {
    switch (type) {
      case 'put':
        return await this.executePutBatch(operations);
      case 'delete':
        return await this.executeDeleteBatch(operations);
      case 'update':
        return await this.executeUpdateBatch(operations);
      case 'query':
        return await this.executeQueryBatch(operations);
      default:
        throw new Error(`Unknown batch type: ${type}`);
    }
  }

  // 执行写入批次
  async executePutBatch(operations) {
    const results = [];

    await this.state.storage.transaction(async (txn) => {
      for (const op of operations) {
        try {
          await txn.put(op.key, op.value, op.options);
          results.push({
            id: op.id,
            success: true,
            key: op.key
          });
        } catch (error) {
          results.push({
            id: op.id,
            success: false,
            error: error.message
          });
        }
      }
    });

    return results;
  }

  // 执行删除批次
  async executeDeleteBatch(operations) {
    const results = [];

    await this.state.storage.transaction(async (txn) => {
      for (const op of operations) {
        try {
          await txn.delete(op.key);
          results.push({
            id: op.id,
            success: true,
            key: op.key
          });
        } catch (error) {
          results.push({
            id: op.id,
            success: false,
            error: error.message
          });
        }
      }
    });

    return results;
  }

  // 执行更新批次
  async executeUpdateBatch(operations) {
    const results = [];

    await this.state.storage.transaction(async (txn) => {
      for (const op of operations) {
        try {
          // 获取现有值
          const existing = await txn.get(op.key);
          if (existing) {
            // 应用更新
            const updated = { ...existing, ...op.updates };
            await txn.put(op.key, updated);
            results.push({
              id: op.id,
              success: true,
              key: op.key,
              updated: updated
            });
          } else {
            results.push({
              id: op.id,
              success: false,
              error: 'Key not found'
            });
          }
        } catch (error) {
          results.push({
            id: op.id,
            success: false,
            error: error.message
          });
        }
      }
    });

    return results;
  }

  // 执行查询批次
  async executeQueryBatch(operations) {
    const results = [];

    for (const op of operations) {
      try {
        const value = await this.state.storage.get(op.key, op.options);
        results.push({
          id: op.id,
          success: true,
          key: op.key,
          value: value
        });
      } catch (error) {
        results.push({
          id: op.id,
          success: false,
          error: error.message
        });
      }
    }

    return results;
  }

  // 检查超时的操作
  async checkTimeouts() {
    const now = Date.now();
    const expired = [];

    for (let i = 0; i < this.batchQueue.length; i++) {
      const op = this.batchQueue[i];
      if (now - op.addedAt > this.maxWaitTime) {
        expired.push(this.batchQueue.splice(i, 1)[0]);
        i--; // 调整索引
      }
    }

    // 强制刷新过期的操作
    if (expired.length > 0) {
      console.warn(`Processing ${expired.length} expired operations`);
      const batch = this.batchQueue.splice(0, expired.length);
      const results = await this.executeBatch('expired', expired);
      return results;
    }

    return [];
  }

  // 获取批次状态
  getStatus() {
    return {
      queueSize: this.batchQueue.length,
      batchSize: this.batchSize,
      flushInterval: this.flushInterval,
      maxWaitTime: this.maxWaitTime
    };
  }

  // 配置批次参数
  configure(config) {
    if (config.batchSize) {
      this.batchSize = config.batchSize;
    }
    if (config.flushInterval) {
      this.flushInterval = config.flushInterval;
    }
    if (config.maxWaitTime) {
      this.maxWaitTime = config.maxWaitTime;
    }
  }
}
```

```javascript
// Durable Object 装饰器基类
export class DurableObjectDecorator {
  constructor(target) {
    this.target = target;
  }

  async fetch(request) {
    // 前置处理
    const processedRequest = await this.before(request);

    // 执行目标
    const response = await this.target.fetch(processedRequest);

    // 后置处理
    const processedResponse = await this.after(response, request);

    return processedResponse;
  }

  async before(request) {
    return request;
  }

  async after(response, originalRequest) {
    return response;
  }
}

// 认证装饰器
export class AuthenticationDecorator extends DurableObjectDecorator {
  async before(request) {
    const url = new URL(request.url);

    // 跳过健康检查等路由
    if (url.pathname === '/health' || url.pathname === '/metrics') {
      return request;
    }

    // 验证认证头
    const authHeader = request.headers.get('Authorization');
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return new Response('Unauthorized', { status: 401 });
    }

    const token = authHeader.substring(7);
    const payload = await this.validateToken(token);

    // 将用户信息添加到请求
    const newRequest = new Request(request, {
      headers: {
        ...Object.fromEntries(request.headers),
        'X-User-ID': payload.userId,
        'X-User-Role': payload.role
      }
    });

    return newRequest;
  }

  async validateToken(token) {
    // 实际的 token 验证逻辑
    try {
      const parts = token.split('.');
      const payload = JSON.parse(atob(parts[1]));
      return payload;
    } catch (error) {
      throw new Error('Invalid token');
    }
  }
}

// 限流装饰器
export class RateLimitDecorator extends DurableObjectDecorator {
  constructor(target, options = {}) {
    super(target);
    this.windowMs = options.windowMs || 60000; // 1分钟
    this.maxRequests = options.maxRequests || 100;
    this.requests = new Map();
  }

  async before(request) {
    const clientId = this.getClientId(request);
    const now = Date.now();

    // 获取客户端请求记录
    let clientRequests = this.requests.get(clientId) || [];

    // 清理过期的请求记录
    clientRequests = clientRequests.filter(time => now - time < this.windowMs);

    // 检查是否超过限制
    if (clientRequests.length >= this.maxRequests) {
      return new Response('Too Many Requests', {
        status: 429,
        headers: {
          'Retry-After': Math.ceil(this.windowMs / 1000).toString()
        }
      });
    }

    // 记录当前请求
    clientRequests.push(now);
    this.requests.set(clientId, clientRequests);

    return request;
  }

  getClientId(request) {
    // 使用 IP 或用户 ID 作为客户端标识
    return request.headers.get('CF-Connecting-IP') ||
           request.headers.get('X-User-ID') ||
           'anonymous';
  }
}

// 缓存装饰器
export class CacheDecorator extends DurableObjectDecorator {
  constructor(target, kvNamespace) {
    super(target);
    this.cache = kvNamespace;
  }

  async before(request) {
    // 只缓存 GET 请求
    if (request.method !== 'GET') {
      return request;
    }

    const cacheKey = this.getCacheKey(request);
    const cached = await this.cache.get(cacheKey);

    if (cached) {
      const response = new Response(cached.body, {
        status: cached.status,
        headers: cached.headers
      });
      response.headers.set('X-Cache', 'HIT');
      return response;
    }

    // 添加缓存标记
    const newRequest = new Request(request, {
      headers: {
        ...Object.fromEntries(request.headers),
        'X-Cache-Key': cacheKey
      }
    });

    return newRequest;
  }

  async after(response, originalRequest) {
    // 只缓存成功的 GET 请求
    if (originalRequest.method !== 'GET' || response.status !== 200) {
      return response;
    }

    const cacheKey = originalRequest.headers.get('X-Cache-Key');
    if (cacheKey) {
      const body = await response.text();
      const cachedResponse = {
        body: body,
        status: response.status,
        headers: Object.fromEntries(response.headers)
      };

      await this.cache.put(cacheKey, JSON.stringify(cachedResponse), {
        expirationTtl: 300 // 5分钟
      });

      // 重建响应
      const newResponse = new Response(body, {
        status: response.status,
        headers: response.headers
      });
      newResponse.headers.set('X-Cache', 'MISS');
      return newResponse;
    }

    return response;
  }

  getCacheKey(request) {
    const url = new URL(request.url);
    return `cache:${url.pathname}${url.search}`;
  }
}

// 日志装饰器
export class LoggingDecorator extends DurableObjectDecorator {
  async before(request) {
    const start = performance.now();

    // 记录请求日志
    console.log(`[${new Date().toISOString()}] ${request.method} ${request.url}`, {
      headers: Object.fromEntries(request.headers),
      cf: request.cf
    });

    // 添加开始时间到请求
    return new Request(request, {
      headers: {
        ...Object.fromEntries(request.headers),
        'X-Start-Time': start.toString()
      }
    });
  }

  async after(response, originalRequest) {
    const startTime = parseFloat(originalRequest.headers.get('X-Start-Time') || '0');
    const duration = performance.now() - startTime;

    // 记录响应日志
    console.log(`[${new Date().toISOString()}] Response ${response.status}`, {
      duration: `${duration.toFixed(2)}ms`,
      size: response.headers.get('Content-Length') || 'unknown'
    });

    return response;
  }
}

// 装饰器工厂
export class DurableObjectFactory {
  static create(baseClass, decorators = []) {
    let decorated = baseClass;

    // 应用装饰器（反向应用，确保正确的执行顺序）
    for (const decorator of decorators.reverse()) {
      decorated = this.applyDecorator(decorated, decorator);
    }

    return decorated;
  }

  static applyDecorator(BaseClass, decoratorClass) {
    return class extends BaseClass {
      constructor(state, env) {
        super(state, env);
        this.decorator = new decoratorClass(this);
      }

      async fetch(request) {
        return await this.decorator.fetch(request);
      }
    };
  }
}

// 使用示例
export class MyDurableObject {
  constructor(state, env) {
    this.state = state;
    this.env = env;
  }

  async fetch(request) {
    const url = new URL(request.url);

    switch (url.pathname) {
      case '/data':
        return this.handleData(request);
      case '/health':
        return Response.json({ status: 'ok' });
      default:
        return new Response('Not Found', { status: 404 });
    }
  }

  async handleData(request) {
    const data = await request.json();
    // 处理数据
    return Response.json({ processed: true, data: data });
  }
}

  ]
```

## 最佳实践

### 1. 选择扩展模式
- **分片模式**: 适用于大规模数据和负载分散
- **缓存模式**: 适用于提高读取性能
- **批处理模式**: 适用于提高写入吞吐量
- **组合模式**: 适用于添加横切关注点

### 2. 设计原则
- **单一职责**: 每个模式专注解决特定问题
- **可组合性**: 模式应该可以组合使用
- **性能考虑**: 避免过度设计
- **可维护性**: 保持代码清晰和可测试

### 3. 实施建议
- **渐进式采用**: 从简单开始，逐步添加复杂性
- **监控指标**: 跟踪扩展模式的性能影响
- **测试策略**: 充分测试扩展场景
- **回滚计划**: 准备降级策略
