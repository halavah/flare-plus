# Durable Objects 实时协作

## 实时协作概述

Durable Objects 的强一致性特性和 WebSocket 支持使其成为构建实时协作应用的理想选择。本章节将详细介绍如何使用 Durable Objects 实现各种实时协作功能。

## 协作基础架构

### 1. 协作空间管理
```javascript
// 协作空间管理器
export class CollaborationSpace {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.participants = new Map();
    this.operations = [];
    this.document = null;
    this.version = 0;
    this.lastActivity = Date.now();
  }

  // 初始化协作空间
  async initialize(spaceId, options = {}) {
    this.spaceId = spaceId;
    this.options = {
      maxParticipants: options.maxParticipants || 50,
      autoSave: options.autoSave !== false,
      conflictResolution: options.conflictResolution || 'last-writer-wins',
      ...options
    };

    // 加载或创建文档
    this.document = await this.loadDocument();

    // 设置定期保存
    if (this.options.autoSave) {
      this.scheduleAutoSave();
    }

    console.log(`Collaboration space ${spaceId} initialized`);
  }

  // 加入协作
  async join(participantInfo) {
    const participantId = participantInfo.id || crypto.randomUUID();

    // 检查人数限制
    if (this.participants.size >= this.options.maxParticipants) {
      throw new Error('Collaboration space is full');
    }

    // 创建参与者对象
    const participant = {
      id: participantId,
      name: participantInfo.name || 'Anonymous',
      role: participantInfo.role || 'editor',
      cursor: { line: 0, column: 0 },
      selection: null,
      color: this.generateParticipantColor(),
      joinedAt: Date.now(),
      lastActivity: Date.now(),
      permissions: this.getPermissions(participantInfo.role)
    };

    // 注册参与者
    this.participants.set(participantId, participant);

    // 更新最后活动时间
    this.updateActivity();

    // 广播加入事件
    await this.broadcastEvent({
      type: 'participant_joined',
      participant: this.sanitizeParticipant(participant),
      timestamp: Date.now()
    });

    // 发送当前状态给新参与者
    await this.sendInitialState(participantId);

    console.log(`Participant ${participantId} joined collaboration space`);

    return {
      participantId: participantId,
      participant: participant,
      document: this.document,
      participants: Array.from(this.participants.values()).map(p => this.sanitizeParticipant(p)),
      version: this.version
    };
  }

  // 离开协作
  async leave(participantId) {
    const participant = this.participants.get(participantId);
    if (!participant) {
      return false;
    }

    // 移除参与者
    this.participants.delete(participantId);

    // 更新最后活动时间
    this.updateActivity();

    // 广播离开事件
    await this.broadcastEvent({
      type: 'participant_left',
      participantId: participantId,
      timestamp: Date.now()
    }, participantId);

    // 如果没有参与者了，保存文档
    if (this.participants.size === 0) {
      await this.saveDocument();
    }

    console.log(`Participant ${participantId} left collaboration space`);

    return true;
  }

  // 应用操作
  async applyOperation(operation, authorId) {
    const participant = this.participants.get(authorId);
    if (!participant || !participant.permissions.canEdit) {
      throw new Error('No permission to edit');
    }

    // 验证操作
    if (!this.validateOperation(operation)) {
      throw new Error('Invalid operation');
    }

    // 版本检查
    if (operation.version !== this.version) {
      // 版本冲突，需要解决
      operation = await this.resolveConflict(operation);
    }

    // 应用操作到文档
    const result = await this.applyToDocument(operation);

    // 更新版本
    this.version++;

    // 记录操作历史
    this.operations.push({
      ...operation,
      authorId: authorId,
      timestamp: Date.now(),
      version: this.version
    });

    // 限制操作历史
    if (this.operations.length > 1000) {
      this.operations = this.operations.slice(-500);
    }

    // 更新最后活动时间
    this.updateActivity();

    // 广播操作
    await this.broadcastEvent({
      type: 'operation_applied',
      operation: operation,
      result: result,
      authorId: authorId,
      version: this.version,
      timestamp: Date.now()
    });

    return result;
  }

  // 更新光标位置
  async updateCursor(participantId, cursor, selection) {
    const participant = this.participants.get(participantId);
    if (!participant) {
      return false;
    }

    participant.cursor = cursor;
    participant.selection = selection;
    participant.lastActivity = Date.now();

    // 广播光标更新
    await this.broadcastEvent({
      type: 'cursor_updated',
      participantId: participantId,
      cursor: cursor,
      selection: selection,
      timestamp: Date.now()
    }, participantId);

    return true;
  }

  // 广播事件
  async broadcastEvent(event, excludeParticipantId = null) {
    const eventString = JSON.stringify(event);

    for (const [participantId, participant] of this.participants) {
      if (participantId !== excludeParticipantId && participant.websocket) {
        try {
          participant.websocket.send(eventString);
        } catch (error) {
          console.error(`Failed to send event to ${participantId}:`, error);
          // 可以考虑移除无效连接
        }
      }
    }
  }

  // 发送初始状态
  async sendInitialState(participantId) {
    const participant = this.participants.get(participantId);
    if (!participant || !participant.websocket) {
      return;
    }

    const initialState = {
      type: 'initial_state',
      document: this.document,
      version: this.version,
      participants: Array.from(this.participants.values())
        .map(p => this.sanitizeParticipant(p)),
      operations: this.operations.slice(-100), // 最近100个操作
      timestamp: Date.now()
    };

    try {
      participant.websocket.send(JSON.stringify(initialState));
    } catch (error) {
      console.error(`Failed to send initial state:`, error);
    }
  }

  // 验证操作
  validateOperation(operation) {
    const requiredFields = ['type', 'position', 'content'];
    return requiredFields.every(field => operation.hasOwnProperty(field));
  }

  // 解决冲突
  async resolveConflict(operation) {
    switch (this.options.conflictResolution) {
      case 'last-writer-wins':
        // 直接应用，覆盖之前的版本
        operation.version = this.version;
        return operation;

      case 'operational-transform':
        // 使用操作转换
        return await this.transformOperation(operation);

      case 'manual':
        // 标记为需要手动解决
        operation.conflict = true;
        return operation;

      default:
        return operation;
    }
  }

  // 操作转换
  async transformOperation(operation) {
    // 找到需要转换的操作
    const conflictingOps = this.operations.filter(op =>
      op.position === operation.position && op.version > operation.version
    );

    // 应用转换
    for (const conflictingOp of conflictingOps) {
      operation = this.transform(operation, conflictingOp);
    }

    operation.version = this.version;
    return operation;
  }

  // 转换操作
  transform(op1, op2) {
    // 简单的转换逻辑
    if (op1.type === 'insert' && op2.type === 'insert') {
      if (op1.position <= op2.position) {
        op2.position += op1.content.length;
      } else {
        op1.position += op2.content.length;
      }
    } else if (op1.type === 'delete' && op2.type === 'delete') {
      if (op1.position < op2.position) {
        op2.position -= op1.length;
      } else if (op1.position > op2.position) {
        op1.position -= op2.length;
      }
    } else if (op1.type === 'insert' && op2.type === 'delete') {
      if (op1.position <= op2.position) {
        op2.position += op1.content.length;
      } else {
        op1.position -= op2.length;
      }
    }

    return op1;
  }

  // 应用到文档
  async applyToDocument(operation) {
    switch (operation.type) {
      case 'insert':
        return this.insertToDocument(operation);
      case 'delete':
        return this.deleteFromDocument(operation);
      case 'replace':
        return this.replaceInDocument(operation);
      case 'format':
        return this.formatDocument(operation);
      default:
        throw new Error(`Unknown operation type: ${operation.type}`);
    }
  }

  // 插入到文档
  insertToDocument(operation) {
    const { position, content, attributes } = operation;

    // 实际的插入逻辑
    this.document = this.document.slice(0, position) + content + this.document.slice(position);

    return {
      success: true,
      newLength: this.document.length,
      operation: operation
    };
  }

  // 从文档删除
  deleteFromDocument(operation) {
    const { position, length } = operation;

    // 实际的删除逻辑
    this.document = this.document.slice(0, position) + this.document.slice(position + length);

    return {
      success: true,
      newLength: this.document.length,
      operation: operation
    };
  }

  // 替换文档内容
  replaceInDocument(operation) {
    const { position, length, content } = operation;

    // 实际的替换逻辑
    this.document = this.document.slice(0, position) + content + this.document.slice(position + length);

    return {
      success: true,
      newLength: this.document.length,
      operation: operation
    };
  }

  // 格式化文档
  formatDocument(operation) {
    const { range, style } = operation;

    // 实际的格式化逻辑
    // 这里应该应用到富文本表示
    return {
      success: true,
      operation: operation
    };
  }

  // 更新活动时间
  updateActivity() {
    this.lastActivity = Date.now();
  }

  // 生成参与者颜色
  generateParticipantColor() {
    const colors = [
      '#FF6B6B', '#4ECDC4', '#45B7D1', '#96CEB4', '#FFEAA7',
      '#DDA0DD', '#98D8C8', '#F7DC6F', '#BB8FCE', '#85C1E9'
    ];
    const usedColors = Array.from(this.participants.values()).map(p => p.color);
    const availableColors = colors.filter(c => !usedColors.includes(c));
    return availableColors.length > 0 ? availableColors[0] : colors[Math.floor(Math.random() * colors.length)];
  }

  // 获取权限
  getPermissions(role) {
    const permissions = {
      owner: { canEdit: true, canDelete: true, canShare: true, canManage: true },
      editor: { canEdit: true, canDelete: false, canShare: true, canManage: false },
      viewer: { canEdit: false, canDelete: false, canShare: false, canManage: false }
    };

    return permissions[role] || permissions.viewer;
  }

  // 清理参与者信息
  sanitizeParticipant(participant) {
    const { websocket, ...sanitized } = participant;
    return sanitized;
  }

  // 加载文档
  async loadDocument() {
    const documentData = await this.state.storage.get('document');
    return documentData ? documentData.content : '';
  }

  // 保存文档
  async saveDocument() {
    await this.state.storage.put('document', {
      content: this.document,
      version: this.version,
      lastSaved: Date.now(),
      participants: Array.from(this.participants.keys())
    });
  }

  // 定时自动保存
  scheduleAutoSave() {
    const saveInterval = 30000; // 30秒
    this.state.storage.setAlarm(Date.now() + saveInterval);
  }

  // 定时器回调
  async alarm() {
    if (this.options.autoSave && this.participants.size > 0) {
      await this.saveDocument();
      this.scheduleAutoSave();
    }
  }
}
```

### 2. 操作协作系统
```javascript
// 操作协作系统
export class OperationalCollaboration {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.history = [];
    this.pending = new Map();
    this.vectorClock = new Map();
  }

  // 生成向量时钟
  generateVectorClock(participantId) {
    const current = this.vectorClock.get(participantId) || 0;
    this.vectorClock.set(participantId, current + 1);
    return new Map(this.vectorClock);
  }

  // 创建操作
  createOperation(type, data, authorId) {
    return {
      id: crypto.randomUUID(),
      type: type,
      data: data,
      authorId: authorId,
      timestamp: Date.now(),
      vectorClock: this.generateVectorClock(authorId),
      dependencies: []
    };
  }

  // 提交操作
  async submitOperation(operation) {
    // 检查依赖
    const missingDeps = await this.checkDependencies(operation);
    if (missingDeps.length > 0) {
      // 等待依赖操作
      operation.dependencies = missingDeps;
      this.pending.set(operation.id, operation);
      return { status: 'pending', missingDependencies: missingDeps };
    }

    // 应用操作
    const result = await this.applyOperation(operation);

    // 检查是否有等待的操作可以执行
    await this.checkPendingOperations();

    return { status: 'applied', result: result };
  }

  // 检查依赖
  async checkDependencies(operation) {
    const requiredOps = this.getRequiredOperations(operation);
    const availableOps = new Set(this.history.map(op => op.id));

    return requiredOps.filter(opId => !availableOps.has(opId));
  }

  // 获取所需操作
  getRequiredOperations(operation) {
    // 根据操作类型和数据确定依赖
    switch (operation.type) {
      case 'edit':
        return operation.data.dependencies || [];
      case 'delete':
        return [operation.data.targetId];
      case 'move':
        return [operation.data.sourceId];
      default:
        return [];
    }
  }

  // 应用操作
  async applyOperation(operation) {
    // 验证操作
    if (!await this.validateOperation(operation)) {
      throw new Error('Invalid operation');
    }

    // 转换操作
    const transformedOp = await this.transformOperation(operation);

    // 执行操作
    const result = await this.executeOperation(transformedOp);

    // 添加到历史
    this.history.push(transformedOp);

    // 限制历史大小
    if (this.history.length > 10000) {
      this.history = this.history.slice(-5000);
    }

    // 持久化
    await this.persistOperation(transformedOp);

    return result;
  }

  // 转换操作
  async transformOperation(operation) {
    // 获取冲突的操作
    const conflictingOps = this.getConflictingOperations(operation);

    // 应用转换
    let transformedOp = { ...operation };

    for (const conflictOp of conflictingOps) {
      transformedOp = this.transform(transformedOp, conflictOp);
    }

    return transformedOp;
  }

  // 获取冲突操作
  getConflictingOperations(operation) {
    return this.history.filter(op =>
      this.operationsConflict(operation, op)
    );
  }

  // 判断操作是否冲突
  operationsConflict(op1, op2) {
    // 简化的冲突检测
    if (op1.type === 'edit' && op2.type === 'edit') {
      return this.rangesOverlap(op1.data.range, op2.data.range);
    }

    if (op1.type === 'delete' && op2.type === 'edit') {
      return this.rangesOverlap(op1.data.range, op2.data.range);
    }

    return false;
  }

  // 判断范围是否重叠
  rangesOverlap(range1, range2) {
    return range1.start < range2.end && range2.start < range1.end;
  }

  // 转换两个操作
  transform(op1, op2) {
    // 根据操作类型进行转换
    if (op1.type === 'edit' && op2.type === 'edit') {
      return this.transformEditEdit(op1, op2);
    }

    if (op1.type === 'edit' && op2.type === 'delete') {
      return this.transformEditDelete(op1, op2);
    }

    return op1;
  }

  // 转换编辑-编辑操作
  transformEditEdit(op1, op2) {
    const transformed = { ...op1 };

    // 根据向量时钟确定顺序
    const op1Time = op1.vectorClock.get(op2.authorId) || 0;
    const op2Time = op2.vectorClock.get(op1.authorId) || 0;

    if (op2Time > op1Time) {
      // op2 更新，调整 op1
      if (this.rangesOverlap(op1.data.range, op2.data.range)) {
        transformed.data.range.start += op2.data.content.length - (op2.data.range.end - op2.data.range.start);
        transformed.data.range.end += op2.data.content.length - (op2.data.range.end - op2.data.range.start);
      }
    }

    return transformed;
  }

  // 转换编辑-删除操作
  transformEditDelete(op1, op2) {
    const transformed = { ...op1 };

    if (this.rangesOverlap(op1.data.range, op2.data.range)) {
      // 重叠部分被删除
      transformed.data.range = {
        start: Math.min(op1.data.range.start, op2.data.range.start),
        end: Math.min(op1.data.range.end, op2.data.range.start)
      };
      transformed.data.content = transformed.data.content.slice(
        0,
        Math.max(0, op2.data.range.start - op1.data.range.start)
      );
    } else if (op2.data.range.start <= op1.data.range.start) {
      // 删除在编辑之前
      const shift = op2.data.range.end - op2.data.range.start;
      transformed.data.range.start -= shift;
      transformed.data.range.end -= shift;
    }

    return transformed;
  }

  // 执行操作
  async executeOperation(operation) {
    // 根据操作类型执行
    switch (operation.type) {
      case 'edit':
        return await this.executeEdit(operation);
      case 'create':
        return await this.executeCreate(operation);
      case 'delete':
        return await this.executeDelete(operation);
      case 'move':
        return await this.executeMove(operation);
      default:
        throw new Error(`Unknown operation type: ${operation.type}`);
    }
  }

  // 执行编辑操作
  async executeEdit(operation) {
    const { itemId, range, content } = operation.data;

    // 获取当前内容
    const item = await this.state.storage.get(`item:${itemId}`);
    if (!item) {
      throw new Error(`Item ${itemId} not found`);
    }

    // 应用编辑
    const newContent = item.content.slice(0, range.start) + content + item.content.slice(range.end);

    // 更新内容
    item.content = newContent;
    item.lastModified = Date.now();
    item.version++;

    await this.state.storage.put(`item:${itemId}`, item);

    return {
      success: true,
      itemId: itemId,
      newContent: newContent,
      version: item.version
    };
  }

  // 执行创建操作
  async executeCreate(operation) {
    const { type, data } = operation.data;
    const itemId = crypto.randomUUID();

    const item = {
      id: itemId,
      type: type,
      data: data,
      createdBy: operation.authorId,
      createdAt: Date.now(),
      lastModified: Date.now(),
      version: 1
    };

    await this.state.storage.put(`item:${itemId}`, item);

    return {
      success: true,
      itemId: itemId,
      item: item
    };
  }

  // 执行删除操作
  async executeDelete(operation) {
    const { itemId } = operation.data;

    await this.state.storage.delete(`item:${itemId}`);

    return {
      success: true,
      itemId: itemId,
      deleted: true
    };
  }

  // 执行移动操作
  async executeMove(operation) {
    const { itemId, newParent, newIndex } = operation.data;

    const item = await this.state.storage.get(`item:${itemId}`);
    if (!item) {
      throw new Error(`Item ${itemId} not found`);
    }

    // 更新位置
    item.parentId = newParent;
    item.index = newIndex;
    item.lastModified = Date.now();
    item.version++;

    await this.state.storage.put(`item:${itemId}`, item);

    return {
      success: true,
      itemId: itemId,
      newParent: newParent,
      newIndex: newIndex
    };
  }

  // 验证操作
  async validateOperation(operation) {
    // 检查必要字段
    const requiredFields = ['id', 'type', 'authorId', 'data', 'timestamp'];
    if (!requiredFields.every(field => operation.hasOwnProperty(field))) {
      return false;
    }

    // 检查权限
    const hasPermission = await this.checkPermission(operation.authorId, operation.type);
    if (!hasPermission) {
      return false;
    }

    return true;
  }

  // 检查权限
  async checkPermission(authorId, operationType) {
    // 获取参与者信息
    const participant = await this.state.storage.get(`participant:${authorId}`);
    if (!participant) {
      return false;
    }

    // 检查操作权限
    const permissions = {
      edit: ['owner', 'editor'],
      create: ['owner', 'editor'],
      delete: ['owner'],
      move: ['owner', 'editor']
    };

    return permissions[operationType].includes(participant.role);
  }

  // 持久化操作
  async persistOperation(operation) {
    const key = `operation:${operation.id}`;
    await this.state.storage.put(key, operation);

    // 更新操作索引
    const index = await this.state.storage.get('operation_index') || [];
    index.push(operation.id);

    // 限制索引大小
    if (index.length > 1000) {
      const toRemove = index.splice(0, 500);
      for (const opId of toRemove) {
        await this.state.storage.delete(`operation:${opId}`);
      }
    }

    await this.state.storage.put('operation_index', index);
  }

  // 检查待处理操作
  async checkPendingOperations() {
    const readyOps = [];

    for (const [opId, operation] of this.pending) {
      const missingDeps = await this.checkDependencies(operation);
      if (missingDeps.length === 0) {
        readyOps.push(operation);
        this.pending.delete(opId);
      }
    }

    // 执行准备好的操作
    for (const operation of readyOps) {
      await this.applyOperation(operation);
    }

    return readyOps.length;
  }

  // 同步状态
  async syncState(participantId, lastKnownVersion) {
    // 获取需要同步的操作
    const missingOps = this.history.filter(op =>
      op.timestamp > lastKnownVersion
    );

    return {
      operations: missingOps,
      currentVersion: Date.now(),
      vectorClock: new Map(this.vectorClock)
    };
  }
}
```

```javascript
// 实时文档编辑器
export class RealTimeEditor {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.document = new DocumentState(state);
    this.cursors = new Map();
    this.selections = new Map();
  }

  // 文档状态类
  class DocumentState {
    constructor(state) {
      this.state = state;
      this.content = '';
      this.version = 0;
      this.operations = [];
    }

    async load() {
      const doc = await this.state.storage.get('document');
      if (doc) {
        this.content = doc.content;
        this.version = doc.version;
        this.operations = doc.operations || [];
      }
    }

    async save() {
      await this.state.storage.put('document', {
        content: this.content,
        version: this.version,
        operations: this.operations,
        lastSaved: Date.now()
      });
    }

    insert(position, text, authorId) {
      this.content = this.content.slice(0, position) + text + this.content.slice(position);
      this.version++;

      const operation = {
        type: 'insert',
        position: position,
        text: text,
        authorId: authorId,
        version: this.version,
        timestamp: Date.now()
      };

      this.operations.push(operation);
      return operation;
    }

    delete(position, length, authorId) {
      const deleted = this.content.slice(position, position + length);
      this.content = this.content.slice(0, position) + this.content.slice(position + length);
      this.version++;

      const operation = {
        type: 'delete',
        position: position,
        length: length,
        text: deleted,
        authorId: authorId,
        version: this.version,
        timestamp: Date.now()
      };

      this.operations.push(operation);
      return operation;
    }

    replace(position, length, text, authorId) {
      const oldText = this.content.slice(position, position + length);
      this.content = this.content.slice(0, position) + text + this.content.slice(position + length);
      this.version++;

      const operation = {
        type: 'replace',
        position: position,
        length: length,
        oldText: oldText,
        newText: text,
        authorId: authorId,
        version: this.version,
        timestamp: Date.now()
      };

      this.operations.push(operation);
      return operation;
    }
  }

  // 初始化编辑器
  async initialize() {
    await this.document.load();
  }

  // 应用编辑操作
  async applyEdit(operation, participantId) {
    // 转换操作
    const transformedOp = await this.transformOperation(operation);

    // 应用到文档
    let result;
    switch (transformedOp.type) {
      case 'insert':
        result = this.document.insert(
          transformedOp.position,
          transformedOp.text,
          participantId
        );
        break;
      case 'delete':
        result = this.document.delete(
          transformedOp.position,
          transformedOp.length,
          participantId
        );
        break;
      case 'replace':
        result = this.document.replace(
          transformedOp.position,
          transformedOp.length,
          transformedOp.text,
          participantId
        );
        break;
    }

    // 保存文档
    await this.document.save();

    // 广播变更
    await this.broadcastChange(result);

    return result;
  }

  // 转换操作
  async transformOperation(operation) {
    let transformed = { ...operation };

    // 获取并发操作
    const concurrentOps = this.document.operations.filter(op =>
      op.timestamp > operation.lastSyncTime && op.authorId !== operation.authorId
    );

    // 应用转换
    for (const concurrentOp of concurrentOps) {
      transformed = this.transform(transformed, concurrentOp);
    }

    return transformed;
  }

  // 转换操作
  transform(op1, op2) {
    if (op1.type === 'insert' && op2.type === 'insert') {
      return this.transformInsertInsert(op1, op2);
    } else if (op1.type === 'insert' && op2.type === 'delete') {
      return this.transformInsertDelete(op1, op2);
    } else if (op1.type === 'delete' && op2.type === 'insert') {
      return this.transformDeleteInsert(op1, op2);
    } else if (op1.type === 'delete' && op2.type === 'delete') {
      return this.transformDeleteDelete(op1, op2);
    }

    return op1;
  }

  // 转换插入-插入
  transformInsertInsert(op1, op2) {
    if (op1.position <= op2.position) {
      // op1 在 op2 之前
      op2.position += op1.text.length;
    } else {
      // op2 在 op1 之前
      op1.position += op2.text.length;
    }

    return op1;
  }

  // 转换插入-删除
  transformInsertDelete(op1, op2) {
    if (op1.position <= op2.position) {
      op2.position += op1.text.length;
    } else if (op1.position >= op2.position + op2.length) {
      op1.position -= op2.length;
    } else {
      // 在删除范围内
      op1.position = op2.position;
    }

    return op1;
  }

  // 转换删除-插入
  transformDeleteInsert(op1, op2) {
    if (op2.position <= op1.position) {
      op1.position += op2.text.length;
    } else if (op2.position >= op1.position + op1.length) {
      op2.position -= op1.length;
    }

    return op1;
  }

  // 转换删除-删除
  transformDeleteDelete(op1, op2) {
    if (op1.position + op1.length <= op2.position) {
      op2.position -= op1.length;
    } else if (op2.position + op2.length <= op1.position) {
      op1.position -= op2.length;
    } else {
      // 重叠的删除
      const start = Math.min(op1.position, op2.position);
      const end = Math.max(op1.position + op1.length, op2.position + op2.length);
      op1.position = start;
      op1.length = end - start;
    }

    return op1;
  }

  // 更新光标
  async updateCursor(participantId, position) {
    this.cursors.set(participantId, {
      position: position,
      timestamp: Date.now()
    });

    await this.broadcastCursorUpdate(participantId, position);
  }

  // 更新选择
  async updateSelection(participantId, start, end) {
    this.selections.set(participantId, {
      start: start,
      end: end,
      timestamp: Date.now()
    });

    await this.broadcastSelectionUpdate(participantId, start, end);
  }

  // 广播变更
  async broadcastChange(operation) {
    const event = {
      type: 'document_change',
      operation: operation,
      version: this.document.version,
      timestamp: Date.now()
    };

    await this.broadcast(event);
  }

  // 广播光标更新
  async broadcastCursorUpdate(participantId, position) {
    const event = {
      type: 'cursor_update',
      participantId: participantId,
      position: position,
      timestamp: Date.now()
    };

    await this.broadcast(event, participantId);
  }

  // 广播选择更新
  async broadcastSelectionUpdate(participantId, start, end) {
    const event = {
      type: 'selection_update',
      participantId: participantId,
      start: start,
      end: end,
      timestamp: Date.now()
    };

    await this.broadcast(event, participantId);
  }

  // 广播事件
  async broadcast(event, excludeParticipantId = null) {
    const eventString = JSON.stringify(event);
    const participants = await this.getActiveParticipants();

    for (const participant of participants) {
      if (participant.id !== excludeParticipantId && participant.ws.readyState === WebSocket.OPEN) {
        try {
          participant.ws.send(eventString);
        } catch (error) {
          console.error(`Failed to broadcast to ${participant.id}:`, error);
        }
      }
    }
  }

  // 获取活跃参与者
  async getActiveParticipants() {
    const participants = [];
    const keys = await this.state.storage.list({ prefix: 'participant:' });

    for (const key of keys.keys) {
      const participant = await this.state.storage.get(key.name);
      if (participant && participant.active) {
        participants.push(participant);
      }
    }

    return participants;
  }
}
```

```javascript
// 白板协作系统
export class WhiteboardCollaboration {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.elements = new Map();
    this.version = 0;
    this.viewports = new Map();
  }

  // 初始化白板
  async initialize() {
    await this.loadElements();
  }

  // 加载元素
  async loadElements() {
    const elementsData = await this.state.storage.get('whiteboard_elements');
    if (elementsData) {
      this.elements = new Map(Object.entries(elementsData.elements));
      this.version = elementsData.version;
    }
  }

  // 保存元素
  async saveElements() {
    const elementsData = {
      elements: Object.fromEntries(this.elements),
      version: this.version,
      lastSaved: Date.now()
    };

    await this.state.storage.put('whiteboard_elements', elementsData);
  }

  // 添加元素
  async addElement(elementData, authorId) {
    const element = {
      id: crypto.randomUUID(),
      type: elementData.type,
      data: elementData.data,
      position: elementData.position,
      size: elementData.size,
      style: elementData.style || {},
      authorId: authorId,
      createdAt: Date.now(),
      lastModified: Date.now(),
      version: 1
    };

    this.elements.set(element.id, element);
    this.version++;

    await this.saveElements();
    await this.broadcastElementChange('add', element);

    return element;
  }

  // 更新元素
  async updateElement(elementId, updates, authorId) {
    const element = this.elements.get(elementId);
    if (!element) {
      throw new Error(`Element ${elementId} not found`);
    }

    // 检查权限
    if (element.authorId !== authorId) {
      throw new Error('No permission to update element');
    }

    // 应用更新
    Object.assign(element, updates);
    element.lastModified = Date.now();
    element.version++;

    this.version++;

    await this.saveElements();
    await this.broadcastElementChange('update', element);

    return element;
  }

  // 删除元素
  async deleteElement(elementId, authorId) {
    const element = this.elements.get(elementId);
    if (!element) {
      throw new Error(`Element ${elementId} not found`);
    }

    // 检查权限
    if (element.authorId !== authorId) {
      throw new Error('No permission to delete element');
    }

    this.elements.delete(elementId);
    this.version++;

    await this.saveElements();
    await this.broadcastElementChange('delete', { id: elementId });

    return true;
  }

  // 移动元素
  async moveElement(elementId, newPosition, authorId) {
    const element = this.elements.get(elementId);
    if (!element) {
      throw new Error(`Element ${elementId} not found`);
    }

    // 记录移动操作
    const moveOperation = {
      type: 'move',
      elementId: elementId,
      from: element.position,
      to: newPosition,
      authorId: authorId,
      timestamp: Date.now()
    };

    // 应用移动
    element.position = newPosition;
    element.lastModified = Date.now();
    element.version++;

    this.version++;

    await this.saveElements();
    await this.broadcastElementChange('move', element, moveOperation);

    return element;
  }

  // 调整元素大小
  async resizeElement(elementId, newSize, authorId) {
    const element = this.elements.get(elementId);
    if (!element) {
      throw new Error(`Element ${elementId} not found`);
    }

    element.size = newSize;
    element.lastModified = Date.now();
    element.version++;

    this.version++;

    await this.saveElements();
    await this.broadcastElementChange('resize', element);

    return element;
  }

  // 更新视口
  async updateViewport(participantId, viewport) {
    this.viewports.set(participantId, {
      ...viewport,
      timestamp: Date.now()
    });

    await this.broadcastViewportUpdate(participantId, viewport);
  }

  // 广播元素变更
  async broadcastElementChange(type, element, operation = null) {
    const event = {
      type: 'element_change',
      changeType: type,
      element: element,
      operation: operation,
      version: this.version,
      timestamp: Date.now()
    };

    await this.broadcast(event);
  }

  // 广播视口更新
  async broadcastViewportUpdate(participantId, viewport) {
    const event = {
      type: 'viewport_update',
      participantId: participantId,
      viewport: viewport,
      timestamp: Date.now()
    };

    await this.broadcast(event, participantId);
  }

  // 同步白板
  async syncWhiteboard(lastKnownVersion) {
    const elements = Array.from(this.elements.values());
    let changes = [];

    if (lastKnownVersion < this.version) {
      // 获取变更的元素
      changes = elements.filter(el => el.version > lastKnownVersion);
    }

    return {
      elements: lastKnownVersion === 0 ? elements : changes,
      version: this.version,
      viewports: Array.from(this.viewports.entries())
    };
  }

  // 处理操作冲突
  async resolveConflict(operation1, operation2) {
    // 根据时间戳解决冲突
    if (operation1.timestamp > operation2.timestamp) {
      return operation1;
    } else {
      return operation2;
    }
  }

  // 广播事件
  async broadcast(event, excludeParticipantId = null) {
    const eventString = JSON.stringify(event);
    const participants = await this.getActiveParticipants();

    for (const participant of participants) {
      if (participant.id !== excludeParticipantId && participant.ws.readyState === WebSocket.OPEN) {
        try {
          participant.ws.send(eventString);
        } catch (error) {
          console.error(`Failed to broadcast to ${participant.id}:`, error);
        }
      }
    }
  }

    const participants = [];
    const keys = await this.state.storage.list({ prefix: 'participant:' });

    for (const key of keys.keys) {
      const participant = await this.state.storage.get(key.name);
      if (participant && participant.active) {
        participants.push(participant);
      }
    }

    return participants;
  }
}
```

## 最佳实践

### 1. 性能优化
- **批量操作**: 合并多个操作减少网络开销
- **操作压缩**: 压缩操作数据
- **增量同步**: 只同步变更部分
- **本地缓存**: 缓存频繁访问的数据

### 2. 冲突处理
- **操作转换**: 使用 OT 或 CRDT 算法
- **版本控制**: 实施版本管理
- **权限管理**: 控制编辑权限
- **冲突解决**: 提供冲突解决策略

### 3. 用户体验
- **实时反馈**: 即时显示操作结果
- **光标跟踪**: 显示其他用户的光标
- **状态指示**: 显示在线状态
