```javascript
export class LazyLoadingOptimizer {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.loadingPromises = new Map();
  }

  async get(key) {
    // 如果正在加载，返回现有 Promise
    if (this.loadingPromises.has(key)) {
      return await this.loadingPromises.get(key);
    }

    // 创建加载 Promise
    const loadingPromise = this.loadWithTimeout(key);
    this.loadingPromises.set(key, loadingPromise);

    try {
      const result = await loadingPromise;
      return result;
    } finally {
      // 清理 Promise
      this.loadingPromises.delete(key);
    }
  }

  async loadWithTimeout(key, timeout = 5000) {
    return Promise.race([
      this.state.storage.get(key),
      new Promise((_, reject) =>
        setTimeout(() => reject(new Error('Loading timeout')), timeout)
      )
    ]);
  }

  // 批量预加载
  async preload(keys) {
    const promises = keys.map(key => this.get(key));
    await Promise.all(promises);
  }
}
```

```javascript
// 解决方案1：内存池管理
export class MemoryPool {
  constructor(maxSize = 100) {
    this.maxSize = maxSize;
    this.pool = new Map();
    this.accessOrder = [];
  }

  get(key) {
    if (this.pool.has(key)) {
      // 更新访问顺序
      this.updateAccessOrder(key);
      return this.pool.get(key);
    }
    return null;
  }

  set(key, value) {
    // 如果键已存在，更新值和访问顺序
    if (this.pool.has(key)) {
      this.pool.set(key, value);
      this.updateAccessOrder(key);
      return;
    }

    // 如果达到最大大小，移除最少使用的项
    if (this.pool.size >= this.maxSize) {
      const lruKey = this.accessOrder.shift();
      this.pool.delete(lruKey);
    }

    // 添加新项
    this.pool.set(key, value);
    this.accessOrder.push(key);
  }

  updateAccessOrder(key) {
    const index = this.accessOrder.indexOf(key);
    if (index > -1) {
      this.accessOrder.splice(index, 1);
    }
    this.accessOrder.push(key);
  }

  clear() {
    this.pool.clear();
    this.accessOrder = [];
  }
}

// 解决方案2：流式处理大数据
export class StreamProcessor {
  constructor(state, env) {
    this.state = state;
    this.env = env;
  }

  async processLargeData(dataStream) {
    const reader = dataStream.getReader();
    const decoder = new TextDecoder();
    let buffer = '';
    const chunkSize = 1024 * 1024; // 1MB chunks

    while (true) {
      const { done, value } = await reader.read();

      if (done) {
        // 处理最后的数据
        if (buffer) {
          await this.processChunk(buffer);
        }
        break;
      }

      buffer += decoder.decode(value, { stream: true });

      // 处理完整的数据块
      while (buffer.length >= chunkSize) {
        const chunk = buffer.slice(0, chunkSize);
        buffer = buffer.slice(chunkSize);
        await this.processChunk(chunk);
      }
    }
  }

  async processChunk(chunk) {
    // 处理数据块
    const processedData = await this.transformChunk(chunk);

    // 分片存储
    const chunkId = crypto.randomUUID();
    await this.state.storage.put(`chunk:${chunkId}`, processedData);

    return chunkId;
  }

  async transformChunk(chunk) {
    // 数据转换逻辑
    return chunk.toUpperCase();
  }

  async reconstructFromChunks(chunkIds) {
    let result = '';

    for (const chunkId of chunkIds) {
      const chunk = await this.state.storage.get(`chunk:${chunkId}`);
      if (chunk) {
        result += chunk;
      }
    }

    return result;
  }
}
```

```javascript
// 问题：频繁的存储访问导致性能下降

// 解决方案1：本地缓存策略
export class LocalCacheStrategy {
  constructor(state) {
    this.state = state;
    this.cache = new Map();
    this.cacheTimeout = 30000; // 30秒
    this.maxCacheSize = 1000;
  }

  async get(key) {
    // 检查本地缓存
    const cached = this.cache.get(key);
    if (cached && Date.now() - cached.timestamp < this.cacheTimeout) {
      return cached.value;
    }

    // 从存储读取
    const value = await this.state.storage.get(key);

    // 更新缓存
    this.updateCache(key, value);

    return value;
  }

  async put(key, value, options = {}) {
    // 更新存储
    await this.state.storage.put(key, value, options);

    // 更新缓存
    this.updateCache(key, value);
  }

  updateCache(key, value) {
    // LRU 淘汰
    if (this.cache.size >= this.maxCacheSize) {
      const firstKey = this.cache.keys().next().value;
      this.cache.delete(firstKey);
    }

    this.cache.set(key, {
      value: value,
      timestamp: Date.now()
    });
  }

  // 批量操作优化
  async batchGet(keys) {
    const results = new Map();
    const uncachedKeys = [];

    // 先从缓存获取
    for (const key of keys) {
      const cached = this.cache.get(key);
      if (cached && Date.now() - cached.timestamp < this.cacheTimeout) {
        results.set(key, cached.value);
      } else {
        uncachedKeys.push(key);
      }
    }

    // 批量从存储获取未缓存的键
    if (uncachedKeys.length > 0) {
      const storageResults = await this.state.storage.getMany(uncachedKeys);

      for (const [key, value] of storageResults) {
        results.set(key, value);
        this.updateCache(key, value);
      }
    }

    return results;
  }
}

// 解决方案2：读写分离
export class ReadWriteSeparation {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.readReplica = env.READ_REPLICA; // 只读副本
  }

  async get(key) {
    // 优先从只读副本读取
    try {
      const value = await this.readReplica.get(key);
      return value;
    } catch (error) {
      // 副本不可用时，从主存储读取
      return await this.state.storage.get(key);
    }
  }

  async put(key, value, options = {}) {
    // 写入主存储
    await this.state.storage.put(key, value, options);

    // 异步更新副本（如果支持）
    this.updateReplicaAsync(key, value);
  }

  async updateReplicaAsync(key, value) {
    // 非关键操作，不等待完成
    try {
      await this.readReplica.put(key, value);
    } catch (error) {
      console.warn('Replica update failed:', error);
    }
  }

  // 读写密集型操作优化
  async readIntensiveOperation(keys) {
    // 并发读取
    const promises = keys.map(key => this.get(key));
    return await Promise.all(promises);
  }

  async writeIntensiveOperation(operations) {
    // 批量写入
    return await this.state.storage.transaction(async (txn) => {
      const results = [];
      for (const op of operations) {
        switch (op.type) {
          case 'put':
            await txn.put(op.key, op.value);
            results.push({ success: true, key: op.key });
            break;
          case 'delete':
            await txn.delete(op.key);
            results.push({ success: true, key: op.key });
            break;
        }
      }
      return results;
    });
  }
}
```

```javascript
// 问题：多个请求同时修改同一数据导致竞态条件

// 解决方案1：乐观锁
export class OptimisticLocking {
  constructor(state) {
    this.state = state;
  }

  async updateWithLock(key, updateFn) {
    let retryCount = 0;
    const maxRetries = 3;

    while (retryCount < maxRetries) {
      // 获取当前值和版本
      const current = await this.state.storage.get(key);
      const version = current ? current._version : 0;

      // 应用更新
      const updated = await updateFn(current || {});
      updated._version = version + 1;

      // 尝试原子更新
      const success = await this.atomicUpdate(key, updated, version);

      if (success) {
        return updated;
      }

      // 冲突，重试
      retryCount++;
      await this.backoff(retryCount);
    }

    throw new Error('Update failed after maximum retries');
  }

  async atomicUpdate(key, newValue, expectedVersion) {
    // 使用事务确保原子性
    try {
      await this.state.storage.transaction(async (txn) => {
        const current = await txn.get(key);
        const currentVersion = current ? current._version : 0;

        if (currentVersion !== expectedVersion) {
          throw new Error('Version conflict');
        }

        await txn.put(key, newValue);
      });

      return true;
    } catch (error) {
      return false;
    }
  }

  async backoff(retryCount) {
    const delay = Math.min(100 * Math.pow(2, retryCount), 1000);
    await new Promise(resolve => setTimeout(resolve, delay));
  }
}

// 解决方案2：分布式锁
export class DistributedLock {
  constructor(state) {
    this.state = state;
    this.lockTimeout = 30000; // 30秒
  }

  async acquireLock(resource, timeout = this.lockTimeout) {
    const lockKey = `lock:${resource}`;
    const lockId = crypto.randomUUID();
    const expiresAt = Date.now() + timeout;

    const lockData = {
      id: lockId,
      resource: resource,
      expiresAt: expiresAt,
      acquiredAt: Date.now()
    };

    // 尝试获取锁
    const existing = await this.state.storage.get(lockKey);
    if (existing && existing.expiresAt > Date.now()) {
      throw new Error(`Lock for ${resource} is held`);
    }

    await this.state.storage.put(lockKey, lockData, {
      expirationTtl: Math.ceil(timeout / 1000) + 1
    });

    return lockId;
  }

  async releaseLock(resource, lockId) {
    const lockKey = `lock:${resource}`;
    const existing = await this.state.storage.get(lockKey);

    if (existing && existing.id === lockId) {
      await this.state.storage.delete(lockKey);
      return true;
    }

    return false;
  }

  async withLock(resource, operation, timeout = this.lockTimeout) {
    const lockId = await this.acquireLock(resource, timeout);

    try {
      return await operation();
    } finally {
      await this.releaseLock(resource, lockId);
    }
  }

  // 自动清理过期锁
  async cleanupExpiredLocks() {
    const locks = await this.state.storage.list({ prefix: 'lock:' });
    const now = Date.now();

    for (const lock of locks.keys) {
      const lockData = await this.state.storage.get(lock.name);
      if (lockData && lockData.expiresAt < now) {
        await this.state.storage.delete(lock.name);
      }
    }
  }
}
```

### 2. 死锁问题
```javascript
// 问题：多个资源锁定顺序不一致导致死锁

// 解决方案：统一的资源锁定顺序
export class DeadlockPrevention {
  constructor(state) {
    this.state = state;
    this.lockOrder = new Map(); // 资源锁定顺序
  }

  // 定义资源锁定顺序
  defineResourceOrder(resources) {
    resources.sort(); // 按名称排序确保一致性
    resources.forEach((resource, index) => {
      this.lockOrder.set(resource, index);
    });
  }

  async acquireMultipleLocks(resources, timeout = 30000) {
    // 按预定顺序获取锁
    const sortedResources = resources.sort((a, b) => {
      const orderA = this.lockOrder.get(a) || a;
      const orderB = this.lockOrder.get(b) || b;
      return orderA < orderB ? -1 : 1;
    });

    const locks = [];
    try {
      for (const resource of sortedResources) {
        const lockId = await this.acquireLock(resource, timeout);
        locks.push({ resource, lockId });
      }
      return locks;
    } catch (error) {
      // 获取失败，释放已获取的锁
      for (const lock of locks) {
        await this.releaseLock(lock.resource, lock.lockId);
      }
      throw error;
    }
  }

  async releaseMultipleLocks(locks) {
    // 按相反顺序释放锁
    const sortedLocks = locks.sort((a, b) => {
      const orderA = this.lockOrder.get(a.resource) || a.resource;
      const orderB = this.lockOrder.get(b.resource) || b.resource;
      return orderA > orderB ? -1 : 1;
    });

    for (const lock of sortedLocks) {
      await this.releaseLock(lock.resource, lock.lockId);
    }
  }

  async withMultipleLocks(resources, operation, timeout = 30000) {
    const locks = await this.acquireMultipleLocks(resources, timeout);

    try {
      return await operation();
    } finally {
      await this.releaseMultipleLocks(locks);
    }
  }

  async acquireLock(resource, timeout) {
    const lockKey = `lock:${resource}`;
    const lockId = crypto.randomUUID();

    const lockData = {
      id: lockId,
      resource: resource,
      expiresAt: Date.now() + timeout
    };

    await this.state.storage.put(lockKey, lockData, {
      expirationTtl: Math.ceil(timeout / 1000) + 1
    });

    return lockId;
  }

  async releaseLock(resource, lockId) {
    const lockKey = `lock:${resource}`;
    const existing = await this.state.storage.get(lockKey);

    if (existing && existing.id === lockId) {
      await this.state.storage.delete(lockKey);
      return true;
    }

    return false;
  }
}
```

```javascript
// 问题：WebSocket 连接意外断开

// 解决方案：自动重连机制
export class AutoReconnect {
  constructor(ws, options = {}) {
    this.ws = ws;
    this.reconnectAttempts = 0;
    this.maxReconnectAttempts = options.maxAttempts || 5;
    this.reconnectDelay = options.delay || 1000;
    this.maxReconnectDelay = options.maxDelay || 30000;
    this.reconnectCallback = options.onReconnect || null;
  }

  async handleDisconnect() {
    if (this.reconnectAttempts < this.maxReconnectAttempts) {
      const delay = this.calculateReconnectDelay();

      console.log(`Reconnecting in ${delay}ms (attempt ${this.reconnectAttempts + 1})`);

      await this.sleep(delay);

      try {
        await this.reconnect();
        this.reconnectAttempts = 0;

        if (this.reconnectCallback) {
          await this.reconnectCallback();
        }
      } catch (error) {
        this.reconnectAttempts++;
        await this.handleDisconnect();
      }
    } else {
      console.error('Max reconnection attempts reached');
      throw new Error('Connection failed permanently');
    }
  }

  calculateReconnectDelay() {
    // 指数退避
    const delay = this.reconnectDelay * Math.pow(2, this.reconnectAttempts);
    return Math.min(delay, this.maxReconnectDelay);
  }

  async reconnect() {
    // 实现重连逻辑
    const newWs = new WebSocket(this.ws.url);

    return new Promise((resolve, reject) => {
      newWs.onopen = () => {
        this.ws = newWs;
        this.setupEventHandlers();
        resolve();
      };

      newWs.onerror = (error) => {
        reject(error);
      };
    });
  }

  setupEventHandlers() {
    this.ws.onclose = () => {
      this.handleDisconnect();
    };

    this.ws.onerror = (error) => {
      console.error('WebSocket error:', error);
    };
  }

  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

export class HeartbeatMonitor {
  constructor(ws, options = {}) {
    this.ws = ws;
    this.interval = options.interval || 30000; // 30秒
    this.timeout = options.timeout || 10000; // 10秒
    this.pingTimeout = null;
    this.pongTimeout = null;
    this.alive = true;
  }

  start() {
    this.sendPing();
  }

  stop() {
    if (this.pingTimeout) {
      clearTimeout(this.pingTimeout);
      this.pingTimeout = null;
    }
    if (this.pongTimeout) {
      clearTimeout(this.pongTimeout);
      this.pongTimeout = null;
    }
  }

  sendPing() {
    if (!this.alive) return;

    try {
      this.ws.send(JSON.stringify({ type: 'ping', timestamp: Date.now() }));

      // 设置 pong 超时
      this.pongTimeout = setTimeout(() => {
        console.log('Pong timeout, connection may be dead');
        this.alive = false;
        this.ws.close();
      }, this.timeout);
    } catch (error) {
      console.error('Ping failed:', error);
      this.alive = false;
    }

    // 设置下一次 ping
    this.pingTimeout = setTimeout(() => this.sendPing(), this.interval);
  }

  handlePong() {
    if (this.pongTimeout) {
      clearTimeout(this.pongTimeout);
      this.pongTimeout = null;
    }
    this.alive = true;
  }

  isAlive() {
    return this.alive;
  }
}
```

### 2. 存储空间不足
```javascript
// 问题：存储空间接近限制

// 解决方案：数据清理和压缩
export class StorageManager {
  constructor(state, options = {}) {
    this.state = state;
    this.maxStorage = options.maxStorage || 128 * 1024 * 1024; // 128MB
    this.warningThreshold = options.warningThreshold || 0.9; // 90%
    this.cleanupPolicy = options.cleanupPolicy || 'lru';
  }

  async checkStorageUsage() {
    const usage = await this.getStorageUsage();
    const usageRatio = usage.used / this.maxStorage;

    if (usageRatio > this.warningThreshold) {
      console.warn(`Storage usage critical: ${(usageRatio * 100).toFixed(1)}%`);
      await this.performCleanup();
    }

    return usage;
  }

  async getStorageUsage() {
    const allKeys = await this.state.storage.list();
    let totalSize = 0;
    const details = [];

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value) {
        const size = JSON.stringify(value).length;
        totalSize += size;
        details.push({ key: key.name, size });
      }
    }

    return {
      used: totalSize,
      max: this.maxStorage,
      count: allKeys.keys.length,
      details: details.sort((a, b) => b.size - a.size)
    };
  }

  async performCleanup() {
    switch (this.cleanupPolicy) {
      case 'lru':
        await this.cleanupLRU();
        break;
      case 'ttl':
        await this.cleanupTTL();
        break;
      case 'priority':
        await this.cleanupByPriority();
        break;
    }
  }

  async cleanupLRU() {
    const allKeys = await this.state.storage.list();
    const accessTimes = new Map();

    // 获取访问时间
    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value && value._accessTime) {
        accessTimes.set(key.name, value._accessTime);
      }
    }

    // 按访问时间排序
    const sortedKeys = Array.from(accessTimes.entries())
      .sort((a, b) => a[1] - b[1]);

    // 删除最旧的 20%
    const toDelete = Math.floor(sortedKeys.length * 0.2);
    for (let i = 0; i < toDelete; i++) {
      await this.state.storage.delete(sortedKeys[i][0]);
    }
  }

  async cleanupTTL() {
    const now = Date.now();
    const allKeys = await this.state.storage.list();

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value && value._expiresAt && value._expiresAt < now) {
        await this.state.storage.delete(key.name);
      }
    }
  }

  async cleanupByPriority() {
    const allKeys = await this.state.storage.list();
    const priorities = new Map();

    // 获取优先级
    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value && value._priority !== undefined) {
        priorities.set(key.name, value._priority);
      }
    }

    // 按优先级排序（低优先级先删除）
    const sortedKeys = Array.from(priorities.entries())
      .sort((a, b) => a[1] - b[1]);

    // 删除低优先级数据
    let deletedSize = 0;
    const targetSize = this.maxStorage * 0.7; // 目标使用率 70%

    for (const [key] of sortedKeys) {
      const value = await this.state.storage.get(key);
      if (value) {
        const size = JSON.stringify(value).length;
        await this.state.storage.delete(key);
        deletedSize += size;

        if (deletedSize >= (this.maxStorage - targetSize)) {
          break;
        }
      }
    }
  }

  // 数据压缩
  async compressData(key) {
    const value = await this.state.storage.get(key);
    if (!value) return;

    const originalSize = JSON.stringify(value).length;

    // 简单压缩：移除不必要的字段
    const compressed = {
      ...value,
      _compressed: true,
      _originalSize: originalSize
    };

    // 移除大字段或进行实际压缩
    if (value.largeData) {
      compressed.largeData = await this.compressString(JSON.stringify(value.largeData));
    }

    await this.state.storage.put(key, compressed);

    const newSize = JSON.stringify(compressed).length;
    console.log(`Compressed ${key}: ${originalSize} -> ${newSize} bytes`);
  }

  async compressString(str) {
    // 这里应该使用实际的压缩算法
    // 简化示例：使用 Base64 编码（实际上不压缩）
    return btoa(str);
  }

  // 自动数据老化
  async ageData(key, maxAge) {
    const value = await this.state.storage.get(key);
    if (!value) return;

    const now = Date.now();
    const age = now - (value._createdAt || now);

    if (age > maxAge) {
      // 老化数据：可以删除或降低精度
      if (age > maxAge * 2) {
        await this.state.storage.delete(key);
      } else {
        // 降低精度
        value._aged = true;
        value._age = age;
        await this.state.storage.put(key, value);
      }
    }
  }
}
```

```javascript
// 解决方案：结构化日志
export class StructuredLogger {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.logLevel = env.LOG_LEVEL || 'info';
    this.logs = [];
    this.maxLogs = 1000;
  }

  log(level, message, data = {}) {
    if (!this.shouldLog(level)) return;

    const logEntry = {
      timestamp: new Date().toISOString(),
      level: level,
      message: message,
      data: data,
      id: crypto.randomUUID()
    };

    // 添加到内存日志
    this.logs.push(logEntry);
    if (this.logs.length > this.maxLogs) {
      this.logs.shift();
    }

    // 控制台输出
    console.log(`[${level.toUpperCase()}] ${message}`, data);

    // 持久化重要日志
    if (level === 'error' || level === 'warn') {
      this.persistLog(logEntry);
    }
  }

  shouldLog(level) {
    const levels = { debug: 0, info: 1, warn: 2, error: 3 };
    return levels[level] >= levels[this.logLevel];
  }

  async persistLog(logEntry) {
    const key = `log:${logEntry.timestamp}:${logEntry.id}`;
    await this.state.storage.put(key, logEntry, {
      expirationTtl: 7 * 24 * 60 * 60 // 7天
    });
  }

  debug(message, data) {
    this.log('debug', message, data);
  }

  info(message, data) {
    this.log('info', message, data);
  }

  warn(message, data) {
    this.log('warn', message, data);
  }

  error(message, data) {
    this.log('error', message, data);
  }

  // 性能日志
  async measure(name, operation) {
    const start = performance.now();

    try {
      const result = await operation();
      const duration = performance.now() - start;

      this.info(`${name} completed`, {
        duration: duration,
        success: true
      });

      return result;
    } catch (error) {
      const duration = performance.now() - start;

      this.error(`${name} failed`, {
        duration: duration,
        error: error.message
      });

      throw error;
    }
  }

  // 获取日志历史
  async getLogs(level = null, limit = 100) {
    let logs = this.logs;

    if (level) {
      logs = logs.filter(log => log.level === level);
    }

    return logs.slice(-limit);
  }

  // 清理日志
  async cleanup() {
    this.logs = [];

    // 清理持久化日志
    const cutoffTime = Date.now() - (24 * 60 * 60 * 1000); // 24小时前
    const logKeys = await this.state.storage.list({ prefix: 'log:' });

    for (const key of logKeys.keys) {
      const timestamp = new Date(key.name.split(':')[1]).getTime();
      if (timestamp < cutoffTime) {
        await this.state.storage.delete(key.name);
      }
    }
  }
}

// 解决方案2：状态快照
export class StateSnapshot {
  constructor(state) {
    this.state = state;
  }

  async createSnapshot(label = null) {
    const snapshot = {
      id: crypto.randomUUID(),
      label: label,
      timestamp: Date.now(),
      storage: await this.dumpStorage()
    };

    const key = `snapshot:${snapshot.id}`;
    await this.state.storage.put(key, snapshot);

    // 记录快照历史
    await this.updateSnapshotHistory(snapshot);

    return snapshot.id;
  }

  async dumpStorage() {
    const allKeys = await this.state.storage.list();
    const dump = {};

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value) {
        dump[key.name] = value;
      }
    }

    return dump;
  }

  async restoreFromSnapshot(snapshotId) {
    const snapshot = await this.state.storage.get(`snapshot:${snapshotId}`);
    if (!snapshot) {
      throw new Error(`Snapshot ${snapshotId} not found`);
    }

    // 清空当前存储
    const currentKeys = await this.state.storage.list();
    for (const key of currentKeys.keys) {
      await this.state.storage.delete(key.name);
    }

    // 恢复快照数据
    for (const [key, value] of Object.entries(snapshot.storage)) {
      await this.state.storage.put(key, value);
    }

    return snapshot;
  }

  async updateSnapshotHistory(snapshot) {
    const historyKey = 'snapshot_history';
    let history = await this.state.storage.get(historyKey) || [];

    history.push({
      id: snapshot.id,
      label: snapshot.label,
      timestamp: snapshot.timestamp
    });

    // 限制历史记录数量
    if (history.length > 50) {
      history = history.slice(-25);
    }

    await this.state.storage.put(historyKey, history);
  }

  async getSnapshotHistory() {
    return await this.state.storage.get('snapshot_history') || [];
  }

  async deleteSnapshot(snapshotId) {
    await this.state.storage.delete(`snapshot:${snapshotId}`);

    // 更新历史记录
    const history = await this.getSnapshotHistory();
    const filtered = history.filter(s => s.id !== snapshotId);
    await this.state.storage.put('snapshot_history', filtered);
  }

  // 自动快照
  async autoSnapshot() {
    const label = `auto-${new Date().toISOString()}`;
    const snapshotId = await this.createSnapshot(label);

    // 清理旧快照
    await this.cleanupOldSnapshots(10); // 保留最近10个快照

    return snapshotId;
  }

  async cleanupOldSnapshots(keepCount) {
    const history = await this.getSnapshotHistory();

    if (history.length > keepCount) {
      const toDelete = history.slice(0, history.length - keepCount);

      for (const snapshot of toDelete) {
        await this.deleteSnapshot(snapshot.id);
      }
    }
  }
}
```
