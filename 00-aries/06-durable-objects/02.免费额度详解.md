```javascript
// 存储使用监控器
export class StorageMonitor {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.limits = FREE_TIER_LIMITS.storage;
  }

  // 获取存储使用情况
  async getStorageUsage() {
    const allKeys = await this.state.storage.list();
    let totalSize = 0;
    const keyDetails = [];

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name, { type: 'arrayBuffer' });
      const size = value ? value.byteLength : 0;
      totalSize += size;

      keyDetails.push({
        key: key.name,
        size: size,
        sizeHuman: this.formatBytes(size)
      });
    }

    const usage = {
      totalKeys: allKeys.keys.length,
      totalSize: totalSize,
      totalSizeHuman: this.formatBytes(totalSize),
      maxCapacity: this.parseBytes(this.limits.totalCapacity),
      usagePercentage: (totalSize / this.parseBytes(this.limits.totalCapacity)) * 100,
      keysBySize: keyDetails.sort((a, b) => b.size - a.size).slice(0, 10),
      timestamp: Date.now()
    };

    // 保存使用记录
    await this.state.storage.put('storage_usage', usage);

    return usage;
  }

  // 检查存储限制
  async checkStorageLimits() {
    const usage = await this.getStorageUsage();
    const alerts = [];

    // 检查总容量
    if (usage.usagePercentage > 90) {
      alerts.push({
        level: 'critical',
        type: 'storage_capacity',
        message: `Storage usage at ${usage.usagePercentage.toFixed(1)}% (${usage.totalSizeHuman}/${this.limits.totalCapacity})`,
        action: 'immediate_cleanup_required'
      });
    } else if (usage.usagePercentage > 75) {
      alerts.push({
        level: 'warning',
        type: 'storage_capacity',
        message: `Storage usage at ${usage.usagePercentage.toFixed(1)}% (${usage.totalSizeHuman}/${this.limits.totalCapacity})`,
        action: 'consider_cleanup'
      });
    }

    // 检查对象数量
    if (usage.totalKeys > this.limits.totalObjects * 0.9) {
      alerts.push({
        level: 'warning',
        type: 'object_count',
        message: `Object count near limit: ${usage.totalKeys}/${this.limits.totalObjects}`,
        action: 'consolidate_objects'
      });
    }

    return {
      usage: usage,
      alerts: alerts,
      withinLimits: usage.usagePercentage < 100 && usage.totalKeys < this.limits.totalObjects
    };
  }

  // 自动清理过期数据
  async cleanupExpiredData() {
    const now = Date.now();
    const allKeys = await this.state.storage.list();
    let cleanedKeys = 0;
    let freedSpace = 0;

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value && value.expiresAt && value.expiresAt < now) {
        const size = await this.getKeySize(key.name);
        await this.state.storage.delete(key.name);
        cleanedKeys++;
        freedSpace += size;
      }
    }

    return {
      cleanedKeys: cleanedKeys,
      freedSpace: freedSpace,
      freedSpaceHuman: this.formatBytes(freedSpace)
    };
  }

  // 压缩大对象
  async compressLargeObjects(threshold = 10240) { // 10KB
    const allKeys = await this.state.storage.list();
    const compressed = [];

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value && typeof value === 'object') {
        const jsonString = JSON.stringify(value);
        if (jsonString.length > threshold) {
          const compressedData = await this.compressData(jsonString);
          await this.state.storage.put(key.name, {
            compressed: true,
            originalSize: jsonString.length,
            compressedSize: compressedData.length,
            data: compressedData
          });
          compressed.push({
            key: key.name,
            originalSize: jsonString.length,
            compressedSize: compressedData.length,
            compressionRatio: ((jsonString.length - compressedData.length) / jsonString.length * 100).toFixed(1)
          });
        }
      }
    }

    return compressed;
  }

  // 格式化字节数
  formatBytes(bytes) {
    if (bytes === 0) return '0 Bytes';
    const k = 1024;
    const sizes = ['Bytes', 'KB', 'MB', 'GB'];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
  }

  // 解析字节数
  parseBytes(sizeStr) {
    const units = { 'B': 1, 'KB': 1024, 'MB': 1024*1024, 'GB': 1024*1024*1024 };
    const match = sizeStr.match(/^(\d+(?:\.\d+)?)\s*(B|KB|MB|GB)$/i);
    if (!match) return 0;
    return parseFloat(match[1]) * units[match[2].toUpperCase()];
  }

  // 获取键大小
  async getKeySize(key) {
    const value = await this.state.storage.get(key, { type: 'arrayBuffer' });
    return value ? value.byteLength : 0;
  }

  // 简单数据压缩（示例）
  async compressData(data) {
    // 这里应该使用真正的压缩算法
    // 例如: pako, lz-string 等
    return btoa(data); // Base64 编码作为示例
  }
}
```

```javascript
// 存储优化管理器
export class StorageOptimizer {
  constructor(state, env) {
    this.state = state;
    this.env = env;
  }

  // 分层存储策略
  async implementTieredStorage() {
    const hotDataMaxAge = 7 * 24 * 60 * 60 * 1000; // 7天
    const warmDataMaxAge = 30 * 24 * 60 * 60 * 1000; // 30天
    const now = Date.now();

    const allKeys = await this.state.storage.list();
    const migrationResults = {
      hotToWarm: 0,
      warmToCold: 0,
      coldDeleted: 0
    };

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value && value.lastAccessed) {
        const age = now - value.lastAccessed;

        if (age > warmDataMaxAge) {
          // 移到冷存储（删除）
          await this.state.storage.delete(key.name);
          migrationResults.coldDeleted++;
        } else if (age > hotDataMaxAge) {
          // 移到温存储（压缩）
          await this.moveToWarmStorage(key.name, value);
          migrationResults.hotToWarm++;
        }
      }
    }

    return migrationResults;
  }

  // 移到温存储
  async moveToWarmStorage(key, value) {
    const compressedValue = {
      ...value,
      storageTier: 'warm',
      compressedAt: Date.now()
    };

    // 移除访问频繁的元数据
    delete compressedValue.accessHistory;
    delete compressedValue.tempData;

    await this.state.storage.put(key, compressedValue);
  }

  // 数据去重
  async deduplicateData() {
    const allKeys = await this.state.storage.list();
    const dataHashes = new Map();
    const duplicates = [];

    for (const key of allKeys.keys) {
      const value = await this.state.storage.get(key.name);
      if (value) {
        const hash = await this.hashValue(value);

        if (dataHashes.has(hash)) {
          duplicates.push({
            originalKey: dataHashes.get(hash),
            duplicateKey: key.name,
            hash: hash
          });
        } else {
          dataHashes.set(hash, key.name);
        }
      }
    }

    // 删除重复数据，使用引用
    for (const duplicate of duplicates) {
      const reference = {
        type: 'reference',
        targetKey: duplicate.originalKey,
        duplicateOf: duplicate.hash
      };
      await this.state.storage.put(duplicate.duplicateKey, reference);
    }

    return {
      totalDuplicates: duplicates.length,
      spaceSaved: duplicates.length * 1024 // 估算节省空间
    };
  }

  // 生成数据哈希
  async hashValue(value) {
    const encoder = new TextEncoder();
    const data = encoder.encode(JSON.stringify(value));
    const hashBuffer = await crypto.subtle.digest('SHA-256', data);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
  }
}
```

```javascript
// CPU 使用监控器
export class CPUMonitor {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.limits = FREE_TIER_LIMITS.compute;
    this.requestStats = new Map();
  }

  // 开始监控请求
  startRequestMonitoring(requestId) {
    this.requestStats.set(requestId, {
      startTime: Date.now(),
      startCPUTime: performance.now()
    });
  }

  // 结束监控请求
  endRequestMonitoring(requestId, operation = 'unknown') {
    const stats = this.requestStats.get(requestId);
    if (!stats) return null;

    const endTime = Date.now();
    const endCPUTime = performance.now();

    const metrics = {
      requestId: requestId,
      operation: operation,
      duration: endTime - stats.startTime,
      cpuTime: endCPUTime - stats.startCPUTime,
      timestamp: endTime
    };

    // 记录到历史
    this.recordMetrics(metrics);

    // 清理请求统计
    this.requestStats.delete(requestId);

    return metrics;
  }

  // 记录性能指标
  async recordMetrics(metrics) {
    const dailyKey = `cpu_metrics:${this.getDateKey()}`;
    let dailyMetrics = await this.state.storage.get(dailyKey) || {
      totalRequests: 0,
      totalCPUTime: 0,
      operations: {},
      maxCPUTime: 0,
      avgCPUTime: 0
    };

    dailyMetrics.totalRequests++;
    dailyMetrics.totalCPUTime += metrics.cpuTime;
    dailyMetrics.maxCPUTime = Math.max(dailyMetrics.maxCPUTime, metrics.cpuTime);
    dailyMetrics.avgCPUTime = dailyMetrics.totalCPUTime / dailyMetrics.totalRequests;

    // 按操作类型统计
    if (!dailyMetrics.operations[metrics.operation]) {
      dailyMetrics.operations[metrics.operation] = {
        count: 0,
        totalCPUTime: 0,
        avgCPUTime: 0
      };
    }

    const opStats = dailyMetrics.operations[metrics.operation];
    opStats.count++;
    opStats.totalCPUTime += metrics.cpuTime;
    opStats.avgCPUTime = opStats.totalCPUTime / opStats.count;

    await this.state.storage.put(dailyKey, dailyMetrics);

    // 检查是否接近限制
    await this.checkCPULimits(dailyMetrics);
  }

  // 检查 CPU 限制
  async checkCPULimits(dailyMetrics) {
    const maxCPUTimeMs = this.parseTime(this.limits.cpuTime);
    const alerts = [];

    // 检查平均 CPU 时间
    if (dailyMetrics.avgCPUTime > maxCPUTimeMs * 0.8) {
      alerts.push({
        level: 'warning',
        type: 'avg_cpu_time',
        message: `Average CPU time approaching limit: ${dailyMetrics.avgCPUTime.toFixed(2)}ms`,
        recommendation: 'optimize_operations'
      });
    }

    // 检查最大 CPU 时间
    if (dailyMetrics.maxCPUTime > maxCPUTimeMs) {
      alerts.push({
        level: 'error',
        type: 'max_cpu_time',
        message: `Maximum CPU time exceeded: ${dailyMetrics.maxCPUTime.toFixed(2)}ms > ${maxCPUTimeMs}ms`,
        recommendation: 'break_into_smaller_operations'
      });
    }

    // 保存告警
    if (alerts.length > 0) {
      const alertKey = `cpu_alerts:${Date.now()}`;
      await this.state.storage.put(alertKey, alerts);
    }

    return alerts;
  }

  // 获取 CPU 使用报告
  async getCPUUsageReport(days = 7) {
    const report = {
      daily: [],
      summary: {
        totalRequests: 0,
        totalCPUTime: 0,
        avgCPUTime: 0,
        topOperations: []
      }
    };

    for (let i = 0; i < days; i++) {
      const date = new Date();
      date.setDate(date.getDate() - i);
      const dateKey = this.formatDateKey(date);
      const dailyKey = `cpu_metrics:${dateKey}`;

      const metrics = await this.state.storage.get(dailyKey);
      if (metrics) {
        report.daily.push({
          date: dateKey,
          ...metrics
        });

        report.summary.totalRequests += metrics.totalRequests;
        report.summary.totalCPUTime += metrics.totalCPUTime;
      }
    }

    // 计算平均值
    if (report.daily.length > 0) {
      report.summary.avgCPUTime = report.summary.totalCPUTime / report.summary.totalRequests;

      // 找出最耗时的操作
      const operationStats = new Map();
      for (const day of report.daily) {
        for (const [op, stats] of Object.entries(day.operations)) {
          if (!operationStats.has(op)) {
            operationStats.set(op, { count: 0, totalTime: 0 });
          }
          const opStat = operationStats.get(op);
          opStat.count += stats.count;
          opStat.totalTime += stats.totalTime;
        }
      }

      report.summary.topOperations = Array.from(operationStats.entries())
        .map(([op, stats]) => ({
          operation: op,
          count: stats.count,
          totalTime: stats.totalTime,
          avgTime: stats.totalTime / stats.count
        }))
        .sort((a, b) => b.totalTime - a.totalTime)
        .slice(0, 5);
    }

    return report;
  }

  // 操作优化建议
  async getOptimizationSuggestions() {
    const report = await this.getCPUUsageReport(7);
    const suggestions = [];

    // 分析高 CPU 操作
    for (const operation of report.summary.topOperations) {
      if (operation.avgTime > 20) { // 20ms 阈值
        suggestions.push({
          type: 'cpu_optimization',
          operation: operation.operation,
          issue: `High average CPU time: ${operation.avgTime.toFixed(2)}ms`,
          recommendations: [
            'Break down into smaller operations',
            'Cache frequently accessed data',
            'Use batch operations where possible',
            'Optimize algorithms and data structures'
          ]
        });
      }
    }

    // 分析请求模式
    if (report.summary.totalRequests > 80000) { // 免费额度 100K
      suggestions.push({
        type: 'request_optimization',
        issue: `High request volume: ${report.summary.totalRequests}/100,000`,
        recommendations: [
          'Implement caching strategies',
          'Use Durable Objects for stateful operations',
          'Consider upgrading to paid plan'
        ]
      });
    }

    return suggestions;
  }

  // 获取日期键
  getDateKey(date = new Date()) {
    return date.toISOString().split('T')[0];
  }

  // 格式化日期键
  formatDateKey(date) {
    return date.toISOString().split('T')[0];
  }

  // 解析时间
  parseTime(timeStr) {
    const match = timeStr.match(/^(\d+(?:\.\d+)?)\s*(ms|s|m|h)$/i);
    if (!match) return 0;
    const value = parseFloat(match[1]);
    const unit = match[2].toLowerCase();
    const units = { ms: 1, s: 1000, m: 60000, h: 3600000 };
    return value * units[unit];
  }
}
```

### 2. 内存使用优化
```javascript
// 内存管理器
export class MemoryManager {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.maxMemory = this.parseBytes(FREE_TIER_LIMITS.compute.memory);
    this.memoryPools = new Map();
  }

  // 创建内存池
  createMemoryPool(name, size) {
    if (this.memoryPools.has(name)) {
      throw new Error(`Memory pool ${name} already exists`);
    }

    const pool = {
      name: name,
      maxSize: size,
      currentSize: 0,
      allocations: new Map(),
      cleanupInterval: null
    };

    this.memoryPools.set(name, pool);

    // 设置定期清理
    pool.cleanupInterval = setInterval(() => {
      this.cleanupPool(name);
    }, 60000); // 每分钟清理一次

    return pool;
  }

  // 分配内存
  allocate(poolName, key, data) {
    const pool = this.memoryPools.get(poolName);
    if (!pool) {
      throw new Error(`Memory pool ${poolName} not found`);
    }

    const size = this.estimateSize(data);

    // 检查是否有足够空间
    if (pool.currentSize + size > pool.maxSize) {
      this.evictLRU(pool, size);
    }

    // 分配内存
    pool.allocations.set(key, {
      data: data,
      size: size,
      lastAccessed: Date.now(),
      accessCount: 1
    });

    pool.currentSize += size;

    return { key: key, size: size };
  }

  // 获取内存
  get(poolName, key) {
    const pool = this.memoryPools.get(poolName);
    if (!pool) return null;

    const allocation = pool.allocations.get(key);
    if (!allocation) return null;

    // 更新访问信息
    allocation.lastAccessed = Date.now();
    allocation.accessCount++;

    return allocation.data;
  }

  // 释放内存
  release(poolName, key) {
    const pool = this.memoryPools.get(poolName);
    if (!pool) return false;

    const allocation = pool.allocations.get(key);
    if (!allocation) return false;

    pool.currentSize -= allocation.size;
    pool.allocations.delete(key);

    return true;
  }

  // LRU 淘汰
  evictLRU(pool, requiredSize) {
    const allocations = Array.from(pool.allocations.entries())
      .sort(([,a], [,b]) => a.lastAccessed - b.lastAccessed);

    let freedSpace = 0;
    for (const [key, allocation] of allocations) {
      pool.allocations.delete(key);
      pool.currentSize -= allocation.size;
      freedSpace += allocation.size;

      if (freedSpace >= requiredSize) break;
    }
  }

  // 清理池
  cleanupPool(poolName) {
    const pool = this.memoryPools.get(poolName);
    if (!pool) return;

    const now = Date.now();
    const maxAge = 30 * 60 * 1000; // 30分钟

    for (const [key, allocation] of pool.allocations.entries()) {
      if (now - allocation.lastAccessed > maxAge) {
        pool.currentSize -= allocation.size;
        pool.allocations.delete(key);
      }
    }
  }

  // 获取内存使用统计
  getMemoryStats() {
    const stats = {
      totalPools: this.memoryPools.size,
      totalAllocated: 0,
      pools: []
    };

    for (const [name, pool] of this.memoryPools) {
      const poolStats = {
        name: name,
        maxSize: pool.maxSize,
        currentSize: pool.currentSize,
        usagePercentage: (pool.currentSize / pool.maxSize) * 100,
        allocations: pool.allocations.size
      };

      stats.totalAllocated += pool.currentSize;
      stats.pools.push(poolStats);
    }

    stats.totalUsagePercentage = (stats.totalAllocated / this.maxMemory) * 100;

    return stats;
  }

  // 估算对象大小
  estimateSize(obj) {
    if (typeof obj === 'string') {
      return obj.length * 2; // UTF-16
    } else if (typeof obj === 'number') {
      return 8; // 64位
    } else if (typeof obj === 'boolean') {
      return 4;
    } else if (Array.isArray(obj)) {
      return obj.reduce((sum, item) => sum + this.estimateSize(item), 0) + 24;
    } else if (typeof obj === 'object' && obj !== null) {
      let size = 24; // 对象开销
      for (const [key, value] of Object.entries(obj)) {
        size += key.length * 2 + this.estimateSize(value);
      }
      return size;
    }
    return 0;
  }

  // 解析字节数
  parseBytes(sizeStr) {
    const units = { 'B': 1, 'KB': 1024, 'MB': 1024*1024, 'GB': 1024*1024*1024 };
    const match = sizeStr.match(/^(\d+(?:\.\d+)?)\s*(B|KB|MB|GB)$/i);
    if (!match) return 0;
    return parseFloat(match[1]) * units[match[2].toUpperCase()];
  }
}
```

```javascript
// 网络使用监控器
export class NetworkMonitor {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.limits = FREE_TIER_LIMITS.network;
    this.requestCounts = new Map();
  }

  // 记录请求
  async recordRequest(request, response) {
    const now = Date.now();
    const dailyKey = `network_metrics:${this.getDateKey(now)}`;
    const hourKey = `network_metrics:${this.getHourKey(now)}`;

    // 获取现有指标
    let dailyMetrics = await this.state.storage.get(dailyKey) || this.createEmptyMetrics();
    let hourlyMetrics = await this.state.storage.get(hourKey) || this.createEmptyMetrics();

    // 更新指标
    dailyMetrics.totalRequests++;
    hourlyMetrics.totalRequests++;

    // 计算响应大小
    const responseSize = await this.calculateResponseSize(response);
    dailyMetrics.totalBytes += responseSize;
    hourlyMetrics.totalBytes += responseSize;

    // 按状态码统计
    const statusCode = response.status;
    dailyMetrics.statusCodes[statusCode] = (dailyMetrics.statusCodes[statusCode] || 0) + 1;
    hourlyMetrics.statusCodes[statusCode] = (hourlyMetrics.statusCodes[statusCode] || 0) + 1;

    // 按路径统计
    const path = new URL(request.url).pathname;
    dailyMetrics.paths[path] = (dailyMetrics.paths[path] || 0) + 1;
    hourlyMetrics.paths[path] = (hourlyMetrics.paths[path] || 0) + 1;

    // 按用户代理统计
    const userAgent = request.headers.get('User-Agent') || 'unknown';
    dailyMetrics.userAgents[userAgent] = (dailyMetrics.userAgents[userAgent] || 0) + 1;

    // 保存指标
    await Promise.all([
      this.state.storage.put(dailyKey, dailyMetrics),
      this.state.storage.put(hourKey, hourlyMetrics, { expirationTtl: 7 * 24 * 60 * 60 }) // 7天过期
    ]);

    // 检查限制
    await this.checkNetworkLimits(dailyMetrics);

    return { daily: dailyMetrics, hourly: hourlyMetrics };
  }

  // 检查网络限制
  async checkNetworkLimits(metrics) {
    const alerts = [];
    const maxRequests = this.limits.requests;
    const maxBytes = this.parseBytes(this.limits.bandwidth);

    // 检查请求数限制
    if (metrics.totalRequests > maxRequests * 0.9) {
      alerts.push({
        level: 'critical',
        type: 'request_limit',
        message: `Request limit nearly exceeded: ${metrics.totalRequests}/${maxRequests}`,
        remaining: maxRequests - metrics.totalRequests
      });
    } else if (metrics.totalRequests > maxRequests * 0.75) {
      alerts.push({
        level: 'warning',
        type: 'request_limit',
        message: `Request usage high: ${metrics.totalRequests}/${maxRequests}`,
        percentage: (metrics.totalRequests / maxRequests * 100).toFixed(1)
      });
    }

    // 检查带宽限制
    if (metrics.totalBytes > maxBytes * 0.9) {
      alerts.push({
        level: 'critical',
        type: 'bandwidth_limit',
        message: `Bandwidth limit nearly exceeded: ${this.formatBytes(metrics.totalBytes)}/${this.limits.bandwidth}`,
        remaining: maxBytes - metrics.totalBytes
      });
    } else if (metrics.totalBytes > maxBytes * 0.75) {
      alerts.push({
        level: 'warning',
        type: 'bandwidth_limit',
        message: `Bandwidth usage high: ${this.formatBytes(metrics.totalBytes)}/${this.limits.bandwidth}`,
        percentage: (metrics.totalBytes / maxBytes * 100).toFixed(1)
      });
    }

    return alerts;
  }

  // 获取网络使用报告
  async getNetworkUsageReport(days = 30) {
    const report = {
      daily: [],
      summary: {
        totalRequests: 0,
        totalBytes: 0,
        avgRequestsPerDay: 0,
        topPaths: [],
        errorRate: 0
      }
    };

    for (let i = 0; i < days; i++) {
      const date = new Date();
      date.setDate(date.getDate() - i);
      const dateKey = this.formatDateKey(date);
      const dailyKey = `network_metrics:${dateKey}`;

      const metrics = await this.state.storage.get(dailyKey);
      if (metrics) {
        report.daily.push({
          date: dateKey,
          ...metrics
        });

        report.summary.totalRequests += metrics.totalRequests;
        report.summary.totalBytes += metrics.totalBytes;
      }
    }

    // 计算汇总统计
    if (report.daily.length > 0) {
      report.summary.avgRequestsPerDay = Math.round(report.summary.totalRequests / report.daily.length);

      // 找出热门路径
      const pathCounts = new Map();
      let totalErrorRequests = 0;

      for (const day of report.daily) {
        for (const [path, count] of Object.entries(day.paths || {})) {
          pathCounts.set(path, (pathCounts.get(path) || 0) + count);
        }

        // 统计错误请求
        for (const [statusCode, count] of Object.entries(day.statusCodes || {})) {
          if (statusCode >= 400) {
            totalErrorRequests += count;
          }
        }
      }

      report.summary.topPaths = Array.from(pathCounts.entries())
        .map(([path, count]) => ({ path, count }))
        .sort((a, b) => b.count - a.count)
        .slice(0, 10);

      report.summary.errorRate = report.summary.totalRequests > 0
        ? (totalErrorRequests / report.summary.totalRequests * 100).toFixed(2)
        : 0;
    }

    return report;
  }

  // 预测使用趋势
  async predictUsageTrend() {
    const report = await this.getNetworkUsageReport(7); // 最近7天
    const daily = report.daily.sort((a, b) => new Date(a.date) - new Date(b.date));

    if (daily.length < 3) {
      return { trend: 'insufficient_data' };
    }

    // 计算日均增长率
    const growthRates = [];
    for (let i = 1; i < daily.length; i++) {
      const current = daily[i].totalRequests;
      const previous = daily[i-1].totalRequests;
      const growthRate = previous > 0 ? (current - previous) / previous : 0;
      growthRates.push(growthRate);
    }

    const avgGrowthRate = growthRates.reduce((sum, rate) => sum + rate, 0) / growthRates.length;

    // 预测月底使用量
    const currentDay = new Date().getDate();
    const daysInMonth = new Date(new Date().getFullYear(), new Date().getMonth() + 1, 0).getDate();
    const remainingDays = daysInMonth - currentDay;

    const currentDailyAvg = daily.reduce((sum, day) => sum + day.totalRequests, 0) / daily.length;
    const predictedMonthlyTotal = report.summary.totalRequests + (currentDailyAvg * remainingDays * (1 + avgGrowthRate));

    return {
      trend: avgGrowthRate > 0.05 ? 'increasing' : avgGrowthRate < -0.05 ? 'decreasing' : 'stable',
      growthRate: (avgGrowthRate * 100).toFixed(1),
      predictedMonthlyTotal: Math.round(predictedMonthlyTotal),
      recommendation: predictedMonthlyTotal > this.limits.requests * 0.8 ? 'upgrade_recommended' : 'within_limits'
    };
  }

  // 创建空的指标对象
  createEmptyMetrics() {
    return {
      totalRequests: 0,
      totalBytes: 0,
      statusCodes: {},
      paths: {},
      userAgents: {}
    };
  }

  // 计算响应大小
  async calculateResponseSize(response) {
    // 简化计算，实际中需要根据响应类型计算
    const contentType = response.headers.get('content-type') || '';

    if (contentType.includes('application/json')) {
      try {
        const body = await response.clone().text();
        return body.length;
      } catch (error) {
        return 1024; // 默认大小
      }
    } else if (contentType.includes('text/')) {
      try {
        const body = await response.clone().text();
        return body.length;
      } catch (error) {
        return 512;
      }
    } else {
      return 2048; // 其他类型的默认大小
    }
  }

  // 获取日期键
  getDateKey(date) {
    return date.toISOString().split('T')[0];
  }

  // 获取小时键
  getHourKey(date) {
    return date.toISOString().slice(0, 13).replace('T', '_');
  }

  // 格式化日期键
  formatDateKey(date) {
    return date.toISOString().split('T')[0];
  }

  // 格式化字节数
  formatBytes(bytes) {
    if (bytes === 0) return '0 Bytes';
    const k = 1024;
    const sizes = ['Bytes', 'KB', 'MB', 'GB'];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
  }

  // 解析字节数
  parseBytes(sizeStr) {
    const units = { 'B': 1, 'KB': 1024, 'MB': 1024*1024, 'GB': 1024*1024*1024 };
    const match = sizeStr.match(/^(\d+(?:\.\d+)?)\s*(B|KB|MB|GB)$/i);
    if (!match) return 0;
    return parseFloat(match[1]) * units[match[2].toUpperCase()];
  }
}
```

### 1. 连接限制管理
```javascript
// WebSocket 连接监控器
export class WebSocketMonitor {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.limits = FREE_TIER_LIMITS.websockets;
    this.activeConnections = new Map();
  }

  // 注册连接
  async registerConnection(connectionId, clientInfo) {
    const now = Date.now();

    // 检查当前连接数
    if (this.activeConnections.size >= this.limits.concurrentConnections) {
      throw new Error('Maximum concurrent connections reached');
    }

    // 检查月度连接数
    const monthlyConnections = await this.getMonthlyConnectionCount();
    if (monthlyConnections >= this.limits.totalConnections) {
      throw new Error('Monthly connection limit reached');
    }

    // 注册连接
    const connectionInfo = {
      id: connectionId,
      clientInfo: clientInfo,
      connectedAt: now,
      lastActivity: now,
      messagesSent: 0,
      messagesReceived: 0,
      bytesReceived: 0,
      bytesSent: 0
    };

    this.activeConnections.set(connectionId, connectionInfo);

    // 更新月度统计
    await this.updateMonthlyStats('connections', 1);

    return connectionInfo;
  }

  // 注销连接
  async unregisterConnection(connectionId) {
    const connection = this.activeConnections.get(connectionId);
    if (!connection) return;

    // 计算连接持续时间
    const duration = Date.now() - connection.connectedAt;

    // 更新统计
    await this.updateConnectionStats(connection, duration);

    // 从活跃连接中移除
    this.activeConnections.delete(connectionId);
  }

  // 记录消息活动
  recordMessage(connectionId, messageSize, direction = 'received') {
    const connection = this.activeConnections.get(connectionId);
    if (!connection) return;

    connection.lastActivity = Date.now();

    if (direction === 'received') {
      connection.messagesReceived++;
      connection.bytesReceived += messageSize;
    } else {
      connection.messagesSent++;
      connection.bytesSent += messageSize;
    }
  }

  // 获取连接统计
  getConnectionStats() {
    const now = Date.now();
    const connections = Array.from(this.activeConnections.values());

    return {
      totalConnections: connections.length,
      maxAllowed: this.limits.concurrentConnections,
      usagePercentage: (connections.length / this.limits.concurrentConnections * 100).toFixed(1),
      connectionsByDuration: this.getC_connectionsByDuration(connections, now),
      connectionsByCountry: this.getConnectionsByCountry(connections),
      totalMessagesSent: connections.reduce((sum, conn) => sum + conn.messagesSent, 0),
      totalMessagesReceived: connections.reduce((sum, conn) => sum + conn.messagesReceived, 0),
      totalBytesTransferred: connections.reduce((sum, conn) => sum + conn.bytesReceived + conn.bytesSent, 0)
    };
  }

  // 按持续时间分组连接
  getConnectionsByDuration(connections, now) {
    const buckets = {
      '0-1min': 0,
      '1-5min': 0,
      '5-30min': 0,
      '30min-1h': 0,
      '1h+': 0
    };

    for (const conn of connections) {
      const duration = (now - conn.connectedAt) / (1000 * 60); // 分钟

      if (duration < 1) buckets['0-1min']++;
      else if (duration < 5) buckets['1-5min']++;
      else if (duration < 30) buckets['5-30min']++;
      else if (duration < 60) buckets['30min-1h']++;
      else buckets['1h+']++;
    }

    return buckets;
  }

  // 按国家分组连接
  getConnectionsByCountry(connections) {
    const countries = {};

    for (const conn of connections) {
      const country = conn.clientInfo.country || 'unknown';
      countries[country] = (countries[country] || 0) + 1;
    }

    return countries;
  }

  // 清理非活跃连接
  cleanupInactiveConnections(maxInactiveTime = 30 * 60 * 1000) { // 30分钟
    const now = Date.now();
    const toRemove = [];

    for (const [id, connection] of this.activeConnections) {
      if (now - connection.lastActivity > maxInactiveTime) {
        toRemove.push(id);
      }
    }

    for (const id of toRemove) {
      this.unregisterConnection(id);
    }

    return toRemove.length;
  }

  // 获取月度连接数
  async getMonthlyConnectionCount() {
    const monthKey = `ws_connections:${this.getMonthKey()}`;
    const stats = await this.state.storage.get(monthKey) || { total: 0 };
    return stats.total;
  }

  // 更新月度统计
  async updateMonthlyStats(type, count) {
    const monthKey = `ws_connections:${this.getMonthKey()}`;
    let stats = await this.state.storage.get(monthKey) || { total: 0, [type]: 0 };

    stats.total += count;
    stats[type] = (stats[type] || 0) + count;

    await this.state.storage.put(monthKey, stats, {
      expirationTtl: 32 * 24 * 60 * 60 // 32天过期
    });
  }

  // 更新连接统计
  async updateConnectionStats(connection, duration) {
    const dayKey = `ws_stats:${this.getDateKey()}`;
    let stats = await this.state.storage.get(dayKey) || {
      totalConnections: 0,
      totalDuration: 0,
      totalMessages: 0,
      totalBytes: 0
    };

    stats.totalConnections++;
    stats.totalDuration += duration;
    stats.totalMessages += connection.messagesSent + connection.messagesReceived;
    stats.totalBytes += connection.bytesReceived + connection.bytesSent;

    await this.state.storage.put(dayKey, stats, {
      expirationTtl: 7 * 24 * 60 * 60 // 7天过期
    });
  }

  // 获取月度键
  getMonthKey(date = new Date()) {
    return `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, '0')}`;
  }

  // 获取日期键
  getDateKey(date = new Date()) {
    return date.toISOString().split('T')[0];
  }
}
```
