## 免费套餐概览

Cloudflare KV 提供慷慨的免费额度，使开发者能够构建和部署大规模的键值存储应用而无需前期投入。

```javascript
export async function monitorKVUsage(env) {
  // 从 Analytics Engine 获取使用数据
  const analytics = await env.ANALYTICS.prepare(`
    SELECT
      date,
      SUM(CASE WHEN operation = 'read' THEN 1 ELSE 0 END) as reads,
      SUM(CASE WHEN operation = 'write' THEN 1 ELSE 0 END) as writes,
      SUM(CASE WHEN operation = 'delete' THEN 1 ELSE 0 END) as deletes,
      SUM(CASE WHEN operation = 'list' THEN 1 ELSE 0 END) as lists
    FROM kv_operations
    WHERE date = date('now')
    GROUP BY date
  `).first();

  const usage = {
    reads: analytics?.reads || 0,
    writes: analytics?.writes || 0,
    deletes: analytics?.deletes || 0,
    lists: analytics?.lists || 0
  };

  // 计算使用百分比
  const limits = {
    reads: 100000,
    writes: 1000,
    deletes: 1000,
    lists: 1000
  };

  const usagePercent = {
    reads: (usage.reads / limits.reads * 100).toFixed(2),
    writes: (usage.writes / limits.writes * 100).toFixed(2),
    deletes: (usage.deletes / limits.deletes * 100).toFixed(2),
    lists: (usage.lists / limits.lists * 100).toFixed(2)
  };

  return {
    usage,
    limits,
    usagePercent,
    date: new Date().toISOString()
  };
}
```

```javascript
export function withQuotaWarning(env, threshold = 0.8) {
  return async (context) => {
    const usage = await getCurrentUsage(env);

    // 检查各项操作是否接近限制
    const warnings = [];

    if (usage.reads / 100000 > threshold) {
      warnings.push('Read operations near limit');
    }

    if (usage.writes / 1000 > threshold) {
      warnings.push('Write operations near limit');
    }

    if (usage.deletes / 1000 > threshold) {
      warnings.push('Delete operations near limit');
    }

    if (usage.lists / 1000 > threshold) {
      warnings.push('List operations near limit');

    if (warnings.length > 0) {
      // 记录预警
      await env.MY_KV.put(`quota-warning:${Date.now()}`, JSON.stringify({
        timestamp: new Date().toISOString(),
        usage,
        warnings
      }));

      // 可选：发送告警通知
      await sendAlert(env, warnings);
    }

    return context.next();
  };
}
```

```javascript
// 分析使用趋势
export async function analyzeUsageTrend(env, days = 7) {
  const trends = [];

  for (let i = days - 1; i >= 0; i--) {
    const date = new Date();
    date.setDate(date.getDate() - i);
    const dateStr = date.toISOString().split('T')[0];

    const stats = await env.MY_KV.get(`usage-stats:${dateStr}`, 'json');
    trends.push({
      date: dateStr,
      ...(stats || { reads: 0, writes: 0, deletes: 0, lists: 0 })
    });
  }

  // 计算平均值和趋势
  const avgReads = trends.reduce((sum, d) => sum + d.reads, 0) / trends.length;
  const avgWrites = trends.reduce((sum, d) => sum + d.writes, 0) / trends.length;

  // 预测未来使用
  const projection = {
    tomorrow: {
      reads: avgReads * 1.1, // 假设10%增长
      writes: avgWrites * 1.1
    },
    week: {
      reads: avgReads * 7,
      writes: avgWrites * 7
    }
  };

  return {
    trends,
    averages: {
      reads: Math.round(avgReads),
      writes: Math.round(avgWrites)
    },
    projection
  };
}
```

### 1. 读取优化

```javascript
// 使用缓存减少 KV 读取
class KVCache {
  constructor(kv, cacheTtl = 300000) { // 5分钟
    this.kv = kv;
    this.cache = new Map();
    this.cacheTtl = cacheTtl;
  }

  async get(key, type = 'text') {
    // 检查内存缓存
    const cached = this.cache.get(key);
    if (cached && Date.now() - cached.timestamp < this.cacheTtl) {
      return cached.value;
    }

    // 从 KV 读取
    const value = await this.kv.get(key, type);

    // 更新缓存
    this.cache.set(key, {
      value,
      timestamp: Date.now()
    });

    return value;
  }

  async getBatch(keys) {
    const results = {};
    const uncachedKeys = [];

    // 检查缓存
    for (const key of keys) {
      const cached = this.cache.get(key);
      if (cached && Date.now() - cached.timestamp < this.cacheTtl) {
        results[key] = cached.value;
      } else {
        uncachedKeys.push(key);
      }
    }

    // 批量读取未缓存的键
    if (uncachedKeys.length > 0) {
      const values = await this.kv.get(uncachedKeys);

      for (const key of uncachedKeys) {
        const value = values[key];
        results[key] = value;

        // 更新缓存
        this.cache.set(key, {
          value,
          timestamp: Date.now()
        });
      }
    }

    return results;
  }
}

// 使用示例

// 单次读取

// 批量读取

  `user:${userId1}`,
  `user:${userId2}`,
  `user:${userId3}`
]);
```

### 2. 写入优化
```javascript
// 批量写入优化
export class KVBatchWriter {
  constructor(kv, batchSize = 100) {
    this.kv = kv;
    this.batchSize = batchSize;
    this.pending = [];
    this.flushTimer = null;
  }

  put(key, value, options = {}) {
    this.pending.push(['put', key, value, options]);
    return this.scheduleFlush();
  }

  delete(key) {
    this.pending.push(['delete', key]);
    return this.scheduleFlush();
  }

  scheduleFlush() {
    if (this.pending.length >= this.batchSize) {
      return this.flush();
    }

    if (this.flushTimer) {
      clearTimeout(this.flushTimer);
    }

    this.flushTimer = setTimeout(() => this.flush(), 1000);
    return Promise.resolve();
  }

  async flush() {
    if (this.pending.length === 0) {
      return;
    }

    const operations = this.pending.splice(0);

    // 执行批量操作
    const promises = operations.map(([type, key, value, options]) => {
      if (type === 'put') {
        return this.kv.put(key, value, options);
      } else {
        return this.kv.delete(key);
      }
    });

    const results = await Promise.all(promises);

    // 记录批量操作统计
    await this.recordBatchOperation(operations.length);

    return results;
  }

  async recordBatchOperation(count) {
    const today = new Date().toISOString().split('T')[0];
    const key = `batch-stats:${today}`;

    const stats = await this.kv.get(key, 'json') || { count: 0, batches: 0 };
    stats.count += count;
    stats.batches += 1;

    await this.kv.put(key, JSON.stringify(stats), {
      expirationTtl: 7 * 24 * 3600 // 7天
    });
  }
}
```

### 4. 智能过期策略
```javascript
// 根据访问频率设置不同的过期时间
export class SmartTTL {
  constructor(kv, analytics) {
    this.kv = kv;
    this.analytics = analytics;
  }

  async put(key, value, baseTtl = 3600) {
    // 检查历史访问模式
    const accessPattern = await this.getAccessPattern(key);

    // 根据访问模式调整 TTL
    let ttl = baseTtl;

    if (accessPattern.frequency === 'high') {
      ttl = baseTtl * 4; // 热数据保留更久
    } else if (accessPattern.frequency === 'low') {
      ttl = baseTtl / 2; // 冷数据更快过期
    }

    // 记录写入
    await this.recordAccess(key, 'write');

    await this.kv.put(key, value, { expirationTtl: ttl });
  }

  async get(key, type = 'text') {
    // 记录读取
    await this.recordAccess(key, 'read');

    return await this.kv.get(key, type);
  }

  async getAccessPattern(key) {
    const pattern = await this.kv.get(`pattern:${key}`, 'json');
    return pattern || { frequency: 'unknown', lastAccess: null };
  }

  async recordAccess(key, operation) {
    const pattern = await this.getAccessPattern(key);
    const now = Date.now();

    // 更新访问模式
    if (pattern.lastAccess) {
      const timeSinceLastAccess = now - pattern.lastAccess;
      const timeSinceLastAccessHours = timeSinceLastAccess / (1000 * 60 * 60);

      // 根据访问间隔判断频率
      if (timeSinceLastAccessHours < 1) {
        pattern.frequency = 'high';
      } else if (timeSinceLastAccessHours < 24) {
        pattern.frequency = 'medium';
      } else {
        pattern.frequency = 'low';
      }
    }

    pattern.lastAccess = now;
    pattern.accessCount = (pattern.accessCount || 0) + 1;

    // 存储访问模式（使用较短的 TTL）
    await this.kv.put(`pattern:${key}`, JSON.stringify(pattern), {
      expirationTtl: 7 * 24 * 3600 // 7天
    });
  }
}
```

## 额度超限处理

### 1. 优雅降级
```javascript
// 额度超限时的降级策略
export class GracefulDegradation {
  constructor(kv, fallbackCache) {
    this.kv = kv;
    this.fallbackCache = fallbackCache;
    this.lastQuotaCheck = 0;
    this.quotaStatus = null;
  }

  async get(key, type = 'text') {
    // 检查配额状态
    await this.checkQuotaStatus();

    if (this.quotaStatus && this.quotaStatus.readsRemaining <= 1000) {
      // 配额不足，使用降级策略
      return await this.getWithFallback(key, type);
    }

    try {
      return await this.kv.get(key, type);
    } catch (error) {
      if (error.message.includes('quota')) {
        // 配额超限，使用降级
        return await this.getWithFallback(key, type);
      }
      throw error;
    }
  }

  async getWithFallback(key, type = 'text') {
    // 尝试从备用缓存获取
    const fallbackValue = await this.fallbackCache.get(key);
    if (fallbackValue !== null) {
      console.log(`Using fallback cache for key: ${key}`);
      return fallbackValue;
    }

    // 无法获取数据，返回默认值或 null
    return null;
  }

  async put(key, value, options = {}) {
    // 检查配额状态
    await this.checkQuotaStatus();

    if (this.quotaStatus && this.quotaStatus.writesRemaining <= 100) {
      // 配额不足，延迟写入或拒绝
      console.warn('Write quota nearly exhausted, deferring write');
      await this.deferWrite(key, value, options);
      return;
    }

    try {
      await this.kv.put(key, value, options);
    } catch (error) {
      if (error.message.includes('quota')) {
        // 配额超限，延迟写入
        await this.deferWrite(key, value, options);
      } else {
        throw error;
      }
    }
  }

  async deferWrite(key, value, options) {
    // 将写入操作存储到延迟队列
    const deferred = {
      key,
      value,
      options,
      timestamp: Date.now()
    };

    await this.fallbackCache.set(`deferred:${key}`, deferred);
  }

  async checkQuotaStatus() {
    const now = Date.now();

    // 缓存配额状态 5 分钟
    if (now - this.lastQuotaCheck < 300000) {
      return;
    }

    try {
      // 获取当前使用情况
      const usage = await this.getUsageMetrics();
      const limits = {
        reads: 100000,
        writes: 1000
      };

      this.quotaStatus = {
        readsRemaining: limits.reads - usage.reads,
        writesRemaining: limits.writes - usage.writes,
        lastCheck: now
      };

      this.lastQuotaCheck = now;
    } catch (error) {
      console.error('Failed to check quota status:', error);
    }
  }

  async processDeferredWrites() {
    // 定期处理延迟的写入
    const deferredKeys = await this.fallbackCache.getKeys('deferred:');

    for (const fullKey of deferredKeys) {
      const deferred = await this.fallbackCache.get(fullKey);

      if (deferred && Date.now() - deferred.timestamp > 300000) {
        // 5分钟后尝试写入
        try {
          await this.kv.put(deferred.key, deferred.value, deferred.options);
          await this.fallbackCache.delete(fullKey);
        } catch (error) {
          console.error('Failed to process deferred write:', error);
        }
      }
    }
  }
}
```

```javascript
// 自动配额恢复和报告
export class QuotaRecovery {
  constructor(kv, notification) {
    this.kv = kv;
    this.notification = notification;
  }

  async monitorAndRecover() {
    const quotaStatus = await this.getQuotaStatus();

    // 检查是否需要恢复
    if (this.needsRecovery(quotaStatus)) {
      await this.executeRecovery(quotaStatus);
    }

    // 生成使用报告
    await this.generateReport(quotaStatus);
  }

  needsRecovery(status) {
    return (
      status.readsRemaining < 1000 ||
      status.writesRemaining < 50 ||
      status.deletesRemaining < 50
    );
  }

  async executeRecovery(status) {
    const recoveryActions = [];

    // 增加缓存时间
    if (status.readsRemaining < 1000) {
      await this.increaseCacheTtl();
      recoveryActions.push('Increased cache TTL');
    }

    // 实施批量操作
    if (status.writesRemaining < 50) {
      await this.enableBatchWrites();
      recoveryActions.push('Enabled batch writes');
    }

    // 启用压缩
    if (status.readsRemaining < 1000) {
      await this.enableCompression();
      recoveryActions.push('Enabled compression');
    }

    // 发送通知
    await this.notification.send({
      type: 'quota_recovery',
      actions: recoveryActions,
      status
    });
  }

  async generateReport(status) {
    const report = {
      timestamp: new Date().toISOString(),
      quota: status,
      recommendations: this.generateRecommendations(status)
    };

    // 存储报告
    await this.kv.put(`quota-report:${Date.now()}`, JSON.stringify(report), {
      expirationTtl: 30 * 24 * 3600 // 30天
    });
  }

  generateRecommendations(status) {
    const recommendations = [];

    if (status.readsRemaining < 10000) {
      recommendations.push('Consider increasing cache TTL for read-heavy workloads');
    }

    if (status.writesRemaining < 500) {
      recommendations.push('Implement batch writes to reduce write operations');
    }

    if (status.deleteRemaining < 500) {
      recommendations.push('Use TTL-based expiration instead of manual deletes');
    }

    return recommendations;
  }
}
```

```javascript
export async function analyzeOperationPattern(env, days = 7) {

    readHeavy: [],

        lists: stats.lists,

    patterns,

  };
```

```javascript
// 分析热点键
export async function analyzeHotKeys(env) {
  const hotKeys = await env.HOT_KEYS.get('current', 'json') || {};

  // 分析访问频率
  const analysis = Object.entries(hotKeys)
    .map(([key, stats]) => ({
      key,
      reads: stats.reads || 0,
      writes: stats.writes || 0,
      score: (stats.reads || 0) + (stats.writes || 0) * 2
    }))
    .sort((a, b) => b.score - a.score)
    .slice(0, 50);

  // 识别访问模式
  const patterns = this.identifyAccessPatterns(analysis);

  return {
    hotKeys: analysis,
    patterns,
    recommendations: this.generateHotKeyRecommendations(analysis)
  };
}

function identifyAccessPatterns(hotKeys) {
  const patterns = {
    session: [],
    cache: [],
    config: [],
    other: []
  };

  hotKeys.forEach(({ key }) => {
    if (key.startsWith('session:')) {
      patterns.session.push(key);
    } else if (key.startsWith('cache:')) {
      patterns.cache.push(key);
    } else if (key.startsWith('config:')) {
      patterns.config.push(key);
    } else {
      patterns.other.push(key);
    }
  });

  return patterns;
}
```
