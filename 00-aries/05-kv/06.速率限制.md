# KV 速率限制

## 速率限制概述

速率限制（Rate Limiting）是保护 API 和服务免受滥用的关键机制。使用 KV 存储可以实现全球分布的速率限制器，确保在全球范围内一致地执行限制规则。

## 基础速率限制

### 1. 简单计数器
```javascript
// 基础速率限制实现
export class RateLimiter {
  constructor(kv, options = {}) {
    this.kv = kv;
    this.options = {
      windowMs: options.windowMs || 60 * 1000,    // 1分钟窗口
      maxRequests: options.maxRequests || 100,    // 最大请求数
      keyGenerator: options.keyGenerator || ((req) => {
        // 默认使用 IP 作为键
        return req.headers.get('CF-Connecting-IP') || 'unknown';
      })
    };
  }

  // 检查是否超过限制
  async isAllowed(request) {
    const key = this.options.keyGenerator(request);
    const window = Math.floor(Date.now() / this.options.windowMs);
    const counterKey = `ratelimit:${key}:${window}`;

    // 获取当前计数
    const current = await this.kv.get(counterKey, 'json') || 0;

    // 检查是否超过限制
    if (current >= this.options.maxRequests) {
      return {
        allowed: false,
        limit: this.options.maxRequests,
        remaining: 0,
        resetTime: (window + 1) * this.options.windowMs
      };
    }

    // 增加计数
    const newCount = current + 1;
    await this.kv.put(counterKey, JSON.stringify(newCount), {
      expirationTtl: Math.ceil(this.options.windowMs / 1000) + 1
    });

    return {
      allowed: true,
      limit: this.options.maxRequests,
      remaining: this.options.maxRequests - newCount,
      resetTime: window * this.options.windowMs
    };
  }
}

// 使用示例
const rateLimiter = new RateLimiter(env.RATE_LIMIT_KV, {
  windowMs: 60 * 1000,      // 1分钟
  maxRequests: 100         // 100次请求
});

export default {
  async fetch(request, env, ctx) {
    const result = await rateLimiter.isAllowed(request);

    if (!result.allowed) {
      return new Response('Rate limit exceeded', {
        status: 429,
        headers: {
          'X-RateLimit-Limit': result.limit.toString(),
          'X-RateLimit-Remaining': result.remaining.toString(),
          'X-RateLimit-Reset': result.resetTime.toString(),
          'Retry-After': Math.ceil((result.resetTime - Date.now()) / 1000).toString()
        }
      });
    }

    // 处理请求
    return new Response('Success');
  }
};
```

### 2. 滑动窗口算法
```javascript
// 滑动窗口速率限制
export class SlidingWindowRateLimiter {
  constructor(kv, options = {}) {
    this.kv = kv;
    this.options = {
      windowMs: options.windowMs || 60 * 1000,    // 1分钟窗口
      maxRequests: options.maxRequests || 100,    // 最大请求数
      granularity: options.granularity || 1000    // 1秒粒度
    };
  }

  async isAllowed(request) {
    const key = this.options.keyGenerator(request);
    const now = Date.now();

    // 获取时间窗口内的所有记录
    const windowStart = now - this.options.windowMs;
    const records = await this.getWindowRecords(key, windowStart);

    // 计算当前请求数
    const requestCount = records.length;

    if (requestCount >= this.options.maxRequests) {
      return { allowed: false, count: requestCount };
    }

    // 记录当前请求
    await this.recordRequest(key, now);

    return { allowed: true, count: requestCount + 1 };
  }

  async getWindowRecords(key, windowStart) {
    // 搜索所有可能的时间段
    const startSlot = Math.floor(windowStart / this.options.granularity);
    const endSlot = Math.floor(Date.now() / this.options.granularity);
    const records = [];

    for (let slot = startSlot; slot <= endSlot; slot++) {
      const slotKey = `sliding:${key}:${slot}`;
      const slotRecords = await this.kv.get(slotKey, 'json') || [];

      for (const timestamp of slotRecords) {
        if (timestamp > windowStart) {
          records.push(timestamp);
        }
      }
    }

    return records;
  }

  async recordRequest(key, timestamp) {
    const slot = Math.floor(timestamp / this.options.granularity);
    const slotKey = `sliding:${key}:${slot}`;

    // 获取当前时间段的记录
    const records = await this.kv.get(slotKey, 'json') || [];

    // 添加新记录
    records.push(timestamp);

    // 只保留窗口内的记录
    const windowStart = timestamp - this.options.windowMs;
    const filtered = records.filter(t => t > windowStart);

    // 更新时间段记录
    await this.kv.put(slotKey, JSON.stringify(filtered), {
      expirationTtl: Math.ceil(this.options.windowMs / this.options.granularity / 1000) + 1
    });
  }
}
```

### 1. 多级限制
```javascript
// 多级速率限制器
export class MultiLevelRateLimiter {
  constructor(kv) {
    this.kv = kv;
    this.limiters = new Map();
  }

  // 添加限制规则
  addLimiter(name, options) {
    this.limiters.set(name, new RateLimiter(this.kv, options));
  }

  // 检查所有限制
  async checkAll(request) {
    const results = {};

    for (const [name, limiter] of this.limiters) {
      const result = await limiter.isAllowed(request);
      results[name] = result;

      // 如果任何一个限制被触发，立即返回
      if (!result.allowed) {
        return {
          allowed: false,
          violatedLimiter: name,
          details: results
        };
      }
    }

    return {
      allowed: true,
      details: results
    };
  }
}

// 使用示例

// 添加不同级别的限制
```

### 2. 令牌桶算法
```javascript
// 令牌桶速率限制
export class TokenBucketRateLimiter {
  constructor(kv, options = {}) {
    this.kv = kv;
    this.options = {
      bucketSize: options.bucketSize || 100,       // 桶容量
      refillRate: options.refillRate || 10,         // 每秒补充10个令牌
      refillInterval: options.refillInterval || 1000, // 1秒补充间隔
      keyGenerator: options.keyGenerator || ((req) => {
        return req.headers.get('CF-Connecting-IP');
      })
    };
  }

  async isAllowed(request) {
    const key = this.options.keyGenerator(request);
    const bucketKey = `tokenbucket:${key}`;

    // 获取当前桶状态
    const bucket = await this.getBucket(bucketKey);

    // 尝试消费令牌
    if (bucket.tokens > 0) {
      bucket.tokens--;
      bucket.lastRefill = Date.now();

      await this.saveBucket(bucketKey, bucket);

      return {
        allowed: true,
        tokens: bucket.tokens,
        bucketSize: this.options.bucketSize
      };
    }

    // 检查是否可以补充令牌
    const now = Date.now();
    const timeSinceLastRefill = now - bucket.lastRefill;
    const tokensToAdd = Math.floor(
      timeSinceLastRefill / this.options.refillInterval * this.options.refillRate
    );

    if (tokensToAdd > 0) {
      bucket.tokens = Math.min(
        this.options.bucketSize,
        bucket.tokens + tokensToAdd
      );
      bucket.lastRefill = now;

      // 再次尝试消费
      if (bucket.tokens > 0) {
        bucket.tokens--;
        await this.saveBucket(bucketKey, bucket);

        return {
          allowed: true,
          tokens: bucket.tokens,
          bucketSize: this.options.bucketSize
        };
      }
    }

    // 计算下次可用的令牌时间
    const tokensNeeded = 1;
    const deficit = tokensNeeded - bucket.tokens;
    const waitTime = Math.ceil(deficit / this.options.refillRate * this.options.refillInterval);

    return {
      allowed: false,
      tokens: 0,
      bucketSize: this.options.bucketSize,
      retryAfter: waitTime
    };
  }

  async getBucket(key) {
    const bucket = await this.kv.get(key, 'json');
    if (bucket) {
      return bucket;
    }

    // 创建新桶
    return {
      tokens: this.options.bucketSize,
      lastRefill: Date.now()
    };
  }

  async saveBucket(key, bucket) {
    await this.kv.put(key, JSON.stringify(bucket), {
      expirationTtl: Math.ceil(this.options.bucketSize / this.options.refillRate) + 10
    });
  }
}
```

### 3. 自适应速率限制
```javascript
// 自适应速率限制器
export class AdaptiveRateLimiter {
  constructor(kv, options = {}) {
    this.kv = kv;
    this.options = {
      minLimit: options.minLimit || 10,           // 最小限制
      maxLimit: options.maxLimit || 1000,         // 最大限制
      defaultLimit: options.defaultLimit || 100,  // 默认限制
      adjustmentFactor: options.adjustmentFactor || 0.1,
      keyGenerator: options.keyGenerator || ((req) => {
        return req.headers.get('CF-Connecting-IP');
      })
    };
  }

  async isAllowed(request) {
    const key = this.options.keyGenerator(request);
    const now = Date.now();

    // 获取当前限制和统计
    const state = await this.getState(key);

    // 检查是否超过限制
    if (state.currentCount >= state.currentLimit) {
      // 超过限制，调整限制
      await this.adjustLimit(key, state, false);
      return { allowed: false, limit: state.currentLimit };
    }

    // 记录请求
    state.currentCount++;
    await this.saveState(key, state);

    // 定期调整限制
    if (Math.random() < 0.01) { // 1%概率调整
      await this.adjustLimit(key, state, true);
    }

    return { allowed: true, limit: state.currentLimit };
  }

  async getState(key) {
    const state = await this.kv.get(`adaptive:${key}`, 'json');

    if (state) {
      return state;
    }

    return {
      currentLimit: this.options.defaultLimit,
      currentCount: 0,
      lastAdjustment: Date.now(),
      violationCount: 0,
      successCount: 0
    };
  }

  async saveState(key, state) {
    await this.kv.put(`adaptive:${key}`, JSON.stringify(state), {
      expirationTtl: 3600 // 1小时
    });
  }

  async adjustLimit(key, state, success) {
    const now = Date.now();

    if (success) {
      state.successCount++;

      // 如果成功率高，可以增加限制
      if (state.successCount > 10 && state.violationCount === 0) {
        state.currentLimit = Math.min(
          this.options.maxLimit,
          Math.floor(state.currentLimit * (1 + this.options.adjustmentFactor))
        );
        state.successCount = 0;
      }
    } else {
      state.violationCount++;

      // 如果违规率高，减少限制
      if (state.violationCount > 3) {
        state.currentLimit = Math.max(
          this.options.minLimit,
          Math.floor(state.currentLimit * (1 - this.options.adjustmentFactor))
        );
        state.violationCount = 0;
      }
    }

    state.lastAdjustment = now;
    state.currentCount = 0; // 重置计数

    await this.saveState(key, state);
  }
}
```

```javascript
// API 端点特定的速率限制
export class APIEndpointRateLimiter {
  constructor(kv) {
    this.kv = kv;
    this.endpointLimits = new Map();
  }

  // 配置端点限制
  setEndpointLimit(path, options) {
    this.endpointLimits.set(path, options);
  }

  async checkEndpoint(request) {
    const url = new URL(request.url);
    const path = url.pathname;
    const method = request.method;

    // 查找匹配的限制规则
    const limit = this.findLimit(path, method);
    if (!limit) {
      return { allowed: true };
    }

    // 生成键
    const key = this.generateKey(request, limit);
    const limiter = this.createLimiter(limit);

    return await limiter.isAllowed({ ...request, rateLimitKey: key });
  }

  findLimit(path, method) {
    // 精确匹配
    const exactKey = `${method}:${path}`;
    if (this.endpointLimits.has(exactKey)) {
      return this.endpointLimits.get(exactKey);
    }

    // 通配符匹配
    for (const [key, limit] of this.endpointLimits) {
      if (key.includes(':path:') && key.startsWith(`${method}:`)) {
        const limitPath = key.split(':')[2];
        if (path.startsWith(limitPath)) {
          return limit;
        }
      }
    }

    return null;
  }

  generateKey(request, limit) {
    if (limit.type === 'global') {
      return 'global';
    } else if (limit.type === 'per_ip') {
      return `ip:${request.headers.get('CF-Connecting-IP')}`;
    } else if (limit.type === 'per_user') {
      // 需要从请求中提取用户ID
      const auth = request.auth;
      return auth ? `user:${auth.userId}` : `ip:${request.headers.get('CF-Connecting-IP')}`;
    }
    return 'unknown';
  }

  createLimiter(options) {
    return new RateLimiter(this.kv, {
      windowMs: options.windowMs,
      maxRequests: options.maxRequests,
      keyGenerator: (req) => req.rateLimitKey
    });
  }
}

// 使用示例

  maxRequests: 100
});

  type: 'per_ip',
  windowMs: 15 * 60 * 1000,
  maxRequests: 5
});

apiLimiter.setEndpointLimit('GET:/api/user/profile', {

  maxRequests: 200
```

### 2. 动态速率限制
```javascript
// 动态速率限制器
export class DynamicRateLimiter {
  constructor(kv, configProvider) {
    this.kv = kv;
    this.configProvider = configProvider;
    this.limiterCache = new Map();
    this.cacheTimeout = 300000; // 5分钟缓存
  }

  async isAllowed(request) {
    // 获取动态配置
    const config = await this.configProvider.getConfig(request);
    if (!config) {
      return { allowed: true };
    }

    // 获取或创建限制器
    const limiter = await this.getLimiter(config);
    return await limiter.isAllowed(request);
  }

  async getLimiter(config) {
    const cacheKey = JSON.stringify(config);

    // 检查缓存
    const cached = this.limiterCache.get(cacheKey);
    if (cached && Date.now() - cached.timestamp < this.cacheTimeout) {
      return cached.limiter;
    }

    // 创建新的限制器
    const limiter = new RateLimiter(this.kv, {
      windowMs: config.windowMs,
      maxRequests: config.maxRequests,
      keyGenerator: config.keyGenerator
    });

    // 缓存限制器
    this.limiterCache.set(cacheKey, {
      limiter,
      timestamp: Date.now()
    });

    // 定期清理缓存
    this.cleanupCache();

    return limiter;
  }

  cleanupCache() {
    if (this.cleanupTimer) return;

    this.cleanupTimer = setTimeout(() => {
      const now = Date.now();
      for (const [key, value] of this.limiterCache) {
        if (now - value.timestamp > this.cacheTimeout) {
          this.limiterCache.delete(key);
        }
      }
      this.cleanupTimer = null;
    }, 60000); // 1分钟后清理
  }
}

// 配置提供者示例
class RateLimitConfigProvider {
  async getConfig(request) {
    const url = new URL(request.url);
    const path = url.pathname;

    // 根据不同情况返回不同配置
    if (path.startsWith('/api/v1/')) {
      return {
        windowMs: 60 * 1000,
        maxRequests: 1000,
        keyGenerator: (req) => req.headers.get('CF-Connecting-IP')
      };
    } else if (path.startsWith('/api/v2/')) {
      return {
        windowMs: 60 * 1000,
        maxRequests: 2000,
        keyGenerator: (req) => req.headers.get('CF-Connecting-IP')
      };
    }

    return null;
  }
}
```

### 3. 分布式速率限制
```javascript
// 分布式速率限制协调器
export class DistributedRateLimiter {
  constructor(kv, nodeId, options = {}) {
    this.kv = kv;
    this.nodeId = nodeId;
    this.options = {
      syncInterval: options.syncInterval || 5000,     // 5秒同步间隔
      consistencyWindow: options.consistencyWindow || 1000, // 1秒一致性窗口
      ...options
    };
    this.localCounters = new Map();
    this.startSync();
  }

  async isAllowed(request) {
    const key = this.generateKey(request);
    const now = Date.now();

    // 获取本地计数
    let localCounter = this.localCounters.get(key);
    if (!localCounter) {
      localCounter = { count: 0, lastSync: now };
      this.localCounters.set(key, localCounter);
    }

    // 增加本地计数
    localCounter.count++;

    // 检查是否需要同步
    if (now - localCounter.lastSync > this.options.syncInterval) {
      await this.syncWithGlobal(key, localCounter);
    }

    // 保守估计：本地计数 + 同步延迟期间的全局计数
    const globalKey = `distributed:${key}`;
    const globalCounter = await this.kv.get(globalKey, 'json') || { count: 0 };

    const estimatedTotal = localCounter.count + globalCounter.count;
    const limit = this.options.maxRequests;

    if (estimatedTotal > limit) {
      return {
        allowed: false,
        limit,
        local: localCounter.count,
        global: globalCounter.count,
        estimated: estimatedTotal
      };
    }

    return {
      allowed: true,
      limit,
      local: localCounter.count,
      global: globalCounter.count,
      estimated: estimatedTotal
    };
  }

  async syncWithGlobal(key, localCounter) {
    const globalKey = `distributed:${key}`;
    const globalCounter = await this.kv.get(globalKey, 'json') || { count: 0 };

    // 合并计数
    globalCounter.count += localCounter.count;
    globalCounter.lastUpdate = Date.now();
    globalCounter.updatedBy = this.nodeId;

    // 重置本地计数
    localCounter.count = 0;
    localCounter.lastSync = Date.now();

    // 更新全局计数
    await this.kv.put(globalKey, JSON.stringify(globalCounter), {
      expirationTtl: Math.ceil(this.options.windowMs / 1000) + 10
    });
  }

  generateKey(request) {
    // 生成分布式键
    const identifier = request.headers.get('CF-Connecting-IP') || 'unknown';
    const path = new URL(request.url).pathname;
    return `${identifier}:${path}`;
  }

  startSync() {
    setInterval(async () => {
      // 同步所有本地计数器
      const now = Date.now();
      for (const [key, counter] of this.localCounters) {
        if (now - counter.lastSync > this.options.syncInterval) {
          await this.syncWithGlobal(key, counter);
        }
      }
    }, this.options.syncInterval);
  }
}
```

## 监控和分析

### 1. 速率限制监控
```javascript
// 速率限制监控器
export class RateLimitMonitor {
  constructor(kv) {
    this.kv = kv;
  }

  // 记录速率限制事件
  async recordEvent(identifier, eventType, details = {}) {
    const timestamp = Date.now();
    const event = {
      identifier,
      eventType, // 'allowed', 'blocked', 'adjusted'
      details,
      timestamp,
      date: new Date(timestamp).toISOString().split('T')[0]
    };

    // 存储事件
    const eventKey = `ratelimit-event:${timestamp}:${Math.random()}`;
    await this.kv.put(eventKey, JSON.stringify(event), {
      expirationTtl: 7 * 24 * 3600 // 7天
    });

    // 更新统计
    await this.updateDailyStats(event);
  }

  async updateDailyStats(event) {
    const statsKey = `ratelimit-stats:${event.date}`;
    const stats = await this.kv.get(statsKey, 'json') || {
      totalRequests: 0,
      blockedRequests: 0,
      allowedRequests: 0,
      byHour: new Array(24).fill(0),
      topViolators: {},
      byEndpoint: {}
    };

    // 更新总体统计
    stats.totalRequests++;

    const hour = new Date(event.timestamp).getHours();
    stats.byHour[hour]++;

    if (event.eventType === 'blocked') {
      stats.blockedRequests++;
      stats.topViolators[event.identifier] = (stats.topViolators[event.identifier] || 0) + 1;
    } else if (event.eventType === 'allowed') {
      stats.allowedRequests++;
    }

    // 按端点统计
    if (event.details.endpoint) {
      const endpoint = event.details.endpoint;
      stats.byEndpoint[endpoint] = (stats.byEndpoint[endpoint] || 0) + 1;
    }

    await this.kv.put(statsKey, JSON.stringify(stats), {
      expirationTtl: 30 * 24 * 3600 // 30天
    });
  }

  // 获取速率限制报告
  async getReport(days = 7) {
    const reports = [];
    const endDate = new Date();
    const startDate = new Date(endDate);
    startDate.setDate(startDate.getDate() - days);

    for (let i = 0; i < days; i++) {
      const date = new Date(startDate);
      date.setDate(date.getDate() + i);
      const dateStr = date.toISOString().split('T')[0];

      const stats = await this.kv.get(`ratelimit-stats:${dateStr}`, 'json');
      if (stats) {
        reports.push({
          date: dateStr,
          ...stats,
          blockRate: (stats.blockedRequests / stats.totalRequests * 100).toFixed(2) + '%'
        });
      }
    }

    return reports;
  }

  // 获取实时状态
  async getRealTimeStatus() {
    const now = Date.now();
    const oneHourAgo = now - (60 * 60 * 1000);

    // 搜索最近的事件
    const list = await this.kv.list({
      prefix: 'ratelimit-event:',
      limit: 1000
    });

    let recentEvents = [];
    for (const key of list.keys.slice(0, 100)) {
      const event = await this.kv.get(key.name, 'json');
      if (event && event.timestamp > oneHourAgo) {
        recentEvents.push(event);
      }
    }

    // 统计最近1小时的数据
    const stats = recentEvents.reduce((acc, event) => {
      acc.total++;
      if (event.eventType === 'blocked') acc.blocked++;
      if (event.eventType === 'allowed') acc.allowed++;
      return acc;
    }, { total: 0, blocked: 0, allowed: 0 });

    return {
      lastHour: stats,
      blockRate: stats.total > 0 ? (stats.blocked / stats.total * 100).toFixed(2) : '0',
      recentEvents: recentEvents.slice(-10)
    };
  }
}
```

### 2. 告警系统
```javascript
// 速率限制告警系统
export class RateLimitAlertSystem {
  constructor(kv, notificationService) {
    this.kv = kv;
    this.notification = notificationService;
    this.alerts = {
      highBlockRate: {
        threshold: 50, // 50% 阻塞率
        message: 'High block rate detected'
      },
      unusualPattern: {
        threshold: 1000, // 1分钟内1000次请求
        message: 'Unusual traffic pattern'
      },
      globalLimit: {
        threshold: 0.9, // 90% 全局限制
        message: 'Approaching global rate limit'
      }
    };
  }

  async checkAndAlert() {
    const monitor = new RateLimitMonitor(this.kv);
    const realTimeStatus = await monitor.getRealTimeStatus();

    // 检查高阻塞率
    if (realTimeStatus.lastHour.blocked > 100) {
      const blockRate = parseFloat(realTimeStatus.blockRate);
      if (blockRate > this.alerts.highBlockRate.threshold) {
        await this.sendAlert('high_block_rate', {
          blockRate,
          blocked: realTimeStatus.lastHour.blocked,
          total: realTimeStatus.lastHour.total
        });
      }
    }

    // 检查异常模式
    const recentMinuteEvents = realTimeStatus.recentEvents.filter(
      e => Date.now() - e.timestamp < 60000
    );

    if (recentMinuteEvents.length > this.alerts.unusualPattern.threshold) {
      const topViolators = this.getTopViolators(recentMinuteEvents);
      await this.sendAlert('unusual_pattern', {
        eventCount: recentMinuteEvents.length,
        topViolators
      });
    }
  }

  getTopViolators(events) {
    const violations = {};
    for (const event of events) {
      if (event.eventType === 'blocked') {
        violations[event.identifier] = (violations[event.identifier] || 0) + 1;
      }
    }

    return Object.entries(violations)
      .sort((a, b) => b[1] - a[1])
      .slice(0, 5)
      .map(([id, count]) => ({ identifier: id, count }));
  }

  async sendAlert(type, data) {
    const alert = {
      type,
      message: this.alerts[type].message,
      data,
      timestamp: new Date().toISOString(),
      severity: type === 'high_block_rate' ? 'critical' : 'warning'
    };

    // 存储告警
    await this.kv.put(`alert:${Date.now()}-${type}`, JSON.stringify(alert), {
      expirationTtl: 7 * 24 * 3600
    });

    // 发送通知
    await this.notification.send(alert);
  }
}
```
