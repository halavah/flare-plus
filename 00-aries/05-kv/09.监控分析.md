# KV 监控分析

## 监控概述

Cloudflare KV 提供了内置的监控功能，帮助开发者了解存储使用情况、性能指标和错误率。通过合理的监控和分析，可以优化应用性能、降低成本并提前发现潜在问题。

## 内置监控指标

### 1. 请求指标监控
```javascript
// 请求指标监控实现
export class KVRequestMonitor {
  constructor(env, namespace = 'MONITOR') {
    this.env = env;
    this.namespace = namespace;
    this.metricsKey = 'kv_metrics';
  }

  // 记录请求指标
  async recordRequest(operation, key, duration, success, region = null) {
    const timestamp = Date.now();
    const metrics = await this.getMetrics();

    // 基础指标
    metrics.total.requests++;
    metrics.total.operations[operation] = (metrics.total.operations[operation] || 0) + 1;

    if (success) {
      metrics.total.successful++;
    } else {
      metrics.total.failed++;
    }

    // 性能指标
    metrics.performance.durations.push(duration);
    metrics.performance.avgDuration = this.calculateAverage(metrics.performance.durations);

    // 记录操作类型性能
    if (!metrics.performance.operations[operation]) {
      metrics.performance.operations[operation] = {
        count: 0,
        totalDuration: 0,
        avgDuration: 0,
        maxDuration: 0,
        minDuration: Infinity
      };
    }

    const opMetrics = metrics.performance.operations[operation];
    opMetrics.count++;
    opMetrics.totalDuration += duration;
    opMetrics.avgDuration = opMetrics.totalDuration / opMetrics.count;
    opMetrics.maxDuration = Math.max(opMetrics.maxDuration, duration);
    opMetrics.minDuration = Math.min(opMetrics.minDuration, duration);

    // 区域指标
    if (region) {
      if (!metrics.regions[region]) {
        metrics.regions[region] = { requests: 0, avgDuration: 0 };
      }
      metrics.regions[region].requests++;
    }

    // 时间序列数据
    const timeSlot = Math.floor(timestamp / 60000) * 60000; // 1分钟粒度
    if (!metrics.timeSeries[timeSlot]) {
      metrics.timeSeries[timeSlot] = {
        requests: 0,
        successful: 0,
        failed: 0,
        avgDuration: 0,
        operations: {}
      };
    }

    const timeSlotData = metrics.timeSeries[timeSlot];
    timeSlotData.requests++;
    timeSlotData.successful += success ? 1 : 0;
    timeSlotData.failed += success ? 0 : 1;
    timeSlotData.operations[operation] = (timeSlotData.operations[operation] || 0) + 1;

    // 更新最后更新时间
    metrics.lastUpdated = timestamp;

    // 持久化指标
    await this.saveMetrics(metrics);

    return metrics;
  }

  // 获取指标
  async getMetrics() {
    const metricsValue = await this.env[this.namespace].get(this.metricsKey);

    if (!metricsValue) {
      return this.initializeMetrics();
    }

    try {
      return JSON.parse(metricsValue);
    } catch (error) {
      console.error('Failed to parse metrics:', error);
      return this.initializeMetrics();
    }
  }

  // 初始化指标结构
  initializeMetrics() {
    return {
      total: {
        requests: 0,
        successful: 0,
        failed: 0,
        operations: {}
      },
      performance: {
        durations: [],
        avgDuration: 0,
        operations: {}
      },
      regions: {},
      timeSeries: {},
      created: Date.now(),
      lastUpdated: Date.now()
    };
  }

  // 保存指标
  async saveMetrics(metrics) {
    // 限制时间序列数据量
    const cutoffTime = Date.now() - (24 * 60 * 60 * 1000); // 24小时
    metrics.timeSeries = Object.fromEntries(
      Object.entries(metrics.timeSeries).filter(([timestamp]) =>
        parseInt(timestamp) > cutoffTime
      )
    );

    // 限制性能历史数据
    if (metrics.performance.durations.length > 10000) {
      metrics.performance.durations = metrics.performance.durations.slice(-5000);
    }

    await this.env[this.namespace].put(
      this.metricsKey,
      JSON.stringify(metrics),
      {
        expirationTtl: 7 * 24 * 60 * 60 // 7天
      }
    );
  }

  // 计算平均值
  calculateAverage(numbers) {
    if (numbers.length === 0) return 0;
    return numbers.reduce((sum, num) => sum + num, 0) / numbers.length;
  }

  // 包装 KV 操作以自动记录指标
  async monitoredOperation(operation, key, ...args) {
    const startTime = Date.now();
    let success = false;
    let result = null;
    let error = null;

    try {
      switch (operation) {
        case 'get':
          result = await this.env[this.namespace].get(key, ...args);
          break;
        case 'put':
          result = await this.env[this.namespace].put(key, ...args);
          break;
        case 'delete':
          result = await this.env[this.namespace].delete(key, ...args);
          break;
        case 'list':
          result = await this.env[this.namespace].list(key, ...args);
          break;
        default:
          throw new Error(`Unsupported operation: ${operation}`);
      }

      success = true;
      return result;

    } catch (err) {
      error = err;
      throw err;

    } finally {
      const duration = Date.now() - startTime;
      await this.recordRequest(operation, key, duration, success);
    }
  }
}
```

### 2. 存储使用监控
```javascript
// 存储使用情况监控
export class KVStorageMonitor {
  constructor(env) {
    this.env = env;
    this.usageStatsKey = 'kv_usage_stats';
  }

  // 记录存储使用
  async recordStorageUsage(operation, key, size = 0) {
    const stats = await this.getUsageStats();
    const timestamp = Date.now();

    // 更新总统计
    stats.total.operations++;
    stats.total.bytesTransferred += size;

    if (operation === 'put') {
      stats.total.puts++;
      stats.current.entries++;
      stats.current.totalBytes += size;

      // 记录键大小分布
      this.recordKeySize(stats, key.length);
      this.recordValueSize(stats, size);

    } else if (operation === 'delete') {
      stats.total.deletes++;
      stats.current.entries = Math.max(0, stats.current.entries - 1);
      // 注意：删除时无法知道原值大小，所以这里不更新总字节数

    } else if (operation === 'get') {
      stats.total.gets++;
      stats.current.bytesRead += size;
    }

    // 按键前缀统计
    const prefix = this.extractPrefix(key);
    if (!stats.prefixes[prefix]) {
      stats.prefixes[prefix] = {
        operations: 0,
        entries: 0,
        bytes: 0
      };
    }

    stats.prefixes[prefix].operations++;
    if (operation === 'put') {
      stats.prefixes[prefix].entries++;
      stats.prefixes[prefix].bytes += size;
    }

    // 时间序列数据
    const timeSlot = Math.floor(timestamp / 3600000) * 3600000; // 1小时粒度
    if (!stats.timeSeries[timeSlot]) {
      stats.timeSeries[timeSlot] = {
        operations: 0,
        puts: 0,
        gets: 0,
        deletes: 0,
        bytesTransferred: 0
      };
    }

    const timeSlotData = stats.timeSeries[timeSlot];
    timeSlotData.operations++;
    timeSlotData[operation + 's'] = (timeSlotData[operation + 's'] || 0) + 1;
    timeSlotData.bytesTransferred += size;

    stats.lastUpdated = timestamp;

    await this.saveUsageStats(stats);
  }

  // 记录键大小分布
  recordKeySize(stats, size) {
    const bucket = this.getSizeBucket(size, 'key');
    stats.sizeDistribution.keys[bucket] = (stats.sizeDistribution.keys[bucket] || 0) + 1;
  }

  // 记录值大小分布
  recordValueSize(stats, size) {
    const bucket = this.getSizeBucket(size, 'value');
    stats.sizeDistribution.values[bucket] = (stats.sizeDistribution.values[bucket] || 0) + 1;
  }

  // 获取大小分桶
  getSizeBucket(size, type) {
    const buckets = type === 'key' ? [32, 64, 128, 256] : [1024, 4096, 16384, 65536, 262144];

    for (const bucket of buckets) {
      if (size <= bucket) {
        return `${bucket}B`;
      }
    }

    return '>';
  }

  // 提取键前缀
  extractPrefix(key) {
    const parts = key.split(':');
    return parts.length > 1 ? parts[0] : 'root';
  }

  // 获取使用统计
  async getUsageStats() {
    const statsValue = await this.env.KV_STATS.get(this.usageStatsKey);

    if (!statsValue) {
      return {
        total: {
          operations: 0,
          puts: 0,
          gets: 0,
          deletes: 0,
          bytesTransferred: 0
        },
        current: {
          entries: 0,
          totalBytes: 0,
          bytesRead: 0
        },
        prefixes: {},
        sizeDistribution: {
          keys: {},
          values: {}
        },
        timeSeries: {},
        created: Date.now(),
        lastUpdated: Date.now()
      };
    }

    try {
      return JSON.parse(statsValue);
    } catch (error) {
      console.error('Failed to parse usage stats:', error);
      return this.getUsageStats();
    }
  }

  // 保存使用统计
  async saveUsageStats(stats) {
    // 限制时间序列数据
    const cutoffTime = Date.now() - (30 * 24 * 60 * 60 * 1000); // 30天
    stats.timeSeries = Object.fromEntries(
      Object.entries(stats.timeSeries).filter(([timestamp]) =>
        parseInt(timestamp) > cutoffTime
      )
    );

    await this.env.KV_STATS.put(
      this.usageStatsKey,
      JSON.stringify(stats),
      {
        expirationTtl: 30 * 24 * 60 * 60 // 30天
      }
    );
  }

  // 生成使用报告
  async generateUsageReport() {
    const stats = await this.getUsageStats();

    return {
      summary: {
        totalOperations: stats.total.operations,
        currentEntries: stats.current.entries,
        totalBytes: stats.current.totalBytes,
        avgKeySize: this.calculateAverageKeySize(stats),
        avgValueSize: this.calculateAverageValueSize(stats)
      },
      operations: {
        puts: stats.total.puts,
        gets: stats.total.gets,
        deletes: stats.total.deletes
      },
      topPrefixes: this.getTopPrefixes(stats.prefixes, 10),
      sizeDistribution: stats.sizeDistribution,
      recentTrend: this.getRecentTrend(stats.timeSeries, 24) // 最近24小时
    };
  }

  // 计算平均键大小
  calculateAverageKeySize(stats) {
    // 这里简化处理，实际应该跟踪具体的键大小
    return 50; // 假设平均键大小
  }

  // 计算平均值大小
  calculateAverageValueSize(stats) {
    if (stats.current.entries === 0) return 0;
    return Math.round(stats.current.totalBytes / stats.current.entries);
  }

  // 获取热门前缀
  getTopPrefixes(prefixes, limit = 10) {
    return Object.entries(prefixes)
      .sort(([,a], [,b]) => b.operations - a.operations)
      .slice(0, limit)
      .map(([prefix, data]) => ({ prefix, ...data }));
  }

  // 获取最近趋势
  getRecentTrend(timeSeries, hours) {
    const cutoffTime = Date.now() - (hours * 60 * 60 * 1000);
    const recentData = Object.entries(timeSeries)
      .filter(([timestamp]) => parseInt(timestamp) > cutoffTime)
      .sort(([a], [b]) => parseInt(a) - parseInt(b));

    if (recentData.length === 0) {
      return { trend: 'stable', change: 0 };
    }

    const firstData = recentData[0][1];
    const lastData = recentData[recentData.length - 1][1];

    const change = ((lastData.operations - firstData.operations) / firstData.operations) * 100;

    return {
      trend: change > 10 ? 'increasing' : change < -10 ? 'decreasing' : 'stable',
      change: Math.round(change),
      dataPoints: recentData.length
    };
  }
}
```

### 3. 错误监控和分析
```javascript
// 错误监控和分析
export class KVErrorMonitor {
  constructor(env) {
    this.env = env;
    this.errorStatsKey = 'kv_error_stats';
  }

  // 记录错误
  async recordError(operation, key, error, context = {}) {
    const timestamp = Date.now();
    const errorType = this.classifyError(error);

    const errorRecord = {
      timestamp: timestamp,
      operation: operation,
      key: key,
      errorType: errorType,
      errorMessage: error.message,
      context: context
    };

    // 保存详细错误记录
    await this.saveErrorRecord(errorRecord);

    // 更新错误统计
    const stats = await this.getErrorStats();
    stats.total.errors++;
    stats.types[errorType] = (stats.types[errorType] || 0) + 1;
    stats.operations[operation] = (stats.operations[operation] || 0) + 1;

    // 时间序列数据
    const timeSlot = Math.floor(timestamp / 60000) * 60000; // 1分钟粒度
    if (!stats.timeSeries[timeSlot]) {
      stats.timeSeries[timeSlot] = {
        total: 0,
        types: {},
        operations: {}
      };
    }

    const timeSlotData = stats.timeSeries[timeSlot];
    timeSlotData.total++;
    timeSlotData.types[errorType] = (timeSlotData.types[errorType] || 0) + 1;
    timeSlotData.operations[operation] = (timeSlotData.operations[operation] || 0) + 1;

    stats.lastUpdated = timestamp;
    stats.lastError = errorRecord;

    await this.saveErrorStats(stats);

    return errorRecord;
  }

  // 分类错误类型
  classifyError(error) {
    const message = error.message.toLowerCase();

    if (message.includes('timeout')) {
      return 'timeout';
    } else if (message.includes('not found')) {
      return 'not_found';
    } else if (message.includes('permission') || message.includes('unauthorized')) {
      return 'permission';
    } else if (message.includes('rate limit') || message.includes('quota')) {
      return 'rate_limit';
    } else if (message.includes('capacity') || message.includes('space')) {
      return 'capacity';
    } else if (message.includes('network') || message.includes('connection')) {
      return 'network';
    } else {
      return 'unknown';
    }
  }

  // 保存错误记录
  async saveErrorRecord(errorRecord) {
    const errorKey = `error:${errorRecord.timestamp}:${Math.random()}`;

    await this.env.KV_ERRORS.put(
      errorKey,
      JSON.stringify(errorRecord),
      {
        expirationTtl: 7 * 24 * 60 * 60 // 7天
      }
    );
  }

  // 获取错误统计
  async getErrorStats() {
    const statsValue = await this.env.KV_STATS.get(this.errorStatsKey);

    if (!statsValue) {
      return {
        total: {
          errors: 0
        },
        types: {},
        operations: {},
        timeSeries: {},
        created: Date.now(),
        lastUpdated: Date.now()
      };
    }

    try {
      return JSON.parse(statsValue);
    } catch (error) {
      console.error('Failed to parse error stats:', error);
      return this.getErrorStats();
    }
  }

  // 保存错误统计
  async saveErrorStats(stats) {
    // 限制时间序列数据
    const cutoffTime = Date.now() - (24 * 60 * 60 * 1000); // 24小时
    stats.timeSeries = Object.fromEntries(
      Object.entries(stats.timeSeries).filter(([timestamp]) =>
        parseInt(timestamp) > cutoffTime
      )
    );

    await this.env.KV_STATS.put(
      this.errorStatsKey,
      JSON.stringify(stats),
      {
        expirationTtl: 7 * 24 * 60 * 60 // 7天
      }
    );
  }

  // 生成错误报告
  async generateErrorReport() {
    const stats = await this.getErrorStats();
    const recentErrors = await this.getRecentErrors(100);

    return {
      summary: {
        totalErrors: stats.total.errors,
        errorRate: this.calculateErrorRate(stats),
        mostCommonType: this.getMostCommonErrorType(stats.types),
        mostProblematicOperation: this.getMostProblematicOperation(stats.operations)
      },
      errorTypes: stats.types,
      operationErrors: stats.operations,
      recentErrors: recentErrors.slice(0, 10), // 最近10个错误
      trends: this.getErrorTrends(stats.timeSeries)
    };
  }

  // 计算错误率
  calculateErrorRate(errorStats) {
    // 需要结合总请求数来计算
    // 这里简化处理
    return 0.05; // 5% 错误率示例
  }

  // 获取最常见的错误类型
  getMostCommonErrorType(types) {
    return Object.entries(types)
      .sort(([,a], [,b]) => b - a)
      [0]?.[0] || 'none';
  }

  // 获取问题最多的操作
  getMostProblematicOperation(operations) {
    return Object.entries(operations)
      .sort(([,a], [,b]) => b - a)
      [0]?.[0] || 'none';
  }

  // 获取最近的错误
  async getRecentErrors(limit = 100) {
    const errors = await this.env.KV_ERRORS.list({
      prefix: 'error:',
      limit: limit
    });

    const errorRecords = [];

    for (const key of errors.keys) {
      const errorValue = await this.env.KV_ERRORS.get(key.name);
      if (errorValue) {
        try {
          errorRecords.push(JSON.parse(errorValue));
        } catch (error) {
          console.error('Failed to parse error record:', error);
        }
      }
    }

    return errorRecords.sort((a, b) => b.timestamp - a.timestamp);
  }

  // 获取错误趋势
  getErrorTrends(timeSeries) {
    const timeSlots = Object.keys(timeSeries)
      .map(Number)
      .sort((a, b) => a - b)
      .slice(-12); // 最近12个时间点

    if (timeSlots.length < 2) {
      return { trend: 'insufficient_data' };
    }

    const recentCount = timeSlots.slice(-6).reduce((sum, slot) =>
      sum + timeSeries[slot].total, 0);
    const previousCount = timeSlots.slice(0, -6).reduce((sum, slot) =>
      sum + timeSeries[slot].total, 0);

    const change = previousCount === 0 ? 0 :
      ((recentCount - previousCount) / previousCount) * 100;

    return {
      trend: change > 20 ? 'increasing' : change < -20 ? 'decreasing' : 'stable',
      change: Math.round(change),
      recentCount,
      previousCount
    };
  }
}
```

### 4. 性能基准测试
```javascript
// KV 性能基准测试
export class KVPerformanceBenchmark {
  constructor(env) {
    this.env = env;
    this.resultsKey = 'kv_benchmark_results';
  }

  // 运行性能基准测试
  async runBenchmark(options = {}) {
    const config = {
      iterations: options.iterations || 1000,
      concurrent: options.concurrent || 10,
      keySize: options.keySize || 32,
      valueSize: options.valueSize || 1024,
      namespace: options.namespace || 'BENCHMARK'
    };

    const results = {
      config: config,
      timestamp: Date.now(),
      operations: {}
    };

    // 写入性能测试
    console.log('Running write benchmark...');
    results.operations.write = await this.benchmarkWrite(config);

    // 读取性能测试
    console.log('Running read benchmark...');
    results.operations.read = await this.benchmarkRead(config);

    // 列表性能测试
    console.log('Running list benchmark...');
    results.operations.list = await this.benchmarkList(config);

    // 批量操作性能测试
    console.log('Running batch benchmark...');
    results.operations.batch = await this.benchmarkBatch(config);

    // 保存结果
    await this.saveBenchmarkResults(results);

    return results;
  }

  // 写入性能测试
  async benchmarkWrite(config) {
    const { iterations, concurrent, keySize, valueSize, namespace } = config;

    const testKey = this.generateTestKey(keySize);
    const testValue = this.generateTestValue(valueSize);
    const results = [];

    const startTime = Date.now();

    // 顺序写入测试
    console.log('Sequential writes...');
    const sequentialTimes = [];

    for (let i = 0; i < Math.min(iterations, 100); i++) {
      const key = `${testKey}:${i}`;
      const start = Date.now();

      await this.env[namespace].put(key, testValue);

      const duration = Date.now() - start;
      sequentialTimes.push(duration);
    }

    // 并发写入测试
    console.log('Concurrent writes...');
    const concurrentBatches = [];

    for (let batch = 0; batch < Math.ceil(iterations / concurrent); batch++) {
      const batchPromises = [];

      for (let i = 0; i < concurrent && (batch * concurrent + i) < iterations; i++) {
        const key = `${testKey}:concurrent:${batch * concurrent + i}`;
        const start = Date.now();

        const promise = this.env[namespace].put(key, testValue)
          .then(() => Date.now() - start);

        batchPromises.push(promise);
      }

      const batchResults = await Promise.all(batchPromises);
      concurrentBatches.push(...batchResults);
    }

    const totalTime = Date.now() - startTime;

    return {
      sequential: {
        operations: sequentialTimes.length,
        totalTime: sequentialTimes.reduce((sum, time) => sum + time, 0),
        avgTime: this.average(sequentialTimes),
        minTime: Math.min(...sequentialTimes),
        maxTime: Math.max(...sequentialTimes),
        p95: this.percentile(sequentialTimes.sort((a, b) => a - b), 95)
      },
      concurrent: {
        operations: concurrentBatches.length,
        totalTime: totalTime,
        avgTime: this.average(concurrentBatches),
        minTime: Math.min(...concurrentBatches),
        maxTime: Math.max(...concurrentBatches),
        p95: this.percentile(concurrentBatches.sort((a, b) => a - b), 95)
      },
      throughput: {
        sequentialOpsPerSec: Math.round(sequentialTimes.length / (sequentialTimes.reduce((sum, time) => sum + time, 0) / 1000)),
        concurrentOpsPerSec: Math.round(concurrentBatches.length / (totalTime / 1000))
      }
    };
  }

  // 读取性能测试
  async benchmarkRead(config) {
    const { iterations, concurrent, keySize, valueSize, namespace } = config;

    const testKey = this.generateTestKey(keySize);
    const testValue = this.generateTestValue(valueSize);

    // 预填充测试数据
    console.log('Pre-populating test data...');
    for (let i = 0; i < iterations; i++) {
      await this.env[namespace].put(`${testKey}:read:${i}`, testValue);
    }

    const readTimes = [];
    const startTime = Date.now();

    for (let i = 0; i < iterations; i++) {
      const key = `${testKey}:read:${i}`;
      const start = Date.now();

      await this.env[namespace].get(key);

      const duration = Date.now() - start;
      readTimes.push(duration);
    }

    const totalTime = Date.now() - startTime;

    return {
      operations: readTimes.length,
      totalTime: totalTime,
      avgTime: this.average(readTimes),
      minTime: Math.min(...readTimes),
      maxTime: Math.max(...readTimes),
      p95: this.percentile(readTimes.sort((a, b) => a - b), 95),
      hitRate: 100, // 所有键都应该存在
      throughput: Math.round(readTimes.length / (totalTime / 1000))
    };
  }

  // 列表性能测试
  async benchmarkList(config) {
    const { iterations, namespace } = config;

    const testKey = this.generateTestKey(16);

    // 预填充测试数据
    console.log('Pre-populating test data...');
    for (let i = 0; i < iterations; i++) {
      await this.env[namespace].put(`${testKey}:list:${i}`, `value-${i}`);
    }

    const listTimes = [];
    const listSizes = [];

    // 测试不同的 list 操作
    const testCases = [
      { prefix: `${testKey}:list:`, limit: 10 },
      { prefix: `${testKey}:list:`, limit: 100 },
      { prefix: `${testKey}:list:`, limit: 1000 }
    ];

    for (const testCase of testCases) {
      for (let run = 0; run < 10; run++) {
        const start = Date.now();

        const result = await this.env[namespace].list({
          prefix: testCase.prefix,
          limit: testCase.limit
        });

        const duration = Date.now() - start;
        listTimes.push(duration);
        listSizes.push(result.keys.length);
      }
    }

    return {
      operations: listTimes.length,
      avgTime: this.average(listTimes),
      minTime: Math.min(...listTimes),
      maxTime: Math.max(...listTimes),
      avgResultSize: this.average(listSizes),
      timePerKey: this.average(listTimes.map((time, i) => time / listSizes[i]))
    };
  }

  // 批量操作性能测试
  async benchmarkBatch(config) {
    const { iterations, concurrent, keySize, valueSize, namespace } = config;

    const testKey = this.generateTestKey(keySize);
    const testValue = this.generateTestValue(valueSize);

    // 准备批量操作
    const batchSize = Math.min(100, Math.ceil(iterations / 10)); // 每批最多100个操作
    const batchOperations = [];

    for (let batch = 0; batch < Math.ceil(iterations / batchSize); batch++) {
      const operations = [];

      for (let i = 0; i < batchSize && (batch * batchSize + i) < iterations; i++) {
        const key = `${testKey}:batch:${batch * batchSize + i}`;
        operations.push(this.env[namespace].put(key, testValue));
      }

      batchOperations.push(operations);
    }

    const batchTimes = [];
    const startTime = Date.now();

    for (const operations of batchOperations) {
      const start = Date.now();

      await Promise.all(operations);

      const duration = Date.now() - start;
      batchTimes.push(duration);
    }

    const totalTime = Date.now() - startTime;
    const totalOperations = batchOperations.reduce((sum, batch) => sum + batch.length, 0);

    return {
      batches: batchTimes.length,
      totalOperations: totalOperations,
      totalTime: totalTime,
      avgBatchTime: this.average(batchTimes),
      avgOperationsPerBatch: totalOperations / batchTimes.length,
      throughput: Math.round(totalOperations / (totalTime / 1000))
    };
  }

  // 生成测试键
  generateTestKey(size) {
    const chars = 'abcdefghijklmnopqrstuvwxyz0123456789';
    let key = '';
    for (let i = 0; i < size; i++) {
      key += chars[Math.floor(Math.random() * chars.length)];
    }
    return key;
  }

  // 生成测试值
  generateTestValue(size) {
    const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
    let value = '';
    for (let i = 0; i < size; i++) {
      value += chars[Math.floor(Math.random() * chars.length)];
    }
    return value;
  }

  // 计算平均值
  average(numbers) {
    if (numbers.length === 0) return 0;
    return numbers.reduce((sum, num) => sum + num, 0) / numbers.length;
  }

  // 计算百分位数
  percentile(sortedNumbers, p) {
    if (sortedNumbers.length === 0) return 0;
    const index = Math.ceil((p / 100) * sortedNumbers.length) - 1;
    return sortedNumbers[Math.max(0, index)];
  }

  // 保存基准测试结果
  async saveBenchmarkResults(results) {
    const timestamp = Date.now();
    const resultKey = `benchmark:${timestamp}`;

    await this.env.KV_BENCHMARK.put(
      resultKey,
      JSON.stringify(results),
      {
        expirationTtl: 30 * 24 * 60 * 60 // 30天
      }
    );

    // 更新结果索引
    const indexKey = 'benchmark_index';
    let index = [];

    try {
      const indexValue = await this.env.KV_BENCHMARK.get(indexKey);
      if (indexValue) {
        index = JSON.parse(indexValue);
      }
    } catch (error) {
      console.error('Failed to parse benchmark index:', error);
    }

    index.push(timestamp);
    index = index.slice(-10); // 只保留最近10次测试

    await this.env.KV_BENCHMARK.put(
      indexKey,
      JSON.stringify(index),
      {
        expirationTtl: 30 * 24 * 60 * 60
      }
    );
  }

  // 获取基准测试历史
  async getBenchmarkHistory(limit = 10) {
    const indexKey = 'benchmark_index';
    let timestamps = [];

    try {
      const indexValue = await this.env.KV_BENCHMARK.get(indexKey);
      if (indexValue) {
        timestamps = JSON.parse(indexValue);
      }
    } catch (error) {
      console.error('Failed to parse benchmark index:', error);
    }

    timestamps = timestamps.slice(-limit);
    const results = [];

    for (const timestamp of timestamps) {
      const resultKey = `benchmark:${timestamp}`;
      const resultValue = await this.env.KV_BENCHMARK.get(resultKey);

      if (resultValue) {
        try {
          results.push(JSON.parse(resultValue));
        } catch (error) {
          console.error('Failed to parse benchmark result:', error);
        }
      }
    }

    return results.sort((a, b) => b.timestamp - a.timestamp);
  }
}
```

```javascript
// 计算趋势
function calculateTrend(timeSeries, hours) {
  const cutoffTime = Date.now() - (hours * 60 * 60 * 1000);
  const recentData = Object.entries(timeSeries)
    .filter(([timestamp]) => parseInt(timestamp) > cutoffTime)
    .sort(([a], [b]) => parseInt(a) - parseInt(b));

  if (recentData.length < 2) {
    return { trend: 'stable', change: 0 };
  }

  const firstValue = recentData[0][1].total || recentData[0][1].operations || 0;
  const lastValue = recentData[recentData.length - 1][1].total || recentData[recentData.length - 1][1].operations || 0;

  const change = firstValue === 0 ? 0 : ((lastValue - firstValue) / firstValue) * 100;

  return {
    trend: change > 10 ? 'increasing' : change < -10 ? 'decreasing' : 'stable',
    change: Math.round(change),
    dataPoints: recentData.length
  };
}

// 生成告警
function generateAlerts(metrics, usage, errors) {
  const alerts = [];

  // 高错误率告警
  if (errors.summary.errorRate > 5) {
    alerts.push({
      level: 'error',
      message: `High error rate: ${errors.summary.errorRate}%`,
      type: 'error_rate'
    });
  }

  // 慢响应告警
  if (metrics.performance.avgDuration > 100) {
    alerts.push({
      level: 'warning',
      message: `Slow average response time: ${metrics.performance.avgDuration}ms`,
      type: 'response_time'
    });
  }

  // 存储空间告警
  if (usage.summary.currentEntries > 1000000) {
    alerts.push({
      level: 'warning',
      message: `High number of entries: ${usage.summary.currentEntries}`,
      type: 'storage_usage'
    });
  }

  // 错误趋势告警
  if (errors.trends.trend === 'increasing' && errors.trends.change > 50) {
    alerts.push({
      level: 'error',
      message: `Error rate increasing: ${errors.trends.change}%`,
      type: 'error_trend'
    });
  }

  return alerts;
}
```
