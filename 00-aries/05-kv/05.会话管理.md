# KV 会话管理

## 会话管理概述

在无服务器的 Cloudflare Workers 环境中，使用 KV 存储会话数据是一个高效的解决方案。KV 的全球分布特性确保了用户无论在何处都能快速访问会话数据。

### 2. 会话管理器实现
```javascript
// 会话管理器类
export class SessionManager {
  constructor(kv, options = {}) {
    this.kv = kv;
    this.options = {
      maxSessionsPerUser: options.maxSessionsPerUser || 5,
      sessionTimeout: options.sessionTimeout || 8 * 3600, // 8小时
      inactiveTimeout: options.inactiveTimeout || 30 * 60, // 30分钟
      cleanupInterval: options.cleanupInterval || 60, // 1分钟
      encryptionKey: options.encryptionKey
    };

    // 启动清理任务
    this.startCleanupTask();
  }

  // 创建新会话
  async createSession(userData, request) {
    const sessionId = this.generateSessionId();
    const now = new Date();

    const session = {
      sessionId,
      userId: userData.id,
      email: userData.email,
      role: userData.role || 'user',
      authenticated: true,
      createdAt: now.toISOString(),
      lastAccessed: now.toISOString(),
      expiresAt: new Date(now.getTime() + this.options.sessionTimeout * 1000).toISOString(),
      data: userData.data || {},
      security: {
        ip: request.headers.get('CF-Connecting-IP'),
        userAgent: request.headers.get('User-Agent'),
        lastActivity: now.toISOString()
      }
    };

    // 加密敏感数据
    const encryptedSession = await this.encryptSession(session);

    // 存储会话
    await this.kv.put(
      `session:${sessionId}`,
      JSON.stringify(encryptedSession),
      {
        expirationTtl: this.options.sessionTimeout
      }
    );

    // 更新用户的会话列表
    await this.updateUserSessions(userData.id, sessionId, 'add');

    // 存储快速查找令牌
    if (userData.rememberMe) {
      await this.kv.put(
        `token:${userData.token}`,
        sessionId,
        { expirationTtl: 30 * 24 * 3600 } // 30天
      );
    }

    // 记录创建日志
    await this.logSecurityEvent(sessionId, 'session_created', {
      userId: userData.id,
      ip: session.security.ip
    });

    return { sessionId, session };
  }

  // 获取会话
  async getSession(sessionId, request) {
    const cached = await this.kv.get(`session:${sessionId}`);
    if (!cached) {
      return null;
    }

    // 解密会话数据
    let session;
    try {
      session = await this.decryptSession(cached);
    } catch (error) {
      console.error('Failed to decrypt session:', error);
      await this.destroySession(sessionId);
      return null;
    }

    // 验证会话有效性
    if (!this.isSessionValid(session, request)) {
      await this.destroySession(sessionId);
      return null;
    }

    // 更新最后访问时间
    await this.updateLastAccessed(sessionId);

    return session;
  }

  // 通过令牌获取会话
  async getSessionByToken(token) {
    const sessionId = await this.kv.get(`token:${token}`);
    if (!sessionId) {
      return null;
    }

    // 创建一个模拟请求对象
    const mockRequest = {
      headers: {
        get: () => 'unknown'
      }
    };

    return await this.getSession(sessionId, mockRequest);
  }

  // 更新会话数据
  async updateSession(sessionId, updates) {
    const cached = await this.kv.get(`session:${sessionId}`);
    if (!cached) {
      throw new Error('Session not found');
    }

    let session = await this.decryptSession(cached);

    // 合并更新
    session = {
      ...session,
      ...updates,
      lastAccessed: new Date().toISOString()
    };

    // 加密并存储
    const encryptedSession = await this.encryptSession(session);

    await this.kv.put(
      `session:${sessionId}`,
      JSON.stringify(encryptedSession),
      {
        expirationTtl: this.options.sessionTimeout
      }
    );

    return session;
  }

  // 销毁会话
  async destroySession(sessionId) {
    const session = await this.getSession(sessionId, {});
    if (!session) {
      return;
    }

    // 删除主会话
    await this.kv.delete(`session:${sessionId}`);

    // 更新用户的会话列表
    await this.updateUserSessions(session.userId, sessionId, 'remove');

    // 删除令牌映射
    await this.kv.delete(`token:${session.token || ''}`);

    // 记录销毁日志
    await this.logSecurityEvent(sessionId, 'session_destroyed', {
      userId: session.userId,
      duration: Date.now() - new Date(session.createdAt).getTime()
    });
  }

  // 更新用户会话列表
  async updateUserSessions(userId, sessionId, action) {
    const key = `user:${userId}:sessions`;
    const sessions = await this.kv.get(key, 'json') || [];

    if (action === 'add') {
      sessions.push(sessionId);

      // 限制会话数量
      if (sessions.length > this.options.maxSessionsPerUser) {
        // 删除最旧的会话
        const oldestSession = sessions.shift();
        await this.destroySession(oldestSession);
      }
    } else if (action === 'remove') {
      const index = sessions.indexOf(sessionId);
      if (index > -1) {
        sessions.splice(index, 1);
      }
    }

    await this.kv.put(key, JSON.stringify(sessions));
  }

  // 验证会话有效性
  isSessionValid(session, request) {
    // 检查过期时间
    if (new Date() > new Date(session.expiresAt)) {
      return false;
    }

    // 检查不活跃时间
    const inactiveTime = Date.now() - new Date(session.lastAccessed).getTime();
    if (inactiveTime > this.options.inactiveTimeout * 1000) {
      return false;
    }

    // 检查 IP 地址变化（可选）
    if (session.security.ip && request.headers) {
      const currentIP = request.headers.get('CF-Connecting-IP');
      if (currentIP && currentIP !== session.security.ip) {
        // 记录可疑活动
        this.logSecurityEvent(session.sessionId, 'ip_changed', {
          oldIP: session.security.ip,
          newIP: currentIP
        });
      }
    }

    return true;
  }

  // 更新最后访问时间
  async updateLastAccessed(sessionId) {
    const cached = await this.kv.get(`session:${sessionId}`);
    if (!cached) return;

    let session = await this.decryptSession(cached);
    session.lastAccessed = new Date().toISOString();
    session.security.lastActivity = new Date().toISOString();

    const encryptedSession = await this.encryptSession(session);
    await this.kv.put(`session:${sessionId}`, JSON.stringify(encryptedSession));
  }

  // 生成会话ID
  generateSessionId() {
    const array = new Uint8Array(32);
    crypto.getRandomValues(array);
    return Array.from(array, byte => byte.toString(16).padStart(2, '0')).join('');
  }

  // 加密会话数据
  async encryptSession(session) {
    if (!this.options.encryptionKey) {
      return session;
    }

    // 实现加密逻辑
    const sessionString = JSON.stringify(session);
    const encrypted = await this.encrypt(sessionString, this.options.encryptionKey);
    return { encrypted };
  }

  // 解密会话数据
  async decryptSession(cached) {
    if (typeof cached === 'string') {
      return JSON.parse(cached);
    }

    if (cached.encrypted) {
      const decrypted = await this.decrypt(cached.encrypted, this.options.encryptionKey);
      return JSON.parse(decrypted);
    }

    return cached;
  }

  // 加密方法（示例）
  async encrypt(data, key) {
    // 实际实现应使用 Web Crypto API
    return Buffer.from(data).toString('base64');
  }

  // 解密方法（示例）
  async decrypt(data, key) {
    // 实际实现应使用 Web Crypto API
    return Buffer.from(data, 'base64').toString();
  }

  // 记录安全事件
  async logSecurityEvent(sessionId, eventType, details) {
    const logEntry = {
      sessionId,
      eventType,
      details,
      timestamp: new Date().toISOString()
    };

    const logKey = `security:${sessionId}:log:${Date.now()}`;
    await this.kv.put(logKey, JSON.stringify(logEntry), {
      expirationTtl: 7 * 24 * 3600 // 7天
    });
  }

  // 启动清理任务
  startCleanupTask() {
    setInterval(async () => {
      await this.cleanupExpiredSessions();
    }, this.options.cleanupInterval * 1000);
  }

  // 清理过期会话
  async cleanupExpiredSessions() {
    // 获取所有会话键
    const list = await this.kv.list({
      prefix: 'session:',
      limit: 1000
    });

    const now = Date.now();
    let cleanedCount = 0;

    for (const key of list.keys) {
      try {
        const cached = await this.kv.get(key.name);
        if (!cached) continue;

        let session;
        try {
          session = await this.decryptSession(cached);
        } catch {
          // 无法解密，删除
          await this.kv.delete(key.name);
          cleanedCount++;
          continue;
        }

        // 检查是否过期
        if (new Date(session.expiresAt) < new Date(now)) {
          await this.destroySession(session.sessionId);
          cleanedCount++;
        }
      } catch (error) {
        console.error(`Error cleaning session ${key.name}:`, error);
      }
    }

    if (cleanedCount > 0) {
      console.log(`Cleaned up ${cleanedCount} expired sessions`);
    }
  }
}
```

### 1. 多设备会话管理
```javascript
// 多设备会话管理
export class MultiDeviceSessionManager extends SessionManager {
  async createDeviceSession(userData, request, deviceInfo) {
    const session = await super.createSession(userData, request);

    // 添加设备信息
    await this.updateSession(session.sessionId, {
      device: {
        id: deviceInfo.id || 'unknown',
        type: deviceInfo.type || 'web',
        name: deviceInfo.name || 'Unknown Device',
        platform: deviceInfo.platform,
        trusted: deviceInfo.trusted || false
      }
    });

    // 存储设备映射
    await this.kv.put(
      `device:${userData.id}:${deviceInfo.id}`,
      session.sessionId,
      { expirationTtl: this.options.sessionTimeout }
    );

    return session;
  }

  async getDeviceSessions(userId) {
    const list = await this.kv.list({
      prefix: `device:${userId}:`,
      limit: 10
    });

    const sessions = [];
    for (const key of list.keys) {
      const sessionId = await this.kv.get(key.name);
      if (sessionId) {
        const session = await this.getSession(sessionId, {});
        if (session) {
          sessions.push({
            sessionId,
            device: session.device,
            lastAccessed: session.lastAccessed
          });
        }
      }
    }

    return sessions.sort((a, b) => new Date(b.lastAccessed) - new Date(a.lastAccessed));
  }

  async revokeDevice(userId, deviceId) {
    const sessionId = await this.kv.get(`device:${userId}:${deviceId}`);
    if (sessionId) {
      await this.destroySession(sessionId);
      await this.kv.delete(`device:${userId}:${deviceId}`);
    }
  }

  async revokeAllOtherDevices(userId, currentDeviceId) {
    const sessions = await this.getDeviceSessions(userId);

    for (const session of sessions) {
      if (session.device.id !== currentDeviceId) {
        await this.revokeDevice(userId, session.device.id);
      }
    }
  }
}
```

```javascript
// 会话分析器
export class SessionAnalytics {
  constructor(kv) {
    this.kv = kv;
  }

  // 获取活跃会话统计
  async getActiveSessionStats() {
    const list = await this.kv.list({
      prefix: 'session:',
      limit: 10000
    });

    const now = Date.now();
    const stats = {
      total: list.keys.length,
      active: 0,        // 5分钟内活跃
      recent: 0,        // 1小时内活跃
      expired: 0,       // 过期的会话
      byAge: {
        under1h: 0,
        under6h: 0,
        under24h: 0,
        over24h: 0
      },
      byDevice: {
        web: 0,
        mobile: 0,
        desktop: 0,
        unknown: 0
      }
    };

    const oneHourAgo = now - (60 * 60 * 1000);
    const fiveMinutesAgo = now - (5 * 60 * 1000);

    for (const key of list.keys.slice(0, 100)) { // 采样检查
      try {
        const cached = await this.kv.get(key.name);
        if (!cached) continue;

        let session;
        try {
          session = JSON.parse(cached);
          if (session.encrypted) continue;
        } catch {
          continue;
        }

        const lastAccess = new Date(session.lastAccessed).getTime();
        const age = now - new Date(session.createdAt).getTime();

        // 按活跃时间统计
        if (lastAccess > fiveMinutesAgo) {
          stats.active++;
        }
        if (lastAccess > oneHourAgo) {
          stats.recent++;
        }

        // 按年龄统计
        if (age < 60 * 60 * 1000) {
          stats.byAge.under1h++;
        } else if (age < 6 * 60 * 60 * 1000) {
          stats.byAge.under6h++;
        } else if (age < 24 * 60 * 60 * 1000) {
          stats.byAge.under24h++;
        } else {
          stats.byAge.over24h++;
        }

        // 按设备类型统计
        if (session.device) {
          const deviceType = session.device.type || 'unknown';
          stats.byDevice[deviceType] = (stats.byDevice[deviceType] || 0) + 1;
        }
      } catch (error) {
        console.error(`Error analyzing session ${key.name}:`, error);
      }
    }

    return stats;
  }

  // 获取用户行为分析
  async getUserBehaviorAnalytics(userId, days = 7) {
    const endDate = new Date();
    const startDate = new Date(endDate);
    startDate.setDate(startDate.getDate() - days);

    const analytics = {
      sessionCount: 0,
      totalDuration: 0,
      averageDuration: 0,
      devices: new Set(),
      peakHours: new Array(24).fill(0),
      activities: {
        logins: 0,
        logouts: 0,
        pageViews: 0
      }
    };

    // 获取用户的安全日志
    const securityLogs = await this.getUserSecurityLogs(userId, days);

    for (const log of securityLogs) {
      analytics.sessionCount++;
      analytics.devices.add(log.ip); // 简单的设备标识

      const hour = new Date(log.timestamp).getHours();
      analytics.peakHours[hour]++;

      switch (log.eventType) {
        case 'session_created':
          analytics.activities.logins++;
          break;
        case 'session_destroyed':
          analytics.activities.logouts++;
          break;
      }
    }

    analytics.deviceCount = analytics.devices.size;
    analytics.averageDuration = analytics.sessionCount > 0
      ? analytics.totalDuration / analytics.sessionCount
      : 0;

    return analytics;
  }

  // 获取用户安全日志
  async getUserSecurityLogs(userId, days) {
    const logs = [];
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - days);

    // 搜索安全日志
    const list = await this.kv.list({
      prefix: 'security:',
      limit: 1000
    });

    for (const key of list.keys.slice(0, 100)) {
      try {
        const log = await this.kv.get(key.name, 'json');
        if (log && log.details.userId === userId &&
            new Date(log.timestamp) > cutoffDate) {
          logs.push(log);
        }
      } catch (error) {
        // 忽略错误
      }
    }

    return logs.sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));
  }
}
```

### 3. 安全增强
```javascript
// 安全增强的会话管理
export class SecureSessionManager extends SessionManager {
  constructor(kv, options = {}) {
    super(kv, options);
    this.securityOptions = {
      maxFailedAttempts: options.maxFailedAttempts || 5,
      lockoutDuration: options.lockoutDuration || 15 * 60, // 15分钟
      suspiciousActivityThreshold: options.suspiciousActivityThreshold || 3,
      ...options.security
    };
  }

  // 检查可疑活动
  async checkSuspiciousActivity(sessionId, request) {
    const logs = await this.getRecentSecurityLogs(sessionId, 24); // 24小时内

    const suspiciousIndicators = {
      multipleIPs: new Set(),
      multipleDevices: new Set(),
      rapidAccess: [],
      unusualTimes: []
    };

    for (const log of logs) {
      if (log.details.ip) {
        suspiciousIndicators.multipleIPs.add(log.details.ip);
      }

      if (log.details.device) {
        suspiciousIndicators.multipleDevices.add(log.details.device);
      }

      // 检查快速连续访问
      const accessTimes = logs
        .filter(l => l.eventType === 'session_accessed')
        .map(l => new Date(l.timestamp).getTime())
        .sort();

      for (let i = 1; i < accessTimes.length; i++) {
        const interval = accessTimes[i] - accessTimes[i - 1];
        if (interval < 1000) { // 1秒内多次访问
          suspiciousIndicators.rapidAccess.push(interval);
        }
      }
    }

    const suspiciousScore =
      (suspiciousIndicators.multipleIPs.size - 1) * 10 +
      (suspiciousIndicators.multipleDevices.size - 1) * 5 +
      suspiciousIndicators.rapidAccess.length * 2;

    if (suspiciousScore > this.securityOptions.suspiciousActivityThreshold) {
      await this.handleSuspiciousActivity(sessionId, suspiciousIndicators);
      return { suspicious: true, score: suspiciousScore };
    }

    return { suspicious: false, score: 0 };
  }

  // 处理可疑活动
  async handleSuspiciousActivity(sessionId, indicators) {
    // 记录安全事件
    await this.logSecurityEvent(sessionId, 'suspicious_activity', {
      indicators: {
        multipleIPs: indicators.multipleIPs.size,
        multipleDevices: indicators.multipleDevices.size,
        rapidAccessCount: indicators.rapidAccess.length
      }
    });

    // 可选：要求重新认证
    await this.requireReauthentication(sessionId);
  }

  // 要求重新认证
  async requireReauthentication(sessionId) {
    await this.updateSession(sessionId, {
      requiresReauth: true,
      reauthReason: 'suspicious_activity'
    });
  }

  // 速率限制
  async checkRateLimit(identifier, limit = 10, window = 60) {
    const key = `ratelimit:${identifier}:${Math.floor(Date.now() / (window * 1000))}`;
    const current = await this.kv.get(key, 'json') || 0;

    if (current >= limit) {
      throw new Error('Rate limit exceeded');
    }

    await this.kv.put(key, JSON.stringify(current + 1), {
      expirationTtl: window
    });
  }

  // 失败登录跟踪
  async trackFailedLogin(email, ip) {
    const key = `failed:${email}:${ip}`;
    const attempts = await this.kv.get(key, 'json') || 0;

    if (attempts >= this.securityOptions.maxFailedAttempts) {
      // 锁定账户
      await this.lockAccount(email, ip);
      throw new Error('Account temporarily locked');
    }

    await this.kv.put(key, JSON.stringify(attempts + 1), {
      expirationTtl: this.securityOptions.lockoutDuration
    });
  }

  // 锁定账户
  async lockAccount(email, ip) {
    const lockKey = `lock:${email}`;
    await this.kv.put(lockKey, JSON.stringify({
      email,
      ip,
      lockedAt: new Date().toISOString(),
      reason: 'too_many_failed_attempts'
    }), {
      expirationTtl: this.securityOptions.lockoutDuration
    });
  }

  // 检查账户锁定状态
  async isAccountLocked(email) {
    const lockKey = `lock:${email}`;
    const lockInfo = await this.kv.get(lockKey, 'json');
    return !!lockInfo;
  }
}
```

### 1. 登录流程
```javascript
// 登录处理
export async function handleLogin(request, env) {
  const sessionManager = new SecureSessionManager(env.MY_KV);

  try {
    const { email, password, rememberMe } = await request.json();
    const ip = request.headers.get('CF-Connecting-IP');

    // 检查账户锁定
    if (await sessionManager.isAccountLocked(email)) {
      return new Response(JSON.stringify({
        error: 'Account temporarily locked'
      }), { status: 429 });
    }

    // 验证凭据
    const user = await validateCredentials(email, password);
    if (!user) {
      await sessionManager.trackFailedLogin(email, ip);
      return new Response(JSON.stringify({
        error: 'Invalid credentials'
      }), { status: 401 });
    }

    // 创建会话
    const deviceInfo = {
      type: 'web',
      platform: navigator.platform,
      trusted: false
    };

    const { sessionId, session } = await sessionManager.createDeviceSession(
      user,
      request,
      deviceInfo
    );

    // 设置响应
    const response = new Response(JSON.stringify({
      success: true,
      user: {
        id: user.id,
        email: user.email,
        role: user.role
      }
    }));

    // 设置会话 Cookie
    response.headers.set('Set-Cookie',
      `session=${sessionId}; HttpOnly; Secure; Path=/; ${
        rememberMe ? 'Max-Age=2592000' : ''
      }`
    );

    return response;

  } catch (error) {
    console.error('Login error:', error);
    return new Response(JSON.stringify({
      error: 'Login failed'
    }), { status: 500 });
  }
}

// 中间件：会话验证
export async function requireAuth(request, env) {
  const sessionManager = new SessionManager(env.MY_KV);

  // 从 Cookie 获取会话 ID
  const cookieHeader = request.headers.get('Cookie');
  const sessionId = cookieHeader
    ? cookieHeader.split(';')
        .find(c => c.trim().startsWith('session='))
        ?.split('=')[1]
    : null;

  if (!sessionId) {
    return { authenticated: false };
  }

  // 获取会话
  const session = await sessionManager.getSession(sessionId, request);
  if (!session) {
    return { authenticated: false };
  }

  // 检查是否需要重新认证
  if (session.requiresReauth) {
    return {
      authenticated: false,
      requiresReauth: true,
      reason: session.reauthReason
    };
  }

  // 检查可疑活动
  const securityCheck = await sessionManager.checkSuspiciousActivity(sessionId, request);
  if (securityCheck.suspicious) {
    return {
      authenticated: false,
      suspicious: true
    };
  }

  return {
    authenticated: true,
    session,
    sessionId
  };
}
```

### 2. API 路由保护
```javascript
// API 路由保护示例
export async function handleAPIRequest(request, env) {
  const auth = await requireAuth(request, env);

  if (!auth.authenticated) {
    if (auth.requiresReauth) {
      return new Response(JSON.stringify({
        error: 'Reauthentication required',
        reason: auth.reason
      }), { status: 401 });
    }

    if (auth.suspicious) {
      return new Response(JSON.stringify({
        error: 'Suspicious activity detected'
      }), { status: 403 });
    }

    return new Response(JSON.stringify({
      error: 'Authentication required'
    }), { status: 401 });
  }

  // 继续处理 API 请求
  const url = new URL(request.url);
  const path = url.pathname;

  switch (path) {
    case '/api/user/profile':
      return getUserProfile(auth.session, env);
    case '/api/user/update':
      return updateUserProfile(auth.session, env, request);
    case '/api/logout':
      return handleLogout(auth.sessionId, env);
    default:
      return new Response('Not found', { status: 404 });
  }
}

// 获取用户资料
async function getUserProfile(session, env) {
  const sessionManager = new SessionManager(env.MY_KV);

  // 更新最后访问时间
  await sessionManager.updateLastAccessed(session.sessionId);

  return new Response(JSON.stringify({
    user: {
      id: session.userId,
      email: session.email,
      role: session.role,
      preferences: session.data.preferences || {}
    }
  }));
}

// 更新用户资料
async function updateUserProfile(session, env, request) {
  const updates = await request.json();
  const sessionManager = new SessionManager(env.MY_KV);

  // 更新会话数据
  const updatedSession = await sessionManager.updateSession(
    session.sessionId,
    {
      data: {
        ...session.data,
        ...updates
      }
    }
  );

  return new Response(JSON.stringify({
    success: true,
    message: 'Profile updated'
  }));
}

// 处理登出
async function handleLogout(sessionId, env) {
  const sessionManager = new SessionManager(env.MY_KV);
  await sessionManager.destroySession(sessionId);

  const response = new Response(JSON.stringify({
    success: true,
    message: 'Logged out'
  }));

  // 清除 Cookie
  response.headers.set('Set-Cookie',
    'session=; HttpOnly; Secure; Path=/; Expires=Thu, 01 Jan 1970 00:00:00 GMT'
  );

  return response;
}
```
