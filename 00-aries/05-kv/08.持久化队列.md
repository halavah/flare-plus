# KV 持久化队列

## 队列概述

Cloudflare KV 可以用作持久化队列系统，虽然不是专门的队列服务，但通过合理的设计可以实现可靠的队列功能。KV 的全球分布式特性使其特别适合需要全球处理能力的队列场景。

## 队列模式

### 1. 简单 FIFO 队列
```javascript
// 简单的先进先出队列
export class SimpleKVQueue {
  constructor(env, queueName) {
    this.env = env;
    this.queueName = queueName;
    this.listKey = `queue:${queueName}:list`;
    this.counterKey = `queue:${queueName}:counter`;
    this.processingKey = `queue:${queueName}:processing`;
  }

  // 入队
  async enqueue(data) {
    const messageId = await this.getNextId();
    const message = {
      id: messageId,
      data: data,
      timestamp: Date.now(),
      attempts: 0
    };

    // 使用 KV 的 list 操作添加到队列
    await this.env.QUEUE_KV.list({
      prefix: `${this.queueName}:`,
      limit: 1
    });

    // 存储消息
    await this.env.QUEUE_KV.put(
      `${this.queueName}:msg:${messageId}`,
      JSON.stringify(message),
      {
        expirationTtl: 24 * 60 * 60 // 24小时过期
      }
    );

    // 更新队列索引
    await this.env.QUEUE_KV.put(
      this.listKey,
      `${messageId}:${Date.now()}`,
      {
        metadata: {
          action: 'enqueue',
          messageId: messageId
        }
      }
    );

    return messageId;
  }

  // 出队
  async dequeue() {
    const listValue = await this.env.QUEUE_KV.get(this.listKey);
    if (!listValue) return null;

    const messageId = listValue.split(':')[0];
    const messageKey = `${this.queueName}:msg:${messageId}`;

    const messageValue = await this.env.QUEUE_KV.get(messageKey);
    if (!messageValue) {
      // 消息不存在，从队列中移除
      await this.env.QUEUE_KV.delete(this.listKey);
      return null;
    }

    try {
      const message = JSON.parse(messageValue);

      // 标记为处理中
      message.status = 'processing';
      message.processStarted = Date.now();

      await this.env.QUEUE_KV.put(
        messageKey,
        JSON.stringify(message),
        {
          expirationTtl: 5 * 60 // 5分钟处理时间
        }
      );

      // 从队列列表中移除
      await this.env.QUEUE_KV.delete(this.listKey);

      return { messageId, message };
    } catch (error) {
      console.error('Failed to parse message:', error);
      return null;
    }
  }

  // 完成处理
  async complete(messageId) {
    const messageKey = `${this.queueName}:msg:${messageId}`;
    await this.env.QUEUE_KV.delete(messageKey);
  }

  // 获取下一个消息ID
  async getNextId() {
    const counter = await this.env.QUEUE_KV.get(this.counterKey);
    const nextId = counter ? parseInt(counter) + 1 : 1;

    await this.env.QUEUE_KV.put(
      this.counterKey,
      nextId.toString(),
      {
        expirationTtl: 365 * 24 * 60 * 60 // 1年过期
      }
    );

    return nextId;
  }
}
```

### 2. 优先级队列
```javascript
// 支持优先级的队列
export class PriorityQueue {
  constructor(env, queueName) {
    this.env = env;
    this.queueName = queueName;
    this.priorities = ['high', 'medium', 'low'];
  }

  // 根据优先级入队
  async enqueue(data, priority = 'medium') {
    if (!this.priorities.includes(priority)) {
      priority = 'medium';
    }

    const messageId = await this.generateId();
    const message = {
      id: messageId,
      data: data,
      priority: priority,
      timestamp: Date.now(),
      attempts: 0
    };

    const queueKey = `${this.queueName}:${priority}:queue`;
    const messageKey = `${this.queueName}:msg:${messageId}`;

    // 存储消息
    await this.env.QUEUE_KV.put(
      messageKey,
      JSON.stringify(message),
      {
        expirationTtl: 24 * 60 * 60,
        metadata: {
          priority: priority,
          queue: queueKey
        }
      }
    );

    // 添加到对应优先级队列
    await this.env.QUEUE_KV.put(
      queueKey,
      `${messageId}:${Date.now()}`,
      {
        metadata: {
          messageId: messageId,
          priority: priority
        }
      }
    );

    return messageId;
  }

  // 按优先级出队
  async dequeue() {
    // 按优先级顺序检查队列
    for (const priority of this.priorities) {
      const queueKey = `${this.queueName}:${priority}:queue`;
      const message = await this.dequeueFromQueue(queueKey, priority);
      if (message) return message;
    }

    return null;
  }

  // 从特定优先级队列出队
  async dequeueFromQueue(queueKey, priority) {
    const listValue = await this.env.QUEUE_KV.get(queueKey);
    if (!listValue) return null;

    const messageId = listValue.split(':')[0];
    const messageKey = `${this.queueName}:msg:${messageId}`;

    const messageValue = await this.env.QUEUE_KV.get(messageKey);
    if (!messageValue) {
      await this.env.QUEUE_KV.delete(queueKey);
      return null;
    }

    try {
      const message = JSON.parse(messageValue);

      // 标记为处理中
      message.status = 'processing';
      message.processStarted = Date.now();

      await this.env.QUEUE_KV.put(
        messageKey,
        JSON.stringify(message),
        {
          expirationTtl: 5 * 60
        }
      );

      await this.env.QUEUE_KV.delete(queueKey);

      return { messageId, message };
    } catch (error) {
      console.error('Failed to parse message:', error);
      return null;
    }
  }

  // 批量出队
  async dequeueBatch(maxItems = 10) {
    const messages = [];

    for (let i = 0; i < maxItems; i++) {
      const message = await this.dequeue();
      if (!message) break;
      messages.push(message);
    }

    return messages;
  }
}
```

### 3. 延迟队列
```javascript
// 支持延迟处理的队列
export class DelayedQueue {
  constructor(env, queueName) {
    this.env = env;
    this.queueName = queueName;
    this.delayedKey = `${queueName}:delayed`;
    this.readyKey = `${queueName}:ready`;
  }

  // 延迟入队
  async enqueue(data, delayMs = 0) {
    const messageId = await this.generateId();
    const executeTime = Date.now() + delayMs;

    const message = {
      id: messageId,
      data: data,
      timestamp: Date.now(),
      executeTime: executeTime,
      attempts: 0
    };

    if (delayMs <= 0) {
      // 无延迟，直接进入就绪队列
      await this.addToReadyQueue(messageId, message);
    } else {
      // 添加到延迟队列
      await this.env.QUEUE_KV.put(
        `${this.delayedKey}:${messageId}`,
        JSON.stringify(message),
        {
          expirationTtl: Math.ceil(delayMs / 1000) + 60,
          metadata: {
            executeTime: executeTime.toString(),
            messageId: messageId
          }
        }
      );
    }

    return messageId;
  }

  // 转移延迟消息到就绪队列
  async moveDelayedMessages() {
    const now = Date.now();

    // 获取所有延迟消息
    const delayedMessages = await this.env.QUEUE_KV.list({
      prefix: `${this.delayedKey}:`
    });

    const messagesToMove = [];

    for (const key of delayedMessages.keys) {
      const messageValue = await this.env.QUEUE_KV.get(key.name);
      if (!messageValue) continue;

      try {
        const message = JSON.parse(messageValue);

        if (message.executeTime <= now) {
          messagesToMove.push({
            key: key.name,
            message: message
          });
        }
      } catch (error) {
        console.error('Failed to parse delayed message:', error);
      }
    }

    // 批量转移
    for (const { key, message } of messagesToMove) {
      // 添加到就绪队列
      await this.addToReadyQueue(message.id, message);

      // 从延迟队列删除
      await this.env.QUEUE_KV.delete(key);
    }

    return messagesToMove.length;
  }

  // 添加到就绪队列
  async addToReadyQueue(messageId, message) {
    await this.env.QUEUE_KV.put(
      `${this.queueName}:msg:${messageId}`,
      JSON.stringify(message),
      {
        expirationTtl: 24 * 60 * 60
      }
    );

    await this.env.QUEUE_KV.put(
      this.readyKey,
      messageId,
      {
        metadata: {
          messageId: messageId,
          timestamp: Date.now().toString()
        }
      }
    );
  }

  // 出队就绪消息
  async dequeue() {
    const messageId = await this.env.QUEUE_KV.get(this.readyKey);
    if (!messageId) return null;

    const messageKey = `${this.queueName}:msg:${messageId}`;
    const messageValue = await this.env.QUEUE_KV.get(messageKey);

    if (!messageValue) {
      await this.env.QUEUE_KV.delete(this.readyKey);
      return null;
    }

    try {
      const message = JSON.parse(messageValue);

      // 标记为处理中
      message.status = 'processing';
      message.processStarted = Date.now();

      await this.env.QUEUE_KV.put(
        messageKey,
        JSON.stringify(message),
        {
          expirationTtl: 5 * 60
        }
      );

      await this.env.QUEUE_KV.delete(this.readyKey);

      return { messageId, message };
    } catch (error) {
      console.error('Failed to parse message:', error);
      return null;
    }
  }
}
```

## 死信队列

### 1. 死信队列实现
```javascript
// 死信队列处理
export class DeadLetterQueue {
  constructor(env, queueName) {
    this.env = env;
    this.queueName = queueName;
    this.deadLetterKey = `${queueName}:deadletter`;
    this.maxAttempts = 3;
  }

  // 发送到死信队列
  async sendToDeadLetter(message, reason, error = null) {
    const deadMessage = {
      originalMessage: message,
      failureReason: reason,
      error: error ? error.message : null,
      timestamp: Date.now(),
      originalAttempts: message.attempts || 0
    };

    await this.env.QUEUE_KV.put(
      `${this.deadLetterKey}:${message.id}`,
      JSON.stringify(deadMessage),
      {
        expirationTtl: 7 * 24 * 60 * 60 // 7天保留
      }
    );

    // 记录死信统计
    await this.recordDeadLetterStats(reason);
  }

  // 处理失败消息
  async handleFailure(message, error) {
    message.attempts = (message.attempts || 0) + 1;

    if (message.attempts >= this.maxAttempts) {
      // 发送到死信队列
      await this.sendToDeadLetter(
        message,
        'max_attempts_exceeded',
        error
      );
      return false;
    }

    // 指数退避重试
    const delayMs = Math.min(1000 * Math.pow(2, message.attempts), 30000);
    await this.requeueWithDelay(message, delayMs);

    return true;
  }

  // 延迟重新入队
  async requeueWithDelay(message, delayMs) {
    const retryMessage = {
      ...message,
      retryAt: Date.now() + delayMs
    };

    await this.env.QUEUE_KV.put(
      `${this.queueName}:retry:${message.id}`,
      JSON.stringify(retryMessage),
      {
        expirationTtl: Math.ceil(delayMs / 1000) + 60
      }
    );
  }

  // 检查重试消息
  async checkRetryMessages() {
    const now = Date.now();
    const retryMessages = await this.env.QUEUE_KV.list({
      prefix: `${this.queueName}:retry:`
    });

    const readyMessages = [];

    for (const key of retryMessages.keys) {
      const messageValue = await this.env.QUEUE_KV.get(key.name);
      if (!messageValue) continue;

      try {
        const message = JSON.parse(messageValue);

        if (message.retryAt <= now) {
          readyMessages.push({
            key: key.name,
            message: message
          });
        }
      } catch (error) {
        console.error('Failed to parse retry message:', error);
      }
    }

    // 重新入队
    for (const { key, message } of readyMessages) {
      await this.requeue(message);
      await this.env.QUEUE_KV.delete(key);
    }

    return readyMessages.length;
  }

  // 重新入队
  async requeue(message) {
    const queueKey = `${this.queueName}:ready`;
    const messageKey = `${this.queueName}:msg:${message.id}`;

    // 重置消息状态
    message.status = 'pending';
    delete message.retryAt;

    await this.env.QUEUE_KV.put(
      messageKey,
      JSON.stringify(message),
      {
        expirationTtl: 24 * 60 * 60
      }
    );

    await this.env.QUEUE_KV.put(queueKey, message.id);
  }

  // 获取死信队列统计
  async getDeadLetterStats() {
    const statsKey = `${this.deadLetterKey}:stats`;
    return await this.env.QUEUE_KV.get(statsKey);
  }

  // 记录死信统计
  async recordDeadLetterStats(reason) {
    const statsKey = `${this.deadLetterKey}:stats`;
    const stats = await this.getDeadLetterStats();

    let parsedStats = { total: 0, reasons: {} };

    if (stats) {
      try {
        parsedStats = JSON.parse(stats);
      } catch (error) {
        console.error('Failed to parse stats:', error);
      }
    }

    parsedStats.total++;
    parsedStats.reasons[reason] = (parsedStats.reasons[reason] || 0) + 1;

    await this.env.QUEUE_KV.put(
      statsKey,
      JSON.stringify(parsedStats),
      {
        expirationTtl: 30 * 24 * 60 * 60 // 30天保留
      }
    );
  }

  // 处理死信消息
  async processDeadLetter(messageId) {
    const deadMessageKey = `${this.deadLetterKey}:${messageId}`;
    const deadMessageValue = await this.env.QUEUE_KV.get(deadMessageKey);

    if (!deadMessageValue) return false;

    try {
      const deadMessage = JSON.parse(deadMessageValue);
      const originalMessage = deadMessage.originalMessage;

      // 重置尝试次数
      originalMessage.attempts = 0;
      delete originalMessage.status;
      delete originalMessage.processStarted;

      // 重新入队
      await this.requeue(originalMessage);

      // 从死信队列删除
      await this.env.QUEUE_KV.delete(deadMessageKey);

      return true;
    } catch (error) {
      console.error('Failed to process dead letter:', error);
      return false;
    }
  }
}
```

### 1. 队列统计
```javascript
// 队列监控和统计
export class QueueMonitor {
  constructor(env, queueName) {
    this.env = env;
    this.queueName = queueName;
    this.statsKey = `${queueName}:stats`;
  }

  // 更新队列统计
  async updateStats(type, count = 1) {
    const stats = await this.getStats();

    stats[type] = (stats[type] || 0) + count;
    stats.lastUpdated = Date.now();

    await this.env.QUEUE_KV.put(
      this.statsKey,
      JSON.stringify(stats),
      {
        expirationTtl: 24 * 60 * 60
      }
    );
  }

  // 获取统计信息
  async getStats() {
    const statsValue = await this.env.QUEUE_KV.get(this.statsKey);

    if (!statsValue) {
      return {
        enqueued: 0,
        dequeued: 0,
        processed: 0,
        failed: 0,
        deadLettered: 0,
        created: Date.now(),
        lastUpdated: Date.now()
      };
    }

    try {
      return JSON.parse(statsValue);
    } catch (error) {
      console.error('Failed to parse stats:', error);
      return {};
    }
  }

  // 获取队列大小
  async getQueueSize() {
    const readyKey = `${this.queueName}:ready`;
    const readyMessage = await this.env.QUEUE_KV.get(readyKey);

    return readyMessage ? 1 : 0;
  }

  // 获取延迟消息数量
  async getDelayedCount() {
    const delayedMessages = await this.env.QUEUE_KV.list({
      prefix: `${this.queueName}:delayed:`
    });

    return delayedMessages.keys.length;
  }

  // 获取处理中消息数量
  async getProcessingCount() {
    const processingMessages = await this.env.QUEUE_KV.list({
      prefix: `${this.queueName}:msg:`,
      filter: (key) => key.includes('processing')
    });

    return processingMessages.keys.length;
  }

  // 获取重试消息数量
  async getRetryCount() {
    const retryMessages = await this.env.QUEUE_KV.list({
      prefix: `${this.queueName}:retry:`
    });

    return retryMessages.keys.length;
  }

  // 生成队列报告
  async generateReport() {
    const stats = await this.getStats();
    const queueSize = await this.getQueueSize();
    const delayedCount = await this.getDelayedCount();
    const processingCount = await this.getProcessingCount();
    const retryCount = await this.getRetryCount();

    return {
      queueName: this.queueName,
      stats: stats,
      currentSize: queueSize,
      delayedMessages: delayedCount,
      processingMessages: processingCount,
      retryMessages: retryCount,
      timestamp: Date.now()
    };
  }
}
```

### 2. 性能指标
```javascript
// 队列性能监控
export class QueuePerformanceMonitor {
  constructor(env, queueName) {
    this.env = env;
    this.queueName = queueName;
    this.metricsKey = `${queueName}:metrics`;
  }

  // 记录处理时间
  async recordProcessingTime(durationMs) {
    const metrics = await this.getMetrics();

    metrics.processingTimes.push(durationMs);

    // 只保留最近1000次记录
    if (metrics.processingTimes.length > 1000) {
      metrics.processingTimes = metrics.processingTimes.slice(-1000);
    }

    // 计算统计信息
    const sorted = metrics.processingTimes.sort((a, b) => a - b);
    metrics.avgProcessingTime = this.average(metrics.processingTimes);
    metrics.medianProcessingTime = this.median(sorted);
    metrics.p95ProcessingTime = this.percentile(sorted, 95);
    metrics.p99ProcessingTime = this.percentile(sorted, 99);

    await this.env.QUEUE_KV.put(
      this.metricsKey,
      JSON.stringify(metrics),
      {
        expirationTtl: 7 * 24 * 60 * 60
      }
    );
  }

  // 获取性能指标
  async getMetrics() {
    const metricsValue = await this.env.QUEUE_KV.get(this.metricsKey);

    if (!metricsValue) {
      return {
        processingTimes: [],
        avgProcessingTime: 0,
        medianProcessingTime: 0,
        p95ProcessingTime: 0,
        p99ProcessingTime: 0,
        created: Date.now()
      };
    }

    try {
      return JSON.parse(metricsValue);
    } catch (error) {
      console.error('Failed to parse metrics:', error);
      return {};
    }
  }

  // 计算平均值
  average(numbers) {
    if (numbers.length === 0) return 0;
    return numbers.reduce((sum, num) => sum + num, 0) / numbers.length;
  }

  // 计算中位数
  median(sortedNumbers) {
    if (sortedNumbers.length === 0) return 0;

    const mid = Math.floor(sortedNumbers.length / 2);
    return sortedNumbers.length % 2 === 0
      ? (sortedNumbers[mid - 1] + sortedNumbers[mid]) / 2
      : sortedNumbers[mid];
  }

  // 计算百分位数
  percentile(sortedNumbers, p) {
    if (sortedNumbers.length === 0) return 0;

    const index = Math.ceil((p / 100) * sortedNumbers.length) - 1;
    return sortedNumbers[Math.max(0, index)];
  }
}
```

```javascript
// 邮件发送队列示例
export class EmailQueue {
  constructor(env) {
    this.env = env;
    this.queue = new PriorityQueue(env, 'email');
    this.deadLetter = new DeadLetterQueue(env, 'email');
    this.monitor = new QueueMonitor(env, 'email');
  }

  // 添加邮件到队列
  async sendEmail(to, subject, body, priority = 'medium') {
    const emailData = {
      to: to,
      subject: subject,
      body: body,
      html: body,
      sentAt: null
    };

    const messageId = await this.queue.enqueue(emailData, priority);
    await this.monitor.updateStats('enqueued');

    return messageId;
  }

  // 处理邮件队列
  async processEmailQueue() {
    const message = await this.queue.dequeue();

    if (!message) {
      return { processed: 0, reason: 'no_messages' };
    }

    try {
      const startTime = Date.now();

      // 发送邮件
      await this.sendEmailViaService(message.message.data);

      // 标记完成
      await this.queue.complete(message.messageId);

      const duration = Date.now() - startTime;
      await this.monitor.updateStats('processed');

      return {
        processed: 1,
        messageId: message.messageId,
        duration: duration
      };

    } catch (error) {
      console.error('Email send failed:', error);

      // 处理失败
      const willRetry = await this.deadLetter.handleFailure(
        message.message,
        error
      );

      if (willRetry) {
        await this.monitor.updateStats('retry');
      } else {
        await this.monitor.updateStats('deadLettered');
      }

      await this.monitor.updateStats('failed');

      return {
        processed: 0,
        error: error.message,
        willRetry: willRetry
      };
    }
  }

  // 发送邮件（示例）
  async sendEmailViaService(emailData) {
    // 这里可以集成实际的邮件服务
    // 例如 SendGrid, Mailgun, 或者使用 Cloudflare Email Routing

    console.log('Sending email:', {
      to: emailData.to,
      subject: emailData.subject
    });

    // 模拟发送延迟
    await new Promise(resolve => setTimeout(resolve, 100));

    // 模拟偶尔的失败
    if (Math.random() < 0.1) {
      throw new Error('Email service temporarily unavailable');
    }

    return { success: true, messageId: Date.now() };
  }
}
```

```javascript
// 图像处理队列
export class ImageProcessingQueue {
  constructor(env) {
    this.env = env;
    this.queue = new SimpleKVQueue(env, 'image-processing');

      quality: options.quality || 80,
      format: options.format || 'webp',
      callbackUrl: options.callbackUrl,

    };

  async processImageTask() {

    if (!message) {
      return { processed: 0, reason: 'no_tasks' };

    try {
      const task = message.message;
      const startTime = Date.now();

      // 处理图像
      const result = await this.processImage(task.data);

      // 通知回调
      if (task.data.callbackUrl) {
        await this.notifyCallback(task.data.callbackUrl, result);
      }

      // 标记完成
      await this.queue.complete(message.messageId);

      const duration = Date.now() - startTime;
      await this.monitor.updateStats('processed');

      return {
        processed: 1,
        taskId: message.messageId,
        result: result,
        duration: duration
      };

    } catch (error) {
      console.error('Image processing failed:', error);

      // 延迟重试
      const retryDelay = 5000; // 5秒后重试
      await this.delayedQueue.enqueue(
        message.message,
        retryDelay
      );

      await this.monitor.updateStats('failed');

      return {
        processed: 0,
        error: error.message,
        retryDelay: retryDelay
      };
    }
  }

  // 处理图像（示例）
  async processImage(task) {
    // 这里应该使用实际的图像处理库
    // 例如在 Cloudflare Workers 中使用 Image Resizing API

    console.log('Processing image:', {
      url: task.imageUrl,
      operations: task.operations,
      size: `${task.width}x${task.height}`
    });

    // 模拟处理时间
    const processingTime = 500 + Math.random() * 1500;
    await new Promise(resolve => setTimeout(resolve, processingTime));

    // 模拟处理结果
    return {
      originalUrl: task.imageUrl,
      processedUrl: `${task.imageUrl}?processed=true&w=${task.width}&h=${task.height}`,
      size: {
        width: task.width,
        height: task.height
      },
      format: task.format,
      processedAt: Date.now()
    };
  }

  // 通知回调
  async notifyCallback(callbackUrl, result) {
    try {
      const response = await fetch(callbackUrl, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          success: true,
          result: result
        })
      });

      if (!response.ok) {
        throw new Error(`Callback failed: ${response.status}`);
      }

      return { success: true };
    } catch (error) {
      console.error('Callback notification failed:', error);
      throw error;
    }
  }
}
```

## 最佳实践

### 1. 队列设计原则
- **消息幂等性**: 确保重复处理不会产生副作用
- **处理超时**: 设置合理的消息处理超时时间
- **错误隔离**: 将错误消息与正常消息分离
- **监控告警**: 实施全面的队列监控和告警

### 2. 性能优化
- **批量操作**: 批量出队和处理以提高效率
- **异步处理**: 使用异步处理避免阻塞
- **缓存优化**: 缓存频繁访问的队列信息
- **负载均衡**: 在多个 Worker 间分配处理负载

### 3. 可靠性保证
- **持久化**: 确保消息在处理前持久化
- **重试机制**: 实施智能重试策略
- **死信处理**: 正确处理无法处理的消息
- **状态追踪**: 跟踪消息处理状态

### 4. 扩展性考虑
- **水平扩展**: 支持多个处理实例
