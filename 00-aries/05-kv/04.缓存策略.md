# KV 缓存策略

## 缓存概述

Cloudflare KV 本质上就是一个全球分布式缓存系统。理解并正确实施缓存策略对于构建高性能、低延迟的应用至关重要。

## 缓存层级

### 1. 三层缓存架构
```javascript
// L1: Workers 内存缓存（最快）
const memoryCache = new Map();

// L2: KV 边缘缓存（快）
const kvCache = env.MY_KV;

// L3: 源数据（最慢）
const sourceData = await fetchFromDatabase();

// 缓存读取流程
async function getData(key) {
  // 1. 检查内存缓存
  let data = memoryCache.get(key);
  if (data && Date.now() - data.timestamp < 300000) { // 5分钟
    return { data: data.value, source: 'memory' };
  }

  // 2. 检查 KV 缓存
  data = await kvCache.get(key, 'json');
  if (data) {
    // 回填内存缓存
    memoryCache.set(key, {
      value: data,
      timestamp: Date.now()
    });
    return { data, source: 'kv' };
  }

  // 3. 从源获取
  data = await fetchFromDatabase(key);

  // 存储到所有缓存层
  await kvCache.put(key, JSON.stringify(data), { expirationTtl: 3600 });
  memoryCache.set(key, {
    value: data,
    timestamp: Date.now()
  });

  return { data, source: 'database' };
}
```

### 2. 智能缓存管理器
```javascript
// 智能缓存管理器
export class SmartCacheManager {
  constructor(kv, options = {}) {
    this.kv = kv;
    this.memoryCache = new Map();
    this.options = {
      memoryTtl: options.memoryTtl || 300000,      // 5分钟
      kvTtl: options.kvTtl || 3600,              // 1小时
      maxSize: options.maxSize || 1000,          // 最大缓存项数
      cleanupInterval: options.cleanupInterval || 60000 // 1分钟
    };

    // 定期清理内存缓存
    setInterval(() => this.cleanup(), this.options.cleanupInterval);
  }

  async get(key, fetchFn, options = {}) {
    // 检查内存缓存
    const memoryValue = this.getFromMemory(key);
    if (memoryValue !== null) {
      return { value: memoryValue, source: 'memory' };
    }

    // 检查 KV 缓存
    const kvValue = await this.getFromKV(key);
    if (kvValue !== null) {
      // 回填内存缓存
      this.setToMemory(key, kvValue);
      return { value: kvValue, source: 'kv' };
    }

    // 从源获取
    const freshValue = await fetchFn();

    // 存储到缓存
    await this.set(key, freshValue, options);

    return { value: freshValue, source: 'source' };
  }

  async set(key, value, options = {}) {
    const ttl = options.ttl || this.options.kvTtl;

    // 存储到 KV
    await this.kv.put(key, JSON.stringify({
      value,
      timestamp: Date.now(),
      ttl
    }), {
      expirationTtl: ttl
    });

    // 存储到内存
    this.setToMemory(key, value);
  }

  getFromMemory(key) {
    const cached = this.memoryCache.get(key);
    if (cached && Date.now() - cached.timestamp < this.options.memoryTtl) {
      return cached.value;
    }
    this.memoryCache.delete(key);
    return null;
  }

  async getFromKV(key) {
    const cached = await this.kv.get(key, 'json');
    if (cached && this.isValid(cached)) {
      return cached.value;
    }
    return null;
  }

  setToMemory(key, value) {
    // 检查缓存大小
    if (this.memoryCache.size >= this.options.maxSize) {
      // 删除最旧的项
      const firstKey = this.memoryCache.keys().next().value;
      this.memoryCache.delete(firstKey);
    }

    this.memoryCache.set(key, {
      value,
      timestamp: Date.now()
    });
  }

  isValid(cached) {
    // 检查时间戳和 TTL
    const age = Date.now() - cached.timestamp;
    return age < (cached.ttl * 1000);
  }

  cleanup() {
    const now = Date.now();
    for (const [key, value] of this.memoryCache) {
      if (now - value.timestamp > this.options.memoryTtl) {
        this.memoryCache.delete(key);
      }
    }
  }

  async invalidate(key) {
    this.memoryCache.delete(key);
    await this.kv.delete(key);
  }

  async invalidatePattern(pattern) {
    // 内存缓存
    for (const key of this.memoryCache.keys()) {
      if (key.includes(pattern)) {
        this.memoryCache.delete(key);
      }
    }

    // KV 缓存
    const list = await this.kv.list({ prefix: pattern });
    await Promise.all(
      list.keys.map(key => this.kv.delete(key.name))
    );
  }
}
```

```javascript
// Cache-Aside: 应用管理缓存
class CacheAsideService {
  constructor(kv, database) {
    this.kv = kv;
    this.database = database;
  }

  async getUser(id) {
    // 1. 先查缓存
    const cacheKey = `user:${id}`;

    if (user) {
      console.log('Cache hit for user:', id);

      });
    }

    return user;
  }

  async updateUser(id, userData) {
    // 1. 更新数据库
    const user = await this.database.updateUser(id, userData);

    // 2. 删除或更新缓存
    const cacheKey = `user:${id}`;
    await this.kv.put(cacheKey, JSON.stringify(user), {
      expirationTtl: 1800
    });

    return user;
  }

  async deleteUser(id) {
    // 1. 删除数据库记录
    await this.database.deleteUser(id);

    // 2. 删除缓存

    await this.kv.delete(cacheKey);
```

### 4. Write-Behind 模式
```javascript
// Write-Behind: 异步写入数据库
class WriteBehindCache {
  constructor(kv, database) {
    this.kv = kv;
    this.database = database;
    this.writeQueue = new Map();
  }

  async set(key, value) {
    // 立即写入缓存
    await this.kv.put(key, JSON.stringify(value), {
      expirationTtl: 3600
    });

    // 加入写入队列
    this.writeQueue.set(key, {
      operation: 'set',
      value,
      timestamp: Date.now()
    });

    // 异步刷新队列
    this.scheduleFlush();
  }

  async delete(key) {
    // 立即从缓存删除
    await this.kv.delete(key);

    // 加入写入队列
    this.writeQueue.set(key, {
      operation: 'delete',
      timestamp: Date.now()
    });

    this.scheduleFlush();
  }

  scheduleFlush() {
    if (this.flushTimer) return;

    this.flushTimer = setTimeout(async () => {
      await this.flushQueue();
      this.flushTimer = null;
    }, 1000); // 1秒后批量写入
  }

  async flushQueue() {
    const operations = Array.from(this.writeQueue.entries());
    this.writeQueue.clear();

    // 批量写入数据库
    await Promise.all(
      operations.map(async ([key, operation]) => {
        try {
          if (operation.operation === 'set') {
            await this.database.set(key, operation.value);
          } else {
            await this.database.delete(key);
          }
        } catch (error) {
          console.error('Failed to flush operation:', error);
          // 可以重新加入队列重试
        }
      })
    );
  }
}
```

## 缓存更新策略

### 1. TTL (Time To Live)
```javascript
// 基于 TTL 的缓存
class TTLCache {
  constructor(kv) {
    this.kv = kv;
  }

  // 动态 TTL 策略
  async get(key, fetchFn) {
    const cached = await this.kv.getWithMetadata(key);

    if (cached.value) {
      const age = Date.now() - cached.metadata.creationTime;
      const ttl = cached.metadata.customTtl || 3600;

      // 如果未过期，返回缓存
      if (age < ttl * 1000) {
        return JSON.parse(cached.value);
      }
    }

    // 获取新数据
    const value = await fetchFn();
    const ttl = this.calculateTTL(value, key);

    // 存储时包含 TTL 信息
    await this.kv.put(key, JSON.stringify(value), {
      expirationTtl: ttl,
      metadata: {
        creationTime: Date.now(),
        customTtl: ttl
      }
    });

    return value;
  }

  calculateTTL(value, key) {
    // 根据数据类型和键模式动态计算 TTL
    if (key.startsWith('user:')) {
      return 1800; // 用户数据：30分钟
    }

    if (key.startsWith('config:')) {
      return 3600; // 配置数据：1小时
    }

    if (key.startsWith('cache:temp:')) {
      return 300; // 临时数据：5分钟
    }

    // 根据数据大小
    const size = JSON.stringify(value).length;
    if (size > 10000) {
      return 7200; // 大数据保留更久
    }

    return 3600; // 默认1小时
  }
}
```

### 2. 主动更新
```javascript
// 主动更新缓存
class ProactiveCache {
  constructor(kv) {
    this.kv = kv;
    this.updateSchedule = new Map();
  }

  // 预加载热门数据
  async preloadPopularData(popularKeys) {
    const preloadPromises = popularKeys.map(async key => {
      const value = await this.fetchData(key);
      await this.kv.put(key, JSON.stringify(value), {
        expirationTtl: 3600
      });
    });

    await Promise.all(preloadPromises);
  }

  // 定期更新
  scheduleUpdate(key, interval) {
    // 避免重复调度
    if (this.updateSchedule.has(key)) {
      return;
    }

    this.updateSchedule.set(key, setInterval(async () => {
      try {
        const value = await this.fetchData(key);
        await this.kv.put(key, JSON.stringify(value), {
          expirationTtl: interval
        });
        console.log(`Updated cache for key: ${key}`);
      } catch (error) {
        console.error(`Failed to update cache for key ${key}:`, error);
      }
    }, interval * 1000));
  }

  // 取消更新调度
  cancelUpdate(key) {
    const interval = this.updateSchedule.get(key);
    if (interval) {
      clearInterval(interval);
      this.updateSchedule.delete(key);
    }
  }

  async fetchData(key) {
    // 根据键获取数据
    if (key.startsWith('user:')) {
      const userId = key.split(':')[1];
      return await database.getUser(userId);
    }

    if (key.startsWith('product:')) {
      const productId = key.split(':')[1];
      return await database.getProduct(productId);
    }

    throw new Error(`Unknown key pattern: ${key}`);
  }
}
```

### 3. 版本控制缓存
```javascript
// 基于版本的缓存
export class VersionedCache {
  constructor(kv) {
    this.kv = kv;
    this.localVersions = new Map();
  }

  async get(key, fetchFn) {
    // 获取缓存版本
    const cacheVersion = await this.kv.get(`${key}:version`);
    const localVersion = this.localVersions.get(key);

    // 版本未变化，使用本地缓存
    if (cacheVersion && cacheVersion === localVersion) {
      const value = await this.kv.get(key);
      return value ? JSON.parse(value) : null;
    }

    // 版本变化或首次访问，重新获取
    const value = await fetchFn();
    const newVersion = Date.now().toString();

    // 更新缓存和版本
    await Promise.all([
      this.kv.put(key, JSON.stringify(value)),
      this.kv.put(`${key}:version`, newVersion)
    ]);

    this.localVersions.set(key, newVersion);

    return value;
  }

  async invalidate(key) {
    // 更新版本号使缓存失效
    const newVersion = Date.now().toString();

    await Promise.all([
      this.kv.delete(key),
      this.kv.put(`${key}:version`, newVersion)
    ]);

    this.localVersions.delete(key);
  }

  // 批量失效
  async invalidatePattern(pattern) {
    const list = await this.kv.list({ prefix: pattern });
    const versionUpdates = [];

    for (const key of list.keys) {
      if (key.name.includes(':version')) {
        continue;
      }

      const versionKey = `${key.name}:version`;
      const newVersion = Date.now().toString();

      versionUpdates.push(
        this.kv.delete(key.name),
        this.kv.put(versionKey, newVersion)
      );
    }

    await Promise.all(versionUpdates);
  }
}
```

```javascript
// 基于访问模式的智能预热
class IntelligentWarmer {
  constructor(kv) {
    this.kv = kv;
    this.accessPatterns = new Map();
  }

  // 记录访问模式
  async recordAccess(key) {
    const pattern = await this.getAccessPattern(key);
    pattern.count++;
    pattern.lastAccess = Date.now();
    pattern.accessHistory.push(Date.now());

    // 保留最近 100 次访问
    if (pattern.accessHistory.length > 100) {
      pattern.accessHistory.shift();
    }

    await this.saveAccessPattern(key, pattern);
  }

  // 预测下一个可能访问的键
  async predictNextAccess(currentKey) {
    const pattern = await this.getAccessPattern(currentKey);

    // 基于访问历史预测
    const predictions = [];

    // 如果某个键经常在这个键之后被访问
    for (const [key, data] of this.accessPatterns) {
      if (key !== currentKey && data.accessHistory.some(time =>
        Math.abs(time - pattern.lastAccess) < 5000 // 5秒内
      )) {
        predictions.push({
          key,
          score: data.count / pattern.count
        });
      }
    }

    // 返回预测得分最高的键
    return predictions
      .sort((a, b) => b.score - a.score)
      .slice(0, 5)
      .map(p => p.key);
  }

  // 预热预测的键
  async warmupPredictedKeys(currentKey) {
    const predictions = await this.predictNextAccess(currentKey);

    for (const key of predictions) {
      const cached = await this.kv.get(key);
      if (!cached) {
        try {
          const value = await fetchFromDatabase(key);
          await this.kv.put(key, JSON.stringify(value), {
            expirationTtl: 3600
          });
          console.log(`Predictive warmup: ${key}`);
        } catch (error) {
          console.error(`Failed to warmup ${key}:`, error);
        }
      }
    }
  }
}
```

## 缓存失效策略

### 1. 标签失效
```javascript
// 基于标签的缓存失效
export class TaggedCache {
  constructor(kv) {
    this.kv = kv;
  }

  async put(key, value, tags = []) {
    // 存储数据
    await this.kv.put(key, JSON.stringify(value), {
      expirationTtl: 3600
    });

    // 存储标签映射
    for (const tag of tags) {
      const tagKey = `tag:${tag}`;
      const taggedKeys = await this.getOrSet(tagKey, () => new Set());
      taggedKeys.add(key);

      await this.kv.put(tagKey, JSON.stringify([...taggedKeys]));
    }

    // 存储键的标签（用于查询）
    await this.kv.put(`tags:${key}`, JSON.stringify(tags));
  }

  async invalidateByTag(tag) {
    const tagKey = `tag:${tag}`;
    const taggedKeys = await this.kv.get(tagKey, 'json');

    if (taggedKeys) {
      // 删除所有标记的键
      const deletes = taggedKeys.map(key => this.kv.delete(key));
      await Promise.all(deletes);

      // 删除标签映射
      await this.kv.delete(tagKey);

      console.log(`Invalidated ${taggedKeys.length} keys with tag: ${tag}`);
    }
  }

  async invalidateKey(key) {
    // 获取键的标签
    const tags = await this.kv.get(`tags:${key}`, 'json') || [];

    // 从所有标签中移除这个键
    for (const tag of tags) {
      const tagKey = `tag:${tag}`;
      const taggedKeys = await this.kv.get(tagKey, 'json') || [];
      const filtered = taggedKeys.filter(k => k !== key);

      if (filtered.length > 0) {
        await this.kv.put(tagKey, JSON.stringify(filtered));
      } else {
        await this.kv.delete(tagKey);
      }
    }

    // 删除键和其标签记录
    await Promise.all([
      this.kv.delete(key),
      this.kv.delete(`tags:${key}`)
    ]);
  }

  async getOrSet(key, fn) {
    let value = await this.kv.get(key, 'json');
    if (!value) {
      value = await fn();
      await this.kv.put(key, JSON.stringify(value));
    }
    return value;
  }
}

// 使用示例
const cache = new TaggedCache(env.MY_KV);

// 存储数据并添加标签

// 失效所有用户相关的缓存
```

### 2. 依赖失效
```javascript
// 基于依赖关系的缓存失效
class DependentCache {
  constructor(kv) {
    this.kv = kv;
  }

  async put(key, value, dependencies = []) {
    // 存储数据
    await this.kv.put(key, JSON.stringify(value), {
      expirationTtl: 3600
    });

    // 记录依赖关系
    for (const dep of dependencies) {
      const depKey = `deps:${dep}`;
      const dependentKeys = await this.getOrSet(depKey, () => new Set());
      dependentKeys.add(key);

      await this.kv.put(depKey, JSON.stringify([...dependentKeys]));
    }

    // 存储键的依赖（用于查询）
    await this.kv.put(`deps:${key}`, JSON.stringify(dependencies));
  }

  async invalidateDependency(dependency) {
    const depKey = `deps:${dependency}`;
    const dependentKeys = await this.kv.get(depKey, 'json');

    if (dependentKeys) {
      // 级联失效
      const invalidateQueue = [...dependentKeys];
      const invalidated = new Set();

      while (invalidateQueue.length > 0) {
        const currentKey = invalidateQueue.shift();

        if (invalidated.has(currentKey)) {
          continue;
        }

        // 删除当前键
        await this.kv.delete(currentKey);
        invalidated.add(currentKey);

        // 获取当前键的依赖者
        const currentDeps = await this.kv.get(`deps:${currentKey}`, 'json') || [];

        // 将依赖者加入队列
        for (const dep of currentDeps) {
          if (!invalidated.has(dep)) {
            invalidateQueue.push(dep);
          }
        }
      }

      console.log(`Invalidated ${invalidated.size} keys due to dependency: ${dependency}`);
    }

    // 删除依赖关系
    await this.kv.delete(depKey);
  }

  async getOrSet(key, fn) {
    let value = await this.kv.get(key, 'json');
    if (!value) {
      value = await fn();
      await this.kv.put(key, JSON.stringify(value));
    }
    return value;
  }
}

// 使用示例
const cache = new DependentCache(env.MY_KV);

// 存储用户列表缓存，依赖于用户表更新
await cache.put('users:list', users, ['table:users']);

// 当用户表更新时，自动失效相关缓存
await cache.invalidateDependency('table:users');
```

### 1. 批量缓存操作
```javascript
// 批量缓存优化器
export class BatchCacheOptimizer {
  constructor(kv) {
    this.kv = kv;
    this.readQueue = [];
    this.writeQueue = [];
    this.processing = false;
  }

  // 批量读取
  async batchGet(keys) {
    // 将键加入队列
    this.readQueue.push(...keys);

    if (!this.processing) {
      return await this.processBatchRead();
    }
  }

  async processBatchRead() {
    this.processing = true;

    try {
      // 去重
      const uniqueKeys = [...new Set(this.readQueue)];
      this.readQueue = [];

      // 批量获取
      const results = await this.kv.get(uniqueKeys);

      // 处理结果
      const processed = {};
      uniqueKeys.forEach(key => {
        processed[key] = results[key] || null;
      });

      return processed;
    } finally {
      this.processing = false;
    }
  }

  // 批量写入
  async batchPut(items) {
    // 将项目加入队列
    this.writeQueue.push(...items);

    if (!this.processing) {
      return await this.processBatchWrite();
    }
  }

  async processBatchWrite() {
    this.processing = true;

    try {
      const batch = this.writeQueue.splice(0);

      // 并行写入
      const writes = batch.map(item =>
        this.kv.put(item.key, item.value, item.options)
      );

      await Promise.all(writes);

      return { success: true, count: batch.length };
    } finally {
      this.processing = false;
    }
  }

  // 强制处理队列
  async flush() {
    const readPromise = this.readQueue.length > 0
      ? this.processBatchRead()
      : Promise.resolve({});

    const writePromise = this.writeQueue.length > 0
      ? this.processBatchWrite()
      : Promise.resolve({});

    return Promise.all([readPromise, writePromise]);
  }
}
```

```javascript
// 压缩缓存以节省空间
export class CompressedCache {
  constructor(kv, compressionThreshold = 1024) {
    this.kv = kv;
    this.threshold = compressionThreshold;
  }

  async put(key, value, options = {}) {
    const serialized = JSON.stringify(value);

    let dataToStore;
    let metadata = options.metadata || {};

    // 检查是否需要压缩
    if (serialized.length > this.threshold) {
      dataToStore = await this.compress(serialized);
      metadata.compressed = true;
      metadata.originalSize = serialized.length;
      metadata.compressedSize = dataToStore.length;
    } else {
      dataToStore = serialized;
      metadata.compressed = false;
    }

    await this.kv.put(key, dataToStore, {
      ...options,
      metadata
    });
  }

  async get(key) {
    const { value, metadata } = await this.kv.getWithMetadata(key);

    if (!value) {
      return null;
    }

    // 检查是否需要解压
    if (metadata?.compressed) {
      return JSON.parse(await this.decompress(value));
    }

    return JSON.parse(value);
  }

  async compress(data) {
    // 实现压缩算法
    // 这里是示例，实际可以使用 gzip, brotli 等
    const compressed = Buffer.from(data).toString('base64');
    return compressed;
  }

  async decompress(data) {
    // 实现解压缩
    return Buffer.from(data, 'base64').toString();
  }

  // 压缩统计
  async getCompressionStats() {
    const list = await this.kv.list({ limit: 1000 });

    let totalOriginalSize = 0;
    let totalCompressedSize = 0;
    let compressedCount = 0;

    for (const key of list.keys) {
      const { metadata } = await this.kv.getWithMetadata(key.name);

      if (metadata?.compressed) {
        totalOriginalSize += metadata.originalSize || 0;
        totalCompressedSize += metadata.compressedSize || 0;
        compressedCount++;
      }
    }

    const compressionRatio = totalCompressedSize > 0
      ? (totalOriginalSize / totalCompressedSize).toFixed(2)
      : 0;

    return {
      compressedItems: compressedCount,
      totalItems: list.keys.length,
      compressionRatio,
      spaceSaved: totalOriginalSize - totalCompressedSize
    };
  }
}
```

```javascript
// 缓存监控系统
export class CacheMonitor {
  constructor(kv) {
    this.kv = kv;
  }

  async recordCacheHit(key, hit) {
    const today = new Date().toISOString().split('T')[0];
    const statsKey = `cache-stats:${today}`;

    const stats = await this.kv.get(statsKey, 'json') || {
      hits: 0,
      misses: 0,
      total: 0
    };

    if (hit) {
      stats.hits++;
    } else {
      stats.misses++;
    }
    stats.total++;

    await this.kv.put(statsKey, JSON.stringify(stats), {
      expirationTtl: 7 * 24 * 3600 // 7天
    });
  }

  async getCacheHitRate(days = 7) {
    const end = new Date();
    const start = new Date(end);
    start.setDate(start.getDate() - days);

    let totalHits = 0;
    let totalMisses = 0;

    for (let i = 0; i < days; i++) {
      const date = new Date(start);
      date.setDate(date.getDate() + i);
      const dateStr = date.toISOString().split('T')[0];

      const stats = await this.kv.get(`cache-stats:${dateStr}`, 'json');
      if (stats) {
        totalHits += stats.hits;
        totalMisses += stats.misses;
      }
    }

    const hitRate = totalHits + totalMisses > 0
      ? (totalHits / (totalHits + totalMisses) * 100).toFixed(2)
      : 0;

    return {
      hitRate: parseFloat(hitRate),
      hits: totalHits,
      misses: totalMisses,
      period: `${days} days`
    };
  }

  async analyzeCachePatterns() {
    const list = await this.kv.list({
      prefix: 'cache-stats:',
      limit: 30
    });

    const patterns = [];

    for (const key of list.keys) {
      const date = key.name.split(':')[1];
      const stats = await this.kv.get(key.name, 'json');

      if (stats) {
        patterns.push({
          date,
          hitRate: (stats.hits / stats.total * 100).toFixed(2),
          total: stats.total
        });
      }
    }

    return patterns.sort((a, b) => new Date(b.date) - new Date(a.date));
  }
}
```
