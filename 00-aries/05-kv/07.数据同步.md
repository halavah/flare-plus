# KV 数据同步

## 数据同步概述

在分布式系统中，数据同步是确保数据一致性的关键挑战。KV 的全球分布特性使得数据同步更加重要，需要考虑网络延迟、故障转移、数据一致性等问题。

## 同步策略

### 1. 写入同步
```javascript
// 同步写入管理器
export class SyncManager {
  constructor(kv, options = {}) {
    this.kv = kv;
    this.options = {
      retryAttempts: options.retryAttempts || 3,
      retryDelay: options.retryDelay || 1000,
      consistencyLevel: options.consistencyLevel || 'eventual', // 'strong' or 'eventual'
    };
    this.pendingWrites = new Map();
    this.conflictResolver = options.conflictResolver || this.defaultConflictResolver;
  }

  // 同步写入
  async put(key, value, options = {}) {
    const writeId = crypto.randomUUID();
    const write = {
      key,
      value,
      options,
      writeId,
      timestamp: Date.now(),
      version: options.version || 0
    };

    // 检查是否需要冲突解决
    if (this.options.consistencyLevel === 'strong') {
      return await this.syncPutWithConflictResolution(write);
    } else {
      return await this.syncPutEventual(write);
    }
  }

  // 强一致性写入（带冲突解决）
  async syncPutWithConflictResolution(write) {
    let attempts = 0;

    while (attempts < this.options.retryAttempts) {
      try {
        // 获取当前值
        const { value: currentValue, metadata } = await this.kv.getWithMetadata(write.key);

        // 检查版本冲突
        if (currentValue && metadata) {
          const currentVersion = metadata.version || 0;
          if (write.version <= currentVersion) {
            throw new Error(`Version conflict: expected > ${currentVersion}, got ${write.version}`);
          }

          // 解决冲突
          const resolved = await this.conflictResolver(
            write.key,
            currentValue,
            write.value,
            metadata
          );

          if (resolved.resolved) {
            write.value = resolved.value;
            write.version = resolved.version;
          }
        }

        // 执行写入
        await this.kv.put(write.key, write.value, {
          ...write.options,
          metadata: {
            ...write.options?.metadata,
            version: write.version,
            writeId,
            timestamp: write.timestamp
          }
        });

        return { success: true, version: write.version };

      } catch (error) {
        attempts++;
        if (attempts >= this.options.retryAttempts) {
          throw error;
        }

        // 指数退避
        await new Promise(resolve => setTimeout(resolve, this.options.retryDelay * Math.pow(2, attempts)));
      }
    }
  }

  // 最终一致性写入
  async syncPutEventual(write) {
    try {
      // 直接写入，快速响应
      await this.kv.put(write.key, write.value, {
        ...write.options,
        metadata: {
          ...write.options?.metadata,
          writeId,
          timestamp: write.timestamp,
          consistency: 'eventual'
        }
      });

      // 记录待同步项
      this.pendingWrites.set(write.key, write);

      // 异步同步
      setTimeout(() => this.syncToReplicas(write), 0);

      return { success: true };
    } catch (error) {
      // 加入重试队列
      this.queueRetry(write);
      throw error;
    }
  }

  // 同步到副本
  async syncToReplicas(write) {
    // 这里可以集成其他存储系统
    // 例如：R2、D1、外部数据库等
    const replicas = [
      { name: 'r2', sync: this.syncToR2 },
      { name: 'd1', sync: this.syncToD1 },
      { name: 'backup', sync: this.syncToBackup }
    ];

    for (const replica of replicas) {
      try {
        await replica.sync(this.kv, write);
      } catch (error) {
        console.error(`Failed to sync ${replica.name}:`, error);
        // 加入重试队列
        this.queueRetryForReplica(write, replica.name);
      }
    }

    // 清理待同步项
    this.pendingWrites.delete(write.key);
  }

  // 重试队列管理
  async queueRetry(write) {
    const retryKey = `retry:${write.writeId}`;
    const retry = {
      write,
      attempts: 0,
      nextAttempt: Date.now() + this.options.retryDelay
    };

    await this.kv.put(retryKey, JSON.stringify(retry), {
      expirationTtl: 3600 // 1小时
    });
  }

  async queueRetryForReplica(write, replicaName) {
    const retryKey = `retry:${replicaName}:${write.writeId}`;
    const retry = {
      write,
      replicaName,
      attempts: 0,
      nextAttempt: Date.now() + this.options.retryDelay
    };

    await this.kv.put(retryKey, JSON.stringify(retry), {
      expirationTtl: 3600
    });
  }

  // 处理重试队列
  async processRetries() {
    const list = await this.kv.list({
      prefix: 'retry:',
      limit: 100
    });

    for (const key of list.keys) {
      try {
        const retry = await this.kv.get(key.name, 'json');
        if (!retry) continue;

        if (Date.now() >= retry.nextAttempt) {
          await this.processRetry(retry);
          await this.kv.delete(key.name);
        }
      } catch (error) {
        console.error(`Error processing retry ${key.name}:`, error);
      }
    }
  }

  async processRetry(retry) {
    retry.attempts++;

    if (retry.attempts >= this.options.retryAttempts) {
      console.error(`Retry failed after ${retry.attempts} attempts`);
      return;
    }

    try {
      if (retry.replicaName) {
        // 重试特定副本
        await this.retryReplicaSync(retry);
      } else {
        // 重试主写入
        await this.put(retry.write.key, retry.write.value, retry.write.options);
      }
    } catch (error) {
      // 更新下次重试时间
      retry.nextAttempt = Date.now() + this.options.retryDelay * Math.pow(2, retry.attempts);

      const retryKey = `retry:${retry.writeId}`;
      await this.kv.put(retryKey, JSON.stringify(retry), {
        expirationTtl: 3600
      });
    }
  }

  // 默认冲突解决器
  defaultConflictResolver(key, current, incoming, metadata) {
    // 简单策略：使用更新的值
    return {
      resolved: true,
      value: incoming,
      version: (metadata.version || 0) + 1
    };
  }
}
```

### 2. 冲突解决策略
```javascript
// 冲突解决器工厂
export class ConflictResolverFactory {
  static createLastWriteWins() {
    return (key, current, incoming, metadata) => ({
      resolved: true,
      value: incoming,
      version: (metadata.version || 0) + 1,
      reason: 'last-write-wins'
    });
  }

  static createFirstWriteWins() {
    return (key, current, incoming, metadata) => ({
      resolved: true,
      value: current,
      version: metadata.version || 0,
      reason: 'first-write-wins'
    });
  }

  static createMergeResolver(mergeFn) {
    return async (key, current, incoming, metadata) => {
      try {
        const merged = await mergeFn(current, incoming);
        return {
          resolved: true,
          value: merged,
          version: (metadata.version || 0) + 1,
          reason: 'merged'
        };
      } catch (error) {
        return {
          resolved: false,
          error: error.message
        };
      }
    };
  }

  static createTimestampResolver() {
    return (key, current, incoming, metadata) => {
      const currentTimestamp = metadata.timestamp || 0;
      const incomingTimestamp = incoming.timestamp || 0;

      if (incomingTimestamp > currentTimestamp) {
        return {
          resolved: true,
          value: incoming,
          version: (metadata.version || 0) + 1,
          reason: 'newer-timestamp'
        };
      } else {
        return {
          resolved: true,
          value: current,
          version: metadata.version || 0,
          reason: 'older-timestamp'
        };
      }
    };
  }

  static createCustomResolver(strategy) {
    return async (key, current, incoming, metadata) => {
      return await strategy.resolve(key, current, incoming, metadata);
    };
  }
}

// 自定义策略示例
class CustomMergeStrategy {
  constructor(options = {}) {
    this.options = {
      arrayStrategy: options.arrayStrategy || 'replace',
      objectStrategy: options.objectStrategy || 'merge',
      conflictFields: options.conflictFields || []
    };
  }

  async resolve(key, current, incoming, metadata) {
    try {
      const currentParsed = typeof current === 'string' ? JSON.parse(current) : current;
      const incomingParsed = typeof incoming === 'string' ? JSON.parse(incoming) : incoming;

      const merged = this.mergeData(currentParsed, incomingParsed);

      return {
        resolved: true,
        value: merged,
        version: (metadata.version || 0) + 1
      };
    } catch (error) {
      return {
        resolved: false,
        error: error.message
      };
    }
  }

  mergeData(current, incoming) {
    // 处理对象
    if (typeof current === 'object' && typeof incoming === 'object') {
      const merged = { ...current };

      for (const [key, value] of Object.entries(incoming)) {
        if (this.options.conflictFields.includes(key)) {
          // 冲突字段使用传入值
          merged[key] = value;
        } else if (Array.isArray(merged[key]) && Array.isArray(value)) {
          // 数组处理
          if (this.options.arrayStrategy === 'merge') {
            merged[key] = [...new Set([...merged[key], ...value])];
          } else {
            merged[key] = value;
          }
        } else if (typeof merged[key] === 'object' && typeof value === 'object') {
          // 递归合并对象
          merged[key] = this.mergeData(merged[key], value);
        } else {
          // 其他类型直接覆盖
          merged[key] = value;
        }
      }

      return merged;
    }

    // 处理数组
    if (Array.isArray(current) && Array.isArray(incoming)) {
      if (this.options.arrayStrategy === 'merge') {
        return [...new Set([...current, ...incoming])];
      } else {
        return incoming;
      }
    }

    // 其他类型直接返回
    return incoming;
  }
}
```

### 3. 双向同步
```javascript
// 双向同步管理器
export class BidirectionalSyncManager {
  constructor(kvA, kvB, options = {}) {
    this.kvA = kvA;
    this.kvB = kvB;
    this.options = {
      syncInterval: options.syncInterval || 30000, // 30秒
      conflictResolver: options.conflictResolver || ConflictResolverFactory.createLastWriteWins(),
      priority: options.priority || 'A' // 'A' or 'B'
    };
    this.lastSync = new Map();
    this.syncQueues = {
      AtoB: [],
      BtoA: []
    };
  }

  // 启动双向同步
  startSync() {
    // 定期同步
    setInterval(() => {
      this.performSync('AtoB');
      this.performSync('BtoA');
    }, this.options.syncInterval);

    // 处理同步队列
    setInterval(() => {
      this.processSyncQueue('AtoB');
      this.processSyncQueue('BtoA');
    }, 5000);
  }

  // 从 A 写入到 B（同步）
  async writeToA(key, value, options = {}) {
    // 先写入优先系统
    const primaryKV = this.options.priority === 'A' ? this.kvA : this.kvB;
    const secondaryKV = this.options.priority === 'A' ? this.kvB : this.kvA;

    // 写入优先系统
    await primaryKV.put(key, value, {
      ...options,
      metadata: {
        ...options?.metadata,
        syncedFrom: this.options.priority,
        syncedAt: Date.now()
      }
    });

    // 加入同步队列
    this.syncQueues[this.options.priority === 'A' ? 'AtoB' : 'BtoA'].push({
      key,
      value,
      options,
      timestamp: Date.now()
    });

    return { success: true, primary: this.options.priority };
  }

  // 从优先系统读取
  async readFromPrimary(key) {
    const primaryKV = this.options.priority === 'A' ? this.kvA : this.kvB;

    try {
      const { value, metadata } = await primaryKV.getWithMetadata(key);
      return { value, metadata, source: this.options.priority };
    } catch (error) {
      // 尝试从备选系统读取
      const secondaryKV = this.options.priority === 'A' ? this.kvB : this.kvA;
      try {
        const { value, metadata } = await secondaryKV.getWithMetadata(key);
        return { value, metadata, source: this.options.priority === 'A' ? 'B' : 'A' };
      } catch (secondaryError) {
        throw new Error(`Both systems unavailable: ${error.message}`);
      }
    }
  }

  // 执行同步
  async performSync(direction) {
    const fromKV = direction === 'AtoB' ? this.kvA : this.kvB;
    const toKV = direction === 'AtoB' ? this.kvB : this.kvA;

    try {
      // 获取需要同步的键
      const keysToSync = await this.getKeysToSync(direction);

      for (const key of keysToSync) {
        await this.syncKey(direction, key, fromKV, toKV);
      }
    } catch (error) {
      console.error(`Sync error in direction ${direction}:`, error);
    }
  }

  async getKeysToSync(direction) {
    // 获取最近修改的键
    const lastSyncKey = `lastSync:${direction}`;
    const lastSync = await this.kvA.get(lastSyncKey) || 0;

    const kv = direction === 'AtoB' ? this.kvA : this.kvB;
    const list = await kv.list({
      limit: 1000
    });

    const keys = [];
    for (const key of list.keys) {
      const { metadata } = await kv.getWithMetadata(key.name);
      if (metadata && metadata.timestamp > lastSync) {
        keys.push(key.name);
      }
    }

    return keys;
  }

  async syncKey(direction, key, fromKV, toKV) {
    try {
      const { value, metadata: fromMetadata } = await fromKV.getWithMetadata(key);

      if (!value) {
        // 键不存在，可能已被删除
        await toKV.delete(key);
        return;
      }

      // 检查目标键是否需要更新
      const { metadata: toMetadata } = await toKV.getWithMetadata(key);
      const needsUpdate = !toMetadata ||
        fromMetadata.timestamp > (toMetadata.syncedAt || 0) ||
        fromMetadata.version > (toMetadata.version || 0);

      if (needsUpdate) {
        // 检查冲突
        if (toMetadata && toMetadata.timestamp) {
          const conflictResult = await this.resolveConflict(
            key,
            toMetadata,
            fromMetadata,
            value
          );

          if (conflictResult.resolved) {
            await toKV.put(key, conflictResult.value, {
              metadata: {
                ...conflictResult.metadata,
                syncedFrom: direction.split('to')[0],
                syncedAt: Date.now()
              }
            });
          }
        } else {
          // 无冲突，直接同步
          await toKV.put(key, value, {
            metadata: {
              ...fromMetadata,
              syncedFrom: direction.split('to')[0],
              syncedAt: Date.now()
            }
          });
        }
      }
    } catch (error) {
      console.error(`Failed to sync key ${key}:`, error);
    }
  }

  async resolveConflict(key, toMetadata, fromMetadata, incomingValue) {
    // 使用配置的冲突解决器
    const currentValue = await this.kvB.get(key);
    return await this.options.conflictResolver(key, currentValue, incomingValue, toMetadata);
  }

  // 处理同步队列
  async processSyncQueue(direction) {
    const queue = this.syncQueues[direction];
    if (queue.length === 0) return;

    // 批量处理
    const batch = queue.splice(0, 10); // 处理10个项目

    const toKV = direction === 'AtoB' ? this.kvB : this.kvA;

    for (const item of batch) {
      try {
        await toKV.put(item.key, item.value, {
          ...item.options,
          metadata: {
            ...item.options?.metadata,
            syncedFrom: direction.split('to')[0],
            syncedAt: Date.now(),
            queuedSync: true
          }
        });
      } catch (error) {
        console.error(`Failed to process queued sync for ${item.key}:`, error);
        // 可以选择重新加入队列
      }
    }
  }
}
```

## 实时同步

### 1. 事件驱动同步
```javascript
// 事件驱动的同步系统
export class EventDrivenSync {
  constructor(kv, eventBus, options = {}) {
    this.kv = kv;
    this.eventBus = eventBus;
    this.options = options;
    this.subscribers = new Map();
  }

  // 发布数据变更事件
  async publishChange(type, key, value, metadata = {}) {
    const event = {
      type,
      key,
      value,
      metadata: {
        timestamp: Date.now(),
        source: this.options.source || 'unknown',
        ...metadata
      }
    };

    // 存储事件到 KV（用于重放）
    await this.kv.put(`event:${Date.now()}:${crypto.randomUUID()}`, JSON.stringify(event), {
      eventTtl: 7 * 24 * 3600 // 7天
    });

    // 发布到事件总线
    await this.eventBus.publish('data_change', event);
  }

  // 订阅数据变更
  async onChange(type, handler) {
    const subscription = await this.eventBus.subscribe('data_change', async (event) => {
      if (event.type === type) {
        await handler(event);
      }
    });

    this.subscribers.set(type, subscription);
    return subscription;
  }

  // 处理数据变更事件
  async handleDataChange(event) {
    const { type, key, value, metadata } = event;

    switch (type) {
      case 'create':
        await this.handleCreate(key, value, metadata);
        break;
      case 'update':
        await this.handleUpdate(key, value, metadata);
        break;
      case 'delete':
        await this.handleDelete(key, metadata);
        break;
      case 'merge':
        await this.handleMerge(key, value, metadata);
        break;
    }

    // 触发自定义处理器
    const customHandlers = this.subscribers.get('*');
    if (customHandlers) {
      await customHandlers(event);
    }
  }

  async handleCreate(key, value, metadata) {
    // 创建数据
    await this.kv.put(key, value, {
      metadata: {
        ...metadata,
        operation: 'create',
        createdAt: metadata.timestamp
      }
    });

    console.log(`Created: ${key}`);
  }

  async handleUpdate(key, value, metadata) {
    // 获取现有数据
    const { value: existing, metadata: existingMeta } = await this.kv.getWithMetadata(key);

    if (!existing) {
      // 数据不存在，当作创建处理
      return this.handleCreate(key, value, metadata);
    }

    // 合并更新
    const merged = await this.mergeData(existing, value, metadata);
    await this.kv.put(key, merged.value, {
      metadata: {
        ...merged.metadata,
        operation: 'update',
        updatedAt: metadata.timestamp,
        previousVersion: existingMeta.version || 0
      }
    });

    console.log(`Updated: ${key}`);
  }

  async handleDelete(key, metadata) {
    // 删除数据
    await this.kv.delete(key);

    // 软移到归档（可选）
    if (this.options.archiveOnDelete) {
      await this.archiveData(key, metadata);
    }

    console.log(`Deleted: ${key}`);
  }

  async handleMerge(key, value, metadata) {
    // 获取现有数据
    const { value: existing } = await this.kv.get(key);

    if (!existing) {
      return this.handleCreate(key, value, metadata);
    }

    // 执行合并
    const merged = await this.mergeData(existing, value, metadata);
    await this.kv.put(key, merged.value, {
      metadata: {
        ...merged.metadata,
        operation: 'merge',
        mergedAt: metadata.timestamp
      }
    });
  }

  async mergeData(existing, incoming, metadata) {
    // 实现数据合并逻辑
    const conflictResolver = ConflictResolverFactory.createMergeResolver(
      (current, incoming) => {
        // 自定义合并策略
        if (typeof current === 'object' && typeof incoming === 'object') {
          return {
            ...current,
            ...incoming
          };
        }
        return incoming;
      }
    );

    const resolved = await conflictResolver(key, existing, incoming, metadata);
    return resolved.resolved ? resolved : { value: incoming };
  }

  async archiveData(key, metadata) {
    const archiveKey = `archive:${key}:${Date.now()}`;
    const { value } = await this.kv.get(key);

    if (value) {
      await this.kv.put(archiveKey, JSON.stringify({
        key,
        value,
        originalMetadata: metadata,
        archivedAt: new Date().toISOString()
      }), {
        expirationTtl: 30 * 24 * 3600 // 30天
      });
    }
  }
}
```

### 2. 增量同步
```javascript
// 增量同步器
export class IncrementalSyncer {
  constructor(kv, source, options = {}) {
    this.kv = kv;
    this.source = source;
    this.options = {
      batchSize: options.batchSize || 100,
      syncInterval: options.syncInterval || 30000,
      checkpointInterval: options.checkpointInterval || 300000, // 5分钟
      ...options
    };
    this.checkpoints = new Map();
    this.isRunning = false;
  }

  // 启动增量同步
  async start() {
    if (this.isRunning) {
      throw new Error('Sync is already running');
    }

    this.isRunning = true;
    console.log('Starting incremental sync...');

    while (this.isRunning) {
      try {
        await this.performIncrementalSync();
        await new Promise(resolve => setTimeout(resolve, this.options.syncInterval));
      } catch (error) {
        console.error('Sync error:', error);
        await new Promise(resolve => setTimeout(resolve, 5000)); // 5秒后重试
      }
    }
  }

  // 停止同步
  stop() {
    this.isRunning = false;
    console.log('Stopping incremental sync...');
  }

  // 执行增量同步
  async performIncrementalSync() {
    const checkpoints = await this.getAllCheckpoints();

    for (const [checkpointId, checkpoint] of checkpoints) {
      const changes = await this.source.getChangesSince(
        checkpoint.lastSynced
      );

      if (changes.length > 0) {
        console.log(`Syncing ${changes.length} changes for ${checkpointId}`);
        await this.syncChanges(changes, checkpointId);
        await this.updateCheckpoint(checkpointId, changes);
      }
    }
  }

  // 获取所有检查点
  async getAllCheckpoints() {
    const checkpoints = new Map();

    // 获取检查点配置
    const config = await this.kv.get('sync-checkpoints', 'json');
    if (config) {
      for (const [id, checkpoint] of Object.entries(config)) {
        checkpoints.set(id, checkpoint);
      }
    }

    return checkpoints;
  }

  // 同步变更
  async syncChanges(changes, checkpointId) {
    // 批量处理变更
    for (let i = 0; i < changes.length; i += this.options.batchSize) {
      const batch = changes.slice(i, i + this.options.batchSize);

      await this.processBatch(batch);
    }
  }

  // 处理变更批次
  async processBatch(changes) {
    const writes = [];

    for (const change of changes) {
      switch (change.type) {
        case 'create':
        case 'update':
          writes.push(
            this.kv.put(change.key, change.value, {
              metadata: {
                ...change.metadata,
                syncedAt: Date.now(),
                changeId: change.id
              }
            })
          );
          break;
        case 'delete':
          writes.push(this.kv.delete(change.key));
          break;
      }
    }

    // 执行批量写入
    await Promise.all(writes);
  }

  // 更新检查点
  async updateCheckpoint(checkpointId, changes) {
    const checkpoint = await this.getCheckpoint(checkpointId);

    if (checkpoint) {
      // 使用最新变更的时间戳
      const lastChange = changes.reduce((latest, change) => {
        return change.timestamp > latest ? change : latest;
      }, { timestamp: 0 });

      checkpoint.lastSynced = lastChange.timestamp;
      checkpoint.lastSyncId = lastChange.id;
      checkpoint.syncCount = (checkpoint.syncCount || 0) + changes.length;
    }

    await this.saveCheckpoint(checkpointId, checkpoint);
  }

  // 获取检查点
  async getCheckpoint(checkpointId) {
    const config = await this.kv.get('sync-checkpoints', 'json') || {};
    return config[checkpointId];
  }

  // 保存检查点
  async saveCheckpoint(checkpointId, checkpoint) {
    const config = await this.kv.get('sync-checkpoints', 'json') || {};
    config[checkpointId] = checkpoint;
    await this.kv.put('sync-checkpoints', JSON.stringify(config));
  }

  // 添加检查点
  async addCheckpoint(id, options = {}) {
    const checkpoint = {
      id,
      lastSynced: options.lastSynced || 0,
      lastSyncId: options.lastSyncId || '',
      syncCount: 0,
      createdAt: Date.now(),
      ...options
    };

    await this.saveCheckpoint(id, checkpoint);
  }
}
```

```javascript
export class DataValidator {
  constructor(kv, options = {}) {
    this.kv = kv;
    this.options = {
      enableChecksum: options.enableChecksum !== false,
      enableEncryption: options.enableEncryption || false,
      encryptionKey: options.encryptionKey
    };
  }

  // 存储经过校验的数据
  async putWithValidation(key, value, options = {}) {
    let valueToStore = value;
    let metadata = {};

    // 计算校验和
    if (this.options.enableChecksum) {
      const checksum = this.calculateChecksum(value);
      metadata.checksum = checksum;
    }

    // 加密数据
    if (this.options.enableEncryption) {
      const encrypted = await this.encrypt(value, this.options.encryptionKey);
      valueToStore = encrypted;
      metadata.encrypted = true;
    }

    // 添加元数据
    metadata = {
      ...metadata,
      ...options?.metadata,
      storedAt: Date.now(),
      size: this.getSize(value)
    };

    await this.kv.put(key, valueToStore, {
      ...options,
      metadata
    });

    return { success: true, metadata };
  }

  // 读取并验证数据
  async getWithValidation(key) {
    const { value, metadata } = await this.kv.getWithMetadata(key);

    if (!value || !metadata) {
      return null;
    }

    // 检查加密
    if (metadata.encrypted) {
      try {
        const decrypted = await this.decrypt(value, this.options.encryptionKey);
        value = decrypted;
      } catch (error) {
        console.error('Decryption failed:', error);
        return null;
      }
    }

    // 验证校验和
    if (this.options.enableChecksum && metadata.checksum) {
      const checksum = this.calculateChecksum(value);
      if (checksum !== metadata.checksum) {
        console.error('Checksum validation failed');
        return null;
      }
    }

    return { value, metadata };
  }

        results[key] = {
          valid: result !== null,
          error: null
        };
      } catch (error) {
        results[key] = {
          valid: false,
          error: error.message
        };
      }
    }

    return results;
  }

  async repairData(key) {
    const { value, metadata } = await this.kv.getWithMetadata(key);

    if (!value) {
      return { repaired: false, reason: 'Data not found' };
    }

    try {
      // 尝试解密
      let repairedValue = value;
      if (metadata.encrypted) {
        repairedValue = await this.decrypt(value, this.options.encryptionKey);
      }

      // 重新计算校验和
      if (this.options.enableChecksum) {
        metadata.checksum = this.calculateChecksum(repairedValue);
      }

      // 重新加密
      if (metadata.encrypted) {
        repairedValue = await this.encrypt(repairedValue, this.options.encryptionKey);
      }

      // 保存修复后的数据
      await this.kv.put(key, repairedValue, { metadata });

      return { repaired: true };
    } catch (error) {
```
