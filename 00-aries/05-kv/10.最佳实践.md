# KV 最佳实践

## 概述

Cloudflare KV 是一个全球分布的键值存储系统，了解其特性和限制对于构建高性能、可靠的应用至关重要。本文档总结了使用 KV 的最佳实践，帮助开发者避免常见陷阱并最大化性能。

```javascript
// 键命名约定最佳实践
export class KeyNamingStrategy {
  // 推荐的键结构
  static generateKey(prefix, identifier, version = null) {
    // 格式: {service}:{entity}:{identifier}[:{version}]
    // 例如: user:profile:12345:v2

    const parts = [
      this.sanitize(prefix),
      this.sanitize(identifier)
    ];

    if (version) {
      parts.push(this.sanitize(version));
    }

    return parts.join(':');
  }

  // 用户数据键设计
  static userKeys = {
    profile: (userId) => `user:profile:${userId}`,
    preferences: (userId) => `user:preferences:${userId}`,
    settings: (userId) => `user:settings:${userId}`,
    sessions: (userId, sessionId) => `user:session:${userId}:${sessionId}`,
    cache: (userId, cacheKey) => `user:cache:${userId}:${cacheKey}`
  };

  // 内容数据键设计
  static contentKeys = {
    article: (articleId) => `content:article:${articleId}`,
    metadata: (articleId) => `content:metadata:${articleId}`,
    comments: (articleId) => `content:comments:${articleId}`,
    tags: (articleId) => `content:tags:${articleId}`
  };

  // 临时数据键设计
  static tempKeys = {
    upload: (uploadId) => `temp:upload:${uploadId}`,
    validation: (token) => `temp:validation:${token}`,
    cache: (key) => `temp:cache:${key}`,
    rateLimit: (identifier, window) => `temp:rate:${identifier}:${window}`
  };

  // 清理键名
  static sanitize(key) {
    return key.toString()
      .toLowerCase()
      .replace(/[^a-z0-9:_-]/g, '-')
      .replace(/-+/g, '-')
      .replace(/^[-:]+|[-:]+$/g, '');
  }
}
```

```javascript
// 值结构最佳实践
export class ValueStructure {
  // 标准化值格式
  static createValue(data, metadata = {}) {
    return {
      v: 1, // 版本号
      data: data, // 实际数据
      meta: {
        created: Date.now(),
        updated: Date.now(),
        version: '1.0.0',
        ...metadata
      }
    };
  }

  // 压缩大值
  static async compressValue(data, compressionLevel = 6) {
    const jsonString = JSON.stringify(data);

    // 只对大于1KB的数据进行压缩
    if (jsonString.length < 1024) {
      return jsonString;
    }

    // 使用简单的字符串压缩（示例）
    // 实际中可以使用更好的压缩算法
    const compressed = await this.simpleCompress(jsonString, compressionLevel);

    // 添加压缩标记
    return {
      compressed: true,
      algorithm: 'simple',
      originalSize: jsonString.length,
      compressedSize: compressed.length,
      data: compressed
    };
  }

  // 分片大值
  static async shardLargeValue(data, maxSize = 25 * 1024) { // 25KB per shard
    const jsonString = JSON.stringify(data);

    if (jsonString.length <= maxSize) {
      return jsonString;
    }

    const shards = [];
    const totalShards = Math.ceil(jsonString.length / maxSize);

    for (let i = 0; i < totalShards; i++) {
      const start = i * maxSize;
      const end = Math.min(start + maxSize, jsonString.length);

      shards.push({
        index: i,
        total: totalShards,
        data: jsonString.slice(start, end)
      });
    }

    return {
      sharded: true,
      totalShards: totalShards,
      shards: shards
    };
  }

  // 创建带索引的值
  static createIndexedValue(data, searchFields = []) {
    const value = this.createValue(data);

    // 生成搜索索引
    value.index = {};
    for (const field of searchFields) {
      const fieldValue = this.getNestedValue(data, field);
      if (fieldValue !== null && fieldValue !== undefined) {
        value.index[field] = String(fieldValue).toLowerCase();
      }
    }

    return value;
  }

  // 获取嵌套值
  static getNestedValue(obj, path) {
    return path.split('.').reduce((current, key) => {
      return current && current[key] !== undefined ? current[key] : null;
    }, obj);
  }

  // 简单压缩（示例）
  static async simpleCompress(data, level = 6) {
    // 这里应该使用实际的压缩算法
    // 例如: pako, lz-string 等
    return btoa(data); // Base64 编码作为简单示例
  }
}
```

### 1. 批量操作策略
```javascript
// 批量操作优化
export class BatchOperations {
  constructor(env) {
    this.env = env;
    this.batchSize = 100; // KV 批量操作限制
    this.batchQueue = [];
    this.batchTimeout = null;
    this.flushInterval = 100; // 100ms
  }

  // 批量写入
  async batchPut(items) {
    const results = [];
    const batches = this.chunkArray(items, this.batchSize);

    for (const batch of batches) {
      const operations = batch.map(item =>
        this.env.KV.put(item.key, item.value, item.options)
      );

      const batchResults = await Promise.allSettled(operations);
      results.push(...batchResults);
    }

    return results;
  }

  // 批量读取（并发）
  async batchGet(keys, options = {}) {
    const { concurrency = 10 } = options;
    const results = new Map();

    // 分组并发读取
    const batches = this.chunkArray(keys, concurrency);

    for (const batch of batches) {
      const promises = batch.map(async (key) => {
        const value = await this.env.KV.get(key);
        return { key, value };
      });

      const batchResults = await Promise.allSettled(promises);

      for (const result of batchResults) {
        if (result.status === 'fulfilled') {
          results.set(result.value.key, result.value.value);
        } else {
          console.error('Get failed:', result.reason);
        }
      }
    }

    return results;
  }

  // 智能批量操作（自动收集）
  async smartBatchPut(key, value, options = {}) {
    this.batchQueue.push({ key, value, options });

    // 如果队列达到批大小，立即刷新
    if (this.batchQueue.length >= this.batchSize) {
      await this.flushBatch();
      return;
    }

    // 设置超时刷新
    if (!this.batchTimeout) {
      this.batchTimeout = setTimeout(() => {
        this.flushBatch();
      }, this.flushInterval);
    }
  }

  // 刷新批次
  async flushBatch() {
    if (this.batchQueue.length === 0) {
      return;
    }

    const batch = this.batchQueue.splice(0, this.batchSize);

    // 清除超时
    if (this.batchTimeout) {
      clearTimeout(this.batchTimeout);
      this.batchTimeout = null;
    }

    // 执行批量写入
    await this.batchPut(batch);
  }

  // 数组分块
  chunkArray(array, chunkSize) {
    const chunks = [];
    for (let i = 0; i < array.length; i += chunkSize) {
      chunks.push(array.slice(i, i + chunkSize));
    }
    return chunks;
  }

  // 预热缓存
  async warmupCache(keys, options = {}) {
    const { concurrency = 20 } = options;

    console.log(`Warming up cache for ${keys.length} keys...`);

    const batches = this.chunkArray(keys, concurrency);

    for (const batch of batches) {
      await Promise.allSettled(
        batch.map(key => this.env.KV.get(key))
      );
    }

    console.log('Cache warmup completed');
  }
}
```

### 2. 缓存策略
```javascript
// 多层缓存策略
export class MultiLevelCache {
  constructor(env) {
    this.env = env;
    this.memoryCache = new Map();
    this.memoryCacheMaxSize = 1000;
    this.memoryCacheTTL = 5 * 60 * 1000; // 5分钟
  }

  // 获取数据（多层缓存）
  async get(key, options = {}) {
    const {
      useMemoryCache = true,
      useEdgeCache = true,
      fallbackFn = null
    } = options;

    // L1: 内存缓存
    if (useMemoryCache) {
      const memoryValue = this.getFromMemoryCache(key);
      if (memoryValue !== null) {
        return memoryValue;
      }
    }

    // L2: 边缘缓存
    if (useEdgeCache) {
      const edgeValue = await this.getFromEdgeCache(key);
      if (edgeValue !== null) {
        // 回填内存缓存
        if (useMemoryCache) {
          this.setToMemoryCache(key, edgeValue);
        }
        return edgeValue;
      }
    }

    // L3: 数据源（回源）
    if (fallbackFn) {
      const freshValue = await fallbackFn(key);
      if (freshValue !== null) {
        // 回填所有缓存层
        await this.setToEdgeCache(key, freshValue, options);
        if (useMemoryCache) {
          this.setToMemoryCache(key, freshValue);
        }
        return freshValue;
      }
    }

    return null;
  }

  // 设置数据（多层缓存）
  async set(key, value, options = {}) {
    const {
      useMemoryCache = true,
      useEdgeCache = true,
      edgeTTL = 3600 // 1小时
    } = options;

    // 设置内存缓存
    if (useMemoryCache) {
      this.setToMemoryCache(key, value);
    }

    // 设置边缘缓存
    if (useEdgeCache) {
      await this.setToEdgeCache(key, value, { ttl: edgeTTL });
    }
  }

  // 内存缓存操作
  getFromMemoryCache(key) {
    const cached = this.memoryCache.get(key);
    if (cached && Date.now() - cached.timestamp < this.memoryCacheTTL) {
      return cached.value;
    }

    if (cached) {
      this.memoryCache.delete(key);
    }

    return null;
  }

  setToMemoryCache(key, value) {
    // LRU 淘汰
    if (this.memoryCache.size >= this.memoryCacheMaxSize) {
      const firstKey = this.memoryCache.keys().next().value;
      this.memoryCache.delete(firstKey);
    }

    this.memoryCache.set(key, {
      value: value,
      timestamp: Date.now()
    });
  }

  // 边缘缓存操作
  async getFromEdgeCache(key) {
    try {
      const value = await this.env.KV.get(key);
      return value !== null ? JSON.parse(value) : null;
    } catch (error) {
      console.error('Edge cache get failed:', error);
      return null;
    }
  }

  async setToEdgeCache(key, value, options = {}) {
    try {
      const serializedValue = JSON.stringify(value);
      await this.env.KV.put(key, serializedValue, {
        expirationTtl: options.ttl
      });
    } catch (error) {
      console.error('Edge cache set failed:', error);
    }
  }

  // 缓存失效
  async invalidate(key) {
    // 清除内存缓存
    this.memoryCache.delete(key);

    // 清除边缘缓存
    await this.env.KV.delete(key);
  }

  // 批量失效
  async invalidatePattern(pattern) {
    // 清除匹配模式的内存缓存
    for (const key of this.memoryCache.keys()) {
      if (key.includes(pattern)) {
        this.memoryCache.delete(key);
      }
    }

    // 清除匹配模式的边缘缓存
    const list = await this.env.KV.list({
      prefix: pattern
    });

    for (const key of list.keys) {
      await this.env.KV.delete(key.name);
    }
  }
}
```

```javascript
// 读写分离策略
export class ReadWriteStrategy {
  constructor(env) {
    this.env = env;
    this.readReplicaThreshold = 100; // 读请求数阈值
    this.readCounters = new Map();
  }

  // 智能读取路由
  async smartRead(key, options = {}) {
    const readCount = this.getReadCount(key);
    const { forceFresh = false } = options;

    // 高频读取使用副本
    if (readCount > this.readReplicaThreshold && !forceFresh) {
      return await this.readFromReplica(key, options);
    }

    // 低频或强制新鲜读取使用主存储
    return await this.readFromPrimary(key, options);
  }

  // 从主存储读取
  async readFromPrimary(key, options = {}) {
    const value = await this.env.KV_PRIMARY.get(key, options);
    this.incrementReadCount(key);
    return value;
  }

  // 从副本读取（实现缓存）
  async readFromReplica(key, options = {}) {
    // 实际上使用缓存作为"副本"
    const cache = new MultiLevelCache(this.env);
    return await cache.get(key, options);
  }

  // 写入（只到主存储）
  async write(key, value, options = {}) {
    const result = await this.env.KV_PRIMARY.put(key, value, options);

    // 异步更新缓存
    this.updateCacheAsync(key, value);

    return result;
  }

  // 异步更新缓存
  async updateCacheAsync(key, value) {
    try {
      const cache = new MultiLevelCache(this.env);
      await cache.set(key, value);
    } catch (error) {
      console.error('Cache update failed:', error);
    }
  }

  // 获取读取计数
  getReadCount(key) {
    const count = this.readCounters.get(key) || 0;

    // 定期重置计数（滑动窗口）
    if (Math.random() < 0.01) { // 1% 概率重置
      this.readCounters.set(key, 0);
    }

    return count;
  }

  // 增加读取计数
  incrementReadCount(key) {
    const current = this.readCounters.get(key) || 0;
    this.readCounters.set(key, current + 1);
  }
}
```

## 错误处理和恢复

### 1. 重试机制
```javascript
// 智能重试系统
export class RetryManager {
  constructor(options = {}) {
    this.maxRetries = options.maxRetries || 3;
    this.baseDelay = options.baseDelay || 1000;
    this.maxDelay = options.maxDelay || 30000;
    this.backoffMultiplier = options.backoffMultiplier || 2;
    this.jitter = options.jitter || true;
  }

  // 执行带重试的操作
  async execute(operation, context = {}) {
    let lastError = null;

    for (let attempt = 0; attempt <= this.maxRetries; attempt++) {
      try {
        const result = await operation(attempt, context);

        if (attempt > 0) {
          console.log(`Operation succeeded after ${attempt} retries`);
        }

        return result;

      } catch (error) {
        lastError = error;

        // 检查是否应该重试
        if (!this.shouldRetry(error, attempt)) {
          break;
        }

        // 计算延迟
        const delay = this.calculateDelay(attempt);

        console.warn(`Attempt ${attempt + 1} failed, retrying in ${delay}ms:`, error.message);

        // 等待
        await this.sleep(delay);
      }
    }

    // 所有重试都失败了
    throw new Error(`Operation failed after ${this.maxRetries + 1} attempts. Last error: ${lastError.message}`);
  }

  // 判断是否应该重试
  shouldRetry(error, attempt) {
    // 超过最大重试次数
    if (attempt >= this.maxRetries) {
      return false;
    }

    // 可重试的错误类型
    const retryableErrors = [
      'timeout',
      'connection',
      'network',
      'rate limit',
      'service unavailable',
      'temporary failure'
    ];

    const errorMessage = error.message.toLowerCase();

    return retryableErrors.some(retryableError =>
      errorMessage.includes(retryableError)
    );
  }

  // 计算重试延迟
  calculateDelay(attempt) {
    let delay = this.baseDelay * Math.pow(this.backoffMultiplier, attempt);

    // 限制最大延迟
    delay = Math.min(delay, this.maxDelay);

    // 添加抖动
    if (this.jitter) {
      const jitterAmount = delay * 0.1;
      delay += (Math.random() - 0.5) * jitterAmount;
    }

    return Math.max(0, Math.floor(delay));
  }

  // 睡眠函数
  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

```javascript
// 断路器实现
export class CircuitBreaker {
  constructor(options = {}) {
    this.failureThreshold = options.failureThreshold || 5;
    this.timeout = options.timeout || 60000; // 1分钟
    this.monitoringPeriod = options.monitoringPeriod || 10000; // 10秒

    this.state = 'CLOSED'; // CLOSED, OPEN, HALF_OPEN
    this.failureCount = 0;
    this.lastFailureTime = null;
    this.successCount = 0;
  }

  // 执行受保护的操作
  async execute(operation) {
    if (this.state === 'OPEN') {
      if (this.shouldAttemptReset()) {
        this.state = 'HALF_OPEN';
        this.successCount = 0;
      } else {
        throw new Error('Circuit breaker is OPEN');
      }
    }

    try {
      const result = await operation();
      this.onSuccess();
      return result;

    } catch (error) {
      this.onFailure();
      throw error;
    }
  }

  // 操作成功
  onSuccess() {
    if (this.state === 'HALF_OPEN') {
      this.successCount++;

      // 在半开状态下，需要连续成功才能关闭断路器
      if (this.successCount >= 2) {
        this.close();
      }
    } else {
      this.failureCount = 0;
    }
  }

  // 操作失败
  onFailure() {
    this.failureCount++;
    this.lastFailureTime = Date.now();

    if (this.state === 'HALF_OPEN') {
      this.open();
    } else if (this.failureCount >= this.failureThreshold) {
      this.open();
    }
  }

  // 打开断路器
  open() {
    this.state = 'OPEN';
    console.warn('Circuit breaker opened due to failures');
  }

  // 关闭断路器
  close() {
    this.state = 'CLOSED';
    this.failureCount = 0;
    this.successCount = 0;
    console.log('Circuit breaker closed');
  }

  // 是否应该尝试重置
  shouldAttemptReset() {
    return Date.now() - this.lastFailureTime >= this.timeout;
  }

  // 获取状态
  getState() {
    return {
      state: this.state,
      failureCount: this.failureCount,
      successCount: this.successCount,
      lastFailureTime: this.lastFailureTime
    };
  }
}
```

```javascript
// 服务降级管理
export class DegradationManager {
  constructor(env) {
    this.env = env;
    this.strategies = new Map();
  }

  // 注册降级策略
  register(name, strategy) {
    this.strategies.set(name, strategy);
  }

  // 执行带降级的操作
  async execute(operationName, primaryOperation, fallbackData = null) {
    const strategy = this.strategies.get(operationName);

    try {
      // 尝试主要操作
      return await primaryOperation();

    } catch (error) {
      console.warn(`Primary operation failed: ${operationName}`, error.message);

      // 执行降级策略
      if (strategy) {
        try {
          return await strategy.execute(fallbackData);
        } catch (fallbackError) {
          console.error(`Fallback failed: ${operationName}`, fallbackError.message);
        }
      }

      // 最后的保护：返回基本数据
      return fallbackData || this.getEmptyResponse(operationName);
    }
  }

  // 获取空响应
  getEmptyResponse(operationName) {
    const emptyResponses = {
```

```javascript
// 客户端加密
export class KVEncryption {
  constructor(secretKey) {
    this.secretKey = secretKey;
    this.algorithm = 'AES-GCM';
  }

  // 加密数据
  async encrypt(data, additionalData = null) {
    const encoder = new TextEncoder();
    const dataBuffer = encoder.encode(JSON.stringify(data));

    // 生成随机 IV
    const iv = crypto.getRandomValues(new Uint8Array(12));

    // 生成密钥
    const key = await this.deriveKey();

    // 加密
    const encryptedData = await crypto.subtle.encrypt(
      {
        name: this.algorithm,
        iv: iv,
        additionalData: additionalData ? encoder.encode(additionalData) : undefined
      },
      key,
      dataBuffer
    );

    return {
      algorithm: this.algorithm,
      iv: Array.from(iv),
      data: Array.from(new Uint8Array(encryptedData)),
      timestamp: Date.now()
    };
  }

  // 解密数据
  async decrypt(encryptedPayload, additionalData = null) {
    const key = await this.deriveKey();
    const encoder = new TextEncoder();
    const decoder = new TextDecoder();

    // 解密
    const decryptedData = await crypto.subtle.decrypt(
      {
        name: encryptedPayload.algorithm,
        iv: new Uint8Array(encryptedPayload.iv),
        additionalData: additionalData ? encoder.encode(additionalData) : undefined
      },
      key,
      new Uint8Array(encryptedPayload.data)
    );

    return JSON.parse(decoder.decode(decryptedData));
  }

  // 派生密钥
  async deriveKey() {
    const encoder = new TextEncoder();
    const keyMaterial = await crypto.subtle.importKey(
      'raw',
      encoder.encode(this.secretKey),
      { name: 'PBKDF2' },
      false,
      ['deriveBits', 'deriveKey']
    );

    return crypto.subtle.deriveKey(
      {
        name: 'PBKDF2',
        salt: encoder.encode('kv-encryption-salt'),
        iterations: 100000,
        hash: 'SHA-256'
      },
      keyMaterial,
      { name: 'AES-GCM', length: 256 },
      false,
      ['encrypt', 'decrypt']
    );
  }
}

// 敏感数据处理
export class SensitiveDataHandler {
  constructor(env) {
    this.env = env;
    this.encryption = new KVEncryption(env.ENCRYPTION_SECRET);
    this.sensitiveFields = ['password', 'token', 'secret', 'key'];
  }

  // 存储敏感数据
  async storeSecure(key, data) {
    const securedData = await this.secureData(data);
    return await this.env.KV.put(key, JSON.stringify(securedData));
  }

  // 读取敏感数据
  async retrieveSecure(key) {
    const value = await this.env.KV.get(key);
    if (!value) return null;

    const securedData = JSON.parse(value);
    return await this.unsecureData(securedData);
  }

  // 保护数据
  async secureData(data) {
    const secured = { ...data };

    for (const field of this.sensitiveFields) {
      if (secured[field]) {
        const encrypted = await this.encryption.encrypt(
          secured[field],
          field
        );
        secured[field] = {
          encrypted: true,
          data: encrypted
        };
      }
    }

    return secured;
  }

  // 解除保护
  async unsecureData(securedData) {
    const data = { ...securedData };

    for (const field of this.sensitiveFields) {
      if (data[field] && data[field].encrypted) {
        data[field] = await this.encryption.decrypt(
          data[field].data,
          field
        );
      }
    }

    return data;
  }
}
```

```javascript
// 基于角色的访问控制
export class KVAccessControl {
  constructor(env) {
    this.env = env;
    this.permissions = new Map();
    this.auditLog = [];
  }

  // 设置权限
  setPermission(role, resource, action) {
    if (!this.permissions.has(role)) {
      this.permissions.set(role, new Set());
    }

    this.permissions.get(role).add(`${resource}:${action}`);
  }

  // 检查权限
  async checkPermission(user, resource, action, key = null) {
    const userRole = await this.getUserRole(user);
    const permission = `${resource}:${action}`;

    const hasPermission = this.permissions.has(userRole) &&
      this.permissions.get(userRole).has(permission);

    // 记录访问日志
    this.logAccess(user, resource, action, key, hasPermission);

    return hasPermission;
  }

  // 带权限检查的操作
  async securedOperation(user, resource, action, key, operation) {
    const hasPermission = await this.checkPermission(user, resource, action, key);

    if (!hasPermission) {
      throw new Error(`Access denied: ${action} on ${resource}`);
    }

    return await operation();
  }

  // 获取用户角色
  async getUserRole(user) {
    // 从存储中获取用户角色
    const roleKey = `user:role:${user}`;
    return await this.env.KV.get(roleKey) || 'guest';
  }

  // 记录访问日志
  logAccess(user, resource, action, key, granted) {
    const logEntry = {
      timestamp: Date.now(),
      user: user,
      resource: resource,
      action: action,
      key: key,
      granted: granted
    };

    this.auditLog.push(logEntry);

    // 限制日志大小
    if (this.auditLog.length > 10000) {
      this.auditLog = this.auditLog.slice(-5000);
    }

    // 异步持久化日志
    this.persistAuditLog(logEntry);
  }

  // 持久化审计日志
  async persistAuditLog(logEntry) {
    try {
      const logKey = `audit:${Date.now()}:${Math.random()}`;
      await this.env.KV_AUDIT.put(
        logKey,
        JSON.stringify(logEntry),
        {
          expirationTtl: 30 * 24 * 60 * 60 // 30天
        }
      );
    } catch (error) {
      console.error('Failed to persist audit log:', error);
    }
  }
}

// 初始化访问控制
export function initializeAccessControl(env) {
  const accessControl = new KVAccessControl(env);

  // 管理员权限
  accessControl.setPermission('admin', 'kv', 'read');
  accessControl.setPermission('admin', 'kv', 'write');
  accessControl.setPermission('admin', 'kv', 'delete');
  accessControl.setPermission('admin', 'kv', 'list');

  // 用户权限
  accessControl.setPermission('user', 'user_data', 'read');
  accessControl.setPermission('user', 'user_data', 'write');
  accessControl.setPermission('user', 'content', 'read');

  // 访客权限
  accessControl.setPermission('guest', 'public', 'read');

  return accessControl;
}
```
