# KV API 使用指南

## API 概述

Cloudflare KV 提供简单而强大的 API，支持在 Workers 和 Pages Functions 中进行键值存储操作。API 设计遵循 RESTful 原则，提供直观的接口。

## Workers 中的 KV 绑定

### 1. 配置 KV 命名空间
```toml
# wrangler.toml
name = "my-worker"
compatibility_date = "2023-12-01"

# 单个 KV 命名空间
[[kv_namespaces]]
binding = "MY_KV"
id = "your-kv-namespace-id"
preview_id = "your-preview-kv-id"

# 多个 KV 命名空间
[[kv_namespaces]]
binding = "USERS_KV"
id = "users-kv-id"

[[kv_namespaces]]

id = "cache-kv-id"

# 环境特定的 KV 命名空间
[env.production]
[[env.production.kv_namespaces]]
binding = "PROD_KV"
id = "prod-kv-id"
```

```javascript

export async function advancedWrite(env, key, value) {
  // 设置过期时间（TTL - 秒）

  // 设置过期时间戳
  const expireAt = new Date('2024-12-31');
  await env.MY_KV.put(key, value, { expiration: Math.floor(expireAt.getTime() / 1000) });

  // 添加元数据
  await env.MY_KV.put(key, value, {

      author: 'system',
      version: '1.0',
      'content-type': 'application/json'
    }
  });

  await env.MY_KV.put(key, value, {

      type: 'cache'
    }
  });

  return { success: true };
}

export async function conditionalWrite(env, key, newValue) {
  // 先读取带元数据

  // 检查 ETag（版本控制）
  if (metadata.etag === 'expected-etag') {
    await env.MY_KV.put(key, newValue);
    return { success: true, updated: true };
  } else {
    return { success: false, reason: 'ETag mismatch' };
  }
}
```

```javascript
    // 列出键

      prefix,
      cursor,
      limit: 1000
    });

    await Promise.all(deletes);

    deletedCount += list.keys.length;
    cursor = list.cursor;

  } while (cursor);

  return { success: true, deletedCount };
}
```

### 2. 批量操作优化
```javascript
// 批量操作管理器
export class KVBatchOperation {
  constructor(kv) {
    this.kv = kv;
    this.operations = [];
  }

  addPut(key, value, options = {}) {
    this.operations.push({ type: 'put', key, value, options });
    return this;
  }

  addDelete(key) {
    this.operations.push({ type: 'delete', key });
    return this;
  }

  async execute(concurrency = 10) {
    // 分组并发执行
    const chunks = this.chunkArray(this.operations, concurrency);
    const results = [];

    for (const chunk of chunks) {
      const chunkResults = await Promise.all(
        chunk.map(op => this.executeOperation(op))
      );
      results.push(...chunkResults);
    }

    // 清空操作列表
    this.operations = [];

    return results;
  }

  async executeOperation(operation) {
    try {
      if (operation.type === 'put') {
        await this.kv.put(
          operation.key,
          operation.value,
          operation.options
        );
        return { success: true, key: operation.key };
      } else {
        await this.kv.delete(operation.key);
        return { success: true, key: operation.key, deleted: true };
      }
    } catch (error) {
      return { success: false, key: operation.key, error: error.message };
    }
  }

  chunkArray(array, size) {
    const chunks = [];
    for (let i = 0; i < array.length; i += size) {
      chunks.push(array.slice(i, i + size));
    }
    return chunks;
  }
}

// 使用示例
const batch = new KVBatchOperation(env.MY_KV);

batch
  .addPut('key1', 'value1', { expirationTtl: 3600 })
  .addPut('key2', { data: 'value2' }, { expirationTtl: 7200 })
  .addDelete('oldKey')
  .addPut('key3', 'value3');

const results = await batch.execute(5); // 最多 5 个并发操作
```

```javascript
// 自动压缩大型值
export class CompressedKV {
  constructor(kv, compressionThreshold = 1024) {
    this.kv = kv;
    this.threshold = compressionThreshold;
  }

  async put(key, value, options = {}) {
    const serialized = typeof value === 'string' ? value : JSON.stringify(value);

    // 检查是否需要压缩
    if (serialized.length > this.threshold) {
      const compressed = await this.compress(serialized);

      await this.kv.put(key, compressed, {
        ...options,
        metadata: {
          ...options.metadata,
          compressed: true,
          originalSize: serialized.length
        }
      });
    } else {
      await this.kv.put(key, serialized, {
        ...options,
        metadata: {
          ...options.metadata,
          compressed: false
        }
      });
    }
  }

  async get(key, type = 'text') {
    const { value, metadata } = await this.kv.getWithMetadata(key);

    if (!value) {
      return null;
    }

    // 检查是否需要解压
    if (metadata?.compressed) {
      const decompressed = await this.decompress(value);

      if (type === 'json') {
        return JSON.parse(decompressed);
      }

      return decompressed;
    }

    if (type === 'json') {
      return JSON.parse(value);
    }

    return value;
  }

  async compress(data) {
    // 使用压缩算法
    // 这里是示例，实际应用中可以使用 zlib 等
    return Buffer.from(data).toString('base64');
  }

  async decompress(data) {
    // 解压缩实现
    return Buffer.from(data, 'base64').toString();
  }
}
```

## 错误处理

### 1. API 错误类型
```javascript
// 错误处理示例
export async function safeKVOperation(env, operation) {
  try {
    const result = await operation();
    return { success: true, result };
  } catch (error) {
    console.error('KV operation failed:', error);

    // 分析错误类型
    if (error.message.includes('quota')) {
      return {
        success: false,
        error: 'Quota exceeded',
        type: 'quota_error'
      };
    }

    if (error.message.includes('timeout')) {
      return {
        success: false,
        error: 'Operation timeout',
        type: 'timeout_error'
      };
    }

    if (error.message.includes('not found')) {
      return {
        success: false,
        error: 'Key not found',
        type: 'not_found_error'
      };
    }

    return {
      success: false,
      error: error.message,
      type: 'unknown_error'
    };
  }
}

// 重试机制
export async function retryKVOperation(env, operation, maxRetries = 3) {
  let lastError;

  for (let i = 0; i < maxRetries; i++) {
    try {
      return await operation();
    } catch (error) {
      lastError = error;

      // 指数退避
      const delay = Math.min(1000 * Math.pow(2, i), 10000);
      await new Promise(resolve => setTimeout(resolve, delay));

      console.log(`Retry ${i + 1}/${maxRetries} after error:`, error.message);
    }
  }

  throw lastError;
}
```

### 2. 连接问题处理
```javascript
// 连接降级策略
export class KVWithFallback {
  constructor(kv, fallbackCache) {
    this.kv = kv;
    this.fallback = fallbackCache;
    this.healthy = true;
    this.lastCheck = 0;
  }

  async get(key, type = 'text') {
    // 检查连接健康状态
    await this.checkHealth();

    if (this.healthy) {
      try {
        const value = await this.kv.get(key, type);

        // 更新后备缓存
        await this.fallback.set(key, value);

        return value;
      } catch (error) {
        console.error('KV read failed, using fallback:', error);
        this.healthy = false;
      }
    }

    // 使用后备缓存
    return await this.fallback.get(key);
  }

  async put(key, value, options = {}) {
    // 写入总是先尝试 KV
    try {
      await this.kv.put(key, value, options);

      // 更新后备缓存
      await this.fallback.set(key, value);

      return true;
    } catch (error) {
      console.error('KV write failed, queuing for later:', error);

      // 将写入加入队列
      await this.queueWrite(key, value, options);

      return false;
    }
  }

  async checkHealth() {
    const now = Date.now();

    // 每 30 秒检查一次
    if (now - this.lastCheck < 30000) {
      return;
    }

    try {
      await this.kv.get('health-check');
      this.healthy = true;
    } catch (error) {
      this.healthy = false;
    }

    this.lastCheck = now;
  }

  async queueWrite(key, value, options) {
    // 将写入操作存储到队列
    const queued = {
      key,
      value,
      options,
      timestamp: Date.now()
    };

    await this.fallback.set(`queue:${key}`, queued);
  }

  async processQueue() {
    // 处理队列中的写入操作
    const queueKeys = await this.fallback.getKeys('queue:');

    for (const fullKey of queueKeys) {
      const queued = await this.fallback.get(fullKey);

      if (queued && Date.now() - queued.timestamp > 5000) {
        try {
          await this.kv.put(queued.key, queued.value, queued.options);
          await this.fallback.delete(fullKey);
        } catch (error) {
          console.error('Failed to process queued write:', error);
        }
      }
    }
  }
}
```

```javascript
// 多层缓存系统
export class MultiLevelCache {
  constructor(kv, l1Cache, l2Cache) {
    this.kv = kv;        // L3: KV
    this.l1 = l1Cache;   // L1: 内存 (最快)
    this.l2 = l2Cache;   // L2: 本地存储
  }

  async get(key) {
    // L1: 内存缓存
    let value = await this.l1.get(key);
    if (value !== null) {
      return { value, level: 'L1' };
    }

    // L2: 本地缓存
    value = await this.l2.get(key);
    if (value !== null) {
      // 回填 L1
      await this.l1.set(key, value, { ttl: 300 }); // 5分钟
      return { value, level: 'L2' };
    }

    // L3: KV
    value = await this.kv.get(key);
    if (value !== null) {
      // 回填 L2 和 L1
      await this.l2.set(key, value, { ttl: 3600 }); // 1小时
      await this.l1.set(key, value, { ttl: 300 });
      return { value, level: 'L3' };
    }

    return { value: null, level: 'None' };
  }

  async put(key, value, options = {}) {
    // 同时写入所有层
    await Promise.all([
      this.kv.put(key, value, options),
      this.l2.set(key, value, { ttl: 3600 }),
      this.l1.set(key, value, { ttl: 300 })
    ]);
  }
}
```

```javascript
// KV 操作日志记录
export class KVLogger {
  constructor(kv, logPrefix = 'kv-log') {
    this.kv = kv;
    this.logPrefix = logPrefix;
  }

  async log(operation, key, details = {}) {
    const logEntry = {
      timestamp: new Date().toISOString(),
      operation,
      key: this.maskSensitiveKey(key),
      details,
      requestId: crypto.randomUUID()
    };

    // 存储日志
    await this.kv.put(
      `${this.logPrefix}:${Date.now()}-${logEntry.requestId}`,
      JSON.stringify(logEntry),
      { expirationTtl: 7 * 24 * 3600 } // 7天
    );

    return logEntry;
  }

  maskSensitiveKey(key) {
    // 掩码敏感键
    const sensitivePatterns = [
      /password/i,
      /token/i,
      /secret/i,
      /key$/i
    ];

    for (const pattern of sensitivePatterns) {
      if (pattern.test(key)) {
        return key.replace(/./g, '*');
      }
    }

    return key;
  }

  async getLogs(limit = 100) {
    const list = await this.kv.list({
      prefix: `${this.logPrefix}:`,
      limit
    });

    const logs = await Promise.all(
      list.keys.map(async key => {
        const log = await this.kv.get(key.name, 'json');
        return log;
      })
    );

    return logs.filter(log => log).sort((a, b) =>
      new Date(b.timestamp) - new Date(a.timestamp)
    );
  }
}
```

```javascript
// KV 性能分析器
export class KVPerformanceAnalyzer {
  constructor(kv) {
    this.kv = kv;
  }

  async analyze(key, operation, fn) {
    const start = performance.now();
    const operationId = crypto.randomUUID();

      const duration = performance.now() - start;

      await this.recordPerformance({

  }

  async recordPerformance(data) {
    const key = `perf:${data.operation}:${Date.now()}-${data.operationId}`;
    await this.kv.put(key, JSON.stringify(data), {
      expirationTtl: 7 * 24 * 3600 // 7天
    });
  }

  async getPerformanceReport(operation, hours = 24) {
    const since = new Date(Date.now() - hours * 3600000);
    const list = await this.kv.list({
      prefix: `perf:${operation}:`,
      limit: 1000
    });

    const records = [];
    for (const key of list.keys) {
      const record = await this.kv.get(key.name, 'json');
      if (record && new Date(record.timestamp) > since) {
        records.push(record);
      }
    }

    // 计算统计信息
    const successful = records.filter(r => r.success);
    const failed = records.filter(r => !r.success);

    const avgDuration = successful.reduce((sum, r) => sum + r.duration, 0) / successful.length;
    const p95Duration = this.calculatePercentile(successful, 0.95);
    const p99Duration = this.calculatePercentile(successful, 0.99);

    return {
      operation,
      timeRange: `${hours}h`,
      totalOperations: records.length,
      successRate: (successful.length / records.length * 100).toFixed(2) + '%',
      avgDuration: avgDuration.toFixed(2),
      p95Duration,
      p99Duration,
      errors: failed.length
    };
  }

  calculatePercentile(data, percentile) {
    if (data.length === 0) return 0;

    const sorted = data.map(r => r.duration).sort((a, b) => a - b);
    const index = Math.ceil(sorted.length * percentile) - 1;
    return sorted[index];
  }
}

// 使用示例
const analyzer = new KVPerformanceAnalyzer(env.MY_KV);

// 分析读取性能
const { result, duration } = await analyzer.analyze('user:123', 'read', async () => {
  return await env.MY_KV.get('user:123');
});

// 获取性能报告
const report = await analyzer.getPerformanceReport('read', 24);
console.log('Read performance:', report);
```

```javascript
// 推荐的键命名模式
const keyPatterns = {

  // 速率限制

const avoidPatterns = {
```
