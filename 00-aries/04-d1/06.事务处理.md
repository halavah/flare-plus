# D1 事务处理

## 事务概述

事务是数据库操作的基本单位，确保一组操作要么全部成功，要么全部失败。D1 基于 SQLite，支持完整的 ACID 事务特性。

## ACID 特性

### 1. 原子性 (Atomicity)
```javascript
// 事务中的所有操作要么全部成功，要么全部失败
export async function transferFunds(env, fromAccountId, toAccountId, amount) {
  try {
    const result = await env.DB.batch([
      // 扣除转出账户
      env.DB.prepare(`
        UPDATE accounts
        SET balance = balance - ?
        WHERE id = ? AND balance >= ?
      `).bind(amount, fromAccountId, amount),

      // 增加转入账户
      env.DB.prepare(`
        UPDATE accounts
        SET balance = balance + ?
        WHERE id = ?
      `).bind(amount, toAccountId),

      // 记录交易
      env.DB.prepare(`
        INSERT INTO transactions (from_account, to_account, amount, type)
        VALUES (?, ?, ?, 'transfer')
      `).bind(fromAccountId, toAccountId, amount)
    ]);

    // 检查转账是否成功
    if (result[0].changes === 0) {
      throw new Error('Insufficient balance or account not found');
    }

    return { success: true, transactionId: result[2].meta.last_row_id };

  } catch (error) {
    // 事务会自动回滚
    console.error('Transfer failed:', error);
    throw error;
  }
}
```

### 2. 一致性 (Consistency)
```javascript
// 确保数据库始终处于一致状态
export async function createOrderWithItems(env, orderData, items) {
  // 计算订单总额
  const total = items.reduce((sum, item) => sum + (item.price * item.quantity), 0);

  const result = await env.DB.batch([
    // 创建订单
    env.DB.prepare(`
      INSERT INTO orders (customer_id, total, status, created_at)
      VALUES (?, ?, 'pending', CURRENT_TIMESTAMP)
    `).bind(orderData.customerId, total),

    // 批量插入订单项
    ...items.map(item =>
      env.DB.prepare(`
        INSERT INTO order_items (order_id, product_id, quantity, unit_price)
        VALUES (?, ?, ?, ?)
      `).bind(
        /* order_id will be set after first insert */,
        item.productId,
        item.quantity,
        item.price
      )
    )
  ]);

  // 更新订单项的 order_id
  const orderId = result[0].meta.last_row_id;
  const itemResults = result.slice(1);

  await env.DB.batch(
    itemResults.map(itemResult =>
      env.DB.prepare(`
        UPDATE order_items
        SET order_id = ?
        WHERE id = ?
      `).bind(orderId, itemResult.meta.last_row_id)
    )
  );

  return { orderId, total };
}
```

### 3. 隔离性 (Isolation)
```javascript
// 隔离级别：默认使用 SERIALIZABLE
export async function updateProductStock(env, productId, quantity, operation) {
  // 使用 FOR UPDATE 确保排他锁 (SQLite 通过事务隔离实现)
  const currentStock = await env.DB.prepare(`
    SELECT stock FROM products WHERE id = ? FOR UPDATE
  `).bind(productId).first();

  if (!currentStock) {
    throw new Error('Product not found');
  }

  const newStock = operation === 'add'
    ? currentStock.stock + quantity
    : currentStock.stock - quantity;

  if (newStock < 0) {
    throw new Error('Insufficient stock');
  }

  await env.DB.prepare(`
    UPDATE products
    SET stock = ?, updated_at = CURRENT_TIMESTAMP
    WHERE id = ?
  `).bind(newStock, productId).run();

  // 记录库存变动
  await env.DB.prepare(`
    INSERT INTO stock_movements (product_id, quantity, operation, created_at)
    VALUES (?, ?, ?, CURRENT_TIMESTAMP)
  `).bind(productId, quantity, operation).run();

  return newStock;
}
```

## 事务模式

### 1. 自动提交模式
```javascript
// 默认模式：每个语句都是独立事务
export async function autoCommitExample(env) {
  // 每个执行都是独立事务
  await env.DB.prepare(`
    INSERT INTO logs (message, level) VALUES ('test', 'info')
  `).run();

  // 失败不影响之前的操作
  try {
    await env.DB.prepare(`
      INSERT INTO users (name, email) VALUES (?, ?)
    `).bind('John', 'invalid-email').run();
  } catch (error) {
    // 只有这个操作失败
    console.error('Insert failed:', error);
  }

  // 后续操作继续执行
  await env.DB.prepare(`
    INSERT INTO logs (message, level) VALUES ('continuing', 'info')
  `).run();
}
```

### 2. 显式事务
```javascript
// 使用 batch() 执行事务
export async function explicitTransaction(env) {
  try {
    // 所有操作在单个事务中
    const results = await env.DB.batch([
      env.DB.prepare(`
        INSERT INTO users (name, email) VALUES (?, ?)
      `).bind('Alice', 'alice@example.com'),

      env.DB.prepare(`
        INSERT INTO profiles (user_id, bio) VALUES (?, ?)
      `).bind(1, 'Software Developer'),

      env.DB.prepare(`
        INSERT INTO user_settings (user_id, theme) VALUES (?, 'dark')
      `).bind(1)
    ]);

    // 所有操作都成功
    console.log('Transaction completed:', results);
    return { success: true };

  } catch (error) {
    // 自动回滚所有操作
    console.error('Transaction rolled back:', error);
    throw error;
  }
}
```

### 3. 嵌套事务 (Savepoints)
```javascript
// 使用 SAVEPOINT 实现部分回滚
export async function nestedTransaction(env) {
  try {
    // 开始主事务
    await env.DB.prepare('BEGIN TRANSACTION').run();

    try {
      // 第一个操作块
      await env.DB.prepare(`
        INSERT INTO users (name, email) VALUES (?, ?)
      `).bind('Bob', 'bob@example.com').run();

      // 创建保存点
      await env.DB.prepare('SAVEPOINT user_inserted').run();

      try {
        // 可能有问题的操作
        await env.DB.prepare(`
          INSERT INTO orders (user_id, total) VALUES (?, ?)
        `).bind(1, 100).run();

        // 提交保存点
        await env.DB.prepare('RELEASE SAVEPOINT user_inserted').run();

      } catch (orderError) {
        // 回滚到保存点
        await env.DB.prepare('ROLLBACK TO SAVEPOINT user_inserted').run();
        console.log('Order failed, but user kept:', orderError);

        // 继续其他操作
        await env.DB.prepare(`
          INSERT INTO user_preferences (user_id) VALUES (?)
        `).bind(1).run();
      }

      // 提交主事务
      await env.DB.prepare('COMMIT').run();

    } catch (error) {
      // 回滚主事务
      await env.DB.prepare('ROLLBACK').run();
      throw error;
    }

  } catch (error) {
    console.error('Transaction failed:', error);
    throw error;
  }
}
```

## 事务隔离级别

### 1. SERIALIZABLE (默认)
```javascript
// 最高隔离级别：完全隔离，防止所有并发问题
export async function serializableExample(env) {
  // D1 默认使用 SERIALIZABLE 隔离级别

  // 场景：库存扣减
  const productId = 1;
  const quantityToBuy = 5;

  // 事务1
  const transaction1 = async () => {
    const product = await env.DB.prepare(`
      SELECT stock FROM products WHERE id = ?
    `).bind(productId).first();

    if (product.stock >= quantityToBuy) {
      // 模拟处理时间
      await new Promise(resolve => setTimeout(resolve, 100));

      await env.DB.prepare(`
        UPDATE products SET stock = stock - ? WHERE id = ?
      `).bind(quantityToBuy, productId).run();

      return true;
    }
    return false;
  };

  // 两个并发事务
  const [result1, result2] = await Promise.all([
    transaction1(),
    transaction1()
  ]);

  // 由于序列化隔离，只有一个会成功
  console.log('Results:', { result1, result2 });
}
```

## 事务锁机制

### 1. 锁类型
```javascript
// SQLite 的锁机制
export async function lockingBehavior(env) {
  // 共享锁 (Shared Lock)
  // - 多个读取可以同时持有
  // - 阻止写入
  await env.DB.prepare('SELECT * FROM products').all();

  // 保留锁 (Reserved Lock)
  // - 单个写入准备
  // - 允许继续读取
  // - 阻止其他写入准备

  // 排他锁 (Exclusive Lock)
  // - 只有单个写入
  // - 阻止所有其他操作
  await env.DB.prepare(`
    INSERT INTO products (name, price) VALUES (?, ?)
  `).bind('New Product', 99.99).run();
}
```

### 2. 死锁处理
```javascript
// 避免死锁的策略
export async function avoidDeadlock(env) {
  // 策略1：按固定顺序访问表
  const safeTransfer = async (fromId, toId, amount) => {
    // 确保总是先访问 ID 较小的账户
    const [firstId, secondId] = fromId < toId ? [fromId, toId] : [toId, fromId];

    await env.DB.batch([
      env.DB.prepare(`
        UPDATE accounts SET balance = balance - ? WHERE id = ?
      `).bind(amount, fromId),

      env.DB.prepare(`
        UPDATE accounts SET balance = balance + ? WHERE id = ?
      `).bind(amount, toId)
    ]);
  };

  // 策略2：减少事务大小
  const smallTransactions = async () => {
    // 将大事务拆分为小事务
    const userId = 1;
    const updates = [
      { field: 'name', value: 'John Doe' },
      { field: 'email', value: 'john@example.com' },
      { field: 'phone', value: '123-456-7890' }
    ];

    for (const update of updates) {
      await env.DB.prepare(`
        UPDATE users SET ${update.field} = ? WHERE id = ?
      `).bind(update.value, userId).run();
    }
  };

  // 策略3：使用重试机制
  const withRetry = async (operation, maxRetries = 3) => {
    for (let i = 0; i < maxRetries; i++) {
      try {
        return await operation();
      } catch (error) {
        if (error.message.includes('database is locked') && i < maxRetries - 1) {
          // 随机延迟后重试
          await new Promise(resolve =>
            setTimeout(resolve, Math.random() * 100 * (i + 1))
          );
          continue;
        }
        throw error;
      }
    }
  };
}
```

```javascript
export async function batchInsert(env, records) {
  const batchSize = 1000; // D1 限制
  const results = [];
```

## 错误处理

### 1. 事务回滚
```javascript
// 完整的错误处理示例
export async function robustTransaction(env, operations) {
  const startTime = Date.now();
  let attempts = 0;
  const maxAttempts = 3;

  while (attempts < maxAttempts) {
    attempts++;

    try {
      // 记录事务开始
      console.log(`Transaction attempt ${attempts} started`);

      // 执行事务
      const results = await env.DB.batch(operations);

      // 记录成功
      const duration = Date.now() - startTime;
      console.log(`Transaction completed in ${duration}ms`);

      return {
        success: true,
        results,
        attempts,
        duration
      };

    } catch (error) {
      console.error(`Transaction attempt ${attempts} failed:`, error);

      // 检查是否可以重试
      if (isRetryableError(error) && attempts < maxAttempts) {
        // 指数退避
        const delay = Math.min(1000 * Math.pow(2, attempts - 1), 5000);
        await new Promise(resolve => setTimeout(resolve, delay));
        continue;
      }

      // 不可重试错误或达到最大重试次数
      throw {
        message: `Transaction failed after ${attempts} attempts`,
        originalError: error,
        attempts,
        duration: Date.now() - startTime
      };
    }
  }
}

function isRetryableError(error) {
  const retryableErrors = [
    'database is locked',
    'SQLITE_BUSY',
    'SQLITE_LOCKED',
    'connection timeout'
  ];

  return retryableErrors.some(err =>
    error.message.toLowerCase().includes(err.toLowerCase())
  );
}
```

### 2. 部分失败处理
```javascript
// 处理部分操作失败
export async function partialFailureHandling(env, items) {
  const results = {
    successful: [],
    failed: []
  };

  // 逐项处理，便于精确错误处理
  for (const item of items) {
    try {
      const result = await env.DB.batch([
        env.DB.prepare(`
          INSERT INTO orders (user_id, total, status)
          VALUES (?, ?, 'pending')
        `).bind(item.userId, item.total),

        env.DB.prepare(`
          UPDATE inventory
          SET quantity = quantity - ?
          WHERE product_id = ? AND quantity >= ?
        `).bind(item.quantity, item.productId, item.quantity)
      ]);

      results.successful.push({
        item,
        orderId: result[0].meta.last_row_id
      });

    } catch (error) {
      results.failed.push({
        item,
        error: error.message
      });

      // 记录失败
      await env.DB.prepare(`
        INSERT INTO order_failures (user_id, product_id, error, item_data)
        VALUES (?, ?, ?, ?)
      `).bind(
        item.userId,
        item.productId,
        error.message,
        JSON.stringify(item)
      ).run();
    }
  }

  return results;
}
```

### 1. 事务监控
```javascript
// 监控事务性能
export async function monitorTransaction(env, operation, name) {
  const metrics = {
    name,
    startTime: Date.now(),
    success: false,
    duration: 0,
    error: null
  };

  try {
    const result = await operation();
    metrics.success = true;
    return result;

  } catch (error) {
    metrics.error = error.message;
    throw error;

  } finally {
    metrics.duration = Date.now() - metrics.startTime;

    // 记录事务指标
    await env.DB.prepare(`
      INSERT INTO transaction_metrics (name, duration, success, error, timestamp)
      VALUES (?, ?, ?, ?, ?)
    `).bind(
      metrics.name,
      metrics.duration,
      metrics.success ? 1 : 0,
      metrics.error,
      new Date().toISOString()
    ).run();

    // 慢事务告警
    if (metrics.duration > 1000) {
      console.warn(`Slow transaction detected: ${metrics.name} took ${metrics.duration}ms`);
    }
  }
}
```

```javascript
// 详细的事务日志记录
export async function loggedTransaction(env, operations, context) {
  const transactionId = crypto.randomUUID();
  const log = {
    transactionId,
    context,
    operations: operations.length,
    startTime: new Date().toISOString(),
    steps: []
  };

  try {
    const results = await env.DB.batch(
      operations.map((op, index) => {
        const stepLog = {
          index,
          sql: op.sql,
          bindings: op.bindings?.length || 0,
          startTime: Date.now()
        };

        return {
          ...op,
          afterRun: (result) => {
            stepLog.endTime = Date.now();
            stepLog.duration = stepLog.endTime - stepLog.startTime;
            stepLog.changes = result.changes;
            stepLog.lastRowId = result.meta?.last_row_id;
            log.steps.push(stepLog);
            return result;
          }
        };
      })
    );

    log.endTime = new Date().toISOString();
    log.success = true;

    // 记录成功日志
    await saveTransactionLog(env, log);

    return results;

  } catch (error) {
    log.endTime = new Date().toISOString();
    log.success = false;
    log.error = error.message;

    // 记录失败日志
    await saveTransactionLog(env, log);

    throw error;
  }
}

async function saveTransactionLog(env, log) {
  await env.DB.prepare(`
    INSERT INTO transaction_logs (transaction_id, context, log_data, created_at)
    VALUES (?, ?, ?, CURRENT_TIMESTAMP)
  `).bind(
    log.transactionId,
    JSON.stringify(log.context),
    JSON.stringify(log)
  ).run();
}
```

## 最佳实践

### 1. 事务设计原则
- **保持简短**: 事务时间尽量短，减少锁定时间
- **避免用户交互**: 不要在事务中等待用户输入
- **批量操作**: 使用 batch() 而不是多个单独语句
- **错误处理**: 始终包含适当的错误处理和回滚
- **重试机制**: 为可重试错误实现指数退避

### 2. 性能建议
- **批量大小**: 将大操作分解为小批量
- **索引优化**: 确保事务中使用的查询有适当索引
- **避免长查询**: 复杂查询可能延长事务时间
- **读写分离**: 使用只读事务处理报表查询

### 3. 监控指标
- 事务持续时间
- 事务成功率
- 死锁频率
- 并发事务数
- 回滚率
