# D1 免费额度详解

## 免费套餐概览

Cloudflare D1 提供慷慨的免费额度，适合小型应用、原型开发和测试环境使用。

### 核心免费额度
```javascript
// 每日限额
100,000 读取操作/day
1,000    写入操作/day
5GB      存储空间
5,000,000 行数据

// 请求限制
- 每个查询最多返回 10,000 行
- 每个事务最多执行 100 条语句
- 单个结果集最大 100MB
- 查询超时 30 秒
```

### 计费单位
```javascript
// 读操作 (Reads)
COUNT(*)             = 1 读取
SELECT               = 1 读取
PRAGMA               = 1 读取
事务中的读取          = 每条 1 读取

// 写操作 (Writes)
INSERT               = 1 写入
UPDATE               = 1 写入
DELETE               = 1 写入
CREATE/DROP TABLE    = 1 写入
CREATE INDEX         = 1 写入
ALTER TABLE          = 1 写入
```

## 操作分类详解

### 1. 读取操作 (Read Operations)
```sql
-- 基础查询 = 1 读取
SELECT * FROM users;
SELECT COUNT(*) FROM posts;

-- 子查询 = 每个子查询 1 读取
SELECT * FROM users
WHERE id IN (SELECT user_id FROM posts);

-- 聚合查询 = 1 读取
SELECT category, COUNT(*) as count
FROM products
GROUP BY category;

-- 连接查询 = 1 读取
SELECT u.*, p.title
FROM users u
JOIN posts p ON u.id = p.user_id;

-- 复杂查询 = 1 读取
WITH RECURSIVE categories(id, name, parent_id) AS (
  SELECT id, name, parent_id
  FROM categories
  WHERE parent_id IS NULL
  UNION ALL
  SELECT c.id, c.name, c.parent_id
  FROM categories c
  JOIN categories p ON c.parent_id = p.id
)
SELECT * FROM categories;
```

### 2. 写入操作 (Write Operations)
```sql
-- 单行插入 = 1 写入
INSERT INTO users (name, email) VALUES ('John', 'john@example.com');

-- 批量插入 = 1 写入
INSERT INTO users (name, email) VALUES
  ('Alice', 'alice@example.com'),
  ('Bob', 'bob@example.com'),
  ('Charlie', 'charlie@example.com');

-- 更新操作 = 1 写入
UPDATE users SET last_login = CURRENT_TIMESTAMP WHERE id = ?;

-- 批量更新 = 1 写入
UPDATE products SET price = price * 0.9 WHERE category = 'electronics';

-- 删除操作 = 1 写入
DELETE FROM users WHERE last_login < datetime('now', '-1 year');

-- 表结构操作 = 1 写入
CREATE TABLE orders (...);
ALTER TABLE users ADD COLUMN phone TEXT;
CREATE INDEX idx_users_email ON users(email);
DROP TABLE old_table;
```

### 3. 事务操作
```javascript
// 事务中的操作计数
BEGIN TRANSACTION;
INSERT INTO orders (...) VALUES (...);  -- 1 写入
UPDATE inventory SET ...;               -- 1 写入
SELECT * FROM orders WHERE id = ?;      -- 1 读取
COMMIT;                                  -- 总计: 2 写入 + 1 读取

// 批量事务
const stmts = db.prepare(`
  INSERT INTO audit_log (action, created_at) VALUES (?, CURRENT_TIMESTAMP);
`);
for (let i = 0; i < 50; i++) {
  await stmts.bind('operation').run();  -- 每条 1 写入
}
// 总计: 50 写入操作
```

### 2. 额度预警
```javascript
// 额度检查中间件
export async function checkD1Quota(context) {
  const usage = await getD1Usage();

  // 检查是否接近限额
  const readUsage = usage.reads / 100000;
  const writeUsage = usage.writes / 1000;

  if (readUsage > 0.9 || writeUsage > 0.9) {
    // 返回警告信息
    return new Response(JSON.stringify({
      error: 'Approaching D1 quota limit',
      reads: usage.reads,
      writes: usage.writes,
      limits: {
        reads: 100000,
        writes: 1000
      }
    }), {
      status: 429,
      headers: { 'Content-Type': 'application/json' }
    });
  }

  return context.next();
}
```

### 3. 使用统计
```javascript
// functions/api/stats/d1-usage.js
export async function onRequestGet(context) {
  const { env } = context;

  // 从 KV 获取统计数据
  const stats = await env.D1_STATS_KV.get('daily_usage');
  const data = stats ? JSON.parse(stats) : {};

  // 计算使用率
  const readUsage = (data.reads || 0) / 100000 * 100;
  const writeUsage = (data.writes || 0) / 1000 * 100;
  const storageUsage = (data.storage || 0) / (5 * 1024 * 1024 * 1024) * 100;

  return Response.json({
    usage: {
      reads: {
        current: data.reads || 0,
        limit: 100000,
        percentage: readUsage.toFixed(2)
      },
      writes: {
        current: data.writes || 0,
        limit: 1000,
        percentage: writeUsage.toFixed(2)
      },
      storage: {
        current: formatBytes(data.storage || 0),
        limit: '5GB',
        percentage: storageUsage.toFixed(2)
      },
      rows: {
        current: data.rows || 0,
        limit: 5000000
      }
    },
    resetAt: getNextResetTime()
  });
}

function formatBytes(bytes) {
  const sizes = ['Bytes', 'KB', 'MB', 'GB'];
  if (bytes === 0) return '0 Bytes';
  const i = Math.floor(Math.log(bytes) / Math.log(1024));
  return Math.round(bytes / Math.pow(1024, i) * 100) / 100 + ' ' + sizes[i];
}
```

## 成本优化策略

### 1. 读取优化
```javascript
// 使用缓存减少读取
const cache = new Map();

export async function getPost(id) {
  // 先检查缓存
  if (cache.has(id)) {
    return cache.get(id);
  }

  // 从数据库读取
  const result = await env.DB.prepare(
    'SELECT * FROM posts WHERE id = ?'
  ).bind(id).first();

  // 存储到缓存
  if (result) {
    cache.set(id, result);
    // 设置过期时间
    setTimeout(() => cache.delete(id), 300000); // 5分钟
  }

  return result;
}

// 批量读取优化
export async function getPosts(ids) {
  if (ids.length === 0) return [];

  // 使用 IN 查询替代多次查询
  const placeholders = ids.map(() => '?').join(',');
  const results = await env.DB.prepare(
    `SELECT * FROM posts WHERE id IN (${placeholders})`
  ).bind(...ids).all();

  return results.results;
}
```

### 2. 写入优化
```javascript
// 批量插入优化
export async function createLogs(logs) {
  // 构建批量插入语句
  const values = logs.map(() => '(?, ?, ?)').join(',');
  const params = logs.flatMap(log => [log.level, log.message, log.timestamp]);

  await env.DB.prepare(`
    INSERT INTO logs (level, message, timestamp)
    VALUES ${values}
  `).bind(...params).run();

  // 比单条插入节省写入次数
  // 批量插入: 1 写入
  // 单条插入: logs.length 写入
}

// 延迟写入队列
const writeQueue = [];
let writeTimeout;

export async function queueWrite(sql, params) {
  writeQueue.push({ sql, params });

  // 清除之前的定时器
  if (writeTimeout) clearTimeout(writeTimeout);

  // 设置新的定时器（5秒后执行）
  writeTimeout = setTimeout(async () => {
    const batch = writeQueue.splice(0);
    await executeBatch(batch);
  }, 5000);
}

async function executeBatch(batch) {
  const stmt = env.DB.prepare(batch[0].sql);
  for (const item of batch) {
    await stmt.bind(...item.params).run();
  }
}
```

```javascript
export async function saveMetadata(id, metadata) {

  // 移动旧数据到归档表
  await env.DB.prepare(`
    INSERT INTO posts_archive
    SELECT * FROM posts
    WHERE created_at < datetime('now', '-1 year')
  `).run();

    DELETE FROM posts
    WHERE created_at < datetime('now', '-1 year')
  `).run();
}
```

### 3. 使用分析
```javascript
// 使用趋势分析
export async function analyzeUsage(days = 30) {
  const analysis = {
    read_trend: [],
    write_trend: [],
    storage_trend: [],
    recommendations: []
  };

  // 分析使用趋势
  for (let i = days; i > 0; i--) {
    const date = new Date();
    date.setDate(date.getDate() - i);

    const dayUsage = await getDayUsage(date);
    analysis.read_trend.push(dayUsage.reads);
    analysis.write_trend.push(dayUsage.writes);
  }

  // 生成优化建议
  const avgReads = analysis.read_trend.reduce((a, b) => a + b, 0) / days;
  const avgWrites = analysis.write_trend.reduce((a, b) => a + b, 0) / days;

  if (avgReads > 80000) {
    analysis.recommendations.push('考虑实施读取缓存以减少数据库读取');
  }

  if (avgWrites > 800) {
    analysis.recommendations.push('考虑使用批量写入以优化写入操作');
  }

  return analysis;
}
```
