# D1 最佳实践

## 概述

本文档总结了使用 Cloudflare D1 的最佳实践，涵盖数据库设计、性能优化、安全、监控等各个方面，帮助开发者构建高效、可靠的应用。

## 数据库设计最佳实践

### 1. 数据建模原则

#### 遵循范式设计
```sql
-- ✅ 良好的设计

  id INTEGER PRIMARY KEY,
  email TEXT UNIQUE NOT NULL,
  name TEXT NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE orders (
  id INTEGER PRIMARY KEY,
  user_id INTEGER NOT NULL,
  total DECIMAL(10,2),
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE

-- ❌ 避免反规范化（除非有性能需求）
-- CREATE TABLE users (
--   id INTEGER PRIMARY KEY,
--   email TEXT UNIQUE NOT NULL,
--   name TEXT NOT NULL,
--   orders JSON -- 避免存储数组
-- );
```

#### 合理使用数据类型
```sql
-- ✅ 选择合适的数据类型

  id INTEGER PRIMARY KEY,                    -- 自增主键

  in_stock BOOLEAN DEFAULT 1,               -- 布尔值

-- ✅ 使用约束

  id INTEGER PRIMARY KEY,
  age INTEGER CHECK(age >= 0 AND age <= 120),
```

#### 索引策略
```sql
-- ✅ 为常用查询创建索引

-- ✅ 复合索引优化

-- ✅ 部分索引（通过应用层实现）
-- 实际使用时查询中包含 WHERE active = 1

-- 不要为每个字段都创建索引
-- 不要创建低选择性索引（如性别字段）
```

### 2. 连接查询优化

#### 使用适当的 JOIN 类型
```sql

-- ✅ LEFT JOIN：保留左表所有行

-- ❌ 避免不必要的连接
-- 不要连接不相关的表
```

#### 优化子查询
```sql

FROM orders o
JOIN users u ON o.user_id = u.id
```

## 性能优化最佳实践

### 1. 缓存策略

#### 实现查询缓存
```javascript
class D1Cache {
  constructor(env, ttl = 300000) { // 5分钟默认
    this.env = env;
    this.cache = new Map();
    this.ttl = ttl;
  }

  async query(sql, params = [], useCache = true) {
    const cacheKey = `${sql}:${JSON.stringify(params)}`;

    if (useCache && this.cache.has(cacheKey)) {
      const cached = this.cache.get(cacheKey);
      if (Date.now() - cached.timestamp < this.ttl) {
        return cached.result;
      }
    }

    const result = await this.env.DB.prepare(sql).bind(...params).all();

    this.cache.set(cacheKey, {
      result,
      timestamp: Date.now()
    });

    return result;
  }

  invalidate(pattern) {
    for (const [key] of this.cache) {
      if (key.includes(pattern)) {
        this.cache.delete(key);
      }
    }
  }
}

// 使用示例
const cache = new D1Cache(env);
const posts = await cache.query('SELECT * FROM posts WHERE status = ?', ['published']);
```

```javascript
// 查询计划分析
export async function analyzeQuery(env, sql, params = []) {
  // 获取查询计划
  const explain = await env.DB.prepare(`EXPLAIN QUERY PLAN ${sql}`).all();

  // 执行查询并测量性能
  const start = performance.now();
  const result = await env.DB.prepare(sql).bind(...params).all();
  const duration = performance.now() - start;

  return {
    sql,
    duration,
    plan: explain.results,
    resultCount: result.results?.length || 0,
    recommendations: generateRecommendations(explain.results, duration)
  };
}

function generateRecommendations(plan, duration) {
  const recommendations = [];

  // 检查是否使用了索引
  const usingIndex = plan.some(p => p.detail.includes('USING INDEX'));
  if (!usingIndex && duration > 100) {
    recommendations.push('Consider adding an index to speed up this query');
  }

  // 检查是否进行了全表扫描
  const fullScan = plan.some(p => p.detail.includes('SCAN TABLE'));
  if (fullScan && duration > 500) {
    recommendations.push('Query is performing full table scan - consider optimization');
  }

  return recommendations;
}
```

```javascript
// 加密工具类
class Encryption {
  constructor(secretKey) {
    this.key = secretKey;
  }

  async encrypt(text) {
    const encoder = new TextEncoder();
    const data = encoder.encode(text);

    const key = await crypto.subtle.importKey(
      'raw',
      encoder.encode(this.key),
      { name: 'AES-GCM' },
      false,
      ['encrypt']
    );

    const iv = crypto.getRandomValues(new Uint8Array(12));
    const encrypted = await crypto.subtle.encrypt(
      { name: 'AES-GCM', iv },
      key,
      data
    );

    return {
      data: btoa(String.fromCharCode(...new Uint8Array(encrypted))),
      iv: btoa(String.fromCharCode(...iv))
    };
  }

  async decrypt(encryptedData) {
    const encoder = new TextEncoder();
    const key = await crypto.subtle.importKey(
      'raw',
      encoder.encode(this.key),
      { name: 'AES-GCM' },
      false,
      ['decrypt']
    );

    const data = new Uint8Array(atob(encryptedData.data).split('').map(c => c.charCodeAt(0)));
    const iv = new Uint8Array(atob(encryptedData.iv).split('').map(c => c.charCodeAt(0)));

    const decrypted = await crypto.subtle.decrypt(
      { name: 'AES-GCM', iv },
      key,
      data
    );

    return new TextDecoder().decode(decrypted);
  }
}

// 使用加密

await env.DB.prepare(`
```

```javascript
// 基于角色的访问控制
export async function checkPermission(env, userId, resource, action) {
  const permission = await env.DB.prepare(`
    SELECT p.name
    FROM users u
    JOIN user_roles ur ON u.id = ur.user_id
    JOIN roles r ON ur.role_id = r.id
    JOIN role_permissions rp ON r.id = rp.role_id
    JOIN permissions p ON rp.permission_id = p.id
    WHERE u.id = ? AND p.resource = ? AND p.action = ?
  `).bind(userId, resource, action).first();

  return !!permission;
}

// 中间件示例
export async function withPermission(resource, action) {
  return async (context) => {
    const user = await authenticateUser(context);

    const hasPermission = await checkPermission(
      context.env,
      user.id,
      resource,
      action
    );

    if (!hasPermission) {
      return new Response('Access denied', { status: 403 });
    }

    context.user = user;
    return context.next();
  };
}
```

```javascript
export async function trackUsage(env) {
  // 获取表大小
  const tables = await env.DB.prepare(`
    SELECT name, sql FROM sqlite_master WHERE type='table'
  `).all();

  const tableStats = await Promise.all(
    tables.results.map(async table => {
      const count = await env.DB.prepare(`
        SELECT COUNT(*) as count FROM ${table.name}
      `).first();

      return {
        name: table.name,
        rowCount: count.count
      };
    })
  );

  // 记录使用情况
  await env.DB.prepare(`
    INSERT INTO usage_stats (date, table_count, total_rows, json_data)
    VALUES (date('now'), ?, ?, ?)
  `).bind(
    tableStats.length,
    tableStats.reduce((sum, t) => sum + t.rowCount, 0),
    JSON.stringify(tableStats)
  ).run();

  return tableStats;
}
```

```javascript
export async function createBackup(env) {
  const backupId = crypto.randomUUID();
  const timestamp = new Date().toISOString();

  // 获取所有表
  const tables = await env.DB.prepare(`
    SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'
  `).all();

  const backup = {
    id: backupId,
    timestamp,
    tables: {}
  };

  // 导出每个表
  for (const table of tables.results) {
    const schema = await env.DB.prepare(`
      SELECT sql FROM sqlite_master WHERE name = ?
    `).bind(table.name).first();

    const data = await env.DB.prepare(`
      SELECT * FROM ${table.name}
    `).all();

    backup.tables[table.name] = {
      schema: schema.sql,
      data: data.results
    };
  }

  // 存储备份到 KV
  await env.BACKUP_KV.put(
    `backup_${backupId}`,
    JSON.stringify(backup),
    { expirationTtl: 30 * 24 * 3600 } // 30天
  );

  return backupId;
}
```

## 总结

### 核心原则

1. **设计优先**
   - 花时间设计良好的数据模型
   - 使用适当的数据类型和约束
   - 建立清晰的命名规范

2. **性能第一**
   - 优化查询和索引
   - 实现合理的缓存策略
   - 监控和调优慢查询

3. **安全至上**
   - 防止 SQL 注入
   - 加密敏感数据
   - 实施访问控制

4. **可靠性**
   - 使用事务保证一致性
   - 实现错误处理和重试
   - 建立备份和恢复机制

5. **可维护性**
   - 编写清晰的代码
   - 完善的文档和注释
   - 自动化部署和测试

### 常见陷阱

- ❌ 过度规范化导致过多连接查询
- ❌ 忽略索引导致全表扫描
- ❌ N+1 查询问题
- ❌ 在事务中执行耗时操作

- ❌ 缺乏监控和日志
