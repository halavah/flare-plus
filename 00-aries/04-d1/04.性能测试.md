# D1 性能测试

## 性能测试概述

D1 性能测试旨在评估数据库在不同负载下的表现，包括查询响应时间、并发处理能力、事务吞吐量等关键指标。

```javascript
export async function onRequestPost(context) {
  const { request, env } = context;
  const { table, count } = await request.json();

  const batchSize = 1000;
  let inserted = 0;

  console.log(`开始生成 ${count} 条测试数据到表 ${table}`);

  while (inserted < count) {

    const records = generateRecords(table, batch);

    inserted += batch;

    console.log(`已插入 ${inserted}/${count} 条记录`);
  }

  return Response.json({
    success: true,
    inserted,
    table
  });
}

function generateRecords(table, count) {
  const records = [];

  for (let i = 0; i < count; i++) {
    switch (table) {
      case 'users':
        records.push([
          faker.name.findName(),
          faker.internet.email(),
          faker.address.city(),
          Math.floor(Math.random() * 50) + 18
        ]);
        break;

      case 'products':
        records.push([
          faker.commerce.productName(),
          faker.commerce.department(),
          Math.floor(Math.random() * 1000) + 10,
          Math.floor(Math.random() * 100)
        ]);
        break;

      case 'orders':
        records.push([
          Math.floor(Math.random() * 10000) + 1,
          Math.floor(Math.random() * 100) + 1,
          Math.floor(Math.random() * 5) + 1,
          (Math.random() * 1000).toFixed(2)
        ]);
        break;
    }
  }

  return records;
}

async function insertBatch(db, table, records) {
  let sql;
  switch (table) {
    case 'users':
      sql = 'INSERT INTO users (name, email, city, age) VALUES (?, ?, ?, ?)';
      break;
    case 'products':
      sql = 'INSERT INTO products (name, category, price, stock) VALUES (?, ?, ?, ?)';
      break;
    case 'orders':
      sql = 'INSERT INTO orders (user_id, product_id, quantity, total) VALUES (?, ?, ?, ?)';
      break;
  }

  const stmt = db.prepare(sql);
  for (const record of records) {
    await stmt.bind(...record).run();
  }
}
```

```javascript
async function measureQuery(db, query) {
  const iterations = 100;
  const times = [];

  for (let i = 0; i < iterations; i++) {
    const start = performance.now();

    const stmt = db.prepare(query.sql);
    if (query.params) {
      await stmt.bind(...query.params).all();
    } else {
      await stmt.all();
    }

    const end = performance.now();
    times.push(end - start);
  }

  return {
    avg: (times.reduce((a, b) => a + b, 0) / times.length).toFixed(2),
    min: Math.min(...times).toFixed(2),
    max: Math.max(...times).toFixed(2),
    p95: percentile(times, 0.95).toFixed(2),
    p99: percentile(times, 0.99).toFixed(2)
  };
}

function percentile(arr, p) {
  const sorted = arr.slice().sort((a, b) => a - b);
  const index = Math.ceil(sorted.length * p) - 1;
  return sorted[index];
}
```

```javascript
async function testSingleInserts(db, count) {
  const start = performance.now();
  const times = [];

  for (let i = 0; i < count; i++) {
    const insertStart = performance.now();

    await db.prepare(`
      INSERT INTO products (name, category, price, stock)
      VALUES (?, ?, ?, ?)
    `).bind(
      `Product ${i}`,
      'Test Category',
      Math.random() * 100,
      Math.floor(Math.random() * 100)
    ).run();

    const insertEnd = performance.now();
    times.push(insertEnd - insertStart);
  }

  const end = performance.now();

  return {
    totalTime: (end - start).toFixed(2),
    avgTime: (times.reduce((a, b) => a + b, 0) / times.length).toFixed(2),
    throughput: (count / (end - start) * 1000).toFixed(2),
    times: times.slice(0, 10) // 只返回前10个时间点
  };
}

async function testBatchInserts(db, count) {
  const batchSize = 100;
  const batches = Math.ceil(count / batchSize);
  const start = performance.now();
  const batchTimes = [];

  for (let i = 0; i < batches; i++) {
    const batchStart = performance.now();
    const records = [];

    // 准备批量数据
    for (let j = 0; j < batchSize && (i * batchSize + j) < count; j++) {
      records.push([
        `Product ${i * batchSize + j}`,
        'Test Category',
        Math.random() * 100,
        Math.floor(Math.random() * 100)
      ]);
    }

    // 批量插入
    const values = records.map(() => '(?, ?, ?, ?)').join(',');
    const sql = `INSERT INTO products (name, category, price, stock) VALUES ${values}`;
    const stmt = db.prepare(sql);

    const params = records.flat();
    await stmt.bind(...params).run();

    const batchEnd = performance.now();
    batchTimes.push(batchEnd - batchStart);
  }

  const end = performance.now();

  return {
    totalTime: (end - start).toFixed(2),
    avgBatchTime: (batchTimes.reduce((a, b) => a + b, 0) / batchTimes.length).toFixed(2),
    throughput: (count / (end - start) * 1000).toFixed(2),
    batchCount: batches
  };
}
```

### 3. 并发性能测试
```javascript
// functions/performance/concurrent-test.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { concurrency, requests } = await request.json();

  console.log(`开始并发测试: ${concurrency} 并发, ${requests} 总请求`);

  const results = await runConcurrentTest(env.DB, concurrency, requests);

  return Response.json(results);
}

async function runConcurrentTest(db, concurrency, totalRequests) {
  const startTime = performance.now();
  const promises = [];
  const requestsPerWorker = Math.ceil(totalRequests / concurrency);

  // 创建并发工作器
  for (let i = 0; i < concurrency; i++) {
    promises.push(workerTest(db, requestsPerWorker, `worker-${i}`));
  }

  // 等待所有工作器完成
  const workerResults = await Promise.all(promises);
  const endTime = performance.now();

  // 汇总结果
  const allTimes = workerResults.flat();
  const totalTime = endTime - startTime;
  const successfulRequests = allTimes.length;

  return {
    configuration: {
      concurrency,
      totalRequests,
      requestsPerWorker
    },
    performance: {
      totalTime: totalTime.toFixed(2),
      successfulRequests,
      failedRequests: totalRequests - successfulRequests,
      throughput: (successfulRequests / totalTime * 1000).toFixed(2),
      avgResponseTime: (allTimes.reduce((a, b) => a + b, 0) / allTimes.length).toFixed(2),
      minResponseTime: Math.min(...allTimes).toFixed(2),
      maxResponseTime: Math.max(...allTimes).toFixed(2),
      p95: percentile(allTimes, 0.95).toFixed(2),
      p99: percentile(allTimes, 0.99).toFixed(2)
    },
    distribution: calculateDistribution(allTimes)
  };
}

async function workerTest(db, requestCount, workerId) {
  const times = [];
  const queries = [
    'SELECT * FROM users WHERE id = ?',
    'SELECT * FROM products WHERE category = ?',
    'SELECT COUNT(*) FROM orders',
    'SELECT u.name, COUNT(o.id) as order_count FROM users u LEFT JOIN orders o ON u.id = o.user_id GROUP BY u.id LIMIT 10'
  ];

  for (let i = 0; i < requestCount; i++) {
    const query = queries[Math.floor(Math.random() * queries.length)];
    const params = generateParamsForQuery(query);

    try {
      const start = performance.now();
      const stmt = db.prepare(query);

      if (params) {
        await stmt.bind(...params).all();
      } else {
        await stmt.all();
      }

      const end = performance.now();
      times.push(end - start);
    } catch (error) {
      console.error(`Worker ${workerId} error:`, error);
    }
  }

  return times;
}

function generateParamsForQuery(query) {
  if (query.includes('id = ?')) {
    return [Math.floor(Math.random() * 10000) + 1];
  } else if (query.includes('category = ?')) {
    return ['Test Category'];
  }
  return null;
}

function calculateDistribution(times) {
  const buckets = [0, 10, 50, 100, 200, 500, 1000];
  const distribution = {};

  for (let i = 0; i < buckets.length; i++) {
    const min = buckets[i] || 0;
    const max = buckets[i + 1] || Infinity;
    const label = max === Infinity ? `${min}ms+` : `${min}-${max}ms`;

    distribution[label] = times.filter(t => t >= min && t < max).length;
  }

  return distribution;
}
```

```javascript
        GROUP BY u.id
        LIMIT 100
      `
    },
    {
      name: 'Multiple JOIN with Aggregation',
      sql: `
        SELECT
          c.name as category,
          COUNT(o.id) as total_orders,
          SUM(o.total) as total_revenue,
          AVG(o.total) as avg_order_value
        FROM categories c
        LEFT JOIN products p ON c.id = p.category_id
        LEFT JOIN orders o ON p.id = o.product_id
        GROUP BY c.id
        ORDER BY total_revenue DESC
        LIMIT 50
      `
    }
  ];

  for (const test of joinTests) {
    results[test.name] = await analyzeQueryPerformance(env.DB, test.sql);
  }

  return Response.json(results);
}

async function analyzeQueryPerformance(db, sql, iterations = 10) {
  const results = {
    executionTimes: [],
    explainResults: null,
    rowCount: 0
  };

  // 获取查询计划
  const explain = await db.prepare(`EXPLAIN QUERY PLAN ${sql}`).all();
  results.explainResults = explain.results;

  // 执行多次查询
  for (let i = 0; i < iterations; i++) {
    const start = performance.now();
    const queryResult = await db.prepare(sql).all();
    const end = performance.now();

    results.executionTimes.push(end - start);
    results.rowCount = queryResult.results?.length || 0;
  }

  // 计算统计信息
  results.stats = {
    avg: (results.executionTimes.reduce((a, b) => a + b, 0) / results.executionTimes.length).toFixed(2),
    min: Math.min(...results.executionTimes).toFixed(2),
    max: Math.max(...results.executionTimes).toFixed(2),
    stdDev: calculateStandardDeviation(results.executionTimes).toFixed(2)
  };

  return results;
}

function calculateStandardDeviation(values) {
  const avg = values.reduce((a, b) => a + b, 0) / values.length;
  const squareDiffs = values.map(value => Math.pow(value - avg, 2));
  const avgSquareDiff = squareDiffs.reduce((a, b) => a + b, 0) / squareDiffs.length;
  return Math.sqrt(avgSquareDiff);
}
```

```javascript
async function compareIndexPerformance(db) {
  const queries = [
    {
      name: 'Age Range Query',
      withIndex: 'SELECT * FROM users WHERE age BETWEEN 20 AND 30',
      withoutIndex: 'SELECT * FROM users WHERE age BETWEEN 20 AND 30' // 相同查询，但比较有无索引的性能
    },
    {
      name: 'Price Range Query',
      withIndex: 'SELECT * FROM products WHERE price BETWEEN 50 AND 200',
      withoutIndex: 'SELECT * FROM products WHERE price BETWEEN 50 AND 200'
    },
    {
      name: 'Composite Query',
      withIndex: 'SELECT * FROM orders WHERE user_id = ? AND product_id = ?',
      withoutIndex: 'SELECT * FROM orders WHERE user_id = ? AND product_id = ?'
    }
  ];

  const results = {};

  for (const query of queries) {
    // 测试带索引的查询
    const withIndexResult = await measureQuery(db, query.withIndex,
      query.name === 'Composite Query' ? [1, 1] : []);

    // 模拟无索引情况（通过索引提示）
    const withoutIndexResult = await measureQuery(db, `SELECT * FROM ${extractTableName(query.withIndex)} WHERE 1=1 AND ${extractWhereClause(query.withIndex)}`,
      query.name === 'Composite Query' ? [1, 1] : []);

    results[query.name] = {
      withIndex: withIndexResult,
      withoutIndex: withoutIndexResult,
      improvement: calculateImprovement(withIndexResult.avg, withoutIndexResult.avg)
    };
  }

  return Response.json(results);
}

function extractTableName(sql) {
  const match = sql.match(/FROM (\w+)/);
  return match ? match[1] : '';
}

function extractWhereClause(sql) {
  const match = sql.match(/WHERE (.+)/);
  return match ? match[1] : '';
}

function calculateImprovement(withIndex, withoutIndex) {
  const improvement = ((withoutIndex - withIndex) / withoutIndex * 100);
  return improvement.toFixed(2) + '%';
}
```

### 1. 压力测试
```javascript
// functions/performance/stress-test.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { duration, rampUp, maxConcurrency } = await request.json();

  console.log(`开始压力测试: 持续${duration}秒, ${rampUp}秒内增至${maxConcurrency}并发`);

  const testConfig = {
    duration: duration * 1000, // 转换为毫秒
    rampUp: rampUp * 1000,
    maxConcurrency
  };

  const results = await runStressTest(env.DB, testConfig);

  return Response.json(results);
}

async function runStressTest(db, config) {
  const startTime = Date.now();
  const results = {
    metrics: {
      totalRequests: 0,
      successfulRequests: 0,
      failedRequests: 0,
      responseTimes: [],
      errors: []
    },
    timeline: []
  };

  // 渐进式增加并发
  let currentConcurrency = 1;
  const rampIncrement = Math.ceil(config.maxConcurrency / (config.rampUp / 1000));

  const testRunner = setInterval(async () => {
    const elapsed = Date.now() - startTime;

    if (elapsed >= config.duration) {
      clearInterval(testRunner);
      return;
    }

    // 增加并发数（在 rampUp 期间）
    if (currentConcurrency < config.maxConcurrency && elapsed < config.rampUp) {
      currentConcurrency = Math.min(currentConcurrency + rampIncrement, config.maxConcurrency);
    }

    // 执行当前批次的请求
    const batchPromises = [];
    for (let i = 0; i < currentConcurrency; i++) {
      batchPromises.push(executeRandomQuery(db));
    }

    const batchResults = await Promise.allSettled(batchResults);

    // 记录指标
    batchResults.forEach((result, index) => {
      if (result.status === 'fulfilled') {
        results.metrics.successfulRequests++;
        results.metrics.responseTimes.push(result.value);
      } else {
        results.metrics.failedRequests++;
        results.metrics.errors.push({
          error: result.reason.message,
          timestamp: Date.now()
        });
      }
      results.metrics.totalRequests++;
    });

    // 记录时间线数据
    results.timeline.push({
      timestamp: Date.now(),
      concurrency: currentConcurrency,
      elapsed: elapsed,
      avgResponseTime: results.metrics.responseTimes.length > 0
        ? results.metrics.responseTimes.slice(-currentConcurrency)
          .reduce((a, b) => a + b, 0) / currentConcurrency
        : 0
    });

  }, 1000); // 每秒执行一次

  return new Promise(resolve => {
    setTimeout(() => {
      // 计算最终统计
      results.summary = {
        duration: (Date.now() - startTime) / 1000,
        throughput: (results.metrics.successfulRequests / ((Date.now() - startTime) / 1000)).toFixed(2),
        errorRate: (results.metrics.failedRequests / results.metrics.totalRequests * 100).toFixed(2),
        avgResponseTime: results.metrics.responseTimes.length > 0
          ? (results.metrics.responseTimes.reduce((a, b) => a + b, 0) / results.metrics.responseTimes.length).toFixed(2)
          : 0,
        p95ResponseTime: percentile(results.metrics.responseTimes, 0.95).toFixed(2),
        p99ResponseTime: percentile(results.metrics.responseTimes, 0.99).toFixed(2)
      };

      resolve(results);
    }, config.duration + 1000);
  });
}

async function executeRandomQuery(db) {
  const queries = [
    { sql: 'SELECT COUNT(*) FROM users', params: [] },
    { sql: 'SELECT * FROM users LIMIT 10', params: [] },
    { sql: 'SELECT * FROM products WHERE price > ?', params: [50] },
    { sql: 'SELECT * FROM orders ORDER BY created_at DESC LIMIT 5', params: [] }
  ];

  const query = queries[Math.floor(Math.random() * queries.length)];
  const start = performance.now();

  await db.prepare(query.sql).bind(...query.params).all();

  return performance.now() - start;
}
```

```javascript
async function collectPerformanceMetrics(db, duration) {
  const startTime = Date.now();
  const metrics = {
    duration,
    queryTypes: {},
    responseTimeDistribution: {
      under10: 0,
      under50: 0,
      under100: 0,
      under500: 0,
      over500: 0
    },
    errors: [],
    throughput: 0
  };

  const monitoring = setInterval(async () => {
    // 执行监控查询
    const queries = [
      {
        type: 'simple_select',
        sql: 'SELECT COUNT(*) FROM users'
      },
      {
        type: 'indexed_select',
        sql: 'SELECT * FROM users WHERE email = ?',
        params: ['test@example.com']
      },
      {
        type: 'aggregate',
        sql: 'SELECT category, COUNT(*) FROM products GROUP BY category LIMIT 10'
      },
      {
        type: 'join',
        sql: 'SELECT u.name, COUNT(o.id) FROM users u LEFT JOIN orders o ON u.id = o.user_id GROUP BY u.id LIMIT 10'
      }
    ];

      try {
        const start = performance.now();
        const stmt = db.prepare(query.sql);

        if (query.params) {
          await stmt.bind(...query.params).all();
        } else {
          await stmt.all();

        // 记录查询类型

        metrics.queryTypes[query.type].totalTime += responseTime;
        metrics.queryTypes[query.type].avgTime =
          metrics.queryTypes[query.type].totalTime / metrics.queryTypes[query.type].count;

        // 记录响应时间分布
        if (responseTime < 10) metrics.responseTimeDistribution.under10++;
        else if (responseTime < 50) metrics.responseTimeDistribution.under50++;
        else if (responseTime < 100) metrics.responseTimeDistribution.under100++;
        else if (responseTime < 500) metrics.responseTimeDistribution.under500++;
        else metrics.responseTimeDistribution.over500++;

      } catch (error) {
        metrics.errors.push({
          query: query.type,
          error: error.message,
          timestamp: Date.now()
        });
      }
    }

    const elapsed = (Date.now() - startTime) / 1000;
    if (elapsed >= duration) {
      clearInterval(monitoring);

      // 计算吞吐量
      const totalQueries = Object.values(metrics.queryTypes).reduce((sum, type) => sum + type.count, 0);
      metrics.throughput = (totalQueries / duration).toFixed(2);
    }

  }, 1000); // 每秒执行一次

  return new Promise(resolve => {
    setTimeout(() => resolve(metrics), duration * 1000 + 100);
  });
}
```

```javascript
// functions/performance/report.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { testType, config } = await request.json();

  let testResults;

  switch (testType) {
    case 'baseline':
      testResults = await runBaselineTest(env.DB);
      break;
    case 'load':
      testResults = await runLoadTest(env.DB, config);
      break;
    case 'endurance':
      testResults = await runEnduranceTest(env.DB, config);
      break;
    default:
      return new Response('Invalid test type', { status: 400 });
  }

  // 生成性能报告
  const report = generatePerformanceReport(testResults, testType);

  // 存储报告
  await env.PERFORMANCE_REPORTS_KV.put(
    `report_${Date.now()}`,
    JSON.stringify(report),
    { expirationTtl: 7 * 24 * 3600 } // 7天过期
  );

  return Response.json(report);
}

function generatePerformanceReport(results, testType) {
  const report = {
    metadata: {
      testType,
      timestamp: new Date().toISOString(),
      version: '1.0.0'
    },
    summary: {
      status: results.summary?.status || 'completed',
      duration: results.summary?.duration || 0,
      totalQueries: results.summary?.totalQueries || 0,
      avgResponseTime: results.summary?.avgResponseTime || 0,
      throughput: results.summary?.throughput || 0,
      errorRate: results.summary?.errorRate || 0
    },
    performance: {
      responseTime: {
        min: results.performance?.minResponseTime || 0,
        max: results.performance?.maxResponseTime || 0,
        avg: results.performance?.avgResponseTime || 0,
        p95: results.performance?.p95ResponseTime || 0,
        p99: results.performance?.p99ResponseTime || 0
      },
      distribution: results.performance?.distribution || {},
      queryTypes: results.queryTypes || {}
    },
    recommendations: generateRecommendations(results),
    details: results
  };

  return report;
}

function generateRecommendations(results) {
  const recommendations = [];

  // 响应时间建议
  if (results.performance?.avgResponseTime > 100) {
    recommendations.push({
      priority: 'high',
      category: 'response_time',
      message: '平均响应时间超过100ms，建议优化查询或添加索引',
      details: {
        current: results.performance.avgResponseTime,
        target: '<100ms'
      }
    });
  }

  // 错误率建议
  if (results.summary?.errorRate > 5) {
    recommendations.push({
      priority: 'critical',
      category: 'error_rate',
      message: '错误率过高，需要立即处理',
      details: {
        current: results.summary.errorRate + '%',
        target: '<1%'
      }
    });
  }

  // 吞吐量建议
  if (results.summary?.throughput < 100) {
    recommendations.push({
      priority: 'medium',
      category: 'throughput',
      message: '吞吐量偏低，考虑优化数据库配置或查询',
      details: {
        current: results.summary.throughput + ' qps',
        target: '>100 qps'
      }
    });
  }

  return recommendations;
}
```
