# D1 备份恢复

## 备份概述

Cloudflare D1 提供自动备份和手动备份功能，确保数据安全和可恢复性。备份包括数据文件、索引、触发器等完整的数据库状态。

## 手动备份

### 1. 即时备份
```javascript
// functions/backup/create.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { type, description } = await request.json();

  try {
    // 创建备份任务
    const backupId = await createBackup(env, type, description);

    // 异步执行备份
    context.waitUntil(executeBackup(env, backupId));

    return Response.json({
      backupId,
      status: 'started',
      message: 'Backup started successfully'
    });

  } catch (error) {
    return Response.json({
      error: error.message,
      status: 'failed'
    }, { status: 500 });
  }
}

async function createBackup(env, type, description) {
  const backupId = crypto.randomUUID();

  // 记录备份任务
  await env.BACKUP_TASKS_KV.put(
    `backup_${backupId}`,
    JSON.stringify({
      id: backupId,
      type: type || 'manual',
      description: description || 'Manual backup',
      status: 'pending',
      createdAt: new Date().toISOString(),
      progress: 0
    })
  );

  return backupId;
}

async function executeBackup(env, backupId) {
  try {
    // 更新状态为进行中
    await updateBackupStatus(env, backupId, 'running', 0);

    // 导出所有表
    const tables = await getAllTables(env);
    const backupData = {
      metadata: {
        backupId,
        timestamp: new Date().toISOString(),
        version: '1.0',
        tables: tables.length
      },
      schema: {},
      data: {}
    };

    // 备份每个表
    for (let i = 0; i < tables.length; i++) {
      const table = tables[i];

      // 获取表结构
      const schema = await getTableSchema(env, table.name);
      backupData.schema[table.name] = schema;

      // 导出数据
      const data = await exportTableData(env, table.name);
      backupData.data[table.name] = data;

      // 更新进度
      const progress = Math.round(((i + 1) / tables.length) * 100);
      await updateBackupStatus(env, backupId, 'running', progress);

      // 避免超时
      if (i % 10 === 0) {
        await new Promise(resolve => setTimeout(resolve, 100));
      }
    }

    // 压缩备份数据
    const compressedData = await compressBackupData(backupData);

    // 存储备份
    await storeBackup(env, backupId, compressedData);

    // 更新状态为完成
    await updateBackupStatus(env, backupId, 'completed', 100);

  } catch (error) {
    await updateBackupStatus(env, backupId, 'failed', 0, error.message);
    throw error;
  }
}

async function updateBackupStatus(env, backupId, status, progress, error = null) {
  const key = `backup_${backupId}`;
  const existing = await env.BACKUP_TASKS_KV.get(key);
  const backup = JSON.parse(existing);

  backup.status = status;
  backup.progress = progress;
  backup.updatedAt = new Date().toISOString();

  if (error) {
    backup.error = error;
  }

  await env.BACKUP_TASKS_KV.put(key, JSON.stringify(backup));
}
```

// functions/backup/table.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { tables } = await request.json();

  if (!tables || tables.length === 0) {
    return new Response('No tables specified', { status: 400 });
  }

  const backupResults = [];

  for (const tableName of tables) {
    try {
      const result = await backupTable(env, tableName);
      backupResults.push({
        table: tableName,
        success: true,
        backupId: result.backupId,
        size: result.size,
        rowCount: result.rowCount
      });
    } catch (error) {
      backupResults.push({
        table: tableName,
        success: false,
        error: error.message
      });
    }
  }

  return Response.json({
    results: backupResults,
    timestamp: new Date().toISOString()
  });
}

async function backupTable(env, tableName) {
  // 验证表是否存在
  const tableInfo = await env.DB.prepare(`
    SELECT name FROM sqlite_master WHERE type='table' AND name=?
  `).bind(tableName).first();

  if (!tableInfo) {
    throw new Error(`Table ${tableName} not found`);
  }

  // 获取表结构
  const schema = await getTableSchema(env, tableName);

  // 导出数据
  const data = await exportTableData(env, tableName);

  // 创建备份包
  const backupData = {
    table: tableName,
    schema,
    data,
    metadata: {
      backupId: crypto.randomUUID(),
      timestamp: new Date().toISOString(),
      rowCount: data.length
    }
  };

  // 存储备份
  const backupId = await storeTableBackup(env, tableName, backupData);

  return {
    backupId,
    size: JSON.stringify(backupData).length,
    rowCount: data.length
  };
}

async function exportFullDatabase(env, format) {
  // 获取所有表
  const tables = await getAllTables(env);
  const exportData = {
    metadata: {
      exportedAt: new Date().toISOString(),
      version: '1.0',
      tableCount: tables.length
    },
    tables: {}
  };

  // 导出每个表
  for (const table of tables) {
    exportData.tables[table.name] = {
      schema: await getTableSchema(env, table.name),
      data: await exportTableData(env, table.name)
    };
  }

  // 根据格式转换数据
  switch (format) {
    case 'sql':
      return convertToSQL(exportData);
    case 'csv':
      return convertToCSV(exportData);
    default:
      return JSON.stringify(exportData, null, 2);
  }
}

async function getAllTables(env) {
  const result = await env.DB.prepare(`
    SELECT name FROM sqlite_master
    WHERE type='table' AND name NOT LIKE 'sqlite_%'
    ORDER BY name
  `).all();

  return result.results;
}

async function getTableSchema(env, tableName) {
  // 获取表结构
  const pragma = await env.DB.prepare(`
    PRAGMA table_info(${tableName})
  `).all();

  // 获取索引
  const indexes = await env.DB.prepare(`
    PRAGMA index_list(${tableName})
  `).all();

  return {
    columns: pragma.results,
    indexes: indexes.results
  };
}

async function exportTableData(env, tableName) {
  const result = await env.DB.prepare(`
    SELECT * FROM ${tableName}
  `).all();

  return result.results || [];
}

export async function onRequestPost(context) {

  const { table, filters, limit } = await request.json();

  try {
    const exportData = await exportWithFilters(env, table, filters, limit);

      exportedCount: exportData.length,
      data: exportData,

  } catch (error) {
    return Response.json({
      error: error.message
    }, { status: 500 });
  }
}

async function exportWithFilters(env, table, filters, limit = null) {
  // 构建查询
  let sql = `SELECT * FROM ${table}`;
  const params = [];

  if (filters && filters.length > 0) {
    const conditions = [];

    filters.forEach(filter => {
      switch (filter.operator) {
        case 'equals':
          conditions.push(`${filter.field} = ?`);
          params.push(filter.value);
          break;
        case 'contains':
          conditions.push(`${filter.field} LIKE ?`);
          params.push(`%${filter.value}%`);
          break;
        case 'greater':
          conditions.push(`${filter.field} > ?`);
          params.push(filter.value);
          break;
        case 'less':
          conditions.push(`${filter.field} < ?`);
          params.push(filter.value);
          break;
        case 'between':
          conditions.push(`${filter.field} BETWEEN ? AND ?`);
          params.push(filter.value.start, filter.value.end);
          break;
      }
    });

    if (conditions.length > 0) {
      sql += ' WHERE ' + conditions.join(' AND ');
    }
  }

  if (limit) {
    sql += ` LIMIT ${limit}`;
  }

  const result = await env.DB.prepare(sql).bind(...params).all();
  return result.results || [];
}

## 数据恢复

### 1. 从备份恢复
```javascript
// functions/restore/from-backup.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { backupId, options } = await request.json();

  try {
    // 验证备份
    const backup = await verifyBackup(env, backupId);

    if (!backup) {
      return new Response('Backup not found', { status: 404 });
    }

    // 开始恢复
    const restoreId = crypto.randomUUID();
    context.waitUntil(executeRestore(env, backupId, restoreId, options));

    return Response.json({
      restoreId,
      status: 'started',
      message: 'Restore process started'
    });

  } catch (error) {
    return Response.json({
      error: error.message
    }, { status: 500 });
  }
}

async function executeRestore(env, backupId, restoreId, options) {
  try {
    // 创建当前数据库备份（恢复前）
    if (options?.createPreRestoreBackup) {
      const preBackupId = await createPreRestoreBackup(env);
      console.log(`Pre-restore backup created: ${preBackupId}`);
    }

    // 获取备份数据
    const backupData = await loadBackupData(env, backupId);

    // 清空现有数据（如果要求）
    if (options?.clearExisting) {
      await clearDatabase(env);
    }

    // 恢复架构
    await restoreSchema(env, backupData.schema);

    // 恢复数据
    await restoreData(env, backupData.data, options);

    // 记录恢复完成
    await recordRestore(env, restoreId, 'completed');

  } catch (error) {
    await recordRestore(env, restoreId, 'failed', error.message);
    throw error;
  }
}

async function restoreSchema(env, schema) {
  for (const [tableName, tableSchema] of Object.entries(schema)) {
    // 删除现有表（如果存在）
    await env.DB.prepare(`DROP TABLE IF EXISTS ${tableName}`).run();

    // 重建表结构
    const createSQL = generateCreateTableSQL(tableName, tableSchema.columns);
    await env.DB.prepare(createSQL).run();

    // 创建索引
    for (const index of tableSchema.indexes) {
      if (!index.origin || index.origin === 'c') {
        const indexSQL = generateCreateIndexSQL(tableName, index);
        await env.DB.prepare(indexSQL).run();
      }
    }
  }
}

async function restoreData(env, data, options) {
  for (const [tableName, tableData] of Object.entries(data)) {
    if (tableData.length === 0) continue;

    // 批量插入数据
    const batchSize = options?.batchSize || 1000;

    for (let i = 0; i < tableData.length; i += batchSize) {
      const batch = tableData.slice(i, i + batchSize);

      // 构建批量插入语句
      const columns = Object.keys(batch[0]);
      const placeholders = columns.map(() => '?').join(', ');
      const values = columns.map(col => batch[0][col]);

      const insertSQL = `
        INSERT INTO ${tableName} (${columns.join(', ')})
        VALUES ${batch.map(() => `(${placeholders})`).join(', ')}
      `;

      const params = batch.flatMap(row => columns.map(col => row[col]));

      await env.DB.prepare(insertSQL).bind(...params).run();

      // 避免超时
      if (i % (batchSize * 10) === 0) {
        await new Promise(resolve => setTimeout(resolve, 100));
      }
    }
  }
}
```

// functions/restore/point-in-time.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { timestamp, tables } = await request.json();

  try {
    // 找到最接近的备份
    const backupId = await findClosestBackup(env, timestamp);

    if (!backupId) {
      return new Response('No suitable backup found', { status: 404 });
    }

    // 获取变更日志
    const changes = await getChangesSince(env, backupId, timestamp);

    // 执行恢复
    const result = await restoreToPointInTime(env, backupId, changes, tables);

    return Response.json({
      backupId,
      targetTime: timestamp,
      restoredTables: tables || 'all',
      changesApplied: changes.length,
      timestamp: new Date().toISOString()
    });

  } catch (error) {
    return Response.json({
      error: error.message
    }, { status: 500 });
  }
}

async function getChangesSince(env, backupId, timestamp) {
  // 从审计日志获取变更
  const changes = await env.DB.prepare(`
    SELECT * FROM audit_logs
    WHERE timestamp BETWEEN (
      SELECT created_at FROM backups WHERE id = ?
    ) AND ?
    ORDER BY timestamp ASC
  `).bind(backupId, timestamp).all();

  return changes.results || [];
}

async function restoreToPointInTime(env, backupId, changes, tables) {
  // 恢复到备份点
  await executeRestore(env, backupId, crypto.randomUUID(), { tables });

  // 应用变更到指定时间点
  for (const change of changes) {
    if (tables && !tables.includes(change.table_name)) {
      continue;
    }

    // 恢复单个变更
    await applyChange(env, change);
  }

  return { changesApplied: changes.length };
}

export async function onRequestPost(context) {
  const { request, env } = context;
  const { backupId, table, options } = await request.json();

  try {
    // 获取表备份
    const tableBackup = await getTableBackup(env, backupId, table);

    if (!tableBackup) {
      return new Response('Table backup not found', { status: 404 });
    }

    // 恢复表
    await restoreSingleTable(env, table, tableBackup, options);

    return Response.json({
      table,
      restoredAt: new Date().toISOString(),
      rowCount: tableBackup.data.length,
      options
    });

  } catch (error) {
    return Response.json({
      error: error.message
    }, { status: 500 });
  }
}

async function restoreSingleTable(env, tableName, backup, options) {
  // 创建表（如果不存在）
  if (options?.dropExisting) {
    await env.DB.prepare(`DROP TABLE IF EXISTS ${tableName}`).run();
  }

  // 恢复表结构
  const createSQL = generateCreateTableSQL(tableName, backup.schema.columns);
  await env.DB.prepare(createSQL).run();

  // 恢复数据
  if (backup.data && backup.data.length > 0) {
    const columns = Object.keys(backup.data[0]);
    const placeholders = columns.map(() => '?').join(', ');

    const insertSQL = `
      INSERT INTO ${tableName} (${columns.join(', ')})
      VALUES (${placeholders})
    `;

    const stmt = env.DB.prepare(insertSQL);

    for (const row of backup.data) {
      const values = columns.map(col => row[col]);
      await stmt.bind(...values).run();
    }
  }

  // 恢复索引
  for (const index of backup.schema.indexes) {
    if (!index.origin || index.origin === 'c') {
      const indexSQL = generateCreateIndexSQL(tableName, index);
      await env.DB.prepare(indexSQL).run();
    }
  }
}

### 1. 完整性检查
```javascript
// functions/backup/verify.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { backupId } = await request.json();

  try {
    const verification = await verifyBackupIntegrity(env, backupId);

    return Response.json({
      backupId,
      verification,
      timestamp: new Date().toISOString()
    });

  } catch (error) {
    return Response.json({
      error: error.message
    }, { status: 500 });
  }
}

async function verifyBackupIntegrity(env, backupId) {
  const results = {
    backupId,
    isValid: true,
    checks: {},
    issues: []
  };

  // 加载备份数据
  const backupData = await loadBackupData(env, backupId);

  if (!backupData) {
    results.isValid = false;
    results.issues.push('Backup data not found');
    return results;
  }

  // 检查元数据
  results.checks.metadata = verifyMetadata(backupData.metadata);

  // 检查架构
  results.checks.schema = verifySchema(backupData.schema);

  // 检查数据完整性
  results.checks.data = await verifyDataIntegrity(env, backupData);

  // 检查数据一致性
  results.checks.consistency = await verifyDataConsistency(env, backupData);

  // 汇总结果
  Object.values(results.checks).forEach(check => {
    if (!check.passed) {
      results.isValid = false;
      results.issues.push(...check.issues);
    }
  });

  return results;
}

async function verifyDataIntegrity(env, backupData) {
  const results = {
    passed: true,
    issues: []
  };

  // 检查每个表的行数
  for (const [tableName, tableData] of Object.entries(backupData.data)) {
    // 在备份中计数
    const backupCount = tableData.length;

    // 在原始表中计数（如果表仍然存在）
    try {
      const result = await env.DB.prepare(`
        SELECT COUNT(*) as count FROM ${tableName}
      `).first();

      if (result.count !== backupCount) {
        results.passed = false;
        results.issues.push(
          `Row count mismatch for ${tableName}: backup=${backupCount}, current=${result.count}`
        );
      }
    } catch (error) {
      // 表可能不存在，这可能是正常的
      console.log(`Table ${tableName} not found in current database`);
    }
  }

  return results;
}

async function verifyDataConsistency(env, backupData) {
  const results = {
    passed: true,
    issues: []
  };

  // 检查外键约束
  for (const [tableName, tableData] of Object.entries(backupData.data)) {
    for (const row of tableData) {
      for (const [column, value] of Object.entries(row)) {
        if (column.endsWith('_id') && value) {
          // 检查引用的表是否存在
          const refTable = column.replace('_id', '');

          if (backupData.data[refTable]) {
            const refExists = backupData.data[refTable].some(
              refRow => refRow.id === value
            );

            if (!refExists) {
              results.passed = false;
              results.issues.push(
                `Foreign key violation: ${tableName}.${column}=${value} references non-existent ${refTable}`
              );
            }
          }
        }
      }
    }
  }

  return results;
}

async function testRestore(env, backupId, testDbId, testTables) {
  const results = {
    success: true,
    tablesTested: testTables || 'all',
    details: {},
    issues: []
  };

  // 获取备份数据
  const backupData = await loadBackupData(env, backupId);

  // 测试每个表
  const tablesToTest = testTables || Object.keys(backupData.data);

  for (const tableName of tablesToTest) {
    if (!backupData.data[tableName]) continue;

    const tableResult = await testTableRestore(
      env,
      testDbId,
      tableName,
      backupData.data[tableName]
    );

    results.details[tableName] = tableResult;

    if (!tableResult.success) {
      results.success = false;
      results.issues.push(...tableResult.issues);
    }
  }

  return results;
}

async function testTableRestore(env, testDbId, tableName, tableData) {
  const result = {
    success: true,
    rowsTested: Math.min(100, tableData.length), // 测试前100行
    issues: []
  };

  // 在测试数据库中创建表并插入数据
  try {
    const testTableData = tableData.slice(0, result.rowsTested);

    // 验证数据类型
    for (const row of testTableData) {
      for (const [column, value] of Object.entries(row)) {
        if (value === null || value === undefined) continue;

        // 检查数据类型是否合理
        if (typeof value === 'object' && !Array.isArray(value)) {
          // 可能是序列化的 JSON
          try {
            JSON.parse(value);
          } catch {
            result.issues.push(
              `Invalid JSON in ${tableName}.${column}: ${value}`
            );
          }
        }
      }
    }

    if (result.issues.length > 0) {
      result.success = false;
    }

  } catch (error) {
    result.success = false;
    result.issues.push(`Test restore failed: ${error.message}`);
  }

  return result;
}

// functions/backup/cleanup.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { policy, dryRun = false } = await request.json();

  try {
    const cleanupResult = await executeBackupCleanup(env, policy, dryRun);

    return Response.json({
      dryRun,
      cleanupResult,
      timestamp: new Date().toISOString()
    });

  } catch (error) {
    return Response.json({
      error: error.message
    }, { status: 500 });
  }
}

async function executeBackupCleanup(env, policy, dryRun) {
  const result = {
    backupsToDelete: [],
    spaceFreed: 0,
    errors: []
  };

  // 获取所有备份
  const backups = await listAllBackups(env);

  // 应用清理策略
  const toDelete = selectBackupsForDeletion(backups, policy);

  if (dryRun) {
    result.backupsToDelete = toDelete.map(b => ({
      id: b.id,
      date: b.date,
      size: b.size
    }));

    result.spaceFreed = toDelete.reduce((sum, b) => sum + b.size, 0);

  } else {
    // 实际删除
    for (const backup of toDelete) {
      try {
        await deleteBackup(env, backup.id);
        result.spaceFreed += backup.size;
        result.backupsToDelete.push(backup.id);
      } catch (error) {
        result.errors.push({
          backupId: backup.id,
          error: error.message
        });
      }
    }
  }

  return result;
}
```
