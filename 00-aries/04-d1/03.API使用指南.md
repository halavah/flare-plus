# D1 API 使用指南

## API 概述

Cloudflare D1 提供完整的 REST API 接口，支持数据库管理、查询执行、备份恢复等功能。

## 数据库管理 API

### 1. 创建数据库
```javascript
// 创建新数据库
const createDatabase = async (name) => {
  const response = await fetch(
    `https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database`,
    {
      method: 'POST',
      headers,
      body: JSON.stringify({
        name: name,
        // 可选：初始位置
        location: {
          region: 'wnam' // 西北美
        }
      })
    }
  );

  const data = await response.json();
  return data.result; // 返回数据库信息
};

// 使用示例
const db = await createDatabase('my-app-db');
console.log('Database ID:', db.uuid);
console.log('Database name:', db.name);
```

## SQL 执行 API

### 1. 执行查询
```javascript
// 执行 SQL 查询
const executeQuery = async (databaseId, sql, params = []) => {
  const response = await fetch(
    `https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database/${databaseId}/query`,
    {
      method: 'POST',
      headers,
      body: JSON.stringify({
        sql: sql,
        params: params // 参数化查询
      })
    }
  );

  const data = await response.json();
  return data.result;
};

// 使用示例
const result = await executeQuery(
  'database-uuid',
  'SELECT * FROM users WHERE email = ?',
  ['user@example.com']
);

console.log('查询结果:', result[0].results);
```

### 2. 执行多个语句
```javascript
// 执行多个 SQL 语句
const executeMultiple = async (databaseId, statements) => {
  const response = await fetch(
    `https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database/${databaseId}/query`,
    {
      method: 'POST',
      headers,
      body: JSON.stringify({
        sql: statements.join('; ')
      })
    }
  );

  return response.json();
};

// 使用示例
const results = await executeMultiple(
  'database-uuid',
  [
    'INSERT INTO users (name, email) VALUES (?, ?)',
    'SELECT LAST_INSERT_ROWID() as id'
  ],
  [['John', 'john@example.com']]
);
```

### 3. 事务执行
```javascript
// 在 Workers 中使用事务
export async function onRequestPost(context) {
  const { request, env } = context;
  const { userId, items } = await request.json();

  try {
    // 开始事务
    const result = await env.DB.batch([
      env.DB.prepare('INSERT INTO orders (user_id, total) VALUES (?, ?)')
        .bind(userId, calculateTotal(items)),
      ...items.map(item =>
        env.DB.prepare('UPDATE inventory SET quantity = quantity - 1 WHERE product_id = ?')
          .bind(item.id)
      ),
      env.DB.prepare('SELECT LAST_INSERT_ROWID() as order_id')
    ]);

    return Response.json({
      success: true,
      orderId: result[result.length - 1].results[0].order_id
    });
  } catch (error) {
    return Response.json({
      success: false,
      error: error.message
    }, { status: 500 });
  }
}
```

## 数据操作示例

### 1. CRUD 操作
```javascript
// Create - 插入数据
const createUser = async (user) => {
  return await env.DB.prepare(`
    INSERT INTO users (name, email, created_at)
    VALUES (?, ?, CURRENT_TIMESTAMP)
  `).bind(user.name, user.email).run();
};

// Read - 查询数据
const getUserById = async (id) => {
  return await env.DB.prepare('SELECT * FROM users WHERE id = ?')
    .bind(id).first();
};

const getUsersByPage = async (page = 1, limit = 10) => {
  const offset = (page - 1) * limit;
  return await env.DB.prepare(`
    SELECT * FROM users
    ORDER BY created_at DESC
    LIMIT ? OFFSET ?
  `).bind(limit, offset).all();
};

// Update - 更新数据
const updateUser = async (id, updates) => {
  const fields = Object.keys(updates).map(key => `${key} = ?`).join(', ');
  const values = Object.values(updates);

  return await env.DB.prepare(`
    UPDATE users SET ${fields} WHERE id = ?
  `).bind(...values, id).run();
};

// Delete - 删除数据
const deleteUser = async (id) => {
  return await env.DB.prepare('DELETE FROM users WHERE id = ?')
    .bind(id).run();
};
```

// 聚合查询
const getUserStats = async () => {

const getUserPosts = async (userId) => {

      p.*,
      u.name as author_name
    FROM posts p
    JOIN users u ON p.user_id = u.id
    WHERE p.user_id = ?
    ORDER BY p.created_at DESC

// 子查询
const getPopularPosts = async () => {
  return await env.DB.prepare(`
    SELECT p.*, c.comment_count
    FROM posts p
    LEFT JOIN (
      SELECT post_id, COUNT(*) as comment_count
      FROM comments
      GROUP BY post_id
    ) c ON p.id = c.post_id
    WHERE p.created_at > datetime('now', '-7 days')
    ORDER BY c.comment_count DESC
    LIMIT 10
  `).all();
};

// 窗口函数
const getPostRankings = async () => {

    FROM posts p

// 存储 JSON 数据
const saveProduct = async (product) => {

    INSERT INTO products (name, price, metadata)

    product.name,

// 查询 JSON 数据
const searchProducts = async (criteria) => {
  let sql = 'SELECT * FROM products WHERE 1=1';
  const params = [];

  if (criteria.brand) {
    sql += ' AND json_extract(metadata, "$.brand") = ?';
    params.push(criteria.brand);
  }

  if (criteria.minPrice) {
    sql += ' AND price >= ?';
    params.push(criteria.minPrice);
  }

  if (criteria.maxPrice) {
    sql += ' AND price <= ?';
    params.push(criteria.maxPrice);
  }

  return await env.DB.prepare(sql).bind(...params).all();
};

// JSON 聚合
const getBrandStats = async () => {
  return await env.DB.prepare(`
    SELECT
      json_extract(metadata, '$.brand') as brand,
      COUNT(*) as product_count,
      AVG(price) as avg_price,
      MIN(price) as min_price,
      MAX(price) as max_price
    FROM products
    GROUP BY json_extract(metadata, '$.brand')
    ORDER BY product_count DESC

### 1. 导出数据
```javascript
// 导出整个数据库
const exportDatabase = async (databaseId) => {
  const tables = await getTables(databaseId);
  const exportData = {};

  for (const table of tables) {
    const data = await exportTable(databaseId, table.name);

  return result[0].results;
};

const backup = await exportDatabase('database-uuid');

```

### 2. 导入数据
```javascript
// 导入数据
const importDatabase = async (databaseId, importData) => {
  for (const [tableName, tableData] of Object.entries(importData)) {
    // 创建表
    await executeQuery(databaseId, tableData.schema);

    // 导入数据
    if (tableData.data && tableData.data.length > 0) {
      const columns = Object.keys(tableData.data[0]);
      const placeholders = columns.map(() => '?').join(', ');
      const sql = `INSERT INTO ${tableName} (${columns.join(', ')}) VALUES (${placeholders})`;

      for (const row of tableData.data) {
        const values = columns.map(col => row[col]);
        await executeQuery(databaseId, sql, values);
      }
    }
  }
};
```

### 3. 增量备份
```javascript
// 增量备份
const incrementalBackup = async (databaseId, lastBackupTime) => {
  const tables = await getTables(databaseId);
  const changes = {};

  for (const table of tables) {
    // 查找自上次备份以来的更改
    const result = await executeQuery(
      databaseId,
      `SELECT * FROM ${table.name}
       WHERE created_at > ? OR updated_at > ?`,
      [lastBackupTime, lastBackupTime]
    );

    if (result[0].results.length > 0) {
      changes[table.name] = result[0].results;
    }
  }

  return changes;
};
```

## 性能优化 API

### 1. 查询分析
```javascript
// 分析查询性能
const analyzeQuery = async (databaseId, sql) => {
  // 获取查询计划
  const explainResult = await executeQuery(
    databaseId,
    `EXPLAIN QUERY PLAN ${sql}`
  );

  // 执行查询并记录性能
  const startTime = Date.now();
  const result = await executeQuery(databaseId, sql);
  const endTime = Date.now();

  return {
    query: sql,
    executionTime: endTime - startTime,
    plan: explainResult[0].results,
    resultCount: result[0].results?.length || 0
  };
};

// 使用示例
const analysis = await analyzeQuery(
  'database-uuid',
  'SELECT * FROM users WHERE email = ?'
);

console.log('查询时间:', analysis.executionTime, 'ms');
console.log('查询计划:', analysis.plan);
```

### 2. 索引建议
```javascript
// 分析索引使用
const getIndexUsage = async (databaseId) => {
  const response = await fetch(
    `https://api.cloudflare.com/client/v4/accounts/${accountId}/d1/database/${databaseId}/metrics`,
    {
      method: 'GET',
      headers
    }
  );

  const metrics = await response.json();
  return {
    slowQueries: metrics.slow_queries,
    indexUsage: metrics.index_usage,
    recommendations: generateIndexRecommendations(metrics)
  };
};

const generateIndexRecommendations = (metrics) => {
  const recommendations = [];

  // 分析慢查询
  metrics.slow_queries.forEach(query => {
    if (query.execution_time > 1000) { // 超过1秒
      recommendations.push({
        type: 'index',
        query: query.sql,
        suggestion: 'Consider adding an index on ' + extractQueryColumns(query.sql)
      });
    }
  });

  return recommendations;
};
```

const handleD1Error = (error) => {
  const errorMap = {
    10006: '数据库不存在',
    10007: 'SQL 语法错误',
    10008: '参数错误',
    10009: '超出限额',
    10010: '数据库锁定'

  const errorCode = error.code || 0;
  const errorMessage = errorMap[errorCode] || error.message || '未知错误';

  return {
    success: false,
    error: {
      code: errorCode,
      message: errorMessage
    }
  };
};

// 使用示例

  return { success: true, data: result };
} catch (error) {
  return handleD1Error(error);
}

## 使用最佳实践

### 1. 连接池管理
```javascript
// Workers 中的连接管理
class D1Manager {
  constructor(db) {
    this.db = db;
    this.queryCache = new Map();
  }

  async query(sql, params = []) {
    const cacheKey = `${sql}:${JSON.stringify(params)}`;

    // 检查缓存
    if (this.queryCache.has(cacheKey)) {
      return this.queryCache.get(cacheKey);
    }

    // 执行查询
    const result = await this.db.prepare(sql).bind(...params).all();

    // 缓存结果
    if (sql.startsWith('SELECT')) {
      this.queryCache.set(cacheKey, result);
      setTimeout(() => this.queryCache.delete(cacheKey), 300000); // 5分钟
    }

    return result;
  }
}
```

### 2. 批量操作优化
```javascript
// 批量插入优化
const batchInsert = async (tableName, records) => {
  const batchSize = 100; // D1 限制
  const batches = [];

  for (let i = 0; i < records.length; i += batchSize) {
    batches.push(records.slice(i, i + batchSize));
  }

  for (const batch of batches) {
    const values = batch.map(() => '(?)').join(',');
    const sql = `INSERT INTO ${tableName} VALUES ${values}`;

    await env.DB.prepare(sql).bind(...batch).run();
  }
};
```
