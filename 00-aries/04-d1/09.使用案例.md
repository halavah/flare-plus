# D1 使用案例

## 实际应用场景

D1 作为 Cloudflare 的分布式 SQLite 数据库，在各种应用场景中都能发挥作用。以下详细介绍了 D1 的典型使用案例和实现方案。

## 1. 内容管理系统 (CMS)

### 场景描述
构建一个轻量级 CMS，支持文章管理、用户评论、标签系统等功能。

### 数据模型设计

### 实现：文章管理 API
```javascript
// functions/api/posts.js
export async function onRequestGet(context) {
  const { request, env } = context;
  const url = new URL(request.url);

  const page = parseInt(url.searchParams.get('page')) || 1;
  const limit = parseInt(url.searchParams.get('limit')) || 10;
  const status = url.searchParams.get('status') || 'published';
  const category = url.searchParams.get('category');
  const tag = url.searchParams.get('tag');
  const search = url.searchParams.get('search');

  const offset = (page - 1) * limit;

  // 构建查询
  let sql = `
    SELECT
      p.*,
      u.username as author_name,
      u.avatar_url as author_avatar,
      GROUP_CONCAT(c.name, ',') as categories,
      GROUP_CONCAT(t.name, ',') as tags,
      COUNT(co.id) as comment_count
    FROM posts p
    JOIN users u ON p.author_id = u.id
    LEFT JOIN post_categories pc ON p.id = pc.post_id
    LEFT JOIN categories c ON pc.category_id = c.id
    LEFT JOIN post_tags pt ON p.id = pt.post_id
    LEFT JOIN tags t ON pt.tag_id = t.id
    LEFT JOIN comments co ON p.id = co.post_id AND co.status = 'approved'
    WHERE p.status = ?
  `;

  const params = [status];

  if (category) {
    sql += ` AND EXISTS (
      SELECT 1 FROM post_categories pc2
      WHERE pc2.post_id = p.id AND pc2.category_id = ?
    )`;
    params.push(category);
  }

  if (tag) {
    sql += ` AND EXISTS (
      SELECT 1 FROM post_tags pt2
      WHERE pt2.post_id = p.id AND pt2.tag_id = ?
    )`;
    params.push(tag);
  }

  if (search) {
    sql += ` AND (p.title LIKE ? OR p.content LIKE ? OR p.excerpt LIKE ?)`;
    const searchTerm = `%${search}%`;
    params.push(searchTerm, searchTerm, searchTerm);
  }

  sql += `
    GROUP BY p.id
    ORDER BY p.published_at DESC, p.created_at DESC
    LIMIT ? OFFSET ?
  `;

  params.push(limit, offset);

  const posts = await env.DB.prepare(sql).bind(...params).all();

  // 获取总数
  let countSql = `
    SELECT COUNT(DISTINCT p.id) as total
    FROM posts p
    WHERE p.status = ?
  `;

  const countParams = [status];

  if (category) {
    countSql += ` AND EXISTS (
      SELECT 1 FROM post_categories pc2
      WHERE pc2.post_id = p.id AND pc2.category_id = ?
    )`;
    countParams.push(category);
  }

  const total = await env.DB.prepare(countSql).bind(...countParams).first();

  return Response.json({
    posts: posts.results.map(post => ({
      ...post,
      categories: post.categories ? post.categories.split(',') : [],
      tags: post.tags ? post.tags.split(',') : []
    })),
    pagination: {
      page,
      limit,
      total: total.total,
      totalPages: Math.ceil(total.total / limit)
    }
  });
}

export async function onRequestPost(context) {
  const { request, env } = context;
  const postData = await request.json();
  const user = await authenticateUser(request, env);

  // 创建文章
  const result = await env.DB.prepare(`
    INSERT INTO posts (title, slug, content, excerpt, author_id, status, published_at)
    VALUES (?, ?, ?, ?, ?, ?, ?)
  `).bind(
    postData.title,
    generateSlug(postData.title),
    postData.content,
    postData.excerpt || null,
    user.id,
    postData.status || 'draft',
    postData.status === 'published' ? new Date().toISOString() : null
  ).run();

  const postId = result.meta.last_row_id;

  // 处理分类和标签
  if (postData.categories && postData.categories.length > 0) {
    await env.DB.batch(
      postData.categories.map(catId =>
        env.DB.prepare(`
          INSERT INTO post_categories (post_id, category_id)
          VALUES (?, ?)
        `).bind(postId, catId)
      )
    );
  }

  if (postData.tags && postData.tags.length > 0) {
    await env.DB.batch(
      postData.tags.map(tagId =>
        env.DB.prepare(`
          INSERT INTO post_tags (post_id, tag_id)
          VALUES (?, ?)
        `).bind(postId, tagId)
      )
    );
  }

  return Response.json({
    success: true,
    postId
  });
}
```

### 实现：订单处理流程
```javascript
// functions/api/orders.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { cartToken, shippingAddress, billingAddress, paymentMethod } = await request.json();

  // 获取购物车
  const cart = await env.DB.prepare(`
    SELECT c.*, ci.product_id, ci.variant_id, ci.quantity,
           p.name as product_name, p.price as product_price,
           pv.name as variant_name, pv.price as variant_price
    FROM carts c
    JOIN cart_items ci ON c.id = ci.cart_id
    LEFT JOIN products p ON ci.product_id = p.id
    LEFT JOIN product_variants pv ON ci.variant_id = pv.id
    WHERE c.token = ? AND c.expires_at > datetime('now')
  `).bind(cartToken).all();

  if (!cart.results || cart.results.length === 0) {
    return new Response('Cart not found or expired', { status: 404 });
  }

  try {
    // 开始事务
    const results = await env.DB.batch([
      // 创建订单
      env.DB.prepare(`
        INSERT INTO orders (order_number, email, financial_status, fulfillment_status,
                           subtotal_price, total_price, shipping_address, billing_address)
        VALUES (?, ?, 'pending', 'unfulfilled', ?, ?, ?, ?)
      `).bind(
        generateOrderNumber(),
        cart.results[0].email,
        calculateSubtotal(cart.results),
        calculateTotal(cart.results),
        JSON.stringify(shippingAddress),
        JSON.stringify(billingAddress)
      ),

      // 创建订单项并更新库存
      ...cart.results.map(item =>
        env.DB.prepare(`
          INSERT INTO order_items (order_id, product_id, variant_id, name, sku,
                                  quantity, price, total)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        `).bind(
          /* order_id will be set after first insert */,
          item.product_id,
          item.variant_id,
          item.product_name + (item.variant_name ? ` - ${item.variant_name}` : ''),
          item.sku,
          item.quantity,
          item.variant_price || item.product_price,
          (item.variant_price || item.product_price) * item.quantity
        )
      ),

      // 更新库存
      ...cart.results.map(item => {
        if (item.variant_id) {
          return env.DB.prepare(`
            UPDATE product_variants
            SET inventory_quantity = inventory_quantity - ?
            WHERE id = ? AND inventory_quantity >= ?
          `).bind(item.quantity, item.variant_id, item.quantity);
        } else {
          return env.DB.prepare(`
            UPDATE products
            SET inventory_quantity = inventory_quantity - ?
            WHERE id = ? AND inventory_quantity >= ?
          `).bind(item.quantity, item.product_id, item.quantity);
        }
      }),

      // 清空购物车
      env.DB.prepare(`
        DELETE FROM cart_items WHERE cart_id = ?
      `).bind(cart.results[0].id),

      env.DB.prepare(`
        DELETE FROM carts WHERE id = ?
      `).bind(cart.results[0].id)
    ]);

    // 更新订单项的 order_id
    const orderId = results[0].meta.last_row_id;
    const itemResults = results.slice(1, cart.results.length + 1);

    await env.DB.batch(
      itemResults.map((result, index) =>
        env.DB.prepare(`
          UPDATE order_items SET order_id = ? WHERE id = ?
        `).bind(orderId, result.meta.last_row_id)
      )
    );

    // 处理支付
    const paymentResult = await processPayment(
      paymentMethod,
      calculateTotal(cart.results),
      orderId
    );

    if (paymentResult.success) {
      // 更新订单状态为已支付
      await env.DB.prepare(`
        UPDATE orders SET financial_status = 'paid', updated_at = CURRENT_TIMESTAMP
        WHERE id = ?
      `).bind(orderId).run();

      // 发送确认邮件
      context.waitUntil(sendOrderConfirmation(orderId));
    }

    return Response.json({
      success: true,
      orderId,
      orderNumber: results[0].result[0].order_number,
      paymentStatus: paymentResult.success ? 'paid' : 'pending'
    });

  } catch (error) {
    console.error('Order creation failed:', error);
    return Response.json({
      error: 'Failed to create order'
    }, { status: 500 });
  }
}

// 辅助函数
function generateOrderNumber() {
  const timestamp = Date.now();
  const random = Math.floor(Math.random() * 1000);
  return `ORD${timestamp}${random}`;
}

function calculateSubtotal(cartItems) {
  return cartItems.reduce((total, item) => {
    const price = item.variant_price || item.product_price;
    return total + (price * item.quantity);
  }, 0);
}

function calculateTotal(cartItems, shippingCost = 0, taxRate = 0.08) {
  const subtotal = calculateSubtotal(cartItems);
  const tax = subtotal * taxRate;
  return subtotal + shippingCost + tax;
}

async function processPayment(method, amount, orderId) {
  // 实现支付处理逻辑
  // 这里可以集成 Stripe、PayPal 等支付服务
  return {
    success: true,
    transactionId: `txn_${Date.now()}`,
    amount
  };
}
```

### 实现：实时仪表板
```javascript
// functions/api/analytics.js
export async function onRequestGet(context) {
  const { request, env } = context;
  const url = new URL(request.url);
  const metric = url.searchParams.get('metric');
  const dateRange = url.searchParams.get('range') || '7d'; // 1d, 7d, 30d

  switch (metric) {
    case 'pageviews':
      return await getPageViews(env, dateRange);
    case 'events':
      return await getEvents(env, dateRange);
    case 'funnels':
      return await getFunnelAnalysis(env, dateRange);
    case 'realtime':
      return await getRealtimeStats(env);
    default:
      return await getDashboardOverview(env, dateRange);
  }
}

async function getPageViews(env, range) {
  const { startDate, endDate, groupBy } = getDateRange(range);

  let sql = `
    SELECT
      DATE(created_at) as date,
      COUNT(*) as views,
      COUNT(DISTINCT session_id) as sessions,
      COUNT(DISTINCT user_id) as unique_users,
      AVG(duration) as avg_duration
    FROM page_views
    WHERE created_at BETWEEN ? AND ?
    GROUP BY DATE(created_at)
    ORDER BY date DESC
  `;

  const result = await env.DB.prepare(sql).bind(startDate, endDate).all();

  // 获取热门页面
  const topPages = await env.DB.prepare(`
    SELECT
      url,
      COUNT(*) as views,
      COUNT(DISTINCT session_id) as unique_sessions,
      AVG(duration) as avg_duration
    FROM page_views
    WHERE created_at BETWEEN ? AND ?
    GROUP BY url
    ORDER BY views DESC
    LIMIT 10
  `).bind(startDate, endDate).all();

  return Response.json({
    timeline: result.results,
    topPages: topPages.results,
    range
  });
}

async function getEvents(env, range) {
  const { startDate, endDate } = getDateRange(range);

  // 事件类型分布
  const eventTypes = await env.DB.prepare(`
    SELECT
      event_type,
      event_name,
      COUNT(*) as count,
      COUNT(DISTINCT user_id) as unique_users
    FROM events
    WHERE created_at BETWEEN ? AND ?
    GROUP BY event_type, event_name
    ORDER BY count DESC
    LIMIT 20
  `).bind(startDate, endDate).all();

  // 事件趋势
  const eventTrends = await env.DB.prepare(`
    SELECT
      DATE(created_at) as date,
      event_type,
      COUNT(*) as count
    FROM events
    WHERE created_at BETWEEN ? AND ?
    GROUP BY DATE(created_at), event_type
    ORDER BY date DESC, count DESC
  `).bind(startDate, endDate).all();

  return Response.json({
    eventTypes: eventTypes.results,
    trends: eventTrends.results,
    range
  });
}

async function getRealtimeStats(env) {
  // 当前在线用户
  const onlineUsers = await env.DB.prepare(`
    SELECT COUNT(DISTINCT session_id) as count
    FROM page_views
    WHERE created_at > datetime('now', '-5 minutes')
  `).first();

  // 今日页面访问
  const todayViews = await env.DB.prepare(`
    SELECT COUNT(*) as count
    FROM page_views
    WHERE DATE(created_at) = DATE('now')
  `).first();

  // 今日新增用户
  const todayUsers = await env.DB.prepare(`
    SELECT COUNT(*) as count
    FROM users
    WHERE DATE(created_at) = DATE('now')
  `).first();

  // 实时事件率
  const eventRate = await env.DB.prepare(`
    SELECT COUNT(*) as count
    FROM events
    WHERE created_at > datetime('now', '-1 minute')
  `).first();

  return Response.json({
    onlineUsers: onlineUsers.count,
    todayViews: todayViews.count,
    todayUsers: todayUsers.count,
    eventsPerMinute: eventRate.count,
    timestamp: new Date().toISOString()
  });
}

// 事件收集端点
export async function onRequestPost(context) {
  const { request, env } = context;
  const { events } = await request.json();

  // 批量插入事件
  const insertPromises = events.map(event => {
    if (event.type === 'pageview') {
      return env.DB.prepare(`
        INSERT INTO page_views (url, referrer, user_agent, ip_address,
                                session_id, user_id, duration, country, city,
                                device_type, browser)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
      `).bind(
        event.url,
        event.referrer || null,
        event.userAgent || null,
        request.headers.get('CF-Connecting-IP'),
        event.sessionId || null,
        event.userId || null,
        event.duration || null,
        event.country || null,
        event.city || null,
        event.deviceType || null,
        event.browser || null
      ).run();
    } else {
      return env.DB.prepare(`
        INSERT INTO events (event_type, event_name, properties, user_id,
                           session_id, url, ip_address)
        VALUES (?, ?, ?, ?, ?, ?, ?)
      `).bind(
        event.type,
        event.name,
        JSON.stringify(event.properties || {}),
        event.userId || null,
        event.sessionId || null,
        event.url || null,
        request.headers.get('CF-Connecting-IP')
      ).run();
    }
  });

  const results = await Promise.all(insertPromises);

  // 更新实时统计
  context.waitUntil(updateRealtimeStats(env, events));

  return Response.json({
    success: true,
    inserted: results.length
  });
```

```javascript
async function processJobs(env, queue = 'default', batchSize = 10) {
  // 获取待处理的任务
  const jobs = await env.DB.prepare(`
    SELECT * FROM job_queue
    WHERE queue_name = ?
      AND status = 'pending'
      AND (delay_until IS NULL OR delay_until <= datetime('now'))
    ORDER BY created_at ASC
    LIMIT ?
  `).bind(queue, batchSize).all();

  if (!jobs.results.length) {
    return Response.json({
      processed: 0,
      message: 'No jobs to process'
    });
  }

  // 更新状态为处理中
  await env.DB.batch(
    jobs.results.map(job =>
      env.DB.prepare(`
        UPDATE job_queue
        SET status = 'processing', updated_at = CURRENT_TIMESTAMP
        WHERE job_id = ?
      `).bind(job.job_id)
    )
  );

  // 处理每个任务
  const results = [];
  for (const job of jobs.results) {
    const result = await processSingleJob(env, job);
    results.push(result);
  }

  return Response.json({
    processed: results.length,
    results
  });
}

async function processSingleJob(env, job) {
  const startTime = Date.now();

  try {
    // 根据任务类型处理
    let result;
    switch (job.job_type) {
      case 'send_email':
        result = await handleSendEmail(job.payload);
        break;
      case 'process_image':
        result = await handleProcessImage(job.payload);
        break;
      case 'generate_report':
        result = await handleGenerateReport(job.payload);
        break;
      case 'webhook':
        result = await handleWebhook(job.payload);
        break;
      default:
        throw new Error(`Unknown job type: ${job.job_type}`);
    }

    const duration = Date.now() - startTime;

    // 标记为完成
    await env.DB.batch([
      env.DB.prepare(`
        UPDATE job_queue
        SET status = 'completed', completed_at = CURRENT_TIMESTAMP, updated_at = CURRENT_TIMESTAMP
        WHERE job_id = ?
      `).bind(job.job_id),

      env.DB.prepare(`
        INSERT INTO job_logs (job_id, queue_name, job_type, status, duration, result)
        VALUES (?, ?, ?, 'completed', ?, ?)
      `).bind(job.job_id, job.queue_name, job.job_type, duration, JSON.stringify(result))
    ]);

    return {
      jobId: job.job_id,
      status: 'completed',
      duration,
      result
    };

  } catch (error) {
    const duration = Date.now() - startTime;
    const newAttempts = job.attempts + 1;

    // 检查是否需要重试
    if (newAttempts < job.max_attempts) {
      // 指数退避重试
      const delayMs = Math.pow(2, newAttempts) * 1000; // 1s, 2s, 4s...
      const retryUntil = new Date(Date.now() + delayMs).toISOString();

      await env.DB.batch([
        env.DB.prepare(`
          UPDATE job_queue
          SET status = 'pending', attempts = ?, delay_until = ?, updated_at = CURRENT_TIMESTAMP
          WHERE job_id = ?
        `).bind(newAttempts, retryUntil, job.job_id),

        env.DB.prepare(`
          INSERT INTO job_logs (job_id, queue_name, job_type, status, duration, error)
          VALUES (?, ?, ?, 'retry', ?, ?)
        `).bind(job.job_id, job.queue_name, job.job_type, duration, error.message)
      ]);

      return {
        jobId: job.job_id,
        status: 'retry',
        attempts: newAttempts,
        delay: delayMs
      };
    } else {
      // 标记为失败
      await env.DB.batch([
        env.DB.prepare(`
          UPDATE job_queue
          SET status = 'failed', error_message = ?, error_stack = ?, updated_at = CURRENT_TIMESTAMP
          WHERE job_id = ?
        `).bind(error.message, error.stack, job.job_id),

        env.DB.prepare(`
          INSERT INTO job_logs (job_id, queue_name, job_type, status, duration, error)
          VALUES (?, ?, ?, 'failed', ?, ?)
        `).bind(job.job_id, job.queue_name, job.job_type, duration, error.message)
      ]);

      return {
        jobId: job.job_id,
        status: 'failed',
        attempts: newAttempts,
        error: error.message
      };
    }
  }
}

async function handleWebhook(payload) {
  const { url, method, headers, body, retries = 3 } = payload;

  let attempt = 0;
  while (attempt < retries) {
    try {
      const response = await fetch(url, {
        method,
        headers: new Headers(headers),
        body: body ? JSON.stringify(body) : undefined
      });

      if (!response.ok) {
        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
      }

      const responseData = await response.text();
      return {
        success: true,
        statusCode: response.status,
        response: responseData
      };

    } catch (error) {
      attempt++;
      if (attempt >= retries) {
        throw error;
      }
      await new Promise(resolve => setTimeout(resolve, 1000 * attempt));
    }
  }
```

```javascript
// functions/api/config.js
class ConfigManager {
  constructor(env) {
    this.env = env;
    this.cache = new Map();
    this.cacheTimeout = 5 * 60 * 1000; // 5分钟缓存
  }

  async get(key, environment = 'production', useCache = true) {
    const cacheKey = `${key}:${environment}`;

    // 检查缓存
    if (useCache && this.cache.has(cacheKey)) {
      const cached = this.cache.get(cacheKey);
      if (Date.now() - cached.timestamp < this.cacheTimeout) {
        return this.parseValue(cached.value, cached.type);
      }
    }

    // 从数据库获取
    const config = await this.env.DB.prepare(`
      SELECT value, value_type, is_encrypted
      FROM configurations
      WHERE key = ? AND environment = ? AND is_active = 1
    `).bind(key, environment).first();

    if (!config) {
      throw new Error(`Configuration '${key}' not found for environment '${environment}'`);
    }

    let value = config.value;

    // 解密
    if (config.is_encrypted) {
      value = await this.decrypt(value);
    }

    // 解析值类型
    const parsedValue = this.parseValue(value, config.value_type);

    // 缓存
    this.cache.set(cacheKey, {
      value: config.is_encrypted ? value : parsedValue,
      type: config.value_type,
      timestamp: Date.now()
    });

    return parsedValue;
  }

  async set(key, value, environment = 'production', options = {}) {
    const {
      description = null,
      isEncrypted = false,
      groupIds = [],
      changedBy = null,
      changeReason = null
    } = options;

    // 获取当前配置
    const current = await this.env.DB.prepare(`
      SELECT id, value FROM configurations
      WHERE key = ? AND environment = ?
    `).bind(key, environment).first();

    const valueType = typeof value;
    let dbValue = value;

    // 序列化
    if (valueType === 'object') {
      dbValue = JSON.stringify(value);
    }

    // 加密
    if (isEncrypted) {
      dbValue = await this.encrypt(dbValue);
    }

    let configId;
    let isNew = false;

    if (current) {
      // 更新现有配置
      await this.env.DB.prepare(`
        UPDATE configurations
        SET value = ?, value_type = ?, updated_at = CURRENT_TIMESTAMP, version = version + 1
        WHERE id = ?
      `).bind(dbValue, valueType, current.id).run();

      configId = current.id;
    } else {
      // 创建新配置
      const result = await this.env.DB.prepare(`
        INSERT INTO configurations (key, value, value_type, environment, description, is_encrypted)
        VALUES (?, ?, ?, ?, ?, ?)
      `).bind(key, dbValue, valueType, environment, description, isEncrypted ? 1 : 0).run();

      configId = result.meta.last_row_id;
      isNew = true;
    }

    // 记录历史
    if (!isNew) {
      await this.env.DB.prepare(`
        INSERT INTO configuration_history (config_id, key, old_value, new_value, changed_by, change_reason)
        VALUES (?, ?, ?, ?, ?, ?)
      `).bind(
        configId,
        key,
        current.value,
        dbValue,
        changedBy,
        changeReason
      ).run();
    }

    // 添加到组
    if (groupIds.length > 0) {
      await this.env.DB.batch(
        groupIds.map(groupId =>
          this.env.DB.prepare(`
            INSERT OR IGNORE INTO configuration_members (config_id, group_id)
            VALUES (?, ?)
          `).bind(configId, groupId)
        )
      );
    }

    // 清除缓存
    this.cache.delete(`${key}:${environment}`);

    return {
      success: true,
      configId,
      isNew
    };
  }

  async getGroup(groupName, environment = 'production') {
    const configs = await this.env.DB.prepare(`
      SELECT c.key, c.value, c.value_type, c.is_encrypted
      FROM configurations c
      JOIN configuration_members cm ON c.id = cm.config_id
      JOIN configuration_groups cg ON cm.group_id = cg.id
      WHERE cg.name = ? AND c.environment = ? AND c.is_active = 1
    `).bind(groupName, environment).all();

    const result = {};
    for (const config of configs.results) {
      let value = config.value;

      if (config.is_encrypted) {
        value = await this.decrypt(value);
      }

      result[config.key] = this.parseValue(value, config.value_type);
    }

    return result;
  }

  async getAll(environment = 'production') {
    const configs = await this.env.DB.prepare(`
      SELECT key, value, value_type, is_encrypted
      FROM configurations
      WHERE environment = ? AND is_active = 1
    `).bind(environment).all();

    const result = {};
    for (const config of configs.results) {
      let value = config.value;

      if (config.is_encrypted) {
        value = await this.decrypt(value);
      }

      result[config.key] = this.parseValue(value, config.value_type);
    }

    return result;
  }

  parseValue(value, type) {
    switch (type) {
      case 'number':
        return Number(value);
      case 'boolean':
        return value === 'true';
      case 'json':
        return JSON.parse(value);
      default:
        return value;
    }
  }

  async encrypt(value) {
    // 实现加密逻辑
    // 这里是示例
    return Buffer.from(value).toString('base64');
  }

  async decrypt(value) {
    // 实现解密逻辑
    // 这里是示例
    return Buffer.from(value, 'base64').toString();
  }

  clearCache(key = null, environment = null) {
    if (key && environment) {
      this.cache.delete(`${key}:${environment}`);
    } else {
      this.cache.clear();
    }
  }
}

// API 端点
export async function onRequestGet(context) {
  const { request, env } = context;
  const url = new URL(request.url);
  const configManager = new ConfigManager(env);

  const key = url.searchParams.get('key');
  const group = url.searchParams.get('group');
  const environment = url.searchParams.get('env') || 'production';

  try {
    if (key) {
      const value = await configManager.get(key, environment);
      return Response.json({ key, value, environment });
    } else if (group) {
      const configs = await configManager.getGroup(group, environment);
      return Response.json({ group, configs, environment });
    } else {
      const allConfigs = await configManager.getAll(environment);
      return Response.json({ configs: allConfigs, environment });
    }
  } catch (error) {
    return Response.json({
      error: error.message
    }, { status: 404 });
  }
}

export async function onRequestPost(context) {
  const { request, env } = context;
  const configManager = new ConfigManager(env);
  const { action, data } = await request.json();

  switch (action) {
    case 'set':
      const setResult = await configManager.set(
        data.key,
        data.value,
        data.environment || 'production',
        data.options
      );
      return Response.json(setResult);

    case 'bulk':
      const bulkResults = await Promise.all(
        data.configs.map(config =>
          configManager.set(
            config.key,
            config.value,
            config.environment || 'production',
            config.options
          )
        )
      );
      return Response.json({
        success: true,
        results: bulkResults
      });

    default:
      return new Response('Invalid action', { status: 400 });
  }
}
```
