# D1 数据建模

## 数据建模概述

D1 数据建模是设计和组织数据库结构的过程，旨在高效存储、检索和管理数据，同时保持数据完整性和性能。

## 数据库设计原则

### 1. 规范化设计
```sql
-- 第一范式 (1NF)
-- 原子性：每个字段都是不可分割的
CREATE TABLE users (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,           -- 避免存储 "张三, 李四" 这样的复合值
  email TEXT UNIQUE NOT NULL,   -- 避免存储多个邮箱在一个字段
  phone TEXT,                   -- 单个电话号码
  created_at DATETIME
);

-- 第二范式 (2NF)
-- 完全依赖：非主键字段完全依赖于主键
CREATE TABLE orders (
  id INTEGER PRIMARY KEY,
  user_id INTEGER NOT NULL,     -- 外键，依赖主键
  order_date DATETIME NOT NULL,
  total DECIMAL(10,2),
  status TEXT NOT NULL,
  FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 第三范式 (3NF)
-- 传递依赖：消除传递依赖
CREATE TABLE categories (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  description TEXT
);

CREATE TABLE products (
  id INTEGER PRIMARY KEY,
  category_id INTEGER NOT NULL, -- 而不是直接存储 category_name
  name TEXT NOT NULL,
  price DECIMAL(10,2),
  FOREIGN KEY (category_id) REFERENCES categories(id)
);
```

### 2. 反规范化考虑
```sql
-- 有时为了性能需要适度反规范化
CREATE TABLE orders (
  id INTEGER PRIMARY KEY,
  user_id INTEGER NOT NULL,
  user_name TEXT,               -- 冗余用户名，避免频繁 JOIN
  order_date DATETIME NOT NULL,
  total DECIMAL(10,2),
  status TEXT NOT NULL
);

-- 或使用触发器维护冗余数据
CREATE TRIGGER update_order_user_name
AFTER UPDATE ON users
BEGIN
  UPDATE orders
  SET user_name = NEW.name
  WHERE user_id = NEW.id;
END;
```

### 3. 数据类型选择
```sql
-- 选择合适的数据类型
CREATE TABLE products (
  id INTEGER PRIMARY KEY,                    -- 自增主键
  name TEXT NOT NULL,                       -- 变长文本
  description TEXT,                         -- 长文本
  price DECIMAL(10,2),                     -- 精确货币
  weight REAL,                             -- 浮点数
  in_stock BOOLEAN DEFAULT 1,              -- 布尔值 (0/1)
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  metadata JSON,                           -- 灵活的元数据
  tags TEXT                                -- 逗号分隔的标签
);

-- 使用 CHECK 约束
CREATE TABLE employees (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  age INTEGER CHECK(age >= 18 AND age <= 65),
  email TEXT UNIQUE NOT NULL,
  salary DECIMAL(10,2) CHECK(salary > 0),
  department TEXT CHECK(department IN ('IT', 'Sales', 'HR', 'Marketing'))
);
```

## 表关系设计

### 1. 一对一关系
```sql
-- 用户表
CREATE TABLE users (
  id INTEGER PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  email TEXT UNIQUE NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 用户详细信息表 (一对一)
CREATE TABLE user_profiles (
  id INTEGER PRIMARY KEY,
  user_id INTEGER UNIQUE NOT NULL,
  first_name TEXT,
  last_name TEXT,
  bio TEXT,
  avatar_url TEXT,
  birth_date DATE,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

-- 也可以在 users 表中添加 profile_id
ALTER TABLE users ADD COLUMN profile_id INTEGER UNIQUE;
ALTER TABLE users ADD FOREIGN KEY (profile_id) REFERENCES user_profiles(id);
```

### 2. 一对多关系
```sql
-- 作者表 (一)
CREATE TABLE authors (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT UNIQUE NOT NULL,
  bio TEXT
);

-- 文章表 (多)
CREATE TABLE articles (
  id INTEGER PRIMARY KEY,
  title TEXT NOT NULL,
  content TEXT,
  author_id INTEGER NOT NULL,
  published_at DATETIME,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (author_id) REFERENCES authors(id) ON DELETE CASCADE,
  INDEX idx_author_id (author_id)
);

-- 评论表 (多对多：文章-评论)
CREATE TABLE comments (
  id INTEGER PRIMARY KEY,
  article_id INTEGER NOT NULL,
  user_id INTEGER NOT NULL,
  content TEXT NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (article_id) REFERENCES articles(id) ON DELETE CASCADE,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  INDEX idx_article_id (article_id)
);
```

### 3. 多对多关系
```sql
-- 学生表
CREATE TABLE students (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT UNIQUE NOT NULL,
  enrollment_date DATE
);

-- 课程表
CREATE TABLE courses (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  description TEXT,
  credits INTEGER,
  instructor TEXT
);

-- 学生-课程关联表 (多对多)
CREATE TABLE student_courses (
  student_id INTEGER NOT NULL,
  course_id INTEGER NOT NULL,
  enrollment_date DATE DEFAULT CURRENT_DATE,
  grade TEXT, -- 'A', 'B', 'C', 'D', 'F'
  PRIMARY KEY (student_id, course_id),
  FOREIGN KEY (student_id) REFERENCES students(id) ON DELETE CASCADE,
  FOREIGN KEY (course_id) REFERENCES courses(id) ON DELETE CASCADE,
  INDEX idx_student_id (student_id),
  INDEX idx_course_id (course_id)
);

-- 标签表
CREATE TABLE tags (
  id INTEGER PRIMARY KEY,
  name TEXT UNIQUE NOT NULL,
  color TEXT -- 十六进制颜色代码
);

-- 文章-标签关联表
CREATE TABLE article_tags (
  article_id INTEGER NOT NULL,
  tag_id INTEGER NOT NULL,
  PRIMARY KEY (article_id, tag_id),
  FOREIGN KEY (article_id) REFERENCES articles(id) ON DELETE CASCADE,
  FOREIGN KEY (tag_id) REFERENCES tags(id) ON DELETE CASCADE
);
```

### 4. 层次结构数据
```sql
-- 分类表 (自引用)
CREATE TABLE categories (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  parent_id INTEGER,
  level INTEGER DEFAULT 0,
  sort_order INTEGER DEFAULT 0,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (parent_id) REFERENCES categories(id) ON DELETE CASCADE,
  INDEX idx_parent_id (parent_id)
);

-- 递归查询示例
WITH RECURSIVE category_tree AS (
  SELECT id, name, parent_id, level, name as path
  FROM categories
  WHERE parent_id IS NULL  -- 根分类

  UNION ALL

  SELECT c.id, c.name, c.parent_id, c.level,
         ct.path || ' > ' || c.name as path
  FROM categories c
  JOIN category_tree ct ON c.parent_id = ct.id
)
SELECT * FROM category_tree
ORDER BY level, sort_order;

-- 评论表 (嵌套集模型)
CREATE TABLE comments (
  id INTEGER PRIMARY KEY,
  post_id INTEGER NOT NULL,
  user_id INTEGER NOT NULL,
  content TEXT NOT NULL,
  lft INTEGER NOT NULL,
  rgt INTEGER NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (post_id) REFERENCES posts(id) ON DELETE CASCADE,
  FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 获取某评论的所有子评论
SELECT c.*
FROM comments c
JOIN comments parent ON c.lft > parent.lft AND c.rgt < parent.rgt
WHERE parent.id = ?
ORDER BY c.lft;
```

## 索引策略

### 1. 主键索引
```sql
-- 自增主键
CREATE TABLE orders (
  id INTEGER PRIMARY KEY AUTOINCREMENT, -- 自动创建索引
  order_number TEXT UNIQUE NOT NULL,   -- 业务主键
  customer_id INTEGER NOT NULL,
  total DECIMAL(10,2),
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 复合主键
CREATE TABLE order_items (
  order_id INTEGER NOT NULL,
  product_id INTEGER NOT NULL,
  quantity INTEGER NOT NULL,
  unit_price DECIMAL(10,2),
  PRIMARY KEY (order_id, product_id),   -- 复合主键自动创建索引
  FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
  FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE
);
```

### 2. 唯一索引
```sql
-- 用户邮箱唯一
CREATE UNIQUE INDEX idx_users_email ON users(email);

-- 用户名唯一
CREATE UNIQUE INDEX idx_users_username ON users(username);

-- 订单号唯一
CREATE UNIQUE INDEX idx_orders_number ON orders(order_number);

-- 复合唯一索引 (用户只能有一个活跃订阅)
CREATE UNIQUE INDEX idx_user_active_subscription
ON subscriptions(user_id)
WHERE status = 'active'; -- SQLite 不支持部分索引，使用触发器实现
```

### 3. 复合索引
```sql
-- 用户活跃时间查询
CREATE INDEX idx_users_status_created ON users(status, created_at);

-- 订单查询优化
CREATE INDEX idx_orders_customer_date ON orders(customer_id, created_at DESC);

-- 商品分类和价格查询
CREATE INDEX idx_products_category_price ON products(category_id, price);

-- 文章查询优化
CREATE INDEX idx_articles_status_published ON articles(status, published_at DESC);

-- 用户会话查询
CREATE INDEX idx_sessions_user_expires ON sessions(user_id, expires_at);
```

### 4. 函数索引
```sql
-- 邮箱大小写不敏感搜索
CREATE INDEX idx_users_lower_email ON users(lower(email));

-- 全文搜索
CREATE VIRTUAL TABLE articles_fts USING fts5(
  title,
  content,
  content='articles',
  content_rowid='id'
);

-- 模糊搜索优化
CREATE INDEX idx_products_name ON products(name); -- 支持 LIKE 前缀匹配
```

## 数据完整性约束

### 1. 外键约束
```sql
-- 级联删除
CREATE TABLE posts (
  id INTEGER PRIMARY KEY,
  author_id INTEGER NOT NULL,
  title TEXT NOT NULL,
  content TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (author_id) REFERENCES authors(id) ON DELETE CASCADE
);

-- 限制删除
CREATE TABLE orders (
  id INTEGER PRIMARY KEY,
  customer_id INTEGER NOT NULL,
  total DECIMAL(10,2),
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (customer_id) REFERENCES customers(id) ON DELETE RESTRICT
);

-- 设置 NULL
CREATE TABLE employees (
  id INTEGER PRIMARY KEY,
  department_id INTEGER,
  name TEXT NOT NULL,
  FOREIGN KEY (department_id) REFERENCES departments(id) ON DELETE SET NULL
);
```

### 2. CHECK 约束
```sql
-- 价格检查
CREATE TABLE products (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  price DECIMAL(10,2) CHECK(price > 0),
  stock INTEGER CHECK(stock >= 0),
  discount_rate REAL CHECK(discount_rate >= 0 AND discount_rate <= 1)
);

-- 日期检查
CREATE TABLE events (
  id INTEGER PRIMARY KEY,
  title TEXT NOT NULL,
  start_date DATE,
  end_date DATE,
  CHECK(end_date >= start_date)
);

-- 复合检查
CREATE TABLE bookings (
  id INTEGER PRIMARY KEY,
  room_id INTEGER NOT NULL,
  guest_id INTEGER NOT NULL,
  check_in_date DATE,
  check_out_date DATE,
  total_guests INTEGER,
  CHECK(check_out_date > check_in_date),
  CHECK(total_guests > 0 AND total_guests <= 10)
);
```

### 3. 非空约束
```sql
-- 必填字段
CREATE TABLE users (
  id INTEGER PRIMARY KEY,
  username TEXT NOT NULL,
  email TEXT NOT NULL,
  password_hash TEXT NOT NULL,
  phone TEXT, -- 可选
  birth_date DATE -- 可选
);

-- 修改表添加非空约束
ALTER TABLE users ADD COLUMN email_verified BOOLEAN DEFAULT 0 NOT NULL;
```

### 4. 默认值
```sql
-- 时间戳默认值
CREATE TABLE posts (
  id INTEGER PRIMARY KEY,
  title TEXT NOT NULL,
  content TEXT,
  status TEXT DEFAULT 'draft',
  view_count INTEGER DEFAULT 0,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 更新时间戳自动更新 (通过触发器)
CREATE TRIGGER update_posts_timestamp
AFTER UPDATE ON posts
BEGIN
  UPDATE posts SET updated_at = CURRENT_TIMESTAMP WHERE id = NEW.id;
END;

-- 枚举类型默认值
CREATE TABLE users (

  status TEXT DEFAULT 'active' CHECK(status IN ('active', 'inactive', 'banned'))

```

## 高级数据模型

### 1. 软删除模型
```sql
-- 软删除基础表
CREATE TABLE base_table (
  id INTEGER PRIMARY KEY,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  deleted_at DATETIME -- NULL 表示未删除
);

-- 使用示例
CREATE TABLE products (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  price DECIMAL(10,2),
  stock INTEGER DEFAULT 0,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  deleted_at DATETIME
);

-- 创建包含软删除的索引
CREATE INDEX idx_products_active ON products(deleted_at) WHERE deleted_at IS NULL;

-- 查询活跃产品
SELECT * FROM products WHERE deleted_at IS NULL;

-- 软删除
UPDATE products SET deleted_at = CURRENT_TIMESTAMP WHERE id = ?;

-- 恢复
UPDATE products SET deleted_at = NULL WHERE id = ?;
```

### 2. 审计日志模型
```sql
-- 审计日志表
CREATE TABLE audit_logs (
  id INTEGER PRIMARY KEY,
  table_name TEXT NOT NULL,
  record_id INTEGER NOT NULL,
  action TEXT NOT NULL CHECK(action IN ('INSERT', 'UPDATE', 'DELETE')),
  old_data JSON,
  new_data JSON,
  changed_by TEXT NOT NULL,
  changed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  ip_address TEXT,
  user_agent TEXT
);

-- 创建触发器自动记录变更
CREATE TRIGGER audit_users_insert
AFTER INSERT ON users
BEGIN
  INSERT INTO audit_logs (table_name, record_id, action, new_data, changed_by)
  VALUES ('users', NEW.id, 'INSERT', json_object(NEW), 'system');
END;

CREATE TRIGGER audit_users_update
AFTER UPDATE ON users
BEGIN
  INSERT INTO audit_logs (table_name, record_id, action, old_data, new_data, changed_by)
  VALUES ('users', NEW.id, 'UPDATE', json_object(OLD), json_object(NEW), 'system');
END;

CREATE TRIGGER audit_users_delete
AFTER DELETE ON users
BEGIN
  INSERT INTO audit_logs (table_name, record_id, action, old_data, changed_by)
  VALUES ('users', OLD.id, 'DELETE', json_object(OLD), 'system');
END;
```

### 3. 版本控制模型
```sql
-- 版本控制基础表
CREATE TABLE document_versions (
  id INTEGER PRIMARY KEY,
  document_id INTEGER NOT NULL,
  version INTEGER NOT NULL,
  title TEXT,
  content TEXT,
  created_by INTEGER NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(document_id, version),
  FOREIGN KEY (document_id) REFERENCES documents(id) ON DELETE CASCADE,
  FOREIGN KEY (created_by) REFERENCES users(id)
);

-- 主文档表
CREATE TABLE documents (
  id INTEGER PRIMARY KEY,
  title TEXT NOT NULL,
  current_version INTEGER DEFAULT 1,
  created_by INTEGER NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (created_by) REFERENCES users(id)
);

-- 创建新版本
INSERT INTO document_versions (document_id, version, title, content, created_by)
SELECT id, current_version + 1, title, content, ?
FROM documents
WHERE id = ?;

UPDATE documents
SET current_version = current_version + 1, updated_at = CURRENT_TIMESTAMP
WHERE id = ?;
```

## 数据迁移策略

### 1. 版本化迁移
```sql
-- 迁移版本表
CREATE TABLE schema_migrations (
  version TEXT PRIMARY KEY,
  applied_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 迁移文件命名: 001_create_users.sql
-- 迁移文件内容
CREATE TABLE IF NOT EXISTS users (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT UNIQUE NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 标记迁移
INSERT OR IGNORE INTO schema_migrations (version) VALUES ('001_create_users');
```

### 2. 增量迁移
```sql
-- 添加新列 (可空)
ALTER TABLE users ADD COLUMN phone TEXT;

-- 数据迁移
UPDATE users SET phone = NULL WHERE phone IS NULL;

-- 添加约束 (可选)
-- ALTER TABLE users ALTER COLUMN phone SET NOT NULL; -- SQLite 不支持
```

### 3. 大表迁移
```sql
-- 1. 创建新表
CREATE TABLE users_new (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT UNIQUE NOT NULL,
  phone TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 2. 逐步迁移数据
INSERT INTO users_new SELECT * FROM users LIMIT 1000;

-- 3. 重命名表
ALTER TABLE users RENAME TO users_old;
ALTER TABLE users_new RENAME TO users;
```

## 性能优化建议

### 1. 查询优化
```sql
-- 避免使用 SELECT *
-- ❌ 不推荐
SELECT * FROM users WHERE id = ?;

-- ✅ 推荐
SELECT id, name, email FROM users WHERE id = ?;

-- 使用 LIMIT 限制结果
SELECT * FROM posts ORDER BY created_at DESC LIMIT 10;

-- 避免在 WHERE 子句中使用函数
-- ❌ 不推荐
SELECT * FROM users WHERE YEAR(created_at) = 2023;

-- ✅ 推荐
SELECT * FROM users
WHERE created_at >= '2023-01-01' AND created_at < '2024-01-01';
```

### 2. 索引优化
```sql
-- 为常用查询创建索引
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_posts_status_date ON posts(status, published_at DESC);

-- 避免过度索引
-- 只为真正需要的查询创建索引

-- 定期分析索引使用情况
EXPLAIN QUERY PLAN SELECT * FROM users WHERE email = ?;
```

### 3. 数据归档
```sql
-- 创建归档表
CREATE TABLE orders_archive (
  LIKE orders INCLUDING ALL
);

-- 归档旧数据
INSERT INTO orders_archive
SELECT * FROM orders
WHERE created_at < date('now', '-1 year');

-- 删除已归档数据
DELETE FROM orders
WHERE created_at < date('now', '-1 year');
```

## 最佳实践

### 1. 命名规范
```sql
-- 表名: 小写，复数

-- 字段名: 小写，下划线分隔
first_name, created_at, updated_at

idx_users_email, idx_orders_customer_date
```
