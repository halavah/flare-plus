# D1 全文搜索

## FTS 概述

Cloudflare D1 支持 SQLite 的 FTS5 扩展，提供强大的全文搜索功能。FTS5 为文本数据创建专门的索引，实现快速、高效的全文搜索查询。

## FTS5 基础

### 1. 创建 FTS 表
```sql
-- 创建虚拟 FTS 表
CREATE VIRTUAL TABLE articles_fts USING fts5(
  title,              -- 文章标题
  content,            -- 文章内容
  author,             -- 作者
  tags,               -- 标签
  content='articles', -- 关联到源表
  content_rowid='id'  -- 行ID映射
);

-- 创建文章表（源表）
CREATE TABLE articles (
  id INTEGER PRIMARY KEY,
  title TEXT NOT NULL,
  content TEXT,
  author TEXT,
  tags TEXT,          -- JSON数组格式
  published_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  status TEXT DEFAULT 'draft'
);

-- 创建触发器自动同步
CREATE TRIGGER articles_ai AFTER INSERT ON articles BEGIN
  INSERT INTO articles_fts(rowid, title, content, author, tags)
  VALUES (new.id, new.title, new.content, new.author, new.tags);
END;

CREATE TRIGGER articles_ad AFTER DELETE ON articles BEGIN
  INSERT INTO articles_fts(articles_fts, rowid, title, content, author, tags)
  VALUES('delete', old.id, old.title, old.content, old.author, old.tags);
END;

CREATE TRIGGER articles_au AFTER UPDATE ON articles BEGIN
  INSERT INTO articles_fts(articles_fts, rowid, title, content, author, tags)
  VALUES('delete', old.id, old.title, old.content, old.author, old.tags);
  INSERT INTO articles_fts(rowid, title, content, author, tags)
  VALUES (new.id, new.title, new.content, new.author, new.tags);
END;
```

```javascript
// 生成包含高亮的片段
export async function searchWithHighlight(env, query, options = {}) {
  const {
    snippetLength = 64,
    startTag = '<mark>',
    endTag = '</mark>',
    ellipsis = '...'
  } = options;

  const result = await env.DB.prepare(`
    SELECT
      a.id,
      a.title,
      a.author,
      a.published_at,
      snippet(articles_fts, 0, ?, ?, ?, ?) as title_snippet,
      snippet(articles_fts, 1, ?, ?, ?, ?) as content_snippet,
      afts.rank
    FROM articles_fts afts
    JOIN articles a ON afts.rowid = a.id
    WHERE articles_fts MATCH ?
    ORDER BY afts.rank
    LIMIT 20
  `).bind(
    startTag, endTag, ellipsis, snippetLength,  // title snippet
    startTag, endTag, ellipsis, snippetLength,  // content snippet
    query
  ).all();

  return result.results.map(item => ({
    ...item,
    snippet: item.content_snippet,
    title_highlighted: item.title_snippet
  }));
}

// 使用示例
```

### 2. 上下文片段
```javascript
// 获取匹配词的上下文
export async function searchWithContext(env, query, contextSize = 3) {
  const result = await env.DB.prepare(`
    SELECT
      a.id,
      a.title,
      a.content,
      snippet(articles_fts, 1, '<mark>', '</mark>', '...', 64) as snippet,
      afts.rank
    FROM articles_fts afts
    JOIN articles a ON afts.rowid = a.id
    WHERE articles_fts MATCH ?
    ORDER BY afts.rank
  `).bind(query).all();

  return result.results.map(item => {
    // 提取完整句子作为上下文
    const sentences = item.content.split(/[.!?]+/);
    const highlightPos = item.content.toLowerCase().indexOf(query.toLowerCase());

    let contextStart = 0;
    for (let i = 0; i < sentences.length; i++) {
      const sentenceEnd = item.content.indexOf(sentences[i], contextStart) + sentences[i].length;
      if (highlightPos < sentenceEnd) {
        contextStart = Math.max(0, i - contextSize);
        contextEnd = Math.min(sentences.length, i + contextSize + 1);
        break;
      }
      contextStart = sentenceEnd;
    }

    return {
      ...item,
      context: sentences.slice(contextStart, contextEnd).join('. '),
      snippet: item.snippet
    };
  });
}
```

```javascript
// 创建搜索建议表
export async function createSuggestionsTable(env) {
  await env.DB.prepare(`
    CREATE VIRTUAL TABLE search_suggestions USING fts5(
      term,
      frequency,
      tokenize='porter'
    )
  `).run();

// 更新搜索频率
export async function updateSearchFrequency(env, term) {
  // 检查术语是否已存在
  const existing = await env.DB.prepare(`
    SELECT rowid, frequency FROM search_suggestions WHERE term MATCH ?
  `).bind(`"${term}"`).first();

  if (existing) {
    // 更新频率
    await env.DB.prepare(`
      UPDATE search_suggestions SET frequency = frequency + 1 WHERE rowid = ?
    `).bind(existing.rowid).run();
  } else {
    // 插入新术语
    await env.DB.prepare(`
      INSERT INTO search_suggestions (term, frequency) VALUES (?, 1)
    `).bind(term).run();
  }
}

// 获取搜索建议
export async function getSearchSuggestions(env, prefix, limit = 10) {
  const query = `${prefix}*`;

  const result = await env.DB.prepare(`
    SELECT term, frequency
    FROM search_suggestions
    WHERE search_suggestions MATCH ?
    ORDER BY frequency DESC, term
    LIMIT ?
  `).bind(query, limit).all();

  return result.results;
}

// 使用示例
const suggestions = await getSearchSuggestions(env, 'java');
```

### 2. 自动完成功能
```javascript
// 实现自动完成
export async function autoComplete(env, partial, options = {}) {
  const {
    minLength = 2,
    maxResults = 10,
    includeFrequency = true
  } = options;

  if (partial.length < minLength) {
    return [];
  }

  let sql = `
    SELECT DISTINCT term
    FROM search_suggestions
    WHERE term LIKE ?
    ORDER BY term
    LIMIT ?
  `;

  let params = [`${partial}%`, maxResults];

  if (includeFrequency) {
    sql = `
      SELECT term, frequency
      FROM search_suggestions
      WHERE term LIKE ?
      ORDER BY frequency DESC, term
      LIMIT ?
    `;
  }

  const result = await env.DB.prepare(sql).bind(...params).all();
  return result.results;
}
```

```javascript
// 热门搜索查询
export async function getPopularSearches(env, days = 7, limit = 20) {
  const result = await env.DB.prepare(`
    SELECT
      query,
      COUNT(*) as search_count,
      AVG(result_count) as avg_results,
      COUNT(DISTINCT user_id) as unique_users
    FROM search_logs
    WHERE timestamp > datetime('now', '-${days} days')
    GROUP BY query
    HAVING search_count > 1
    ORDER BY search_count DESC
    LIMIT ?
  `).bind(limit).all();

  return result.results;
}

// 搜索趋势分析
export async function getSearchTrends(env, days = 30) {
  const result = await env.DB.prepare(`
    SELECT
      DATE(timestamp) as date,
      COUNT(*) as total_searches,
      COUNT(DISTINCT query) as unique_queries,
      COUNT(DISTINCT user_id) as unique_users
    FROM search_logs
    WHERE timestamp > datetime('now', '-${days} days')
    GROUP BY DATE(timestamp)
    ORDER BY date DESC
  `).all();

  return result.results;
}

// 零结果查询分析
export async function getNoResultQueries(env, days = 7, limit = 50) {
  const result = await env.DB.prepare(`
    SELECT
      query,
      COUNT(*) as count,
      GROUP_CONCAT(DISTINCT user_id) as users
    FROM search_logs
    WHERE result_count = 0
      AND timestamp > datetime('now', '-${days} days')
    GROUP BY query
    HAVING count > 1
    ORDER BY count DESC
    LIMIT ?
  `).bind(limit).all();

  return result.results.map(item => ({
    ...item,
    users: item.users ? item.users.split(',').length : 0
  }));
}
```

```javascript
export async function checkFTSHealth(env, tableName) {
```

```javascript
// 分页搜索优化
export async function paginatedSearch(env, query, page = 1, pageSize = 20) {
  const offset = (page - 1) * pageSize;

  // 使用 LIMIT 和 OFFSET
  const result = await env.DB.prepare(`
    SELECT
      a.*,
      afts.rank
    FROM articles_fts afts
    JOIN articles a ON afts.rowid = a.id
    WHERE articles_fts MATCH ?
    ORDER BY afts.rank, a.id
    LIMIT ? OFFSET ?
  `).bind(query, pageSize, offset).all();

  // 获取总数（用于分页信息）
  const count = await env.DB.prepare(`
    SELECT COUNT(*) as total
    FROM articles_fts
    WHERE articles_fts MATCH ?
  `).bind(query).first();

  return {
    results: result.results,
    pagination: {
      page,
      pageSize,
      total: count.total,
      totalPages: Math.ceil(count.total / pageSize)
    }
  };
}

// 使用游标分页（更高效）
export async function cursorSearch(env, query, cursor = null, limit = 20) {
  let sql = `
    SELECT
      a.*,
      afts.rank,
      a.id as cursor
    FROM articles_fts afts
    JOIN articles a ON afts.rowid = a.id
    WHERE articles_fts MATCH ?
  `;

  const params = [query];

  if (cursor) {
    sql += ` AND a.id > ?`;
    params.push(cursor);
  }

  sql += ` ORDER BY afts.rank, a.id LIMIT ?`;
  params.push(limit);

  const result = await env.DB.prepare(sql).bind(...params).all();

  return {
    results: result.results,
    nextCursor: result.results.length > 0
      ? result.results[result.results.length - 1].cursor
      : null
  };
}
```

### 2. 语言检测和路由
```javascript
// 简单的语言检测
function detectLanguage(text) {
  const chineseRegex = /[\u4e00-\u9fff]/;
  const englishRegex = /^[a-zA-Z\s\d\W]+$/;

  if (chineseRegex.test(text)) {
    return 'zh';
  } else if (englishRegex.test(text)) {
    return 'en';
  }

  return 'en'; // 默认英语
}

// 多语言搜索
export async function multiLanguageSearch(env, query, language = null) {
  // 自动检测查询语言
  const detectedLang = language || detectLanguage(query);

  let ftsTable;
  switch (detectedLang) {
    case 'zh':
      ftsTable = 'articles_fts_zh';
      break;
    case 'en':
    default:
      ftsTable = 'articles_fts_en';
      break;
  }

  const sql = `
    SELECT a.*, afts.rank
    FROM ${ftsTable} afts
    JOIN articles a ON afts.rowid = a.id
    WHERE afts MATCH ? AND a.language = ?
    ORDER BY afts.rank
    LIMIT 20
  `;

  const result = await env.DB.prepare(sql).bind(query, detectedLang).all();
  return result.results;
}
```

```javascript
// functions/api/search.js
export async function onRequestGet(context) {
  const { request, env } = context;
  const url = new URL(request.url);

  const query = url.searchParams.get('q');
  const language = url.searchParams.get('lang');
  const page = parseInt(url.searchParams.get('page')) || 1;
  const pageSize = parseInt(url.searchParams.get('size')) || 20;

  if (!query) {
    return new Response('Query parameter "q" is required', { status: 400 });
  }

  try {
    // 记录搜索
    const startTime = Date.now();

    // 执行搜索
    const results = await multiLanguageSearch(env, query, language);
    const paginatedResults = paginateResults(results, page, pageSize);

    // 计算耗时
    const duration = Date.now() - startTime;

    // 记录搜索日志
    context.waitUntil(logSearch(env, query, null, results));

    return Response.json({
      query,
      results: paginatedResults.items,
      pagination: paginatedResults.pagination,
      meta: {
        total: results.length,
        duration,
        language: language || 'auto'
      }
    });

  } catch (error) {
    return Response.json({
      error: error.message
    }, { status: 500 });
  }
}

export async function onRequestPost(context) {
  const { request, env } = context;
  const { action, data } = await request.json();

  switch (action) {
    case 'suggest':
      return Response.json({
        suggestions: await getSearchSuggestions(env, data.query, data.limit || 10)
      });

    case 'autocomplete':
      return Response.json({
        completions: await autoComplete(env, data.partial, data.options)
      });

    case 'highlight':
      return Response.json({
        results: await searchWithHighlight(env, data.query, data.options)
      });

    default:
      return new Response('Invalid action', { status: 400 });
  }
}
```

```javascript
async function rebuildFTSIndex(env) {
  // 删除现有索引
  await env.DB.prepare('DROP TABLE IF EXISTS articles_fts').run();

  // 重新创建
  await env.DB.prepare(`
    CREATE VIRTUAL TABLE articles_fts USING fts5(
      title, content, author, tags,
      content='articles',
      content_rowid='id'
    )
  `).run();

  // 重新填充
  await env.DB.prepare(`
    INSERT INTO articles_fts(rowid, title, content, author, tags)
    SELECT id, title, content, author, tags FROM articles
  `).run();
}
```

## 最佳实践

### 1. FTS 使用建议
- **合理选择字段**: 只索引需要搜索的文本字段
- **定期优化**: 定期运行 OPTIMIZE 命令
- **监控性能**: 监控搜索查询性能
- **分词器选择**: 根据语言选择合适的分词器
- **限制结果**: 使用 LIMIT 限制返回结果

### 2. 搜索优化
- **使用缓存**: 缓存热门搜索结果
- **预处理查询**: 清理和标准化查询
- **分页优化**: 使用游标分页而不是 OFFSET
- **异步处理**: 对于大型索引，考虑异步重建

### 3. 用户体验
- **搜索建议**: 提供搜索建议和自动完成
- **结果高亮**: 高亮匹配的关键词
- **相关性排序**: 使用合适的排序算法
- **零结果处理**: 优雅处理无结果的情况
