# Email Routing 概述

## 什么是 Email Routing

Cloudflare Email Routing 是 Cloudflare 提供的邮件路由和转发服务，允许开发者接收、处理和转发电子邮件。它可以将发送到自定义域名的电子邮件路由到指定的目的地，包括 Workers、HTTP 端点或其他邮件服务器。

## 核心特性

```javascript
// 邮件处理函数
async function processEmail(message, env) {
  const { from, to, subject, text } = message;

  // 邮件验证和过滤
  if (!isValidEmail(from)) {
    throw new Error('Invalid sender email');
  }

  // 根据收件人路由到不同处理器
  const handler = getHandlerByRecipient(to);
  return await handler(message, env);
}

function isValidEmail(email) {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return emailRegex.test(email);
}

function getHandlerByRecipient(to) {
  const domain = to.split('@')[1];

  switch (domain) {
    case 'support.example.com':
      return handleSupportEmail;
    case 'noreply.example.com':
      return handleNoreplyEmail;
    default:
      return handleDefaultEmail;
  }
}

async function handleSupportEmail(message, env) {
  // 处理客户支持邮件
  const ticket = await createSupportTicket(message);

  // 自动回复
  await sendAutoReply(message, 'support', ticket.id);

  // 通知客服团队
  await notifySupportTeam(message, ticket.id);

  return new Response('Support email processed', { status: 200 });
}

async function handleNoreplyEmail(message, env) {
  // 处理noreply邮件
  console.log('Received noreply email, discarding');
  return new Response('Noreply email discarded', { status: 200 });
}

async function handleDefaultEmail(message, env) {
  // 默认邮件处理
  await forwardEmail(message, env.DEFAULT_FORWARD_ADDRESS);
  return new Response('Email forwarded', { status: 200 });
```

### 2. 自定义邮件处理
```javascript
// 高级邮件处理器
export class EmailProcessor {
  constructor(env) {
    this.env = env;
    this.rules = new Map();
    this.filters = [];
    this.handlers = new Map();
    this.analytics = new EmailAnalytics(env);
  }

  // 添加处理规则
  addRule(name, condition, action) {
    this.rules.set(name, { condition, action });
  }

  // 添加邮件过滤器
  addFilter(filter) {
    this.filters.push(filter);
  }

  // 添加邮件处理器
  addHandler(type, handler) {
    this.handlers.set(type, handler);
  }

  // 处理邮件
  async process(message) {
    // 记录分析数据
    await this.analytics.recordReceived(message);

    // 应用过滤器
    const filtered = await this.applyFilters(message);
    if (!filtered) {
      return { action: 'dropped', reason: 'filtered' };
    }

    // 应用规则
    const action = await this.applyRules(message);

    // 执行处理动作
    return await this.executeAction(action, message);
  }

  // 应用过滤器
  async applyFilters(message) {
    for (const filter of this.filters) {
      if (!await filter.filter(message)) {
        await this.analytics.recordFiltered(message, filter.name);
        return false;
      }
    }
    return true;
  }

  // 应用规则
  async applyRules(message) {
    for (const [name, rule] of this.rules) {
      if (await rule.condition(message)) {
        await this.analytics.recordRuleMatch(message, name);
        return rule.action;
      }
    }

    // 默认动作
    return { type: 'forward' };
  }

  // 执行动作
  async executeAction(action, message) {
    const handler = this.handlers.get(action.type);
    if (!handler) {
      throw new Error(`No handler for action type: ${action.type}`);
    }

    const result = await handler(message, action);

    // 记录处理结果
    await this.analytics.recordProcessed(message, action.type, result);

    return result;
  }
}

// 邮件分析器
export class EmailAnalytics {
  constructor(env) {
    this.env = env;
  }

  // 记录收到的邮件
  async recordReceived(message) {
    const record = {
      timestamp: Date.now(),
      from: message.from,
      to: message.to,
      subject: message.subject,
      size: this.calculateSize(message),
      attachments: message.attachments ? message.attachments.length : 0,
      type: this.detectEmailType(message)
    };

    await this.saveRecord('received', record);
  }

  // 记录过滤的邮件
  async recordFiltered(message, filterName) {
    const record = {
      timestamp: Date.now(),
      from: message.from,
      to: message.to,
      filter: filterName,
      reason: 'filtered'
    };

    await this.saveRecord('filtered', record);
  }

  // 记录规则匹配
  async recordRuleMatch(message, ruleName) {
    const record = {
      timestamp: Date.now(),
      from: message.from,
      to: message.to,
      rule: ruleName
    };

    await this.saveRecord('rule_match', record);
  }

  // 记录处理的邮件
  async recordProcessed(message, actionType, result) {
    const record = {
      timestamp: Date.now(),
      from: message.from,
      to: message.to,
      action: actionType,
      result: result,
      duration: result.duration || null
    };

    await this.saveRecord('processed', record);
  }

  // 计算邮件大小
  calculateSize(message) {
    let size = 0;

    if (message.text) size += message.text.length;
    if (message.html) size += message.html.length;
    if (message.subject) size += message.subject.length;
    if (message.attachments) {
      message.attachments.forEach(att => {
        size += att.content ? att.content.length : 0;
      });
    }

    return size;
  }

  // 检测邮件类型
  detectEmailType(message) {
    const subject = (message.subject || '').toLowerCase();
    const text = (message.text || '').toLowerCase();

    if (subject.includes('bounce') || text.includes('delivery failed')) {
      return 'bounce';
    }

    if (subject.includes('auto-reply') || subject.includes('out of office')) {
      return 'auto-reply';
    }

    if (subject.includes('spam') || text.includes('unsolicited')) {
      return 'spam';
    }

    if (message.attachments && message.attachments.length > 0) {
      return 'attachment';
    }

    return 'normal';
  }

  // 保存记录
  async saveRecord(type, record) {
    const key = `analytics:${type}:${Date.now()}:${Math.random()}`;
    await this.env.ANALYTICS.put(key, record, {
      expirationTtl: 30 * 24 * 60 * 60 // 30天
    });
  }

  // 获取统计报告
  async getReport(days = 7) {
    const cutoffTime = Date.now() - (days * 24 * 60 * 60 * 1000);
    const report = {
      total: 0,
      byType: {},
      byAction: {},
      byHour: {},
      topSenders: new Map(),
      topRecipients: new Map()
    };

    const types = ['received', 'filtered', 'rule_match', 'processed'];

    for (const type of types) {
      const keys = await this.env.ANALYTICS.list({
        prefix: `analytics:${type}:`
      });

      for (const key of keys.keys) {
        const timestamp = parseInt(key.name.split(':')[1]);
        if (timestamp > cutoffTime) {
          const record = await this.env.ANALYTICS.get(key.name);
          if (record) {
            report.total++;

            // 按类型统计
            if (record.type) {
              report.byType[record.type] = (report.byType[record.type] || 0) + 1;
            }

            // 按动作统计
            if (record.action) {
              report.byAction[record.action] = (report.byAction[record.action] || 0) + 1;
            }

            // 按小时统计
            const hour = new Date(record.timestamp).getHours();
            report.byHour[hour] = (report.byHour[hour] || 0) + 1;

            // 发件人统计
            if (record.from) {
              report.topSenders.set(
                record.from,
                (report.topSenders.get(record.from) || 0) + 1
              );
            }

            // 收件人统计
            if (record.to) {
              report.topRecipients.set(
                record.to,
                (report.topRecipients.get(record.to) || 0) + 1
              );
            }
          }
        }
      }
    }

    // 转换 Map 为数组并排序
    report.topSenders = Array.from(report.topSenders.entries())
      .sort((a, b) => b[1] - a[1])
      .slice(0, 10);

    report.topRecipients = Array.from(report.topRecipients.entries())
      .sort((a, b) => b[1] - a[1])
      .slice(0, 10);

    return report;
  }
}
```

## 邮件过滤和安全

### 1. 垃圾邮件过滤
```javascript
// 垃圾邮件过滤器
export class SpamFilter {
  constructor(env) {
    this.env = env;
    this.blacklist = new Set();
    this.whitelist = new Set();
    this.spamKeywords = new Set();
    this.reputationCache = new Map();
  }

  // 初始化过滤器
  async initialize() {
    await this.loadLists();
    await this.loadKeywords();
  }

  // 加载黑白名单
  async loadLists() {
    const blacklistData = await this.env.BLACKLIST.get('addresses') || '';
    const whitelistData = await this.env.WHITELIST.get('addresses') || '';

    if (blacklistData) {
      blacklistData.split(',').forEach(email => {
        this.blacklist.add(email.trim());
      });
    }

    if (whitelistData) {
      whitelistData.split(',').forEach(email => {
        this.whitelist.add(email.trim());
      });
    }
  }

  // 加载关键词
  async loadKeywords() {
    const keywordsData = await this.env.SPAM_KEYWORDS.get('words') || '';
    if (keywordsData) {
      keywordsData.split(',').forEach(keyword => {
        this.spamKeywords.add(keyword.toLowerCase().trim());
      });
    }
  }

  // 检查是否为垃圾邮件
  async isSpam(message) {
    const score = await this.calculateSpamScore(message);
    const threshold = 50; // 垃圾邮件阈值

    return score >= threshold;
  }

  // 计算垃圾邮件分数
  async calculateSpamScore(message) {
    let score = 0;

    // 黑名单检查
    if (this.blacklist.has(message.from)) {
      score += 80;
    }

    // 白名单检查
    if (this.whitelist.has(message.from)) {
      score = Math.max(0, score - 50);
    }

    // 发件人声誉检查
    const reputation = await this.checkSenderReputation(message.from);
    score += (100 - reputation) * 0.5;

    // 关键词检查
    score += this.checkKeywords(message);

    // 邮件结构检查
    score += this.checkEmailStructure(message);

    // 附件检查
    if (message.attachments) {
      score += this.checkAttachments(message.attachments);
    }

    // 内容分析
    score += await this.analyzeContent(message);

    return Math.min(100, score);
  }

  // 检查发件人声誉
  async checkSenderReputation(sender) {
    const cached = this.reputationCache.get(sender);
    if (cached && Date.now() - cached.timestamp < 3600000) { // 1小时缓存
      return cached.reputation;
    }

    let reputation = 50; // 默认声誉

    // 查询发件人历史
    const history = await this.getSenderHistory(sender);
    if (history) {
      const totalEmails = history.sent + history.spam;
      if (totalEmails > 0) {
        reputation = (history.sent / totalEmails) * 100;
      }
    }

    // 缓存结果
    this.reputationCache.set(sender, {
      reputation: reputation,
      timestamp: Date.now()
    });

    return reputation;
  }

  // 检查关键词
  checkKeywords(message) {
    let score = 0;
    const text = `${message.subject || ''} ${message.text || ''}`.toLowerCase();

    for (const keyword of this.spamKeywords) {
      if (text.includes(keyword)) {
        score += 5;
      }
    }

    return score;
  }

  // 检查邮件结构
  checkEmailStructure(message) {
    let score = 0;

    // 缺少主题扣分
    if (!message.subject || message.subject.trim().length === 0) {
      score += 10;
    }

    // 主题全大写扣分
    if (message.subject && message.subject === message.subject.toUpperCase()) {
      score += 5;
    }

    // 缺少正文扣分
    if (!message.text && !message.html) {
      score += 15;
    }

    return score;
  }

  // 检查附件
  checkAttachments(attachments) {
    let score = 0;

    for (const attachment of attachments) {
      // 可疑文件扩展名
      const suspiciousExtensions = ['.exe', '.bat', '.scr', '.pif'];
      const ext = attachment.filename.split('.').pop().toLowerCase();

      if (suspiciousExtensions.includes(`.${ext}`)) {
        score += 20;
      }

      // 大附件
      if (attachment.size && attachment.size > 10 * 1024 * 1024) { // 10MB
        score += 10;
      }
    }

    return score;
  }

  // 内容分析
  async analyzeContent(message) {
    let score = 0;

    // 简单的内容分析
    const text = `${message.subject || ''} ${message.text || ''}`;

    // 检查过度重复
    const repeated = this.checkRepetition(text);
    score += repeated * 5;

    // 检查可疑链接
    const suspiciousLinks = this.checkSuspiciousLinks(text);
    score += suspiciousLinks * 10;

    return score;
  }

  // 检查重复内容
  checkRepetition(text) {
    const words = text.split(/\s+/);
    const wordCounts = new Map();

    for (const word of words) {
      if (word.length > 2) {
        wordCounts.set(word, (wordCounts.get(word) || 0) + 1);
      }
    }

    let repetitions = 0;
    for (const count of wordCounts.values()) {
      if (count > 3) {
        repetitions++;
      }
    }

    return repetitions;
  }

  // 检查可疑链接
  checkSuspiciousLinks(text) {
    const urlRegex = /https?:\/\/[^\s]+/gi;
    const urls = text.match(urlRegex) || [];

    let suspiciousCount = 0;
    for (const url of urls) {
      // 检查短链接服务
      const shorteners = ['bit.ly', 'tinyurl.com', 'goo.gl'];
      if (shorteners.some(shortener => url.includes(shortener))) {
        suspiciousCount++;
      }

      // 检查IP地址链接
      const ipRegex = /\b(?:\d{1,3}\.){3}\d{1,3}\b/g;
      if (ipRegex.test(url)) {
        suspiciousCount++;
      }
    }

    return suspiciousCount;
  }

  // 获取发件人历史
  async getSenderHistory(sender) {
    const key = `sender_history:${sender}`;
    return await this.env.SENDER_HISTORY.get(key);
  }

  // 更新发件人历史
  async updateSenderHistory(sender, isSpam) {
    const key = `sender_history:${sender}`;
    let history = await this.getSenderHistory(sender) || { sent: 0, spam: 0 };

    if (isSpam) {
      history.spam++;
    } else {
      history.sent++;
    }

    history.lastUpdate = Date.now();

    await this.env.SENDER_HISTORY.put(key, history, {
      expirationTtl: 90 * 24 * 60 * 60 // 90天
    });
  }
}
```

### 2. 病毒邮件检测
```javascript
// 病毒扫描器
export class VirusScanner {
  constructor(env) {
    this.env = env;
    this.virusDefinitions = new Map();
    this.scanQueue = [];
    this.scanResults = new Map();
  }

  // 初始化扫描器
  async initialize() {
    await this.loadVirusDefinitions();
  }

  // 加载病毒定义
  async loadVirusDefinitions() {
    const definitions = await this.env.VIRUS_DEFINITIONS.get('definitions') || '';
    if (definitions) {
      const defs = JSON.parse(definitions);
      for (const def of defs) {
        this.virusDefinitions.set(def.signature, def);
      }
    }
  }

  // 扫描邮件
  async scan(message) {
    const scanId = crypto.randomUUID();

    // 检查是否已扫描过
    const messageHash = this.calculateMessageHash(message);
    if (this.scanResults.has(messageHash)) {
      return this.scanResults.get(messageHash);
    }

    // 添加到扫描队列
    this.scanQueue.push({
      id: scanId,
      message: message,
      hash: messageHash,
      status: 'pending'
    });

    // 执行扫描
    const result = await this.performScan(message, scanId);

    // 缓存结果
    this.scanResults.set(messageHash, result);

    return result;
  }

  // 执行实际扫描
  async performScan(message, scanId) {
    const result = {
      id: scanId,
      clean: true,
      threats: [],
      scannedAt: Date.now()
    };

    try {
      // 扫描邮件内容
      const textContent = `${message.subject || ''} ${message.text || ''}`;
      const threats = await this.scanTextContent(textContent);

      // 扫描附件
      if (message.attachments) {
        for (const attachment of message.attachments) {
          const attachmentThreats = await this.scanAttachment(attachment);
          threats.push(...attachmentThreats);
        }
      }

      // 更新扫描结果
      if (threats.length > 0) {
        result.clean = false;
        result.threats = threats;
      }

    } catch (error) {
      console.error('Virus scan error:', error);
      result.error = error.message;
    }

    return result;
  }

  // 扫描文本内容
  async scanTextContent(content) {
    const threats = [];

    for (const [signature, virus] of this.virusDefinitions) {
      if (content.includes(signature)) {
        threats.push({
          type: 'text',
          name: virus.name,
          signature: signature,
          severity: virus.severity
        });
      }
    }

    return threats;
  }

  // 扫描附件
  async scanAttachment(attachment) {
    const threats = [];

    try {
      // 检查文件名
      const filenameThreats = await this.scanFilename(attachment.filename);
      threats.push(...filenameThreats);

      // 检查文件内容
      if (attachment.content) {
        const contentThreats = await this.scanFileContent(
          attachment.content,
          attachment.filename
        );
        threats.push(...contentThreats);
      }

    } catch (error) {
      console.error('Attachment scan error:', error);
    }

    return threats;
  }

  // 扫描文件名
  async scanFilename(filename) {
    const threats = [];
    const suspiciousPatterns = [
      /\.exe$/i,
      /\.bat$/i,
      /\.scr$/i,
      /\.pif$/i,
      /\.com$/i,
      /\.vbs$/i
    ];

    for (const pattern of suspiciousPatterns) {
      if (pattern.test(filename)) {
        threats.push({
          type: 'filename',
          name: 'Suspicious file extension',
          pattern: pattern.toString(),
          severity: 'medium'
        });
      }
    }

    return threats;
  }

  // 扫描文件内容
  async scanFileContent(content, filename) {
    const threats = [];

    // 检查PE文件头
    if (this.isPEFile(content)) {
      threats.push({
        type: 'file',
        name: 'PE file detected',
        severity: 'high'
      });
    }

    // 检查脚本内容
    if (this.isScript(content, filename)) {
      threats.push({
        type: 'file',
        name: 'Script file detected',
        severity: 'low'
      });
    }

    // 检查病毒签名
    const contentStr = content.toString('binary');
    for (const [signature, virus] of this.virusDefinitions) {
      if (contentStr.includes(signature)) {
        threats.push({
          type: 'file',
          name: virus.name,
          signature: signature,
          severity: virus.severity
        });
      }
    }

    return threats;
  }

  // 检查是否为PE文件
  isPEFile(content) {
    if (content.length < 2) return false;
    return content[0] === 0x4D && content[1] === 0x5A; // MZ header
  }

  // 检查是否为脚本文件
  isScript(content, filename) {
    const scriptExtensions = ['.js', '.vbs', '.ps1', '.py', '.rb', '.php'];
    const ext = filename.split('.').pop().toLowerCase();

    return scriptExtensions.includes(`.${ext}`) ||
           content.toString().includes('<script');
  }

  // 计算邮件哈希
  calculateMessageHash(message) {
    const content = [
      message.from || '',
      message.to || '',
      message.subject || '',
      message.text || '',
      message.html || ''
    ].join('|');

    // 简单哈希实现（实际应用中应使用更强的哈希算法）
    let hash = 0;
    for (let i = 0; i < content.length; i++) {
      const char = content.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // 转换为32位整数
    }
    return hash.toString();
  }

  // 清理扫描队列
  cleanupQueue() {
    this.scanQueue = [];
    this.scanResults.clear();
  }

  // 获取扫描统计
  async getScanStats() {
    const totalScans = this.scanResults.size;
    const infected = Array.from(this.scanResults.values())
      .filter(result => !result.clean).length;

    return {
      total: totalScans,
      clean: totalScans - infected,
      infected: infected,
      infectionRate: totalScans > 0 ? (infected / totalScans * 100).toFixed(2) : 0
    };
  }
}
```

## 使用场景

### 1. 邮件转发服务
```javascript
// 邮件转发服务
export class EmailForwardingService {
  constructor(env) {
    this.env = env;
  }

  async email(message, env) {
    const { from, to, subject } = message;

    // 查找转发规则
    const rule = await this.findForwardingRule(to);

    if (!rule) {
      return new Response('No forwarding rule found', { status: 404 });
    }

    try {
      // 执行转发
      const result = await this.forwardEmail(message, rule);

      // 记录转发日志
      await this.logForward(message, rule, result);

      return new Response('Email forwarded', { status: 200 });

    } catch (error) {
      console.error('Email forwarding failed:', error);
      return new Response('Forwarding failed', { status: 500 });
    }
  }

  async findForwardingRule(email) {
    const domain = email.split('@')[1];
    const username = email.split('@')[0];

    // 精确匹配
    let rule = await this.env.FORWARDING_RULES.get(email);
    if (rule) return rule;

    // 域名匹配
    rule = await this.env.FORWARDING_RULES.get(`@${domain}`);
    if (rule) return rule;

    // 通配符匹配
    rule = await this.env.FORWARDING_RULES.get('*');
    return rule;
  }

  async forwardEmail(message, rule) {
    const targets = rule.targets || [];

    // 并发转发到多个目标
    const promises = targets.map(target => {
      return this.sendEmail(target, message);
    });

    const results = await Promise.allSettled(promises);
    const successful = results.filter(r => r.status === 'fulfilled').length;

    return {
      targets: targets.length,
      successful: successful,
      failed: targets.length - successful
    };
  }

  async sendEmail(to, originalMessage) {
    // 使用外部邮件服务发送邮件
    const sendgridApiKey = this.env.SENDGRID_API_KEY;

    const emailData = {
      personalizations: [{
        to: [{ email: to }],
        subject: `Fwd: ${originalMessage.subject}`,
        content: [{
          type: 'text/plain',
          value: `Forwarded message from: ${originalMessage.from}\n\n${originalMessage.text}`
        }]
      }]
    };

    const response = await fetch('https://api.sendgrid.com/v3/mail/send', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${sendgridApiKey}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(emailData)
    });

    return response;
  }

  async logForward(message, rule, result) {
    const log = {
      id: crypto.randomUUID(),
      original: {
        from: message.from,
        to: message.to,
        subject: message.subject
      },
      rule: rule,
      result: result,
      timestamp: Date.now()
    };

    const key = `forwarding_log:${log.id}`;
    await this.env.LOGS.put(key, log, {
      expirationTtl: 30 * 24 * 60 * 60 // 30天
    });
  }
}
```
