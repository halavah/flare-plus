# Email Routing 邮件过滤

## 🔍 过滤系统概述

Cloudflare Email Routing 提供强大的邮件过滤功能，可以有效拦截垃圾邮件、恶意内容和病毒。通过多层过滤机制，确保只有安全、合规的邮件到达用户邮箱。

### 过滤层次结构
```
输入邮件
    ↓
[网络层过滤] - IP声誉检查
    ↓
[协议层过滤] - SMTP验证
    ↓
[内容层过滤] - 垃圾邮件检测
    ↓
[附件层过滤] - 病毒扫描
    ↓
[规则层过滤] - 用户自定义规则
    ↓
输出邮件 / 拦截
```

export default {
  async email(message, env, ctx) {

    // 多维度垃圾邮件检测

    if (spamScore > SPAM_THRESHOLD) {
      await logSpam(emailData, spamScore, env);
      return new Response('Spam blocked', { status: 200 });

    // 继续处理邮件
    return await processValidEmail(message, env);
  }

// 计算垃圾邮件分数
async function calculateSpamScore(emailData, env) {
  let score = 0;

  // 发件人信誉检查
  score += await checkSenderReputation(emailData.from, env);

  // 内容分析
  score += await analyzeContent(emailData.subject + emailData.body, env);

  // 邮件结构检查
  score += await analyzeEmailStructure(emailData, env);

  // 历史行为分析
  score += await analyzeSenderHistory(emailData.from, env);

  return score;
}

// 发件人信誉检查
async function checkSenderReputation(sender, env) {
  const domain = sender.split('@')[1];

  // 检查域名黑名单
  const blacklisted = await env.BLACKLIST_KV.get(`domain:${domain}`);
  if (blacklisted) return 100;

  // 检查发件人历史
  const history = await getSenderHistory(sender, env);
  if (history.rejectedCount > 10) return 50;

  // 检查 SPF/DKIM/DMARC
  const authResult = await checkEmailAuthentication(sender);
  if (!authResult.valid) return 30;

  return 0;
}

// 内容分析
async function analyzeContent(content, env) {
  let score = 0;

  // 关键词检测
  const spamKeywords = await env.SPAM_KEYWORDS_KV.get('keywords', 'json');
  for (const keyword of spamKeywords) {
    const count = (content.match(new RegExp(keyword, 'gi')) || []).length;
    score += count * keyword.weight || 5;
  }

  // URL检测
  const urls = extractUrls(content);
  for (const url of urls) {
    if (await isSuspiciousUrl(url)) {
      score += 20;
    }
  }

  // 文本特征检测
  score += analyzeTextFeatures(content);

  return score;
}

// 文本特征分析
function analyzeTextFeatures(content) {
  let score = 0;

  // 大写字母比例
  const upperCaseRatio = (content.match(/[A-Z]/g) || []).length / content.length;
  if (upperCaseRatio > 0.3) score += 10;

  // 特殊字符比例
  const specialCharRatio = (content.match(/[^a-zA-Z0-9\s]/g) || []).length / content.length;
  if (specialCharRatio > 0.1) score += 5;

  // 重复字符检测
  if (/(.)\1{3,}/.test(content)) score += 5;

  // 邮件长度检测
  if (content.length < 10 || content.length > 100000) score += 10;

  return score;
}

// virus-scanner.js
export default {
  async email(message, env, ctx) {
    const attachments = message.files;

    if (attachments.length === 0) {
      // 无附件，继续处理
      return await processEmail(message, env);
    }

    // 扫描所有附件
    const scanResults = await Promise.allSettled(
      attachments.map(file => scanAttachment(file, env))
    );

    const hasVirus = scanResults.some(result =>
      result.status === 'fulfilled' && result.value.hasVirus
    );

    if (hasVirus) {
      await logVirusDetection(message, scanResults, env);
      return new Response('Virus detected', { status: 200 });
    }

    // 继续处理干净的邮件
    return await processEmail(message, env);
  }
};

// 附件病毒扫描
async function scanAttachment(file, env) {
  try {
    // 获取文件内容
    const buffer = await file.arrayBuffer();
    const bytes = new Uint8Array(buffer);

    // 文件类型检查
    const fileType = await detectFileType(bytes);
    if (isExecutable(fileType)) {
      return { hasVirus: true, reason: 'Executable file type' };
    }

    // 大小检查
    if (bytes.length > MAX_ATTACHMENT_SIZE) {
      return { hasVirus: true, reason: 'File too large' };
    }

    // 病毒特征码扫描
    const signatureResult = await scanWithSignatures(bytes, env);
    if (signatureResult.hasVirus) {
      return signatureResult;
    }

    // 启发式扫描
    const heuristicResult = await heuristicScan(bytes, env);
    if (heuristicResult.hasVirus) {
      return heuristicResult;
    }

    return { hasVirus: false };

  } catch (error) {
    console.error('Virus scanning error:', error);
    // 安全起见，出现错误时视为可疑
    return { hasVirus: true, reason: 'Scan error' };
  }
}

// 文件类型检测
function detectFileType(bytes) {
  // 检查文件魔数
  const signatures = {
    'MZ': { type: 'exe', executable: true }, // Windows EXE
    '\x7fELF': { type: 'elf', executable: true }, // Linux ELF
    '\xCA\xFE\xBA\xBE': { type: 'class', executable: true }, // Java Class
    'PK\x03\x04': { type: 'zip', executable: false },
    '\x89PNG': { type: 'png', executable: false },
    '\xFF\xD8\xFF': { type: 'jpg', executable: false }
  };

  for (const [signature, info] of Object.entries(signatures)) {
    const bytesSignature = String.fromCharCode(...bytes.slice(0, signature.length));
    if (bytesSignature === signature) {
      return info;
    }
  }

  return { type: 'unknown', executable: false };
}

// 病毒特征码扫描
async function scanWithSignatures(bytes, env) {
  // 获取病毒特征码
  const signatures = await env.VIRUS_SIGNATURES_KV.get('signatures', 'json');

  for (const signature of signatures) {
    if (bytes.length >= signature.pattern.length) {
      const match = bytes.every((byte, index) =>
        byte === signature.pattern[index] ||
        signature.pattern[index] === '?'
      );

      if (match) {
        return {
          hasVirus: true,
          reason: `Virus detected: ${signature.name}`,
          severity: signature.severity
        };
      }
    }
  }

  return { hasVirus: false };
}

// 启发式扫描
async function heuristicScan(bytes, env) {
  let riskScore = 0;

  // 检查可执行代码特征
  riskScore += checkExecutablePatterns(bytes);

  // 检查脚本特征
  riskScore += checkScriptPatterns(bytes);

  // 检查加密特征
  riskScore += checkEncryptionPatterns(bytes);

  // 检查可疑字符串
  riskScore += checkSuspiciousStrings(bytes);

  return {
    hasVirus: riskScore > 50,
    reason: riskScore > 50 ? `Heuristic detection: ${riskScore}` : null,
    score: riskScore
  };
}

// 检查可执行代码模式
function checkExecutablePatterns(bytes) {
  let score = 0;

  // 检查PE头
  if (bytes[0] === 0x4D && bytes[1] === 0x5A) { // MZ
    score += 40;
  }

  // 检查常见的恶意代码模式
  const maliciousPatterns = [
    [0x90, 0x90, 0x90, 0x90], // NOP滑动
    [0x68, 0x00, 0x00, 0x00, 0x00], // PUSH 0
    [0xE8, 0x00, 0x00, 0x00, 0x00] // CALL near
  ];

  for (const pattern of maliciousPatterns) {
    if (containsPattern(bytes, pattern)) {
      score += 10;
    }
  }

  return score;
}

// 模式匹配函数
function containsPattern(bytes, pattern) {
  for (let i = 0; i <= bytes.length - pattern.length; i++) {
    let match = true;
    for (let j = 0; j < pattern.length; j++) {
      if (bytes[i + j] !== pattern[j]) {
        match = false;
        break;
      }
    }
    if (match) return true;
  }
  return false;
}

// content-filter.js
export default {
  async email(message, env, ctx) {
    const emailData = await extractEmailData(message);

    // 应用内容过滤规则
    const filterResult = await applyContentFilters(emailData, env);

    if (filterResult.blocked) {
      await logContentBlock(emailData, filterResult, env);
      return new Response('Content filtered', { status: 200 });
    }

    if (filterResult.quarantine) {
      await quarantineEmail(message, filterResult.reason, env);
      return new Response('Email quarantined', { status: 200 });
    }

    // 标记和修改邮件
    if (filterResult.modified) {
      await modifyEmail(message, filterResult.modifications, env);
    }

    return await processEmail(message, env);
  }
};

// 应用内容过滤规则
async function applyContentFilters(emailData, env) {
  const result = {
    blocked: false,
    quarantine: false,
    modified: false,
    reason: null,
    modifications: []
  };

  // 获取过滤规则
  const filters = await getActiveFilters(env);

  // 逐条应用规则
  for (const filter of filters) {
    const filterResult = await applyFilter(emailData, filter, env);

    if (filterResult.blocked) {
      result.blocked = true;
      result.reason = filterResult.reason;
      break;
    }

    if (filterResult.quarantine) {
      result.quarantine = true;
      result.reason = filterResult.reason;
    }

    if (filterResult.modified) {
      result.modified = true;
      result.modifications.push(...filterResult.modifications);
    }
  }

  return result;
}

// 应用单个过滤规则
async function applyFilter(emailData, filter, env) {
  const result = { blocked: false, quarantine: false, modified: false };

  switch (filter.type) {
    case 'keyword':
      return await applyKeywordFilter(emailData, filter);

    case 'regex':
      return await applyRegexFilter(emailData, filter);

    case 'url':
      return await applyUrlFilter(emailData, filter);

    case 'attachment':
      return await applyAttachmentFilter(emailData, filter);

    case 'size':
      return await applySizeFilter(emailData, filter);

    default:
      return result;
  }
}

// 关键词过滤
async function applyKeywordFilter(emailData, filter) {
  const content = (emailData.subject + ' ' + emailData.body).toLowerCase();
  const keywords = filter.keywords.map(k => k.toLowerCase());

  const matchedKeywords = keywords.filter(keyword =>
    content.includes(keyword)
  );

  if (matchedKeywords.length > 0) {
    return {
      blocked: filter.action === 'block',
      quarantine: filter.action === 'quarantine',
      reason: `Blocked keywords: ${matchedKeywords.join(', ')}`
    };
  }

  return { blocked: false, quarantine: false };
}

// 正则表达式过滤
async function applyRegexFilter(emailData, filter) {
  const content = emailData.subject + ' ' + emailData.body;
  const regex = new RegExp(filter.pattern, filter.flags || 'i');

  const matches = content.match(regex);
  if (matches) {
    return {
      blocked: filter.action === 'block',
      quarantine: filter.action === 'quarantine',
      reason: `Pattern matched: ${filter.pattern}`
    };
  }

  return { blocked: false, quarantine: false };
}

// URL过滤
async function applyUrlFilter(emailData, filter) {
  const urls = extractUrls(emailData.subject + ' ' + emailData.body);

  for (const url of urls) {
    const domain = extractDomain(url);

    // 检查域名黑名单
    if (filter.blacklistedDomains.includes(domain)) {
      return {
        blocked: true,
        reason: `Blacklisted domain: ${domain}`
      };
    }

    // 检查URL信誉
    if (filter.checkReputation) {
      const reputation = await checkUrlReputation(url);
      if (reputation.score < filter.reputationThreshold) {
        return {
          blocked: true,
          reason: `Suspicious URL: ${url}`
        };
      }
    }
  }

  return { blocked: false };
}

async function createFilter(filterData, env) {
  // 验证规则数据
  const validation = validateFilter(filterData);
  if (!validation.valid) {
    return new Response(
      JSON.stringify({ error: validation.error }),
      { status: 400, headers: { 'Content-Type': 'application/json' } }
    );
  }

  // 存储规则
  const filterId = crypto.randomUUID();
  const filter = {
    id: filterId,
    ...filterData,
    createdAt: new Date().toISOString(),
    enabled: true
  };

  await env.DB.prepare(`
    INSERT INTO email_filters (id, name, type, config, action, enabled, created_at)

    filter.id,
    filter.name,
    filter.type,
    JSON.stringify(filter.config),
    filter.action,
    filter.enabled ? 1 : 0,
    filter.createdAt
  ).run();

  // 缓存规则
  await cacheFilter(filter, env);

  return new Response(
    JSON.stringify(filter),
    { status: 201, headers: { 'Content-Type': 'application/json' } }
  );
}

// filter-optimization.js
// 动态规则优先级调整
class RuleOptimizer {
  constructor(env) {
    this.env = env;
  }

  async optimizeRules() {
    // 获取规则性能数据
    const performanceData = await this.getRulePerformance();

    // 计算优化建议
    const optimizations = this.calculateOptimizations(performanceData);

    // 应用优化
    for (const opt of optimizations) {
      await this.applyOptimization(opt);
    }

    return optimizations;
  }

  calculateOptimizations(performanceData) {
    const optimizations = [];

    // 识别低效规则
    const slowRules = performanceData.filter(rule =>
      rule.avgExecutionTime > 100 && rule.matchRate < 0.01
    );

    for (const rule of slowRules) {
      optimizations.push({
        type: 'de_prioritize',
        ruleId: rule.id,
        reason: 'Low match rate with high execution time'
      });
    }

    // 识别高匹配率规则
    const highMatchRules = performanceData.filter(rule =>
      rule.matchRate > 0.5
    );

    for (const rule of highMatchRules) {
      optimizations.push({
        type: 'prioritize',
        ruleId: rule.id,
        reason: 'High match rate should be processed early'
      });
    }

    return optimizations;
  }
}
