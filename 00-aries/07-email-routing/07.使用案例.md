### 案例1：部门邮件自动分发

// department-routing.js
export default {
  async email(message, env, ctx) {
    const emailData = await extractEmailData(message);

    // 根据主题和内容智能分发
    const department = await identifyDepartment(emailData, env);

    if (department) {
      // 获取部门邮箱配置
      const deptConfig = await getDepartmentConfig(department, env);

      // 根据规则处理邮件
      await routeToDepartment(message, deptConfig, env);

      // 创建工单（如果是支持请求）
      if (department === 'support') {
        await createSupportTicket(emailData, env);
      }

      // 发送确认邮件
      await sendDepartmentConfirmation(emailData, department, env);
    }

    return new Response('Email routed to department', { status: 200 });
  }
};

// 识别部门
async function identifyDepartment(emailData, env) {
  const content = (emailData.subject + ' ' + emailData.body).toLowerCase();

  // 部门关键词映射
  const departmentKeywords = {
    'hr': ['招聘', '面试', '入职', '离职', '薪资', '福利', 'hr', 'human resources'],
    'it': ['it支持', '系统', '网络', '软件', '硬件', '技术支持', 'bug', '系统故障'],
    'finance': ['财务', '发票', '报销', '付款', '预算', '会计', '财务部'],
    'sales': ['销售', '报价', '合同', '订单', '产品咨询', '价格'],
    'marketing': ['市场', '营销', '推广', '广告', '品牌', '活动'],
    'legal': ['法律', '法务', '合同审核', '合规', '知识产权'],
    'support': ['客服', '支持', '投诉', '建议', '帮助', '问题']
  };

  // 计算每个部门的匹配分数
  const scores = {};
  for (const [dept, keywords] of Object.entries(departmentKeywords)) {
    scores[dept] = keywords.filter(keyword =>
      content.includes(keyword)
    ).length;
  }

  // 返回得分最高的部门
  const bestMatch = Object.entries(scores).sort((a, b) => b[1] - a[1])[0];

  return bestMatch[1] > 0 ? bestMatch[0] : null;
}

// 获取部门配置
async function getDepartmentConfig(department, env) {
  const config = await env.DB.prepare(`
    SELECT * FROM department_configs
    WHERE department = ? AND active = 1
  `).bind(department).first();

  return config;
}

// 部门路由
async function routeToDepartment(message, config, env) {
  // 添加部门标签
  const taggedMessage = await addDepartmentTag(message, config.department);

  // 转发到部门邮箱
  await taggedMessage.forward(config.email);

  // 根据重要性进行额外处理
  if (await isUrgentEmail(message, env)) {
    // 发送紧急通知到相关负责人
    await sendUrgentNotification(message, config, env);
  }

  // 存档到部门文件夹
  await archiveToDepartmentFolder(message, config.department, env);
}

// executive-filter.js
export default {

// 处理高管邮件
async function processExecutiveEmail(emailData, env) {
  const result = {
    blocked: false,
    flagged: false,
    summarized: false,
    requiresAssistantReview: false,
    reason: null
  };

  // 垃圾邮件检查
  if (await isSpam(emailData, env)) {
    result.blocked = true;
    result.reason = 'Spam detected';
    return result;
  }

  // 发件人信誉检查
  const senderRep = await checkSenderReputation(emailData.from, env);
  if (senderRep.score < 30) {
    result.flagged = true;
    result.requiresAssistantReview = true;
    result.reason = 'Low sender reputation';
  }

  // 内容敏感性分析
  const sensitivity = await analyzeContentSensitivity(emailData, env);
  if (sensitivity.level > 7) {
    result.flagged = true;
    result.requiresAssistantReview = true;
    result.reason = `High sensitivity: ${sensitivity.type}`;
  }

  // 长邮件生成摘要
  if (emailData.body.length > 2000) {
    result.summarized = true;
    result.requiresAssistantReview = true;
  }

  // 非工作时间邮件需要预审
  if (isNonWorkingHours()) {
    result.requiresAssistantReview = true;
    result.reason = 'Non-working hours';
  }

  return result;
}

// 内容敏感性分析
async function analyzeContentSensitivity(emailData, env) {
  const content = (emailData.subject + ' ' + emailData.body).toLowerCase();

  const sensitivePatterns = {
    'legal': ['诉讼', '法律', '合同纠纷', '法务', '律师', '法院'],
    'financial': ['财务', '预算', '投资', '收购', '并购', '财务报告'],
    'hr_confidential': ['人事机密', '薪资调整', '人事变动', '裁员', '升职'],
    'strategic': ['战略', '商业计划', '机密项目', '新产品', '竞争对手']
  };

  let maxSensitivity = 0;
  let sensitiveType = null;

  for (const [type, keywords] of Object.entries(sensitivePatterns)) {
    const matches = keywords.filter(keyword => content.includes(keyword));
    if (matches.length > 0) {
      const sensitivity = Math.min(10, matches.length * 2);
      if (sensitivity > maxSensitivity) {
        maxSensitivity = sensitivity;
        sensitiveType = type;
      }
    }
  }

  return {
    level: maxSensitivity,
    type: sensitiveType
  };
}

// order-automation.js
export default {
  async email(message, env, ctx) {

// 处理订单邮件
async function processOrderEmail(emailData, env) {
  // 解析订单信息
  const orderInfo = await parseOrderInfo(emailData, env);

  if (!orderInfo) {
    // 无法解析，转发给人工处理
    await forwardToManualProcessing(message, env);
    return;
  }

  // 根据订单状态处理
  switch (orderInfo.type) {
    case 'new_order':
      await handleNewOrder(orderInfo, env);
      break;

    case 'payment_confirmation':
      await handlePaymentConfirmation(orderInfo, env);
      break;

    case 'shipping_notification':
      await handleShippingNotification(orderInfo, env);
      break;

    case 'return_request':
      await handleReturnRequest(orderInfo, env);
      break;

    case 'customer_inquiry':
      await handleCustomerInquiry(orderInfo, env);
      break;
  }
}

// 处理新订单
async function handleNewOrder(orderInfo, env) {
  // 验证订单信息
  const validation = await validateOrder(orderInfo, env);
  if (!validation.valid) {
    await sendOrderValidationError(orderInfo, validation.errors, env);
    return;
  }

  // 创建订单记录
  const orderId = await createOrderRecord(orderInfo, env);

  // 发送订单确认
  await sendOrderConfirmation(orderInfo, orderId, env);

  // 通知仓库
  await notifyWarehouse(orderInfo, orderId, env);

  // 更新库存
  await updateInventory(orderInfo.items, env);

  // 记录日志
  await logOrderEvent(orderId, 'created', orderInfo, env);
}

// 解析订单信息
async function parseOrderInfo(emailData, env) {
  // 使用正则表达式和关键词提取订单信息
  const content = emailData.subject + ' ' + emailData.body;

  const patterns = {
    orderNumber: /(?:订单号|order(?:\s+)?#?)[\s:]*([A-Z0-9-]+)/i,
    customerEmail: /(?:客户|customer)[\s:]*([^\s@]+@[^\s@]+\.[^\s]+)/i,
    totalAmount: /(?:总额|total)[\s:]*\$?(\d+(?:\.\d{2})?)/i,
    items: /(?:商品|产品|item)[\s:]*([^]+)/i
  };

  const orderInfo = {
    type: identifyOrderType(emailData),
    orderNumber: extractPattern(content, patterns.orderNumber),
    customerEmail: extractPattern(content, patterns.customerEmail),
    totalAmount: parseFloat(extractPattern(content, patterns.totalAmount) || 0),
    items: parseItems(content)
  };

  // 验证必要信息
  if (!orderInfo.orderNumber || !orderInfo.customerEmail) {
    return null;
  }

  return orderInfo;
}

// 识别订单类型
function identifyOrderType(emailData) {
  const subject = emailData.subject.toLowerCase();
  const body = emailData.body.toLowerCase();

  if (subject.includes('新订单') || body.includes('new order')) {
    return 'new_order';
  }
  if (subject.includes('支付确认') || body.includes('payment confirmed')) {
    return 'payment_confirmation';
  }
  if (subject.includes('发货') || body.includes('shipped')) {
    return 'shipping_notification';
  }
  if (subject.includes('退货') || body.includes('return')) {
    return 'return_request';
  }

  return 'customer_inquiry';
}

// 分析客服请求
async function analyzeServiceRequest(emailData, env) {
  const content = (emailData.subject + ' ' + emailData.body).toLowerCase();

  const request = {
    category: null,
    urgency: 1,
    priority: 'normal',
    autoReply: true,
    estimatedResponseTime: 24 // hours
  };

  // 分类分析
  const categories = {
    'technical': ['技术问题', 'bug', '故障', '无法', '错误', 'technical'],
    'billing': ['账单', '付款', '退款', '发票', 'billing', 'payment'],
    'product': ['产品', '功能', '使用方法', 'product', 'feature'],
    'account': ['账户', '密码', '登录', '注册', 'account', 'password'],
    'shipping': ['配送', '物流', '快递', 'shipping', 'delivery'],
    'complaint': ['投诉', '不满', '抱怨', 'complaint']
  };

  for (const [category, keywords] of Object.entries(categories)) {
    const matches = keywords.filter(keyword => content.includes(keyword));
    if (matches.length > 0) {
      request.category = category;
      request.urgency += matches.length;
      break;
    }
  }

  // 紧急程度分析
  const urgentKeywords = ['紧急', 'urgent', 'asap', '立即', '马上'];
  const urgentMatches = urgentKeywords.filter(keyword => content.includes(keyword));
  if (urgentMatches.length > 0) {
    request.urgency += 5;
    request.priority = 'high';
    request.estimatedResponseTime = 2;
  }

  // VIP客户检查
  const isVIP = await checkVIPCustomer(emailData.from, env);
  if (isVIP) {
    request.priority = 'high';
    request.estimatedResponseTime = 1;
  }

  // 重复问题检测
  const isRepeated = await checkRepeatedIssue(emailData, env);
  if (isRepeated) {
    request.autoReply = false; // 不自动回复重复问题
    request.urgency += 2;
  }

  return request;
}

// 路由到客服团队
async function routeToServiceTeam(message, request, env) {
  // 获取团队配置
  const teamConfig = await getServiceTeamConfig(request.category, env);

  if (teamConfig) {
    // 根据负载均衡选择客服
    const agent = await selectAvailableAgent(teamConfig.teamId, env);

    if (agent) {
      // 转发给指定客服
      await message.forward(agent.email);

      // 更新客服负载
      await updateAgentLoad(agent.id, env);
    } else {
      // 没有可用客服，加入队列
      await addToQueue(message, teamConfig.teamId, env);
    }

    // 通知团队主管（如果是高优先级）
    if (request.priority === 'high') {
      await notifyTeamSupervisor(request, teamConfig, env);
    }
  } else {
    // 未分类问题，转发给通用客服
    await message.forward('general-support@company.com');
  }
}

// 选择可用客服
async function selectAvailableAgent(teamId, env) {
  // 获取团队客服列表
  const agents = await env.DB.prepare(`
    SELECT * FROM service_agents
    WHERE team_id = ? AND status = 'active'
    ORDER BY current_load ASC, last_response_time ASC
    LIMIT 5
  `).bind(teamId).all();

  // 检查每个客服的可用性
  for (const agent of agents.results) {
    const isAvailable = await checkAgentAvailability(agent.id, env);
    if (isAvailable) {
      return agent;
    }
  }

  return null;
}

async function handleSubscription(emailData, env) {
  const email = emailData.from;

  // 验证邮箱格式
  if (!isValidEmail(email)) {
    await sendSubscriptionError(email, 'Invalid email format', env);
    return;
  }

  // 检查是否已订阅
  const isSubscribed = await checkSubscription(email, env);
  if (isSubscribed) {
    await sendSubscriptionNotice(email, 'already_subscribed', env);
    return;

  // 生成验证码
  const verificationCode = generateVerificationCode();

  // 存储待验证订阅

      id, email, verification_code, created_at, expires_at
    ) VALUES (?, ?, ?, ?, ?)
  `).bind(
    crypto.randomUUID(),
    email,
    verificationCode,
    new Date().toISOString(),
    new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString() // 24小时后过期
  ).run();

  // 发送验证邮件
  await sendVerificationEmail(email, verificationCode, env);

  // 记录日志
  await logSubscriptionEvent(email, 'verification_sent', env);
}

async function handleUnsubscription(emailData, env) {

        }
      };

      personalizedEmails.push(personalized);

    } catch (error) {
      console.error(`Personalization failed for ${subscriber.email}:`, error);
    }
  }

// workflow-trigger.js
export default {
  async email(message, env, ctx) {
    const emailData = await extractEmailData(message);

    // 解析工作流触发条件
    const triggers = await parseWorkflowTriggers(emailData, env);

    // 执行匹配的工作流
    for (const trigger of triggers) {
      await executeWorkflow(trigger, emailData, env);
    }

    return new Response('Workflows executed', { status: 200 });
  }
};

// 解析工作流触发器
async function parseWorkflowTriggers(emailData, env) {
  const triggers = [];

  // 获取所有活跃的工作流
  const workflows = await env.DB.prepare(`
    SELECT * FROM email_workflows
    WHERE active = 1
  `).all();

  for (const workflow of workflows.results) {
    // 检查触发条件
    const conditions = JSON.parse(workflow.trigger_conditions);

    if (await evaluateConditions(emailData, conditions)) {
      triggers.push(workflow);
    }
  }

  return triggers;
}

// 执行工作流
async function executeWorkflow(workflow, emailData, env) {
  // 创建工作流实例
  const instanceId = crypto.randomUUID();

  await env.DB.prepare(`
    INSERT INTO workflow_instances (
      id, workflow_id, email_data, status, started_at
    ) VALUES (?, ?, ?, ?, ?)
  `).bind(
    instanceId,
    workflow.id,
    JSON.stringify(emailData),
    'running',
    new Date().toISOString()
  ).run();

  try {
    // 执行工作流步骤
    const steps = JSON.parse(workflow.steps);

    for (const step of steps) {
      await executeWorkflowStep(step, emailData, instanceId, env);
    }

    // 标记为完成
    await updateWorkflowInstance(instanceId, 'completed', env);

  } catch (error) {
    console.error('Workflow execution error:', error);
    await updateWorkflowInstance(instanceId, 'failed', env);
    await logWorkflowError(instanceId, error, env);
  }
}

// 执行工作流步骤
async function executeWorkflowStep(step, emailData, instanceId, env) {
  switch (step.type) {
    case 'send_email':
      await executeSendEmailStep(step, emailData, env);
      break;

    case 'create_task':
      await executeCreateTaskStep(step, emailData, env);
      break;

    case 'update_database':
      await executeUpdateDatabaseStep(step, emailData, env);
      break;

    case 'call_api':
      await executeCallAPIStep(step, emailData, env);
      break;

    case 'delay':
      await executeDelayStep(step, env);
      break;

    case 'condition':
      await executeConditionStep(step, emailData, instanceId, env);
      break;

    default:
      throw new Error(`Unknown step type: ${step.type}`);
  }

  // 记录步骤执行
  await logWorkflowStep(instanceId, step, 'success', env);
}

// 发送邮件步骤

// 创建任务步骤
async function executeCreateTaskStep(step, emailData, env) {
  const taskData = {
    title: renderTemplate(step.title, emailData),
    description: renderTemplate(step.description, emailData),
    assignee: step.assignee,
    priority: step.priority,
    due_date: calculateDueDate(step.due_in_days),
    source_email_id: emailData.id
  };

  // 调用任务管理API
  const response = await fetch('https://tasks.company.com/api/tasks', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${env.TASKS_API_TOKEN}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify(taskData)
  });

  if (!response.ok) {
    throw new Error('Failed to create task');
  }

async function collectEmailMetrics(emailData, env) {
  const metrics = {
    timestamp: new Date().toISOString(),
    from_domain: emailData.from.split('@')[1],
    to_domain: emailData.to.split('@')[1],
    subject_length: emailData.subject.length,

    keywords: await extractKeywords(emailData, env),
    response_priority: await calculateResponsePriority(emailData, env)
  };

  // 存储到分析数据库
  await env.ANALYTICS_DB.prepare(`
    INSERT INTO email_metrics (
      email_id, timestamp, from_domain, to_domain,
      subject_length, body_length, has_attachments,
      attachment_count, spam_score, sentiment,
      keywords, response_priority
    ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
  `).bind(
    emailData.id,
    metrics.timestamp,
    metrics.from_domain,
    metrics.to_domain,
    metrics.subject_length,
    metrics.body_length,
    metrics.has_attachments ? 1 : 0,
    metrics.attachment_count,
    metrics.spam_score,

    JSON.stringify(metrics.keywords),
    metrics.response_priority
  ).run();

  // 更新实时统计
  await updateRealTimeStats(metrics, env);

// 情感分析
async function analyzeSentiment(emailData, env) {
  const content = emailData.subject + ' ' + emailData.body;

  // 使用预训练的情感分析模型
  const sentimentModel = await getSentimentModel(env);
  const analysis = await sentimentModel.analyze(content);

  return {
    score: analysis.score, // -1 to 1
    label: analysis.label, // positive, negative, neutral
    confidence: analysis.confidence
  };
}

// 关键词提取
async function extractKeywords(emailData, env) {
  const content = (emailData.subject + ' ' + emailData.body).toLowerCase();

  // 移除停用词
  const stopWords = await env.STOP_WORDS_KV.get('en', 'json');

  // 分词和统计
  const words = content.split(/\W+/).filter(word =>
    word.length > 3 && !stopWords.includes(word)
  );

  const wordCount = {};
  words.forEach(word => {
    wordCount[word] = (wordCount[word] || 0) + 1;
  });

  // 返回前10个高频词
  return Object.entries(wordCount)
    .sort((a, b) => b[1] - a[1])
    .slice(0, 10)
    .map(([word, count]) => ({ word, count }));
}

// 生成每日报告
async function generateDailyReport(env) {
  const yesterday = new Date();
  yesterday.setDate(yesterday.getDate() - 1);
  const yesterdayStr = yesterday.toISOString().split('T')[0];

  // 获取昨日数据
  const dailyStats = await env.ANALYTICS_DB.prepare(`
    SELECT
      COUNT(*) as total_emails,
      AVG(body_length) as avg_body_length,
      AVG(spam_score) as avg_spam_score,
      COUNT(CASE WHEN has_attachments = 1 THEN 1 END) as emails_with_attachments,
      AVG(response_priority) as avg_priority
    FROM email_metrics
    WHERE DATE(timestamp) = ?
  `).bind(yesterdayStr).first();

  // 获取发件人统计
  const topSenders = await env.ANALYTICS_DB.prepare(`
    SELECT from_domain, COUNT(*) as count
    FROM email_metrics
    WHERE DATE(timestamp) = ?
    GROUP BY from_domain
    ORDER BY count DESC
    LIMIT 10
  `).bind(yesterdayStr).all();

  // 生成报告
  const report = {
    date: yesterdayStr,
    summary: dailyStats,
    top_senders: topSenders.results,
    trends: await calculateTrends(yesterdayStr, env),
    recommendations: await generateRecommendations(dailyStats, env)
  };

  // 发送报告
  await sendAnalyticsReport(report, 'daily', env);

  // 存储报告
  await env.REPORTS_BUCKET.put(
    `reports/daily/${yesterdayStr}.json`,
    JSON.stringify(report)
  );
}
