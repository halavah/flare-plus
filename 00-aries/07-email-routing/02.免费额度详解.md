// 邮件接收配额监控器
export class EmailQuotaMonitor {
  constructor(env) {
    this.env = env;
    this.dailyLimit = 1000;  // 日峰限制
    this.monthlyLimit = 10000; // 月度限制
    this.alertsEnabled = true;
  }

  // 检查是否可以接收邮件
  async canReceiveEmail() {
    const dailyUsage = await this.getDailyUsage();
    const monthlyUsage = await this.getMonthlyUsage();

    // 检查日限制
    if (dailyUsage >= this.dailyLimit) {
      await this.triggerAlert('daily_limit_exceeded', {
        usage: dailyUsage,
        limit: this.dailyLimit,
        resetTime: this.getDailyResetTime()
      });
      return false;
    }

    // 检查月度限制
    if (monthlyUsage >= this.monthlyLimit) {
      await this.triggerAlert('monthly_limit_exceeded', {
        usage: monthlyUsage,
        limit: this.monthlyLimit,
        resetTime: this.getMonthlyResetTime()
      });
      return false;
    }

    return true;
  }

  // 获取日使用量
  async getDailyUsage() {
    const today = new Date().toISOString().slice(0, 10);
    const key = `daily_usage:${today}`;

    const usage = await this.env.QUOTA_TRACKER.get(key);
    return usage ? parseInt(usage) : 0;
  }

  // 获取月度使用量
  async getMonthlyUsage() {
    const currentMonth = new Date().toISOString().slice(0, 7);
    const key = `monthly_usage:${currentMonth}`;

    const usage = await this.env.QUOTA_TRACKER.get(key);
    return usage ? parseInt(usage) : 0;
  }

  // 增加使用计数
  async incrementUsage() {
    const today = new Date().toISOString().slice(0, 10);
    const currentMonth = new Date().toISOString().slice(0, 7);

    // 更新日使用量
    const dailyKey = `daily_usage:${today}`;
    const currentDailyUsage = await this.env.QUOTA_TRACKER.get(dailyKey);
    const newDailyUsage = currentDailyUsage ? parseInt(currentDailyUsage) + 1 : 1;

    await this.env.QUOTA_TRACKER.put(dailyKey, newDailyUsage.toString(), {
      expirationTtl: 2 * 24 * 60 * 60 // 2天
    });

    // 更新月度使用量
    const monthlyKey = `monthly_usage:${currentMonth}`;
    const currentMonthlyUsage = await this.env.QUOTA_TRACKER.getMonthlyKey);
    const newMonthlyUsage = currentMonthlyUsage ? parseInt(currentMonthlyUsage) + 1 : 1;

    await this.env.QUOTA_TRACKER.put(monthlyKey, newMonthlyUsage.toString(), {
      expirationTtl: 32 * 24 * 60 * 60 // 32天
    });

    // 检查并通知告警
    await this.checkUsageAlerts(newDailyUsage, newMonthlyUsage);

    return { daily: newDailyUsage, monthly: newMonthlyUsage };
  }

  // 检查使用量告警
  async checkUsageAlerts(dailyUsage, monthlyUsage) {
    const dailyPercent = (dailyUsage / this.dailyLimit) * 100;
    const monthlyPercent = (monthlyUsage / this.monthlyLimit) * 100;

    // 日用量警告
    if (dailyPercent >= 80) {
      await this.triggerAlert('daily_usage_warning', {
        percentage: dailyPercent,
        usage: dailyUsage,
        limit: this.dailyLimit
      });
    }

    // 月度用量警告
    if (monthlyPercent >= 80) {
      await this.triggerAlert('monthly_usage_warning', {
        percentage: monthlyPercent,
        usage: monthlyUsage,
        limit: this.monthlyLimit
      });
    }

    // 接近限制的严重警告
    if (dailyPercent >= 95 || monthlyPercent >= 95) {
      await this.triggerAlert('usage_critical', {
        daily: {
          percentage: dailyPercent,
          usage: dailyUsage,
          limit: this.dailyLimit
        },
        monthly: {
          percentage: monthlyPercent,
          usage: monthlyUsage,
          limit: this.monthlyLimit
        }
      });
    }
  }

  // 获取日重置时间
  getDailyResetTime() {
    const tomorrow = new Date();
    tomorrow.setDate(tomorrow.getDate() + 1);
    tomorrow.setHours(0, 0, 0, 0);

    return tomorrow.toISOString();
  }

  // 获取月度重置时间
  getMonthlyResetTime() {
    const nextMonth = new Date();
    nextMonth.setMonth(nextMonth.getMonth() + 1);
    nextMonth.setDate(1);
    nextMonth.setHours(0, 0, 0, 0);

    return nextMonth.toISOString();
  }

  // 触发告警
  async triggerAlert(type, data) {
    if (!this.alertsEnabled) return;

    const alert = {
      type: type,
      data: data,
      timestamp: Date.now(),
      id: crypto.randomUUID()
    };

    // 保存告警
    const alertKey = `alert:${alert.id}`;
    await this.env.ALERTS.put(alertKey, alert, {
      expirationTtl: 7 * 24 * 60 * 60 // 7天
    });

    // 可以集成外部告警系统
    await this.sendExternalAlert(alert);
  }

  // 发送外部告警
  async sendExternalAlert(alert) {
    // 示例：发送到 Slack
    if (this.env.SLACK_WEBHOOK) {
      const slackMessage = {
        text: `📧 Email Routing Alert: ${alert.type}`,
        attachments: [{
          color: alert.type.includes('critical') ? 'danger' : 'warning',
          fields: [
            {
              title: 'Type',
              value: alert.type,
              short: true
            },
            {
              title: 'Data',
              value: JSON.stringify(alert.data, null, 2),
              short: false
            },
            {
              title: 'Time',
              value: new Date(alert.timestamp).toISOString(),
              short: true
            }
          ]
        }]
      };

      try {
        await fetch(this.env.SLACK_WEBHOOK, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(slackMessage)
        });
      } catch (error) {
        console.error('Failed to send Slack alert:', error);
      }
    }

    // 示例：发送邮件通知
    if (this.env.ALERT_EMAIL) {
      await this.sendEmailAlert(alert);
    }
  }

Email Routing Alert: ${alert.type}

Time: ${new Date(alert.timestamp).toISOString()}

Data: ${JSON.stringify(alert.data, null, 2)}
      `
    };

    // 这里需要实现实际的邮件发送逻辑

  }

// 配额优化管理器
export class QuotaOptimizationManager {
  constructor(env) {
    this.env = env;
    this.optimizationStrategies = new Map();
    this.analytics = new EmailAnalytics(env);
  }

  // 注册优化策略
  registerStrategy(name, strategy) {
    this.optimizationStrategies.set(name, strategy);
  }

  // 优化邮件处理
  async optimizeEmail(message) {
    const optimizationResults = [];

    // 应用所有优化策略
    for (const [name, strategy] of this.optimizationStrategies) {
      try {
        const result = await strategy.optimize(message);
        if (result.optimized) {
          optimizationResults.push({
            strategy: name,
            originalSize: result.originalSize,
            optimizedSize: result.optimizedSize,
            savings: result.savings
          });
        }
      } catch (error) {
        console.error(`Optimization strategy ${name} failed:`, error);
      }
    }

    // 记录优化结果
    await this.recordOptimization(optimizationResults);

    return optimizationResults;
  }

  // 压缩优化
  async setupCompressionStrategy() {
    this.registerStrategy('compression', {
      optimize: async (message) => {
        const originalSize = this.calculateEmailSize(message);
        const compressedMessage = await this.compressMessage(message);
        const compressedSize = this.calculateEmailSize(compressedMessage);

        const savings = originalSize - compressedSize;
        const optimized = savings > 1024; // 只对节省超过1KB的邮件进行压缩

        return {
          optimized: optimized,
          originalSize: originalSize,
          optimizedSize: optimizedSize,
          savings: savings,
          compressedMessage: compressedMessage
        };
      }
    });
  }

  // 压缩邮件
  async compressMessage(message) {
    // 压缩文本内容
    if (message.text) {
      message.text = await this.compressText(message.text);
    }

    // 压缩 HTML 内容
    if (message.html) {
      message.html = await this.compressText(message.html);
    }

    // 压缩附件
    if (message.attachments) {
      message.attachments = await Promise.all(
        message.attachments.map(att => this.compressAttachment(att))
      );
    }

    return message;
  }

  // 压缩文本
  async compressText(text) {
    // 简单的文本压缩（实际应用中应使用更好的压缩算法）
    return text
      .replace(/\s+/g, ' ')
      .replace(/([a-z])\1+/gi, '$1')
      .trim();
  }

  // 压缩附件
  async compressAttachment(attachment) {
    // 检查是否需要压缩
    if (attachment.size < 1024) {
      return attachment; // 小附件不压缩
    }

    // 对文本类型附件进行压缩
    if (this.isTextAttachment(attachment)) {
      const compressedContent = await this.compressText(
        attachment.content.toString()
      );

      return {
        ...attachment,
        content: new TextEncoder().encode(compressedContent),
        compressed: true,
        originalSize: attachment.size,
        compressedSize: compressedContent.length
      };
    }

    return attachment;
  }

  // 检查是否为文本附件
  isTextAttachment(attachment) {
    const textTypes = [
      'text/plain',
      'text/html',
      'text/csv',
      'application/json',
      'application/xml',
      'application/javascript'
    ];

    return textTypes.includes(attachment.contentType);
  }

  // 批处理优化
  async setupBatchProcessingStrategy() {
    this.registerStrategy('batch', {
      optimize: async (message) => {
        // 检查是否可以批量处理
        if (this.canBatchProcess(message)) {
          return await this.addToBatch(message);
        }

        return {
          optimized: false,
          message: 'Cannot be batched'
        };
      }
    });
  }

  canBatchProcess(message) {
    // 检查邮件是否适合批量处理
    const batchableTypes = ['auto-reply', 'notification', 'newsletter'];
    const emailType = this.detectEmailType(message);

    return batchableTypes.includes(emailType);
  }

  detectEmailType(message) {
    const subject = (message.subject || '').toLowerCase();

    if (subject.includes('auto-reply') || subject.includes('out of office')) {
      return 'auto-reply';
    }

    if (subject.includes('notification') || subject.includes('alert')) {
      return 'notification';
    }

    if (subject.includes('newsletter') || subject.includes('marketing')) {
      return 'newsletter';
    }

    return 'normal';
  }

  async addToBatch(message) {
    const batchType = this.detectEmailType(message);
    const batchKey = `batch:${batchType}`;

    // 获取当前批次
    const batch = await this.env.EMAIL_BATCHES.get(batchKey) || {
      messages: [],
      createdAt: Date.now(),
      maxSize: 100
    };

    // 添加到批次
    batch.messages.push(message);

    // 批次已满或超时，处理批次
    const shouldProcess = batch.messages.length >= batch.maxSize ||
                           Date.now() - batch.createdAt > 30000; // 30秒超时

    if (shouldProcess) {
      await this.processBatch(batchType, batch);
      await this.env.EMAIL_BATCHES.delete(batchKey);
    } else {
      // 更新批次
      await this.env.EMAIL_BATCHES.put(batchKey, batch, {
        expirationTtl: 3600 // 1小时
      });
    }

    return {
      optimized: true,
      batched: true,
      batchId: batchKey,
      batchSize: batch.messages.length
    };
  }

  async processBatch(batchType, batch) {

        break;

    await this.analytics.recordBatchProcessed(batchType, batch.messages.length);
  }

  }

  async processNotificationBatch(messages) {

    // 可以使用队列系统处理
  }

  async processNewsletterBatch(messages) {
    // 批量处理邮件通讯邮件
    // 适合使用邮件服务提供商的批量API

    for (const message of messages) {
      await this.env.EMAIL_WORKER.fetch(

      date: new Date().toISOString().slice(0, 10),
      optimizations: results,
      totalSavings: totalSavings,
      timestamp: Date.now()

    await this.env.OPTIMIZATION_RECORDS.put(key, record, {
      expirationTtl: 30 * 24 * 60 * 60 // 30天
    });

  calculateEmailSize(message) {
    let size = 0;

    if (message.from) size += message.from.length * 2; // UTF-16
    if (message.to) size += Array.isArray(message.to) ? message.to.length * 2 : message.to.length * 2;
    if (message.cc) size += Array.isArray(message.cc) ? message.cc.length * 2 : message.cc.length * 2;
    if (message.subject) size += message.subject.length * 2;
    if (message.text) size += message.text.length * 2;
    if (message.html) size += message.html.length * 2;

    // 计算附件大小
    if (message.attachments) {
      for (const attachment of message.attachments) {
        size += attachment.size || 0;
      }
    }

    return size;
  }

  // 简单哈希函数
  hashString(str) {
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      const char = str.charCodeAt(i);
      hash = ((hash << 5) - hash) + char);
      hash = hash & hash;
    }
    return hash.toString();
  }

// 转发配额管理器
export class ForwardingQuotaManager {
  constructor(env) {
    this.env = env;
    this.forwardingRules = new Map();
    this.rateLimiters = new Map();
    this.globalLimit = {
      daily: 1000,
      monthly: 10000
    };
  }

  // 初始化转发规则
  async initialize() {
    await this.loadForwardingRules();
    await this.initializeRateLimiters();
  }

  // 加载转发规则
  async loadForwardingRules() {
    const rules = await this.env.FORWARDING_RULES.get('rules') || [];

    for (const rule of rules) {
      this.forwardingRules.set(rule.email, rule);
    }
  }

  // 初始化速率限制器
  async initializeRateLimiters() {
    // 为每个域名设置速率限制器
    const domains = new Set();
    for (const rule of this.forwardingRules.values()) {
      if (rule.targets) {
        rule.targets.forEach(target => {
          const domain = target.split('@')[1];
          if (domain) {
            domains.add(domain);
          }
        });
      }
    }

    for (const domain of domains) {
      this.rateLimiters.set(domain, {
        daily: new Map(),
        monthly: new Map()
      });
    }
  }

  // 检查是否可以转发
  async canForward(message, rule) {
    const { to } = message;

    // 检查全局限制
    if (!(await this.checkGlobalLimits())) {
      return {
        allowed: false,
        reason: 'global_limit_exceeded'
      };
    }

    // 检查规则限制
    if (!this.checkRuleLimits(rule)) {
      return {
        allowed: false,
        reason: 'rule_limit_exceeded'
      };
    }

    // 检查速率限制
    const domains = this.getTargetDomains(rule);
    for (const domain of domains) {
      if (!(await this.checkRateLimit(domain))) {
        return {
          allowed: false,
          reason: `rate_limit_exceeded:${domain}`
        };
      }
    }

    return { allowed: true };
  }

  // 检查全局限制
  async checkGlobalLimits() {
    const dailyUsage = await this.getGlobalDailyUsage();
    const monthlyUsage = await this.getGlobalMonthlyUsage();

    return dailyUsage < this.globalLimit.daily &&
           monthlyUsage < this.globalLimit.monthly;
  }

  // 检查规则限制
  checkRuleLimits(rule) {
    // 检查每个目标的限制
    if (rule.targets) {
      for (const target of rule.targets) {
        const dailyLimit = rule.dailyLimit || 100;
        const dailyUsage = this.getDailyUsage(target);

        if (dailyUsage >= dailyLimit) {
          return false;
        }
      }
    }

    return true;
  }

  // 获取目标域名
  getTargetDomains(rule) {
    const domains = new Set();

    if (rule.targets) {
      for (const target of rule.targets) {
        const domain = target.split('@')[1];
        if (domain) {
          domains.add(domain);
        }
      }
    }

    return Array.from(domains);
  }

  // 检查速率限制
  async checkRateLimit(domain) {
    const dailyUsage = this.getDailyDomainUsage(domain);
    const monthlyUsage = this.getMonthlyDomainUsage(domain);

    const rateLimiter = this.rateLimiters.get(domain);
    if (!rateLimiter) {
      return true; // 没有限制器时允许
    }

    const limits = {
      daily: rateLimiter.daily.get('limit') || 100,
      monthly: rateLimiter.monthly.get('limit') || 1000
    };

    return dailyUsage < limits.daily && monthlyUsage < limits.monthly;
  }

  // 获取全局日使用量
  async getGlobalDailyUsage() {
    const today = new Date().toISOString().slice(0, 10);
    const key = `global_daily_usage:${today}`;

    const usage = await this.env.FORWARDING_STATS.get(key);
    return usage ? parseInt(usage) : 0;
  }

  // 获取全局月使用量
  async getGlobalMonthlyUsage() {
    const currentMonth = new Date().toISOString().slice(0, 7);
    const key = `global_monthly_usage:${currentMonth}`;

    const usage = await this.env.FORWARDING_STATS.get(key);
    return usage ? parseInt(usage) : 0;
  }

  // 获取目标日使用量
  getDailyUsage(target) {
    const today = new Date().toISOString().slice(0, 10);
    const key = `daily_usage:${target}:${today}`;

    const usage = this.env.FORWARDING_STATS.get(key);
    return usage ? parseInt(usage) : 0;
  }

  // 获取目标月使用量
  getMonthlyUsage(target) {
    const currentMonth = new Date().toISOString().slice(0, 7);
    const key = `monthly_usage:${target}:${currentMonth}`;

    const usage = this.env.FORWARDING_STATS.get(key);
    return usage ? parseInt(usage) : 0;
  }

  // 获取域名日使用量
  getDailyDomainUsage(domain) {
    const today = new Date().toISOString().slice(0, 10);
    const key = `domain_daily_usage:${domain}:${today}`;

    const usage = this.rateLimiters.get(domain)?.daily.get(Date.now()) || 0;
    return usage;
  }

  // 获取域名月使用量
  getMonthlyDomainUsage(domain) {
    const currentMonth = new Date().toISOString().slice(0, 7);

    const rateLimiter = this.rateLimiters.get(domain);
    if (!rateLimiter) return 0;

    const monthlyUsage = rateLimiter.monthly.get(Date.now()) || 0;
    return monthlyUsage;
  }

  // 增加使用统计
  async incrementUsage(target) {
    const today = new Date().toISOString().slice(0, 10);
    const currentMonth = new Date().toISOString().slice(0, 7);
    const domain = target.split('@')[1];

    // 更新全局统计
    await this.updateGlobalDailyUsage();
    await this.updateGlobalMonthlyUsage();

    // 更新目标统计
    await this.updateTargetUsage(target);

    // 更新域名统计
    if (domain) {
      await this.updateDomainUsage(domain);
    }

    return {
      daily: await this.getGlobalDailyUsage(),
      monthly: await this.getGlobalMonthlyUsage()
    };
  }

  // 更新全局日统计
  async updateGlobalDailyUsage() {
    const today = new Date().toISOString().slice(0, 10);
    const key = `global_daily_usage:${today}`;

    const currentUsage = await this.env.FORWARDING_STATS.get(key);
    const newUsage = currentUsage ? parseInt(currentUsage) + 1 : 1;

    await this.env.FORWARDING_STATS.put(key, newUsage.toString(), {
      expirationTtl: 2 * 24 * 60 * 60
    });
  }

  // 更新全局月度统计
  async updateGlobalMonthlyUsage() {
    const currentMonth = new Date().toISOString().slice(0, 7);
    const key = `global_monthly_usage:${currentMonth}`;

    const currentUsage = await this.env.FORWARDING_STATS.get(key);
    const newUsage = currentUsage ? parseInt(currentUsage) + 1 : 1;

    await this.env.FORWARDING_STATS.put(key, newUsage.toString(), {
      expirationTtl: 32 * 24 * 60 * 60
    });
  }

  // 更新目标使用统计
  async updateTargetUsage(target) {
    const today = new Date().toISOString().slice(0, 10);
    const currentMonth = new Date().toISOString().slice(0, 7);
    const domain = target.split('@')[1];

    // 更新日统计
    const dailyKey = `daily_usage:${target}:${today}`;
    const currentDailyUsage = await this.env.FORWARDING_STATS.get(dailyKey);
    const newDailyUsage = currentDailyUsage ? parseInt(currentDailyUsage) + 1 : 1;

    await this.env.FORWARDING_STATS.put(dailyKey, newDailyUsage.toString(), {
      expirationTtl: 2 * 24 * 60 * 60
    });

    // 更新月度统计
    const monthlyKey = `monthly_usage:${target}:${currentMonth}`;
    const currentMonthlyUsage = await this.env.FORWARDING_STATS.get(monthlyKey);
    const newMonthlyUsage = currentMonthlyUsage ? parseInt(currentMonthlyUsage) + 1 : 1;

    await this.env.FORWARDING_STATS.put(monthlyKey, newMonthlyUsage.toString(), {
      expirationTtl: 32 * 24 * 60 * 60
    });

    // 更新域名统计
    if (domain) {
      await this.updateDomainUsage(domain);
    }
  }

  // 更新域名使用统计
  async updateDomainUsage(domain) {
    const rateLimiter = this.rateLimiters.get(domain);
    if (!rateLimiter) {
      return;
    }

    const now = Date.now();
    const today = new Date(now.getFullYear()).toString().slice(0, 5);

    // 更新日统计
    const currentDailyCount = rateLimiter.daily.get(now) || 0;
    rateLimiter.daily.set(now, currentDailyCount + 1);

    // 更新月度统计
    const currentMonthlyCount = rateLimiter.monthly.get(now) || 0;
    rateLimiter.monthly.set(now, currentMonthlyCount + 1);

    // 清理旧数据（保留31天）
    this.cleanupOldUsageData(domain);
  }

  // 清理旧的使用数据
  cleanupOldUsageData(domain) {
    const rateLimiter = this.rateLimiters.get(domain);
    if (!rateLimiter) return;

    const now = Date.now();
    const cutoffTime = now - (31 * 24 * 60 * 60 * 1000); // 31天前

    // 清理日统计
    for (const [timestamp] of rateLimiter.daily.keys()) {
      if (parseInt(timestamp) < cutoffTime) {
        rateLimiter.daily.delete(parseInt(timestamp));
      }
    }

    // 清理月度统计
    const cutoffMonth = now - (31 * 24 * 60 * 60 * 1000); // 31天前
    const currentMonth = new Date(now.getFullYear(), now.getMonth()).toString().slice(0, 7);

    for (const [timestamp] of rateLimiter.monthly.keys()) {
      if (timestamp < currentMonth) {
        rateLimiter.monthly.delete(parseInt(timestamp));
      }
    }
  }
}
