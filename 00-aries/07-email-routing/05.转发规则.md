# Email Routing 转发规则

## 📧 转发规则概述

Email Routing 的转发规则允许你根据发件人、收件人、主题等条件自动将邮件转发到指定的邮箱地址。支持复杂的条件匹配、多目标转发和智能路由策略。

### 转发规则结构
```json

    }

      "type": "forward",

  "created_at": "2024-01-01T00:00:00Z"
}
```

### 1. 简单邮件转发

```javascript
// simple-forward.js
export default {
  async email(message, env, ctx) {
    const to = message.to;

    // 简单转发规则配置
    const forwardRules = {
      'contact@example.com': ['contact-forward@example.com'],
      'support@example.com': ['support-team@example.com', 'manager@example.com'],
      'sales@example.com': ['sales-department@example.com']
    };

    // 检查是否需要转发
    const forwardTo = forwardRules[to];
    if (forwardTo && forwardTo.length > 0) {
      console.log(`转发邮件到: ${forwardTo.join(', ')}`);

      // 并发转发到多个地址
      const forwardPromises = forwardTo.map(address =>
        message.forward(address)
      );

      await Promise.allSettled(forwardPromises);

      // 记录转发日志
      await logForward(message, forwardTo, env);
    }

    return new Response('Email processed', { status: 200 });
  }
};

// 记录转发日志
async function logForward(message, forwardTo, env) {
  await env.DB.prepare(`
    INSERT INTO forward_logs (from_email, to_email, forward_to, subject, forwarded_at)
    VALUES (?, ?, ?, ?, ?)
  `).bind(
    message.from,
    message.to,
    JSON.stringify(forwardTo),
    message.subject,
    new Date().toISOString()
  ).run();
}
```

```javascript
// conditional-forward.js
export default {
  async email(message, env, ctx) {
    const emailData = await extractEmailData(message);

    // 评估转发条件
    const conditions = await evaluateForwardConditions(emailData, env);

    if (conditions.shouldForward && conditions.destinations.length > 0) {
      // 应用转发规则
      await applyForwardRules(message, conditions, env);
    }

    return new Response('Email processed', { status: 200 });
  }
};

// 评估转发条件
async function evaluateForwardConditions(emailData, env) {
  const conditions = {
    shouldForward: false,
    destinations: [],
    reason: null
  };

  // 获取转发规则
  const rules = await getForwardRules(env);

  for (const rule of rules) {
    if (await matchRule(emailData, rule)) {
      conditions.shouldForward = true;
      conditions.destinations.push(...rule.actions[0].value);
      conditions.reason = `Matched rule: ${rule.name}`;
    }
  }

  // 去重目标地址
  conditions.destinations = [...new Set(conditions.destinations)];

  return conditions;
}

// 规则匹配
async function matchRule(emailData, rule) {
  if (!rule.enabled) return false;

  // 检查所有匹配条件
  for (const matcher of rule.matchers) {
    if (!(await matchCondition(emailData, matcher))) {
      return false;
    }
  }

  return true;
}

// 单个条件匹配
async function matchCondition(emailData, matcher) {
  const fieldValue = getField(emailData, matcher.field);

  switch (matcher.type) {
    case 'literal':
      return fieldValue.toLowerCase() === matcher.value.toLowerCase();

    case 'contains':
      return fieldValue.toLowerCase().includes(matcher.value.toLowerCase());

    case 'regex':
      const regex = new RegExp(matcher.value, 'i');
      return regex.test(fieldValue);

    case 'wildcard':
      return matchWildcard(fieldValue, matcher.value);

    case 'list':
      return matcher.value.includes(fieldValue.toLowerCase());

    default:
      return false;
  }
}

// 获取字段值
function getField(emailData, field) {
  const fieldMap = {
    'from': emailData.from,
    'to': emailData.to,
    'cc': emailData.cc || '',
    'bcc': emailData.bcc || '',
    'subject': emailData.subject,
    'body': emailData.body,
    'headers': JSON.stringify(emailData.headers)
  };

  return fieldMap[field] || '';
}

// 通配符匹配
function matchWildcard(text, pattern) {
  const regexPattern = pattern
    .replace(/\*/g, '.*')
    .replace(/\?/g, '.');

  const regex = new RegExp(`^${regexPattern}$`, 'i');
  return regex.test(text);
}
```

```javascript
// 动态计算转发目标
async function calculateForwardDestinations(emailData, env) {
  const destinations = [];

  // 基于时间路由
  const timeBasedDest = await getTimeBasedDestination(emailData, env);
  if (timeBasedDest) {
    destinations.push(timeBasedDest);
  }

  // 基于地理位置路由
  const geoBasedDest = await getGeoBasedDestination(emailData, env);
  if (geoBasedDest) {
    destinations.push(geoBasedDest);
  }

  // 基于负载均衡路由
  const loadBalanceDest = await getLoadBalancedDestination(emailData, env);
  if (loadBalanceDest) {
    destinations.push(loadBalanceDest);
  }

  // 基于部门路由
  const deptBasedDest = await getDepartmentDestination(emailData, env);
  if (deptBasedDest) {
    destinations.push(deptBasedDest);
  }

  return [...new Set(destinations)];
}

// 基于时间的路由
async function getTimeBasedDestination(emailData, env) {
  const now = new Date();
  const hour = now.getHours();
  const dayOfWeek = now.getDay();

  // 工作时间配置
  const workHours = { start: 9, end: 17 };
  const workDays = [1, 2, 3, 4, 5]; // 周一到周五

  const isWorkTime = workDays.includes(dayOfWeek) &&
                    hour >= workHours.start &&
                    hour < workHours.end;

  // 获取时间路由配置
  const timeRoutes = await env.DB.prepare(`
    SELECT destination FROM time_routes
    WHERE is_work_time = ?
    ORDER BY priority DESC
    LIMIT 1
  `).bind(isWorkTime ? 1 : 0).first();

  return timeRoutes?.destination;
}

// 基于地理位置的路由
async function getGeoBasedDestination(emailData, env) {
  try {
    // 解析发件人IP（如果有）
    const senderIP = emailData.headers['x-originating-ip'] ||
                     emailData.headers['x-real-ip'];

    if (!senderIP) {
      return null;
    }

    // 查询地理位置
    const geoResponse = await fetch(
      `http://ip-api.com/json/${senderIP}?fields=country,city`
    );
    const geoData = await geoResponse.json();

    if (geoData.status !== 'success') {
      return null;
    }

    // 查找地区路由规则
    const route = await env.DB.prepare(`
      SELECT destination FROM geo_routes
      WHERE country = ? OR (country = '*' AND city = ?)
      ORDER BY country DESC, priority DESC
      LIMIT 1
    `).bind(geoData.country, geoData.city).first();

    return route?.destination;

  } catch (error) {
    console.error('Geo routing error:', error);
    return null;
  }
}

// 基于负载均衡的路由
async function getLoadBalancedDestination(emailData, env) {
  // 获取可用目标
  const targets = await env.DB.prepare(`
    SELECT destination, current_load, max_load, last_used
    FROM load_balance_targets
    WHERE enabled = 1
    ORDER BY (current_load / max_load) ASC
  `).all();

  if (targets.results.length === 0) {
    return null;
  }

  // 选择负载最低的目标
  const selectedTarget = targets.results[0];

  // 更新负载统计
  await env.DB.prepare(`
    UPDATE load_balance_targets
    SET current_load = current_load + 1,
        last_used = CURRENT_TIMESTAMP
    WHERE destination = ?
  `).bind(selectedTarget.destination).run();

  return selectedTarget.destination;
}

// 基于部门的路由
async function getDepartmentDestination(emailData, env) {
  // 从主题或内容中识别部门
  const departmentKeywords = {
    '技术部': ['技术', '开发', 'bug', 'feature', '系统', '代码'],
    '销售部': ['销售', '报价', '订单', '价格', '合同', '商务'],
    '人事部': ['人事', '招聘', '面试', '薪资', '福利', '培训'],
    '财务部': ['财务', '发票', '付款', '报销', '预算', '账单']
  };

  const content = (emailData.subject + ' ' + emailData.body).toLowerCase();

  for (const [dept, keywords] of Object.entries(departmentKeywords)) {
    const matched = keywords.some(keyword => content.includes(keyword));

    if (matched) {
      // 获取部门邮箱
      const deptEmail = await env.DB.prepare(`
        SELECT email_address FROM department_emails
        WHERE department = ?
      `).bind(dept).first();

      if (deptEmail) {
        return deptEmail.email_address;
      }
    }
  }

  return null;
}
```

```javascript
// intelligent-router.js
export default {
  async email(message, env, ctx) {
    const emailData = await extractEmailData(message);

    // 分析邮件优先级
    const priority = await analyzeEmailPriority(emailData, env);

    // 智能路由决策

    await executeRoutingStrategy(message, routingDecision, env);

// 分析邮件优先级
async function analyzeEmailPriority(emailData, env) {
  let priority = 0; // 0-100, 越高越重要

  // 紧急关键词
  const urgentKeywords = ['紧急', 'urgent', 'asap', 'immediately'];
  const hasUrgent = urgentKeywords.some(keyword =>
    emailData.subject.toLowerCase().includes(keyword)
  );
  if (hasUrgent) priority += 40;

  // 发件人重要性
  const senderImportance = await getSenderImportance(emailData.from, env);
  priority += senderImportance;

  // 历史响应时间
  const avgResponseTime = await getAverageResponseTime(emailData.from, env);
  if (avgResponseTime < 60) priority += 20; // 快速响应者加分

  // 附件数量
  const attachmentCount = emailData.attachments?.length || 0;
  priority += Math.min(attachmentCount * 5, 15);

  return Math.min(priority, 100);
}

// 路由决策
async function makeRoutingDecision(emailData, priority, env) {
  const decision = {
    strategy: 'normal',
    destinations: [],
    delay: 0,
    notify: []
  };

  // 高优先级邮件
  if (priority > 70) {
    decision.strategy = 'high_priority';
    decision.destinations = await getHighPriorityDestinations(emailData, env);
    decision.notify = ['manager@example.com'];
  }
  // 中优先级邮件
  else if (priority > 40) {
    decision.strategy = 'normal_priority';
    decision.destinations = await getNormalDestinations(emailData, env);
  }
  // 低优先级邮件
  else {
    decision.strategy = 'low_priority';
    decision.destinations = await getLowPriorityDestinations(emailData, env);
    decision.delay = 300; // 延迟5分钟处理
  }

  return decision;
}

// 执行路由策略
async function executeRoutingStrategy(message, decision, env) {
  // 记录路由决策
  await logRoutingDecision(message, decision, env);

  // 延迟处理（如果需要）
  if (decision.delay > 0) {
    await new Promise(resolve => setTimeout(resolve, decision.delay));
  }

  // 执行转发
  if (decision.destinations.length > 0) {
    await forwardWithStrategy(message, decision.destinations, decision.strategy, env);
  }

  // 发送通知
  for (const notifyAddress of decision.notify) {
    await sendRoutingNotification(message, notifyAddress, decision, env);
  }
}

// 策略化转发
async function forwardWithStrategy(message, destinations, strategy, env) {
  switch (strategy) {
    case 'high_priority':
      // 立即转发到所有目标
      await Promise.all(destinations.map(addr => message.forward(addr)));
      break;

    case 'normal_priority':
      // 正常转发
      for (const addr of destinations) {
        await message.forward(addr);
      }
      break;

    case 'low_priority':
      // 批量转发（减少频率）
      const batches = chunkArray(destinations, 3);
      for (const batch of batches) {
        await Promise.all(batch.map(addr => message.forward(addr)));
        await new Promise(resolve => setTimeout(resolve, 1000));
      }
      break;
  }
}
```

```javascript
export default {
  async email(message, env, ctx) {

    const learnedPattern = await learnForwardPattern(emailData, env);

    // 执行转发
    if (adaptiveRule.action === 'forward') {
      await message.forward(adaptiveRule.destination);
    }

    // 更新学习模型

  }
};

// 学习转发模式
async function learnForwardPattern(emailData, env) {
  // 获取相似邮件的历史转发记录
  const similarEmails = await env.DB.prepare(`
    SELECT forward_to, action_result, response_time
    FROM forward_history
    WHERE
      ABS(LENGTH(subject) - ?) < 20 OR
      from_email LIKE ?
    ORDER BY created_at DESC
    LIMIT 10
  `).bind(
    emailData.subject.length,
    emailData.from.split('@')[1]
  ).all();

  // 分析成功模式
  const successfulPatterns = similarEmails.results.filter(
    email => email.action_result === 'success'
  );

  if (successfulPatterns.length > 0) {
    // 选择最常见的成功模式
    const patternCounts = {};
    for (const pattern of successfulPatterns) {
      const dest = pattern.forward_to;
      patternCounts[dest] = (patternCounts[dest] || 0) + 1;
    }

    const mostCommonPattern = Object.entries(patternCounts)
      .sort((a, b) => b[1] - a[1])[0];

    return {
      destination: mostCommonPattern[0],
      confidence: mostCommonPattern[1] / successfulPatterns.length
    };
  }

  return null;
}

// 生成自适应规则
async function generateAdaptiveRule(emailData, learnedPattern, env) {
  const rule = {
    action: 'ignore',
    destination: null,
    confidence: 0
  };

  // 如果有学习模式且置信度足够高
  if (learnedPattern && learnedPattern.confidence > 0.7) {
    rule.action = 'forward';
    rule.destination = learnedPattern.destination;
    rule.confidence = learnedPattern.confidence;
  } else {
    // 回退到规则引擎
    const ruleBasedDest = await getRuleBasedDestination(emailData, env);
    if (ruleBasedDest) {
      rule.action = 'forward';
      rule.destination = ruleBasedDest;
      rule.confidence = 0.5;
    }
  }

  return rule;
}
```

```javascript
// 创建转发规则
async function createRule(ruleData, env) {
  // 验证规则数据
  const validation = validateRuleData(ruleData);
  if (!validation.valid) {
    return new Response(
      JSON.stringify({ error: validation.error }),
      { status: 400, headers: { 'Content-Type': 'application/json' } }
    );
  }

  // 生成规则ID
  const ruleId = crypto.randomUUID();

  // 保存规则到数据库
  await env.DB.prepare(`
    INSERT INTO forward_rules (
      id, name, description, enabled, priority,
      matchers, actions, created_at, updated_at
    ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
  `).bind(
    ruleId,
    ruleData.name,
    ruleData.description || '',
    ruleData.enabled !== false,
    ruleData.priority || 1,
    JSON.stringify(ruleData.matchers),
    JSON.stringify(ruleData.actions),
    new Date().toISOString(),
    new Date().toISOString()
  ).run();

  // 更新缓存
  await updateRulesCache(env);

  // 返回创建的规则
  const createdRule = await env.DB.prepare(`
    SELECT * FROM forward_rules WHERE id = ?
  `).bind(ruleId).first();

  return new Response(
    JSON.stringify(createdRule),
    { status: 201, headers: { 'Content-Type': 'application/json' } }
  );
}
```

```javascript
// 分析规则性能
async function analyzeRulePerformance(env) {
  const now = new Date();
  const oneHourAgo = new Date(now.getTime() - 3600000);

  // 获取规则执行统计
  const stats = await env.DB.prepare(`
    SELECT
      rule_id,
      COUNT(*) as execution_count,
      COUNT(DISTINCT email_id) as unique_emails,
      AVG(execution_time) as avg_execution_time,
      MAX(execution_time) as max_execution_time,
      SUM(CASE WHEN matched = 1 THEN 1 ELSE 0 END) as match_count,
      SUM(CASE WHEN error = 1 THEN 1 ELSE 0 END) as error_count
    FROM rule_execution_logs
    WHERE executed_at BETWEEN ? AND ?
    GROUP BY rule_id
  `).bind(oneHourAgo.toISOString(), now.toISOString()).all();

  // 分析每个规则
  for (const stat of stats.results) {
    const analysis = await analyzeRuleStat(stat, env);

    // 更新规则性能指标
    await env.DB.prepare(`
      UPDATE forward_rules
      SET
        last_execution_at = CURRENT_TIMESTAMP,
        execution_count = execution_count + ?,
        match_count = match_count + ?,
        error_count = error_count + ?,
        avg_execution_time = ?,
        performance_score = ?
      WHERE id = ?
    `).bind(
      stat.execution_count,
      stat.match_count,
      stat.error_count,
      stat.avg_execution_time,
      analysis.score,
      stat.rule_id
    ).run();

    // 性能告警
    if (analysis.alert) {
      await sendPerformanceAlert(stat, analysis, env);
    }
  }
}

// 分析单个规则统计
async function analyzeRuleStat(stat, env) {
  const analysis = {
    score: 100,
    alert: null
  };

  // 匹配率分析
  const matchRate = stat.match_count / stat.execution_count;
  if (matchRate < 0.01 && stat.execution_count > 100) {
    analysis.score -= 20;
    analysis.alert = 'Very low match rate - rule may be too specific';
  }

  // 执行时间分析
  if (stat.avg_execution_time > 50) {
    analysis.score -= 15;
    analysis.alert = 'Slow execution time - rule may need optimization';
  }

  // 错误率分析
  const errorRate = stat.error_count / stat.execution_count;
  if (errorRate > 0.1) {
    analysis.score -= 30;
    analysis.alert = 'High error rate - rule may be broken';
  }

  return analysis;
}
```
