// 意图识别
async function recognizeIntentAI(emailData, env) {
  const content = `${emailData.subject} ${emailData.body}`;

  // 预定义的意图模式
  const intents = {
    'inquiry': ['询问', '咨询', '想了解', '请问', 'help', 'question'],
    'complaint': ['投诉', '不满', '问题', 'complaint', 'issue', 'problem'],
    'purchase': ['购买', '订购', '价格', 'buy', 'purchase', 'order'],
    'support': ['支持', '帮助', '技术', 'support', 'technical', 'assistance'],
    'cancellation': ['取消', '退订', 'cancel', 'unsubscribe', 'terminate'],
    'compliment': ['感谢', '赞', '好', 'thanks', 'great', 'excellent']
  };

  const scores = {};
  for (const [intent, keywords] of Object.entries(intents)) {
    scores[intent] = keywords.filter(keyword =>
      content.toLowerCase().includes(keyword.toLowerCase())
    ).length / keywords.length;
  }

  const bestIntent = Object.entries(scores)
    .sort((a, b) => b[1] - a[1])[0];

  return {
    intent: bestIntent[0],
    confidence: bestIntent[1],
    allScores: scores
  };
}

async function shouldAutoReply(emailData, env) {
  // 检查发件人历史
  const senderHistory = await getSenderHistory(emailData.from, env);

  // 自动回复条件
  const conditions = [
    !isAutoReply(emailData), // 不是自动回复邮件
    !isMarketingEmail(emailData), // 不是营销邮件
    senderHistory.totalEmails < 10, // 新发件人
    emailData.subject.includes('问'), // 包含疑问
    hasQuickAnswer(emailData) // 有快速答案
  ];

  return conditions.filter(c => c).length >= 3;
}

async function searchKnowledgeBase(keyInfo, env) {
  // 使用向量搜索
  const queryVector = await generateEmbedding(keyInfo.question, env);

    body: JSON.stringify({

      topK: 3,
      includeMetadata: true

async function generateReplyDraft(emailData, knowledgeBase, env) {

    Subject: ${emailData.subject}
    From: ${emailData.from}
    Content: ${emailData.body}

    Knowledge Base:
    ${knowledgeBase.map(kb => `- ${kb.text}`).join('\n')}
  `;

  const response = await fetch('https://api.openai.com/v1/completions', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${env.OPENAI_API_KEY}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      model: 'text-davinci-003',
      prompt: `Generate a helpful email reply:\n\n${context}\n\nReply:`,
      max_tokens: 500,
      temperature: 0.7
    })
  });

  const result = await response.json();
  return result.choices[0].text.trim();
}

async function assignTicket(ticketId, emailData, env) {

    .sort((a, b) => b.score - a.score)[0];

          assignee_id: bestAgent.agent.id

    // 通知客服
    await notifyAgent(bestAgent.agent, ticketId, emailData, env);
  }
}

// 计算客服匹配分数
async function calculateAgentScore(agent, emailData, env) {
  let score = 0;

  // 技能匹配
  const skillsMatch = await matchSkills(agent.skills, emailData, env);
  score += skillsMatch * 0.4;

  // 当前负载
  const loadScore = (10 - agent.current_load) / 10;
  score += loadScore * 0.3;

  // 历史处理时间
  const avgResponseTime = await getAgentAvgResponseTime(agent.id, env);
  const responseTimeScore = Math.max(0, (24 - avgResponseTime) / 24);
  score += responseTimeScore * 0.2;

  // 语言匹配
  const languageMatch = await matchLanguage(agent.languages, emailData, env);
  score += languageMatch * 0.1;

  return score;
}

// streaming-processor.js
export default {
  async email(message, env, ctx) {
    // 创建数据流
    const emailStream = createEmailStream(message);

    // 实时处理流
    const processedStream = emailStream.pipeThrough(
      new TransformStream({
        transform: async (chunk, controller) => {
          // 实时分析每个chunk
          const analysis = await analyzeChunk(chunk, env);

          // 发送到实时处理管道
          await sendToProcessingPipeline(analysis, env);

          controller.enqueue(chunk);
        }
      })
    );

    // 存储处理后的流
    await storeStream(processedStream, env);

    return new Response('Stream processing complete', { status: 200 });
  }
};

// 创建邮件流
function createEmailStream(message) {
  const chunks = [];
  let offset = 0;
  const chunkSize = 1024; // 1KB chunks

  return new ReadableStream({
    async pull(controller) {
      if (offset >= message.raw.size) {
        controller.close();
        return;
      }

      const chunk = message.raw.slice(offset, offset + chunkSize);
      chunks.push(chunk);
      offset += chunkSize;

      controller.enqueue(chunk);
    }
  });
}

// 分析数据块
async function analyzeChunk(chunk, env) {
  const text = new TextDecoder().decode(chunk);

  return {
    timestamp: Date.now(),
    size: chunk.byteLength,
    hash: await hashChunk(chunk),
    patterns: await detectPatterns(text),
    anomalies: await detectAnomalies(text, env),
    sentiment: await analyzeChunkSentiment(text),
    keywords: await extractChunkKeywords(text)
  };
}

// 发送到处理管道
async function sendToProcessingPipeline(analysis, env) {

    method: 'POST',
    headers: {

      'Authorization': `Bearer ${env.ANALYTICS_TOKEN}`
    },

      type: 'email_chunk_processed',
      data: analysis
    })
  });

  // 发送到 Durable Object 进行实时处理
  const DO_ID = env.EMAIL_PROCESSOR_DO.idFromName('realtime');
  const DO = env.EMAIL_PROCESSOR_DO.get(DO_ID);

    method: 'POST',

}

// Durable Object 实时处理器
export class EmailProcessorDO {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.metrics = new Map();
  }

  async fetch(request) {
    const data = await request.json();

    switch (new URL(request.url).pathname) {
      case '/process':
        return this.processRealtime(data);
      case '/metrics':
        return this.getMetrics();
      case '/reset':
        return this.resetMetrics();
      default:
        return new Response('Not Found', { status: 404 });
    }
  }

  async processRealtime(data) {
    // 更新实时指标
    this.updateMetrics(data);

    // 检查阈值
    const alerts = this.checkThresholds(data);

    // 触发告警
    if (alerts.length > 0) {
      await this.triggerAlerts(alerts);
    }

    // 保存到时序数据库
    await this.saveTimeSeries(data);

    return new Response('OK');
  }

  updateMetrics(data) {
    const key = `${data.timestamp - (data.timestamp % 60000)}`; // 按分钟聚合

    if (!this.metrics.has(key)) {
      this.metrics.set(key, {
        count: 0,
        totalSize: 0,
        anomalies: 0,
        sentimentSum: 0
      });
    }

    const metric = this.metrics.get(key);
    metric.count++;
    metric.totalSize += data.size;
    if (data.anomalies.length > 0) metric.anomalies++;
    if (data.sentiment) metric.sentimentSum += data.sentiment.score;
  }

  checkThresholds(data) {
    const alerts = [];

    // 异常检测
    if (data.anomalies.length > 0) {
      alerts.push({
        type: 'anomaly_detected',
        severity: 'medium',
        message: `Anomalies detected: ${data.anomalies.length}`,
        data: data.anomalies
      });
    }

    // 大小异常
    if (data.size > 1024 * 1024) { // 1MB
      alerts.push({
        type: 'large_chunk',
        severity: 'low',
        message: `Large email chunk: ${data.size} bytes`
      });
    }

    // 负面情绪
    if (data.sentiment && data.sentiment.score < -0.7) {
      alerts.push({
        type: 'negative_sentiment',
        severity: 'high',
        message: 'Strong negative sentiment detected'
      });
    }

    return alerts;
  }
}

// 实时分析
async function realTimeAnalysis(emailData, env) {
  const analysis = {
    timestamp: Date.now(),
    emailId: emailData.id,
    from: emailData.from,
    subject: emailData.subject,
    urgency: await calculateUrgency(emailData, env),
    category: await quickCategorize(emailData, env),

    keywords: []

  for (const pattern of urgentPatterns) {
    if (content.includes(pattern.toLowerCase())) {
      analysis.requiresAttention = true;
      analysis.urgency = Math.max(analysis.urgency, 8);
      analysis.keywords.push(pattern);
    }
  }

  return analysis;
}

export class NotificationWebSocket {
  constructor(state, env) {

    this.env = env;
    this.sockets = new Map();
  }

  async fetch(request) {

    const [client, server] = pair;

    await this.state.acceptWebSocket(server);

    // 添加到连接池
    const socketId = crypto.randomUUID();
    this.sockets.set(socketId, server);

    // 设置消息处理
    server.addEventListener('message', async (event) => {
      const message = JSON.parse(event.data);
      await this.handleClientMessage(socketId, message);
    });

    server.addEventListener('close', () => {
      this.sockets.delete(socketId);
    });

    return new Response(null, {
      status: 101,
      webSocket: client
    });
  }

  async handleBroadcast(request) {
    const data = await request.json();

    // 广播给所有连接的客户端
    const message = JSON.stringify(data);

    for (const [id, socket] of this.sockets) {
      try {
        socket.send(message);
      } catch (error) {
        // 连接已关闭，移除
        this.sockets.delete(id);
      }
    }

    return new Response('Broadcast complete');
  }

  async handleClientMessage(socketId, message) {
    switch (message.type) {
      case 'subscribe':
        await this.handleSubscribe(socketId, message.filters);
        break;

      case 'unsubscribe':
        await this.handleUnsubscribe(socketId, message.filters);
        break;

      case 'ping':
        // 发送pong响应
        const socket = this.sockets.get(socketId);
        if (socket) {
          socket.send(JSON.stringify({ type: 'pong' }));
        }
        break;
    }
  }

  async handleSubscribe(socketId, filters) {
    // 存储订阅过滤条件
    await this.state.storage.put(`subscription:${socketId}`, filters);
  }

  async handleUnsubscribe(socketId, filters) {
    // 移除特定订阅
    const subscription = await this.state.storage.get(`subscription:${socketId}`);
    if (subscription) {
      // 更新过滤条件
      const updated = subscription.filter(f => !filters.includes(f));
      await this.state.storage.put(`subscription:${socketId}`, updated);
    }
  }
}

  const pendingDeployments = await env.DB.prepare(`

    extensions: {},
    system: {}
  };

  // 收集扩展指标
  const extensions = await env.DB.prepare(`
    SELECT id, name FROM extensions WHERE active = 1
  `).all();

    memory_usage: await getMetric('memory_usage', env)
  };

// 检测异常
async function detectAnomalies(metrics, env) {
  const anomalies = [];

  // 性能异常检测
  for (const [extId, extMetrics] of Object.entries(metrics.extensions)) {
    // 响应时间异常
    if (extMetrics.avg_response_time > 5000) { // 5秒
      anomalies.push({
        type: 'performance',
        severity: 'high',
        extension: extId,
        metric: 'response_time',
        value: extMetrics.avg_response_time,
        threshold: 5000
      });
    }

    // 错误率异常
    if (extMetrics.error_rate > 0.05) { // 5%
      anomalies.push({
        type: 'reliability',
        severity: 'medium',
        extension: extId,
        metric: 'error_rate',
        value: extMetrics.error_rate,
        threshold: 0.05
      });
    }

    // 内存使用异常
    if (extMetrics.memory_usage > 128 * 1024 * 1024) { // 128MB
      anomalies.push({
        type: 'resource',
        severity: 'high',
        extension: extId,
        metric: 'memory_usage',
        value: extMetrics.memory_usage,
        threshold: 128 * 1024 * 1024
      });
    }
  }

  // 系统级异常
  if (metrics.system.error_rate > 0.1) {
    anomalies.push({
      type: 'system',
      severity: 'critical',
      metric: 'overall_error_rate',
      value: metrics.system.error_rate,
      threshold: 0.1
    });
  }

  return anomalies;
}
