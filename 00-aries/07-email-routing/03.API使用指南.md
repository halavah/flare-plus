# Email Routing API使用指南

## 🚀 API 概览

Cloudflare Email Routing API 提供了完整的邮件路由管理功能，支持规则配置、邮件处理和监控。

### API 端点
```
基础URL: https://api.cloudflare.com/client/v4/accounts/{account_id}
```

### 认证方式
```bash
# API Token 认证（推荐）
curl -X GET "https://api.cloudflare.com/client/v4/accounts" \
  -H "Authorization: Bearer YOUR_API_TOKEN"

# Global Key 认证
curl -X GET "https://api.cloudflare.com/client/v4/accounts" \
  -H "X-Auth-Email: your.email@example.com" \
  -H "X-Auth-Key: YOUR_GLOBAL_KEY"
```

## 📧 Workers API 集成

### 1. 邮件处理 Worker

```javascript
// email-processor.js
export default {
  async email(message, env, ctx) {
    try {
      // 获取邮件信息
      const from = message.from;
      const to = message.to;
      const subject = message.subject;
      const headers = Object.fromEntries(message.headers.entries());

      // 解析邮件内容
      let body = '';
      if (message.isPlainText()) {
        body = await message.text();
      } else if (message.isHtml()) {
        body = await message.html();
      }

      // 读取附件
      const attachments = [];
      for (const file of message.files) {
        const data = await file.arrayBuffer();
        attachments.push({
          name: file.name,
          type: file.type,
          size: data.byteLength,
          data: data
        });
      }

      // 处理逻辑
      const processed = await processEmail({
        from,
        to,
        subject,
        body,
        headers,
        attachments
      }, env);

      // 根据处理结果执行相应操作
      if (processed.forwardTo) {
        await message.forward(processed.forwardTo);
      }

      if (processed.store) {
        await storeEmail(processed.data, env);
      }

      if (processed.sendReply) {
        await sendAutoReply(processed.reply, message);
      }

      return new Response('Email processed successfully', { status: 200 });

    } catch (error) {
      console.error('Email processing error:', error);
      return new Response('Processing failed', { status: 500 });
    }
  }
};

// 邮件处理函数
async function processEmail(emailData, env) {
  const result = {
    forwardTo: null,
    store: true,
    data: emailData,
    sendReply: false,
    reply: null
  };

  // 垃圾邮件检测
  if (await isSpam(emailData, env)) {
    result.store = false;
    return result;
  }

  // 病毒扫描
  if (await hasVirus(emailData, env)) {
    result.store = false;
    return result;
  }

  // 自动回复
  if (emailData.subject.toLowerCase().includes('auto reply')) {
    result.sendReply = true;
    result.reply = {
      to: emailData.from,
      subject: 'Re: ' + emailData.subject,
      body: 'Thank you for your message. We will respond soon.',
      contentType: 'text/plain'
    };
  }

  // 转发规则
  const forwardedTo = await checkForwardingRules(emailData, env);
  if (forwardedTo) {
    result.forwardTo = forwardedTo;
  }

  return result;
}
```

### 2. 邮件存储 Worker

```javascript
// email-storage.js
export default {
  async email(message, env, ctx) {
    try {
      // 生成唯一ID
      const emailId = crypto.randomUUID();

      // 构建邮件对象
      const emailRecord = {
        id: emailId,
        from: message.from,
        to: message.to,
        subject: message.subject,
        receivedAt: new Date().toISOString(),
        size: message.raw.size,
        hasAttachments: message.files.length > 0,
        status: 'received'
      };

      // 存储到 R2
      await env.EMAIL_BUCKET.put(`emails/${emailId}/raw.eml`, message.raw);

      // 存储元数据到 D1
      await env.DB.prepare(`
        INSERT INTO emails (id, from_email, to_email, subject, received_at, size, has_attachments, status)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
      `).bind(
        emailRecord.id,
        emailRecord.from,
        emailRecord.to,
        emailRecord.subject,
        emailRecord.receivedAt,
        emailRecord.size,
        emailRecord.hasAttachments ? 1 : 0,
        emailRecord.status
      ).run();

      // 处理附件
      for (const file of message.files) {
        const attachmentId = crypto.randomUUID();
        await env.EMAIL_BUCKET.put(
          `emails/${emailId}/attachments/${attachmentId}`,
          await file.arrayBuffer()
        );

        await env.DB.prepare(`
          INSERT INTO email_attachments (id, email_id, name, type, size)
          VALUES (?, ?, ?, ?, ?)
        `).bind(
          attachmentId,
          emailId,
          file.name,
          file.type,
          file.size
        ).run();
      }

      // 更新配额
      await updateQuota(env);

      return new Response('Email stored successfully', { status: 200 });

    } catch (error) {
      console.error('Storage error:', error);
      return new Response('Storage failed', { status: 500 });
    }
  }
};

// 更新配额
async function updateQuota(env) {
  const today = new Date().toISOString().split('T')[0];

  await env.DB.prepare(`
    INSERT INTO daily_quota (date, emails_received, total_size)
    VALUES (?, 1, ?)
    ON CONFLICT(date) DO UPDATE SET
      emails_received = emails_received + 1,
      total_size = total_size + ?
  `).bind(today, 1024, 1024).run(); // 假设平均邮件大小为1KB
}
```

### 3. 邮件转发 Worker

```javascript
// email-forwarder.js
export default {
  async email(message, env, ctx) {
    try {
      // 获取转发配置
      const forwardConfig = await getForwardConfig(message.to, env);

      if (!forwardConfig) {
        console.log('No forwarding rule for:', message.to);
        return new Response('No forwarding rule', { status: 200 });
      }

      // 记录转发
      await logForward(message, forwardConfig, env);

      // 执行转发
      const forwardPromises = forwardConfig.forwardTo.map(address =>
        message.forward(address)
      );

      await Promise.allSettled(forwardPromises);

      return new Response('Email forwarded successfully', { status: 200 });

    } catch (error) {
      console.error('Forwarding error:', error);
      return new Response('Forwarding failed', { status: 500 });
    }
  }
};

// 获取转发配置
async function getForwardConfig(toEmail, env) {
  // 从 KV 或 D1 获取转发配置
  const result = await env.DB.prepare(`
    SELECT * FROM forwarding_rules WHERE email_address = ?
  `).bind(toEmail).first();

  return result;
}

// 记录转发
async function logForward(message, config, env) {
  await env.DB.prepare(`
    INSERT INTO forward_logs (from_email, to_email, forward_to, subject, forwarded_at)
    VALUES (?, ?, ?, ?, ?)
  `).bind(
    message.from,
    message.to,
    JSON.stringify(config.forwardTo),
    message.subject,
    new Date().toISOString()
  ).run();
}
```

## 🛠️ API 错误处理

### 1. 错误响应处理

```javascript
class EmailRoutingAPIError extends Error {
  constructor(message, code, details) {
    super(message);
    this.name = 'EmailRoutingAPIError';
    this.code = code;
    this.details = details;
  }
}

async function handleAPIResponse(response) {
  const data = await response.json();

  if (!data.success) {
    throw new EmailRoutingAPIError(
      data.errors?.[0]?.message || 'API request failed',
      data.errors?.[0]?.code || 'UNKNOWN_ERROR',
      data.errors?.[0] || {}
    );
  }

  return data.result;
}

// 使用示例
try {
  const rules = await handleAPIResponse(
    await fetch(apiUrl, {
      headers: { 'Authorization': `Bearer ${api_token}` }
    })
  );
  console.log('获取规则成功:', rules);
} catch (error) {
  if (error instanceof EmailRoutingAPIError) {
    console.error(`API错误 ${error.code}: ${error.message}`);
    // 根据错误码进行相应处理
    switch (error.code) {
      case '1000':
        console.log('认证失败，请检查API token');
        break;
      case '1001':
        console.log('权限不足');
        break;
      case '1002':
        console.log('请求频率过高');
        break;
      default:
        console.log('未知错误');
    }
  }
}
```

### 2. 重试机制

```javascript
async function retryAPICall(apiCall, maxRetries = 3, delay = 1000) {
  let lastError;

  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await apiCall();
    } catch (error) {
      lastError = error;

      // 判断是否可重试的错误
      if (!isRetryableError(error)) {
        throw error;
      }

      if (attempt === maxRetries) {
        throw error;
      }

      // 指数退避
      const backoffDelay = delay * Math.pow(2, attempt - 1);
      console.log(`第 ${attempt} 次重试失败，${backoffDelay}ms 后重试`);

      await new Promise(resolve => setTimeout(resolve, backoffDelay));
    }
  }

  throw lastError;
}

// 判断是否为可重试错误
function isRetryableError(error) {
  if (error instanceof EmailRoutingAPIError) {
    return [
      '1002', // 请求频率过高
      '1003', // 服务暂时不可用
      '1004'  // 网络超时
    ].includes(error.code);
  }

  if (error.name === 'TypeError' && error.message.includes('fetch')) {
    return true; // 网络错误
  }

  return false;
}
```

```javascript
// 批量创建规则的优化版本
async function createRulesOptimized(rules, batchSize = 10) {
  const results = [];

  // 分批处理
  for (let i = 0; i < rules.length; i += batchSize) {
    const batch = rules.slice(i, i + batchSize);

    // 并发创建当前批次
    const batchPromises = batch.map(async (rule, index) => {
      try {
        const result = await createRoutingRule(rule);
        return { index: i + index, success: true, data: result };
      } catch (error) {
        return { index: i + index, success: false, error };
      }
    });

    const batchResults = await Promise.all(batchPromises);
    results.push(...batchResults);

    // 添加延迟避免速率限制
    if (i + batchSize < rules.length) {
      await new Promise(resolve => setTimeout(resolve, 100));
    }
  }

  return results;
}
```

```javascript
// 带缓存的规则获取
class EmailRoutingCache {
  constructor(kvNamespace, ttl = 300) {
    this.kv = kvNamespace;
    this.ttl = ttl;
  }

  async getRules(accountId) {
    const cacheKey = `routing_rules:${accountId}`;

    // 尝试从缓存获取
    const cached = await this.kv.get(cacheKey, 'json');
    if (cached) {
      console.log('从缓存获取规则');
      return cached;
    }

    // 从API获取
    const rules = await getEmailRouting(accountId);

    // 存储到缓存
    await this.kv.put(cacheKey, JSON.stringify(rules), {
      expirationTtl: this.ttl
    });

    console.log('从API获取规则并缓存');
    return rules;
  }

  async invalidateCache(accountId) {
    const cacheKey = `routing_rules:${accountId}`;
    await this.kv.delete(cacheKey);
  }
}

// 使用示例
const cache = new EmailRoutingCache(env.ROUTING_CACHE);
```

```javascript
// API 调用监控
class APIMonitor {
  constructor(kvNamespace) {
    this.kv = kvNamespace;
  }

  async recordCall(endpoint, success, duration, error = null) {
    const timestamp = Date.now();
    const date = new Date().toISOString().split('T')[0];

    // 记录调用详情
    await this.kv.put(
      `api_calls:${timestamp}`,
      JSON.stringify({
        endpoint,
        success,
        duration,
        error: error?.message,
        timestamp: new Date().toISOString()
      }),
      { expirationTtl: 86400 * 7 } // 保留7天
    );

    // 更新统计
    await this.updateStats(date, endpoint, success, duration);
  }

  async updateStats(date, endpoint, success, duration) {
    const statsKey = `api_stats:${date}:${endpoint}`;

    const current = await this.kv.get(statsKey, 'json') || {
      total: 0,
      success: 0,
      errors: 0,
      totalDuration: 0
    };

    current.total++;
    current.totalDuration += duration;

    if (success) {
      current.success++;
    } else {
      current.errors++;
    }

    current.avgDuration = current.totalDuration / current.total;
    current.successRate = (current.success / current.total) * 100;

    await this.kv.put(statsKey, JSON.stringify(current), {
      expirationTtl: 86400 * 30 // 保留30天
    });
  }
}

// 使用示例
const monitor = new APIMonitor(env.MONITOR_KV);

const startTime = Date.now();
try {

  await monitor.recordCall('/email/routing/rules', true, Date.now() - startTime);
  return result;
} catch (error) {
  await monitor.recordCall('/email/routing/rules', false, Date.now() - startTime, error);
  throw error;
}
```
