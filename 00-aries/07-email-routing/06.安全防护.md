```javascript
// api-key-manager.js
export default {
  async fetch(request, env, ctx) {
    // 验证 API 密钥
    const apiKey = request.headers.get('X-API-Key');
    if (!apiKey) {
      return new Response('API key required', { status: 401 });
    }

    const keyValidation = await validateAPIKey(apiKey, env);
    if (!keyValidation.valid) {
      return new Response('Invalid API key', { status: 403 });
    }

    // 检查权限
    const requiredPermission = getRequiredPermission(request);
    if (!hasPermission(keyValidation.permissions, requiredPermission)) {
      return new Response('Insufficient permissions', { status: 403 });
    }

    // 记录 API 调用
    await logAPICall(request, keyValidation, env);

    // 继续处理请求
    return await handleRequest(request, env);
  }
};

// 验证 API 密钥
async function validateAPIKey(apiKey, env) {
  // 从缓存或数据库获取密钥信息
  let keyInfo = await env.API_KEYS_KV.get(apiKey, 'json');

  if (!keyInfo) {
    // 从数据库查询
    keyInfo = await env.DB.prepare(`
      SELECT * FROM api_keys
      WHERE key_hash = ? AND status = 'active'
    `).bind(await hashKey(apiKey)).first();

    if (keyInfo) {
      // 缓存密钥信息
      await env.API_KEYS_KV.put(apiKey, JSON.stringify(keyInfo), {
        expirationTtl: 300 // 5分钟缓存
      });
    }
  }

  if (!keyInfo) {
    return { valid: false };
  }

  // 检查密钥是否过期
  if (keyInfo.expires_at && new Date(keyInfo.expires_at) < new Date()) {
    return { valid: false };
  }

  // 检查使用限制
  if (await checkRateLimit(keyInfo.id, env)) {
    return { valid: false, reason: 'Rate limit exceeded' };
  }

  return {
    valid: true,
    id: keyInfo.id,
    name: keyInfo.name,
    permissions: keyInfo.permissions,
    rateLimit: keyInfo.rate_limit
  };
}

// 密钥哈希
async function hashKey(key) {
  const encoder = new TextEncoder();
  const data = encoder.encode(key + env.API_KEY_SALT);
  const hashBuffer = await crypto.subtle.digest('SHA-256', data);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
}

// 检查速率限制
async function checkRateLimit(keyId, env) {
  const now = Date.now();
  const window = 60000; // 1分钟窗口
  const windowStart = Math.floor(now / window) * window;

  const windowKey = `rate_limit:${keyId}:${windowStart}`;

  // 获取当前计数
  const currentCount = await env.RATE_LIMIT_KV.get(windowKey) || '0';

  // 增加计数
  const newCount = parseInt(currentCount) + 1;

  // 设置新的计数（带过期时间）
  await env.RATE_LIMIT_KV.put(windowKey, newCount.toString(), {
    expirationTtl: Math.ceil(window / 1000) + 10
  });

  // 获取密钥的速率限制
  const keyInfo = await env.DB.prepare(`
    SELECT rate_limit FROM api_keys WHERE id = ?
  `).bind(keyId).first();

  const rateLimit = keyInfo?.rate_limit || 100;

  return newCount > rateLimit;
}
```

### 2. JWT Token 认证

```javascript
// jwt-auth.js
export default {
  async fetch(request, env, ctx) {
    const authHeader = request.headers.get('Authorization');

    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return new Response('Authorization header required', { status: 401 });
    }

    const token = authHeader.substring(7);
    const tokenValidation = await validateJWT(token, env);

    if (!tokenValidation.valid) {
      return new Response('Invalid token', { status: 403 });
    }

    // 将用户信息附加到请求
    request.user = tokenValidation.payload;

    return await handleRequest(request, env);
  }
};

// 验证 JWT Token
async function validateJWT(token, env) {
  try {
    // 分割 Token
    const [headerB64, payloadB64, signatureB64] = token.split('.');

    // 解码 Header 和 Payload
    const header = JSON.parse(atob(headerB64));
    const payload = JSON.parse(atob(payloadB64));

    // 验证 Token 结构
    if (header.alg !== 'HS256') {
      return { valid: false, error: 'Unsupported algorithm' };
    }

    // 验证 Token 是否过期
    if (payload.exp && payload.exp < Math.floor(Date.now() / 1000)) {
      return { valid: false, error: 'Token expired' };
    }

    // 验证签名
    const expectedSignature = await sign(`${headerB64}.${payloadB64}`, env.JWT_SECRET);

    if (signatureB64 !== expectedSignature) {
      return { valid: false, error: 'Invalid signature' };
    }

    // 验证签发者
    if (payload.iss !== 'cloudflare-email-routing') {
      return { valid: false, error: 'Invalid issuer' };
    }

    return { valid: true, payload };

  } catch (error) {
    console.error('JWT validation error:', error);
    return { valid: false, error: 'Validation failed' };
  }
}

// HMAC-SHA256 签名
async function sign(data, secret) {
  const encoder = new TextEncoder();
  const key = await crypto.subtle.importKey(
    'raw',
    encoder.encode(secret),
    { name: 'HMAC', hash: 'SHA-256' },
    false,
    ['sign']
  );

  const signature = await crypto.subtle.sign('HMAC', key, encoder.encode(data));
  const signatureArray = Array.from(new Uint8Array(signature));
  return signatureArray.map(b => b.toString(16).padStart(2, '0')).join('');
}

// 生成 JWT Token
async function generateJWT(payload, env, expiresIn = '1h') {
  const header = {
    alg: 'HS256',
    typ: 'JWT'
  };

  // 设置过期时间
  const exp = Math.floor(Date.now() / 1000) + parseExpiration(expiresIn);

  const tokenPayload = {
    ...payload,
    iss: 'cloudflare-email-routing',
    iat: Math.floor(Date.now() / 1000),
    exp: exp
  };

  const headerB64 = btoa(JSON.stringify(header));
  const payloadB64 = btoa(JSON.stringify(tokenPayload));

  const signature = await sign(`${headerB64}.${payloadB64}`, env.JWT_SECRET);

  return `${headerB64}.${payloadB64}.${signature}`;
}

// 解析过期时间
function parseExpiration(expiresIn) {
  const units = {
    's': 1,
    'm': 60,
    'h': 3600,
    'd': 86400
  };

  const match = expiresIn.match(/^(\d+)([smhd])$/);
  if (!match) {
    throw new Error('Invalid expiration format');
  }

  const [, amount, unit] = match;
  return parseInt(amount) * units[unit];
}
```

```javascript
// 权限检查中间件
export class RBACMiddleware {
  constructor(env) {
    this.env = env;
  }

  async checkPermission(userId, requiredPermission) {
    // 获取用户角色
    const user = await this.env.DB.prepare(`
      SELECT role FROM users WHERE id = ?
    `).bind(userId).first();

    if (!user) {
      return false;
    }

    // 检查角色权限
    const rolePermissions = ROLES[user.role]?.permissions || [];
    return rolePermissions.includes(requiredPermission);
  }

  async checkResourceAccess(userId, resourceType, resourceId, action) {
    // 检查基础权限
    const permission = `${resourceType}.${action}`;
    const hasPermission = await this.checkPermission(userId, permission);

    if (!hasPermission) {
      return false;
    }

    // 检查资源级权限（如果需要）
    if (resourceType === 'rules') {
      return await this.checkRuleAccess(userId, resourceId, action);
    }

    return true;
  }

  async checkRuleAccess(userId, ruleId, action) {
    // 管理员可以访问所有规则
    const user = await this.env.DB.prepare(`
      SELECT role FROM users WHERE id = ?
    `).bind(userId).first();

    if (user?.role === 'admin') {
      return true;
    }

    // 检查规则所有者
    const rule = await this.env.DB.prepare(`
      SELECT created_by FROM forward_rules WHERE id = ?
    `).bind(ruleId).first();

    return rule?.created_by === userId;
  }
}

// 使用示例
export default {
  async fetch(request, env, ctx) {
    const rbac = new RBACMiddleware(env);
    const user = request.user;

    // 获取请求的资源信息
    const resourceInfo = getResourceInfo(request);

    // 检查权限
    const hasAccess = await rbac.checkResourceAccess(
      user.id,
      resourceInfo.type,
      resourceInfo.id,
      resourceInfo.action
    );

    if (!hasAccess) {
      return new Response('Access denied', { status: 403 });
    }

    return await handleRequest(request, env);
  }
};

// 解析资源信息
function getResourceInfo(request) {
  const url = new URL(request.url);
  const path = url.pathname;
  const method = request.method;

  // 解析路径：/api/rules/123 -> { type: 'rules', id: '123', action: 'read' }
  const match = path.match(/^\/api\/(\w+)\/?(\d+)?/);

  if (!match) {
    return null;
  }

  const [, resourceType, resourceId] = match;
  const actionMap = {
    'GET': 'read',
    'POST': 'create',
    'PUT': 'update',
    'DELETE': 'delete'
  };

  return {
    type: resourceType,
    id: resourceId,
    action: actionMap[method] || 'read'
  };
}
```

### 2. IP 白名单与黑名单

```javascript
// ip-filter.js
export default {
  async fetch(request, env, ctx) {
    const clientIP = request.headers.get('CF-Connecting-IP') ||
                     request.headers.get('X-Forwarded-For');

    // IP 白名单检查
    if (!await checkWhitelist(clientIP, env)) {
      await logBlockedIP(clientIP, 'not_in_whitelist', env);
      return new Response('IP not allowed', { status: 403 });
    }

    // IP 黑名单检查
    if (await checkBlacklist(clientIP, env)) {
      await logBlockedIP(clientIP, 'in_blacklist', env);
      return new Response('IP blocked', { status: 403 });
    }

    // 地理位置检查
    if (!await checkGeoLocation(clientIP, env)) {
      await logBlockedIP(clientIP, 'geo_blocked', env);
      return new Response('Geographic location not allowed', { status: 403 });
    }

    return await handleRequest(request, env);
  }
};

// 检查白名单
async function checkWhitelist(ip, env) {
  // 从缓存获取白名单
  let whitelist = await env.IP_WHITELIST_KV.get('whitelist', 'json');

  if (!whitelist) {
    // 从数据库获取
    const results = await env.DB.prepare(`
      SELECT ip_range FROM ip_whitelist WHERE enabled = 1
    `).all();

    whitelist = results.results.map(row => row.ip_range);

    // 缓存白名单
    await env.IP_WHITELIST_KV.put('whitelist', JSON.stringify(whitelist), {
      expirationTtl: 300
    });
  }

  // 如果白名单为空，允许所有IP
  if (whitelist.length === 0) {
    return true;
  }

  // 检查IP是否在白名单中
  for (const range of whitelist) {
    if (isIPInRange(ip, range)) {
      return true;
    }
  }

  return false;
}

// 检查黑名单
async function checkBlacklist(ip, env) {
  // 检查动态黑名单
  const dynamicBlocked = await env.DYNAMIC_BLACKLIST_KV.get(ip);
  if (dynamicBlocked) {
    return true;
  }

  // 检查静态黑名单
  const blacklist = await env.IP_BLACKLIST_KV.get('blacklist', 'json') || [];

  for (const range of blacklist) {
    if (isIPInRange(ip, range)) {
      return true;
    }
  }

  return false;
}

// IP范围检查
function isIPInRange(ip, range) {
  // 支持单个IP和CIDR格式
  if (range.includes('/')) {
    return isIPInCIDR(ip, range);
  } else {
    return ip === range;
  }
}

// CIDR 范围检查
function isIPInCIDR(ip, cidr) {
  const [network, prefixLength] = cidr.split('/');
  const ipInt = ipToInt(ip);
  const networkInt = ipToInt(network);
  const mask = (0xFFFFFFFF << (32 - parseInt(prefixLength))) >>> 0;

  return (ipInt & mask) === (networkInt & mask);
}

// IP转整数
function ipToInt(ip) {
  return ip.split('.').reduce((acc, octet) => (acc << 8) + parseInt(octet), 0) >>> 0;
}

// 地理位置检查
async function checkGeoLocation(ip, env) {
  try {
    // 获取允许的国家列表
    const allowedCountries = await env.GEO_RULES_KV.get('allowed_countries', 'json') || [];

    if (allowedCountries.length === 0) {
      return true; // 没有限制则允许
    }

    // 使用 Cloudflare 的地理位置数据
    const country = request.cf?.country;

    if (!country) {
      return false; // 无法确定地理位置
    }

    return allowedCountries.includes(country);
  } catch (error) {
    console.error('Geo check error:', error);
    return true; // 出错时允许
  }
}
```

### 2. 数据存储加密

```javascript
// encryption.js
export class EmailEncryption {
  constructor(env) {
    this.env = env;
    this.keyCache = new Map();
  }

  async generateKey() {
    return await crypto.subtle.generateKey(
      {
        name: 'AES-GCM',
        length: 256
      },
      true,
      ['encrypt', 'decrypt']
    );
  }

  async encryptEmail(emailContent, keyId = null) {
    try {
      // 获取加密密钥
      const key = await this.getEncryptionKey(keyId);

      // 生成随机 IV
      const iv = crypto.getRandomValues(new Uint8Array(12));

      // 加密内容
      const encoder = new TextEncoder();
      const data = encoder.encode(emailContent);

      const encryptedData = await crypto.subtle.encrypt(
        {
          name: 'AES-GCM',
          iv: iv
        },
        key,
        data
      );

      // 组合 IV 和加密数据
      const result = new Uint8Array(iv.length + encryptedData.byteLength);
      result.set(iv);
      result.set(new Uint8Array(encryptedData), iv.length);

      // 返回 Base64 编码的结果
      return btoa(String.fromCharCode(...result));

    } catch (error) {
      console.error('Encryption error:', error);
      throw new Error('Encryption failed');
    }
  }

  async decryptEmail(encryptedContent, keyId = null) {
    try {
      // 获取解密密钥
      const key = await this.getEncryptionKey(keyId);

      // 解码 Base64
      const encryptedData = Uint8Array.from(
        atob(encryptedContent),
        c => c.charCodeAt(0)
      );

      // 提取 IV 和加密数据
      const iv = encryptedData.slice(0, 12);
      const data = encryptedData.slice(12);

      // 解密数据
      const decryptedData = await crypto.subtle.decrypt(
        {
          name: 'AES-GCM',
          iv: iv
        },
        key,
        data
      );

      // 解码为文本
      const decoder = new TextDecoder();
      return decoder.decode(decryptedData);

    } catch (error) {
      console.error('Decryption error:', error);
      throw new Error('Decryption failed');
    }
  }

  async getEncryptionKey(keyId = null) {
    // 如果指定了密钥ID，使用该密钥
    if (keyId) {
      const cachedKey = this.keyCache.get(keyId);
      if (cachedKey) {
        return cachedKey;
      }

      // 从存储获取密钥
      const keyData = await this.env.ENCRYPTION_KEYS_KV.get(keyId);
      if (keyData) {
        const key = await crypto.subtle.importKey(
          'raw',
          Uint8Array.from(atob(keyData), c => c.charCodeAt(0)),
          'AES-GCM',
          false,
          ['encrypt', 'decrypt']
        );

        this.keyCache.set(keyId, key);
        return key;
      }
    }

    // 使用默认密钥
    const defaultKeyId = 'default_key';
    const cachedDefaultKey = this.keyCache.get(defaultKeyId);
    if (cachedDefaultKey) {
      return cachedDefaultKey;
    }

    // 生成或获取默认密钥
    let defaultKeyData = await this.env.ENCRYPTION_KEYS_KV.get(defaultKeyId);

    if (!defaultKeyData) {
      // 生成新的默认密钥
      const key = await this.generateKey();
      const exportedKey = await crypto.subtle.exportKey('raw', key);
      defaultKeyData = btoa(String.fromCharCode(...new Uint8Array(exportedKey)));

      // 存储密钥
      await this.env.ENCRYPTION_KEYS_KV.put(defaultKeyId, defaultKeyData);
    }

    const defaultKey = await crypto.subtle.importKey(
      'raw',
      Uint8Array.from(atob(defaultKeyData), c => c.charCodeAt(0)),
      'AES-GCM',
      false,
      ['encrypt', 'decrypt']
    );

    this.keyCache.set(defaultKeyId, defaultKey);
    return defaultKey;
  }

  async rotateKey(keyId) {
    // 生成新密钥
    const newKey = await this.generateKey();
    const exportedKey = await crypto.subtle.exportKey('raw', newKey);
    const keyData = btoa(String.fromCharCode(...new Uint8Array(exportedKey)));

    // 存储新密钥
    await this.env.ENCRYPTION_KEYS_KV.put(keyId, keyData);

    // 更新缓存
    this.keyCache.set(keyId, newKey);

    // 标记旧密钥（用于后续数据迁移）
    const oldKeyId = `${keyId}_old`;
    await this.env.ENCRYPTION_KEYS_KV.put(oldKeyId, 'deprecated', {
      expirationTtl: 86400 * 7 // 7天后删除
    });

    return newKey;
  }
}

// 邮件处理中的加密使用
export default {
  async email(message, env, ctx) {
    const encryption = new EmailEncryption(env);

    // 提取邮件内容
    const emailData = await extractEmailData(message);

    // 加密敏感内容

    const encryptedBody = await encryption.encryptEmail(emailData.body);

    // 存储加密的邮件
    await storeEncryptedEmail({

      from: emailData.from,
      to: emailData.to,
      encrypted_subject: encryptedSubject,
      encrypted_body: encryptedBody,
      received_at: new Date().toISOString()
    }, env);

    return new Response('Email stored securely', { status: 200 });
  }
```

```javascript
// 执行安全检查
async function performSecurityChecks(emailData, env) {
  const checks = {
    riskLevel: 0,
    violations: [],
    recommendations: []
  };

  // 检查发件人信誉
  const senderRep = await checkSenderReputation(emailData.from, env);
  if (senderRep.score < 30) {
    checks.riskLevel += 30;
    checks.violations.push('Low sender reputation');
  }

  // 检查邮件内容
  const contentRisk = await analyzeContentRisk(emailData, env);
  checks.riskLevel += contentRisk.score;
  checks.violations.push(...contentRisk.violations);

  // 检查附件安全
  const attachmentRisk = await analyzeAttachmentRisk(emailData.attachments, env);
  checks.riskLevel += attachmentRisk.score;
  checks.violations.push(...attachmentRisk.violations);

  // 检查发送频率
  const frequencyRisk = await checkSendFrequency(emailData.from, env);
  checks.riskLevel += frequencyRisk.score;
  checks.violations.push(...frequencyRisk.violations);

  return checks;
}

// 发件人信誉检查
async function checkSenderReputation(from, env) {
  const domain = from.split('@')[1];

  // 检查域名黑名单
  const blacklisted = await env.DOMAIN_BLACKLIST_KV.get(domain);
  if (blacklisted) {
    return { score: 0 };
  }

  // 计算历史评分
  const history = await env.DB.prepare(`
    SELECT
      COUNT(*) as total_emails,
      SUM(CASE WHEN marked_as_spam = 1 THEN 1 ELSE 0 END) as spam_count,
      SUM(CASE WHEN marked_as_phishing = 1 THEN 1 ELSE 0 END) as phishing_count
    FROM email_history
    WHERE from_domain = ?
    AND received_at > datetime('now', '-30 days')
  `).bind(domain).first();

  if (!history) {
    return { score: 70 }; // 新发件人，给予中等信任
  }

  const spamRate = history.spam_count / history.total_emails;
  const phishingRate = history.phishing_count / history.total_emails;

  let score = 70;
  score -= spamRate * 50;
  score -= phishingRate * 80;

  return Math.max(0, Math.min(100, score));
}

// 内容风险分析
async function analyzeContentRisk(emailData, env) {
  const result = {
    score: 0,
    violations: []
  };

  const content = (emailData.subject + ' ' + emailData.body).toLowerCase();

  // 检查钓鱼关键词
  const phishingKeywords = await env.PHISHING_KEYWORDS_KV.get('keywords', 'json') || [];
  const matchedPhishing = phishingKeywords.filter(keyword =>
    content.includes(keyword.toLowerCase())
  );

  if (matchedPhishing.length > 0) {
    result.score += 25;
    result.violations.push(`Phishing keywords: ${matchedPhishing.join(', ')}`);
  }

  // 检查恶意URL
  const urls = extractUrls(content);
  const maliciousUrls = [];

  for (const url of urls) {
    if (await isMaliciousUrl(url)) {
      maliciousUrls.push(url);
    }
  }

  if (maliciousUrls.length > 0) {
    result.score += 30;
    result.violations.push(`Malicious URLs detected`);
  }

  return result;
}

  }
}
```

```javascript
// threat-detection.js
export default {

// 威胁检测
async function detectThreats(env) {
  const now = new Date();
  const oneMinuteAgo = new Date(now.getTime() - 60000);

  // 检测异常模式
  const anomalies = await detectAnomalies(oneMinuteAgo, env);

  // 处理检测到的威胁
  for (const anomaly of anomalies) {
    await handleThreat(anomaly, env);
  }
}

// 检测异常
async function detectAnomalies(since, env) {
  const anomalies = [];

  // 检测1：异常发送频率
  const frequentSenders = await env.DB.prepare(`
    SELECT from_email, COUNT(*) as count
    FROM email_logs
    WHERE received_at > ?
    GROUP BY from_email
    HAVING count > 20
  `).bind(since.toISOString()).all();

  for (const sender of frequentSenders.results) {
    anomalies.push({
      type: 'high_frequency_sending',
      severity: 'medium',
      details: {
        email: sender.from_email,
        count: sender.count
      }
    });
  }

  // 检测2：大量失败转发
  const failedForwards = await env.DB.prepare(`
    SELECT COUNT(*) as count, error_type
    FROM forward_logs
    WHERE forwarded_at > ? AND status = 'failed'
    GROUP BY error_type
    HAVING count > 10
  `).bind(since.toISOString()).all();

  for (const failure of failedForwards.results) {
    anomalies.push({
      type: 'mass_forward_failure',
      severity: 'high',
      details: {
        errorType: failure.error_type,
        count: failure.count
      }
    });
  }

  // 检测3：异常IP访问
  const suspiciousIPs = await env.DB.prepare(`
    SELECT client_ip, COUNT(*) as count
    FROM api_logs
    WHERE request_time > ?
    GROUP BY client_ip
    HAVING count > 100
  `).bind(since.toISOString()).all();

  for (const ip of suspiciousIPs.results) {
    anomalies.push({
      type: 'suspicious_ip_activity',
      severity: 'high',
      details: {
        ip: ip.client_ip,
        count: ip.count
      }
    });
  }

  return anomalies;
}

// 处理威胁
async function handleThreat(threat, env) {
  // 记录威胁
  await env.DB.prepare(`
    INSERT INTO threat_logs (
      id, type, severity, details, detected_at
    ) VALUES (?, ?, ?, ?, ?)
  `).bind(
    crypto.randomUUID(),
    threat.type,
    threat.severity,
    JSON.stringify(threat.details),
    new Date().toISOString()
  ).run();

  // 根据威胁类型采取行动
  switch (threat.type) {
    case 'high_frequency_sending':
      await blockSender(threat.details.email, env);
      break;

    case 'mass_forward_failure':
      await investigateForwardFailure(threat.details, env);
      break;

    case 'suspicious_ip_activity':
      await blockIP(threat.details.ip, env);
      break;
  }

  // 发送告警
  if (threat.severity === 'high') {
    await sendThreatAlert(threat, env);
  }
}

// 临时阻止发件人
async function blockSender(email, env) {
  const blockDuration = 3600; // 1小时
  const blockUntil = new Date(Date.now() + blockDuration * 1000).toISOString();

  await env.BLOCKED_SENDERS_KV.put(email, blockUntil, {
    expirationTtl: blockDuration
  });
}

// 临时阻止IP
async function blockIP(ip, env) {
  const blockDuration = 7200; // 2小时
  const blockUntil = new Date(Date.now() + blockDuration * 1000).toISOString();

  await env.DYNAMIC_BLACKLIST_KV.put(ip, blockUntil, {
    expirationTtl: blockDuration
  });
}
```
