# R2 API 使用指南

## API 概览

Cloudflare R2 提供与 Amazon S3 完全兼容的 REST API，让你可以使用现有的 S3 工具和库直接与 R2 交互。

```javascript
// 列出所有对象
async function listObjects(bucketName, prefix = '', maxKeys = 1000) {
  const params = {
    Bucket: bucketName,
    Prefix: prefix,
    MaxKeys: maxKeys
  };

  try {
    const data = await s3.listObjectsV2(params).promise();
    return data;
  } catch (error) {
    console.error('Error listing objects:', error);
    throw error;
  }
}

// 递归列出所有对象
async function listAllObjects(bucketName, prefix = '') {
  const allObjects = [];
  let continuationToken = null;

  do {
    const params = {
      Bucket: bucketName,
      Prefix: prefix,
      MaxKeys: 1000
    };

    if (continuationToken) {
      params.ContinuationToken = continuationToken;
    }

    const data = await s3.listObjectsV2(params).promise();
    allObjects.push(...data.Contents);
    continuationToken = data.NextContinuationToken;
  } while (continuationToken);

  return allObjects;
}

// 分页列出
async function listObjectsWithPagination(bucketName, pageSize = 100) {
  let allObjects = [];
  let continuationToken = null;
  let pageCount = 1;

  do {
    const params = {
      Bucket: bucketName,
      MaxKeys: pageSize
    };

    if (continuationToken) {
      params.ContinuationToken = continuationToken;
    }

    const data = await s3.listObjectsV2(params).promise();

    console.log(`Page ${pageCount}: ${data.Contents.length} objects`);
    allObjects.push(...data.Contents);

    continuationToken = data.NextContinuationToken;
    pageCount++;
  } while (continuationToken);

  return allObjects;
}
```

### 1. 多部分上传

```javascript
// 初始化多部分上传
async function initiateMultipartUpload(bucketName, key, options = {}) {
  const params = {
    Bucket: bucketName,
    Key: key,
    ContentType: options.contentType || 'application/octet-stream'
  };

  try {
    const data = await s3.createMultipartUpload(params).promise();
    return data.UploadId;
  } catch (error) {
    console.error('Error initiating multipart upload:', error);
    throw error;
  }
}

// 上传部分
async function uploadPart(bucketName, key, uploadId, partNumber, data) {
  const params = {
    Bucket: bucketName,
    Key: key,
    UploadId: uploadId,
    PartNumber: partNumber,
    Body: data
  };

  try {
    const result = await s3.uploadPart(params).promise();
    return {
      partNumber,
      etag: result.ETag
    };
  } catch (error) {
    console.error('Error uploading part:', error);
    throw error;
  }
}

// 完成多部分上传
async function completeMultipartUpload(bucketName, key, uploadId, parts) {
  const params = {
    Bucket: bucketName,
    Key: key,
    UploadId: uploadId,
    MultipartUpload: {
      Parts: parts
    }
  };

  try {
    const result = await s3.completeMultipartUpload(params).promise();
    return result;
  } catch (error) {
    console.error('Error completing multipart upload:', error);
    throw error;
  }
}

// 完整的多部分上传示例
async function multipartUpload(bucketName, key, file, chunkSize = 5 * 1024 * 1024) {
  const fileSize = file.length;
  const uploadId = await initiateMultipartUpload(bucketName, key, {
    contentType: 'application/octet-stream'
  });

  const parts = [];
  const numParts = Math.ceil(fileSize / chunkSize);

  for (let partNumber = 1; partNumber <= numParts; partNumber++) {
    const start = (partNumber - 1) * chunkSize;
    const end = Math.min(partNumber * chunkSize, fileSize);
    const chunk = file.slice(start, end);

    const part = await uploadPart(bucketName, key, uploadId, partNumber, chunk);
    parts.push(part);

    console.log(`Uploaded part ${partNumber}/${numParts}`);
  }

  parts.sort((a, b) => a.partNumber - b.partNumber);
  return completeMultipartUpload(bucketName, key, uploadId, parts);
}
```

```javascript
async function updateObjectMetadata(bucketName, key, metadata) {

  const params = {
    Bucket: bucketName,
    Key: key,
    Body: object.Body,

  }
```

## 前端集成

### 1. 直接上传

```html
<!DOCTYPE html>
<html>
<head>
  <title>R2 File Upload</title>
  <script src="https://sdk.amazonaws.com/js/aws-sdk-2.1094.0.min.js"></script>
</head>
<body>
  <input type="file" id="fileInput" multiple>
  <div id="progress"></div>
  <div id="results"></div>

  <script>
    // 配置 R2
    AWS.config.update({
      region: 'auto',
      credentials: new AWS.Credentials({
        accessKeyId: 'your-access-key-id',
        secretAccessKey: 'your-secret-access-key'
      })
    });

    const s3 = new AWS.S3({
      endpoint: 'https://your-account-id.r2.cloudflarestorage.com',
      signatureVersion: 'v4'
    });

    // 文件上传
    document.getElementById('fileInput').addEventListener('change', async (e) => {
      const files = Array.from(e.target.files);
      const results = document.getElementById('results');
      const progress = document.getElementById('progress');

      results.innerHTML = '';
      progress.innerHTML = '';

      for (const file of files) {
        const resultDiv = document.createElement('div');
        resultDiv.textContent = `Uploading ${file.name}...`;
        results.appendChild(resultDiv);

        const progressBar = document.createElement('progress');
        progress.appendChild(progressBar);

        try {
          const params = {
            Bucket: 'my-bucket',
            Key: `uploads/${Date.now()}-${file.name}`,
            Body: file,
            ContentType: file.type
          };

          const upload = s3.upload(params, {
            partSize: 5 * 1024 * 1024, // 5MB parts
            queueSize: 4
          });

          upload.on('httpUploadProgress', (evt) => {
            const percent = Math.round((evt.loaded / evt.total) * 100);
            progressBar.value = percent;
            resultDiv.textContent = `${file.name}: ${percent}%`;
          });

          const result = await upload.promise();
          resultDiv.innerHTML = `<a href="${result.Location}" target="_blank">${file.name}</a> - Upload complete!`;
        } catch (error) {
          resultDiv.textContent = `${file.name} - Error: ${error.message}`;
        }
      }
    });
  </script>
</body>
</html>
```

```html
      const response = await fetch('/api/presigned-url', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          fileName,
          fileType

      const status = document.getElementById('status');
      const file = fileInput.files[0];

      if (!file) {
        status.textContent = 'Please select a file';
        return;

        // 获取预签名 URL

          method: 'PUT',

          body: file

            Upload successful!<br>
            File key: ${key}<br>
            <a href="https://your-bucket.your-account-id.r2.cloudflarestorage.com/${key}" target="_blank">View File</a>

          status.textContent = 'Upload failed: ' + response.statusText;
        }
      } catch (error) {
        status.textContent = 'Error: ' + error.message;
      }
    }
  </script>
```

## 错误处理

### 1. 常见错误

```javascript
// 错误处理示例
class R2ErrorHandler {
  static handle(error) {
    switch (error.code) {
      case 'NoSuchBucket':
        return 'Bucket does not exist';

      case 'NoSuchKey':
        return 'Object not found';

      case 'AccessDenied':
        return 'Access denied - check your credentials';

      case 'InvalidBucketName':
        return 'Invalid bucket name';

      case 'BucketAlreadyExists':
        return 'Bucket already exists';

      case 'EntityTooLarge':
        return 'Object too large';

      case 'InvalidPart':
        return 'Invalid part in multipart upload';

      case 'EntityTooSmall':
        return 'Object too small';

      case 'RequestTimeout':
        return 'Request timeout - please try again';

      default:
        return `Error: ${error.message}`;
    }
  }

  static async withRetry(operation, maxRetries = 3) {
    for (let i = 0; i < maxRetries; i++) {
      try {
        return await operation();
      } catch (error) {
        if (i === maxRetries - 1 || !this.shouldRetry(error)) {
          throw error;
        }

        // 指数退避
        const delay = Math.pow(2, i) * 1000;
        await new Promise(resolve => setTimeout(resolve, delay));
      }
    }
  }

  static shouldRetry(error) {
    const retryableErrors = [
      'RequestTimeout',
      'ServiceUnavailable',
      'InternalError',
      'SlowDown'
    ];

    return retryableErrors.includes(error.code);
  }
}

// 使用示例
try {
  const result = await R2ErrorHandler.withRetry(async () => {
    return await s3.getObject(params).promise();
  });
} catch (error) {
  console.error(R2ErrorHandler.handle(error));
}
```

### 2. 连接管理

```javascript
// R2 客户端管理
class R2Client {
  constructor(credentials) {
    this.s3 = new AWS.S3({
      endpoint: `https://${credentials.accountId}.r2.cloudflarestorage.com`,
      accessKeyId: credentials.accessKeyId,
      secretAccessKey: credentials.secretAccessKey,
      region: 'auto',
      maxRetries: 3,
      retryDelayOptions: {
        customBackoff: function(retryCount) {
          return Math.pow(2, retryCount) * 100; // 100ms, 200ms, 400ms
        }
      }
    });
  }

  async uploadWithRetry(params, maxRetries = 3) {
    let lastError;

    for (let i = 0; i < maxRetries; i++) {
      try {
        return await this.s3.upload(params).promise();
      } catch (error) {
        lastError = error;

        if (i < maxRetries - 1) {
          const delay = Math.pow(2, i) * 1000;
          console.log(`Upload failed, retrying in ${delay}ms...`);
          await new Promise(resolve => setTimeout(resolve, delay));
        }
      }
    }

    throw lastError;
  }
}
```
