```javascript
// performance-test.js
const AWS = require('aws-sdk');

class R2PerformanceTester {
  constructor(credentials) {
    this.s3 = new AWS.S3({
      endpoint: `https://${credentials.accountId}.r2.cloudflarestorage.com`,
      accessKeyId: credentials.accessKeyId,
      secretAccessKey: credentials.secretAccessKey,
      region: 'auto'
    });
  }

  // 测试上传性能
  async testUploadPerformance(bucketName, fileSizeMB = 10) {
    const testFile = Buffer.alloc(fileSizeMB * 1024 * 1024, 'x');
    const key = `test-upload-${Date.now()}.bin`;

    console.log(`\n=== Upload Performance Test (${fileSizeMB}MB) ===`);

    const startTime = Date.now();

    try {
      const result = await this.s3.upload({
        Bucket: bucketName,
        Key: key,
        Body: testFile,
        ContentType: 'application/octet-stream'
      }).promise();

      const endTime = Date.now();
      const duration = (endTime - startTime) / 1000;
      const speedMBps = fileSizeMB / duration;
      const speedMbps = (speedMBps * 8).toFixed(2);

      console.log(`✅ Upload complete`);
      console.log(`   Duration: ${duration}s`);
      console.log(`   Speed: ${speedMBps.toFixed(2)} MB/s (${speedMbps} Mbps)`);
      console.log(`   ETag: ${result.ETag}`);

      return {
        fileSizeMB,
        duration,
        speedMBps,
        speedMbps,
        etag: result.ETag
      };
    } catch (error) {
      console.error('❌ Upload failed:', error.message);
      throw error;
    }
  }

  // 测试下载性能
  async testDownloadPerformance(bucketName, key) {
    console.log(`\n=== Download Performance Test ===`);

    const startTime = Date.now();

    try {
      const result = await this.s3.getObject({
        Bucket: bucketName,
        Key: key
      }).promise();

      // 模拟读取数据
      let downloadedSize = 0;
      for await (const chunk of result.Body) {
        downloadedSize += chunk.length;
      }

      const endTime = Date.now();
      const duration = (endTime - startTime) / 1000;
      const fileSizeMB = downloadedSize / (1024 * 1024);
      const speedMBps = fileSizeMB / duration;
      const speedMbps = (speedMBps * 8).toFixed(2);

      console.log(`✅ Download complete`);
      console.log(`   File size: ${fileSizeMB.toFixed(2)}MB`);
      console.log(`   Duration: ${duration}s`);
      console.log(`   Speed: ${speedMBps.toFixed(2)} MB/s (${speedMbps} Mbps)`);
      console.log(`   ContentLength: ${result.ContentLength}`);

      return {
        fileSizeMB,
        duration,
        speedMBps,
        speedMbps,
        contentLength: result.ContentLength
      };
    } catch (error) {
      console.error('❌ Download failed:', error.message);
      throw error;
    }
  }

  // 并发上传测试
  async testConcurrentUploads(bucketName, concurrency = 10, fileSizeMB = 1) {
    console.log(`\n=== Concurrent Uploads Test (${concurrency} concurrent, ${fileSizeMB}MB each) ===`);

    const testFile = Buffer.alloc(fileSizeMB * 1024 * 1024, 'x');
    const promises = [];

    for (let i = 0; i < concurrency; i++) {
      const key = `concurrent-test-${Date.now()}-${i}.bin`;

      promises.push(
        this.s3.upload({
          Bucket: bucketName,
          Key: key,
          Body: testFile,
          ContentType: 'application/octet-stream'
        }).promise()
      );
    }

    const startTime = Date.now();

    try {
      const results = await Promise.all(promises);
      const endTime = Date.now();
      const duration = (endTime - startTime) / 1000;
      const totalMB = concurrency * fileSizeMB;
      const totalSpeedMBps = totalMB / duration;
      const avgSpeedMBps = totalSpeedMBps / concurrency;

      console.log(`✅ Concurrent uploads complete`);
      console.log(`   Total files: ${concurrency}`);
      console.log(`   Total size: ${totalMB}MB`);
      console.log(`   Total duration: ${duration}s`);
      console.log(`   Total speed: ${totalSpeedMBps.toFixed(2)} MB/s`);
      console.log(`   Average speed per file: ${avgSpeedMBps.toFixed(2)} MB/s`);

      return {
        concurrency,
        fileSizeMB,
        duration,
        totalMB,
        totalSpeedMBps,
        avgSpeedMBps
      };
    } catch (error) {
      console.error('❌ Concurrent upload failed:', error.message);
      throw error;
    }
  }

  // 延迟测试
  async testLatency(bucketName, iterations = 100) {
    console.log(`\n=== Latency Test (${iterations} iterations) ===`);

    const latencies = [];

    for (let i = 0; i < iterations; i++) {
      const startTime = Date.now();

      try {
        // 使用 HEAD 请求测试延迟
        await this.s3.headObject({
          Bucket: bucketName,
          Key: 'latency-test-file'
        }).promise();

        const endTime = Date.now();
        const latency = endTime - startTime;
        latencies.push(latency);

        if (i % 10 === 0) {
          process.stdout.write(`Progress: ${i}/${iterations}\r`);
        }
      } catch (error) {
        // 忽略 404 错误，只关心延迟
        if (error.code !== 'NoSuchKey') {
          console.error(`❌ Error in iteration ${i}:`, error.message);
        }
        continue;
      }
    }

    process.stdout.write('\n');

    const avgLatency = latencies.reduce((a, b) => a + b, 0) / latencies.length;
    const minLatency = Math.min(...latencies);
    const maxLatency = Math.max(...latencies);
    const p50 = calculatePercentile(latencies, 0.5);
    const p95 = calculatePercentile(latencies, 0.95);
    const p99 = calculatePercentile(latencies, 0.99);

    console.log(`✅ Latency test complete`);
    console.log(`   Iterations: ${latencies.length}`);
    console.log(`   Average: ${avgLatency.toFixed(2)}ms`);
    console.log(`   Min: ${minLatency}ms`);
    console.log(`   Max: ${maxLatency}ms`);
    console.log(`   P50: ${p50.toFixed(2)}ms`);
    console.log(`   P95: ${p95.toFixed(2)}ms`);
    console.log(`   P99: ${p99.toFixed(2)}ms`);

    return {
      iterations: latencies.length,
      avgLatency,
      minLatency,
      maxLatency,
      p50,
      p95,
      p99
    };
  }

  // 多部分上传性能测试
  async testMultipartUpload(bucketName, fileSizeMB = 100, partSizeMB = 5) {
    console.log(`\n=== Multipart Upload Test (${fileSizeMB}MB, ${partSizeMB}MB parts) ===`);

    const testFile = Buffer.alloc(fileSizeMB * 1024 * 1024, 'x');
    const key = `multipart-test-${Date.now()}.bin`;
    const partSize = partSizeMB * 1024 * 1024;
    const numParts = Math.ceil(fileSizeMB / partSizeMB);

    console.log(`File size: ${fileSizeMB}MB`);
    console.log(`Part size: ${partSizeMB}MB`);
    console.log(`Number of parts: ${numParts}`);

    const startTime = Date.now();

    try {
      // 1. 初始化多部分上传
      const uploadId = (await this.s3.createMultipartUpload({
        Bucket: bucketName,
        Key: key,
        ContentType: 'application/octet-stream'
      }).promise()).UploadId;

      // 2. 上传各个部分
      const parts = [];
      const uploadPromises = [];

      for (let i = 0; i < numParts; i++) {
        const partNumber = i + 1;
        const start = i * partSize;
        const end = Math.min(start + partSize, testFile.length);
        const partData = testFile.slice(start, end);

        uploadPromises.push(
          (async () => {
            const result = await this.s3.uploadPart({
              Bucket: bucketName,
              Key: key,
              UploadId: uploadId,
              PartNumber: partNumber,
              Body: partData
            }).promise();

            console.log(`  Uploaded part ${partNumber}/${numParts}`);
            return {
              PartNumber: partNumber,
              ETag: result.ETag
            };
          })()
        );

        // 限制并发上传数量
        if (uploadPromises.length >= 5) {
          const results = await Promise.all(uploadPromises);
          parts.push(...results);
          uploadPromises.length = 0;
        }
      }

      // 上传剩余的部分
      if (uploadPromises.length > 0) {
        const results = await Promise.all(uploadPromises);
        parts.push(...results);
      }

      // 3. 完成多部分上传
      const result = await this.s3.completeMultipartUpload({
        Bucket: bucketName,
        Key: key,
        UploadId: uploadId,
        MultipartUpload: {
          Parts: parts.sort((a, b) => a.PartNumber - b.PartNumber)
        }
      }).promise();

      const endTime = Date.now();
      const duration = (endTime - startTime) / 1000;
      const speedMBps = fileSizeMB / duration;
      const speedMbps = (speedMBps * 8).toFixed(2);

      console.log(`✅ Multipart upload complete`);
      console.log(`   Duration: ${duration}s`);
      console.log(`   Speed: ${speedMBps.toFixed(2)} MB/s (${speedMbps} Mbps)`);
      console.log(`   Parts uploaded: ${numParts}`);

      return {
        fileSizeMB,
        partSizeMB,
        numParts,
        duration,
        speedMBps,
        speedMbps
      };
    } catch (error) {
      console.error('❌ Multipart upload failed:', error.message);
      throw error;
    }
  }

  // 综合性能测试
  async runFullTestSuite(bucketName) {
    console.log('🚀 Starting R2 Performance Test Suite\n');
    console.log(`Bucket: ${bucketName}`);
    console.log(`Time: ${new Date().toISOString()}\n`);

    const results = {
      upload: {},
      download: {},
      concurrent: {},
      latency: {},
      multipart: {}
    };

    try {
      // 1. 上传测试
      results.upload['1MB'] = await this.testUploadPerformance(bucketName, 1);
      results.upload['10MB'] = await this.testUploadPerformance(bucketName, 10);
      results.upload['100MB'] = await this.testUploadPerformance(bucketName, 100);

      // 2. 下载测试
      results.download['1MB'] = await this.testDownloadPerformance(bucketName, results.upload['1MB'].key);
      results.download['10MB'] = await this.testDownloadPerformance(bucketName, results.upload['10MB'].key);

      // 3. 并发测试
      results.concurrent['10x1MB'] = await this.testConcurrentUploads(bucketName, 10, 1);
      results.concurrent['100x1MB'] = await this.testConcurrentUploads(bucketName, 100, 1);

      // 4. 延迟测试
      results.latency = await this.testLatency(bucketName);

      // 5. 多部分上传测试
      results.multipart = await this.testMultipartUpload(bucketName, 100, 10);

      // 生成报告
      await this.generateReport(results);

      return results;
    } catch (error) {
      console.error('❌ Test suite failed:', error.message);
      throw error;
    }
  }

  // 生成测试报告
  async generateReport(results) {
    const report = {
      timestamp: new Date().toISOString(),
      results: results,
      summary: this.generateSummary(results)
    };

    // 保存到文件
    const fs = require('fs');
    fs.writeFileSync('r2-performance-report.json', JSON.stringify(report, null, 2));

    console.log('\n📊 Performance Report Generated');
    console.log('File: r2-performance-report.json\n');

    // 显示摘要
    console.log('=== Performance Summary ===');
    console.log(`Upload Speed (1MB): ${results.upload['1MB'].speedMbps} Mbps`);
    console.log(`Upload Speed (10MB): ${results.upload['10MB'].speedMbps} Mbps`);
    console.log(`Download Speed (1MB): ${results.download['1MB'].speedMbps} Mbps`);
    console.log(`Concurrent Uploads (10x1MB): ${results.concurrent['10x1MB'].totalSpeedMBps.toFixed(2)} MB/s`);
    console.log(`Average Latency: ${results.latency.avgLatency.toFixed(2)}ms`);
    console.log(`P95 Latency: ${results.latency.p95.toFixed(2)}ms`);
    console.log(`Multipart Upload Speed: ${results.multipart.speedMbps} Mbps`);
  }

  generateSummary(results) {
    return {
      uploadSpeeds: {
        '1MB': results.upload['1MB']?.speedMbps,
        '10MB': results.upload['10MB']?.speedMbps,
        '100MB': results.upload['100MB']?.speedMbps
      },
      downloadSpeeds: {
        '1MB': results.download['1MB']?.speedMbps,
        '10MB': results.download['10MB']?.speedMbps
      },
      latency: {
        average: results.latency?.avgLatency,
        p95: results.latency?.p95
      }
    };
  }
}

// 百分位计算
function calculatePercentile(numbers, percentile) {
  const sorted = numbers.sort((a, b) => a - b);
  const index = Math.ceil(percentile * sorted.length) - 1;
  return sorted[index];
}

// 使用示例

}

if (require.main === module) {

module.exports = R2PerformanceTester;
```

## 性能优化

### 1. 优化上传策略

```javascript
// 智能上传策略
class OptimizedUploader {
  constructor(r2Client) {
    this.r2Client = r2Client;
  }

  async upload(bucketName, file, options = {}) {
    const fileSizeMB = file.length / (1024 * 1024);
    const optimalChunkSize = this.calculateOptimalChunkSize(fileSizeMB);

    if (fileSizeMB < 5) {
      // 小文件直接上传
      return this.directUpload(bucketName, file, options);
    } else if (fileSizeMB < 100) {
      // 中等文件使用小分块
      return this.multipartUpload(bucketName, file, optimalChunkSize, options);
    } else {
      // 大文件使用大分块
      return this.multipartUpload(bucketName, file, Math.max(optimalChunkSize, 10), options);
    }
  }

  calculateOptimalChunkSize(fileSizeMB) {
    if (fileSizeMB < 10) return 1;      // 1MB
    if (fileSizeMB < 100) return 5;     // 5MB
    if (fileSizeMB < 1000) return 10;   // 10MB
    return 20;                          // 20MB
  }

  async directUpload(bucketName, file, options) {
    const params = {
      Bucket: bucketName,
      Key: options.key || `upload-${Date.now()}`,
      Body: file,
      ContentType: options.contentType || 'application/octet-stream'
    };

    return this.r2Client.s3.upload(params).promise();
  }

  async multipartUpload(bucketName, file, partSizeMB, options) {
    const partSize = partSizeMB * 1024 * 1024;
    const numParts = Math.ceil(file.length / partSize);

    // 初始化多部分上传
    const uploadParams = {
      Bucket: bucketName,
      Key: options.key || `multipart-${Date.now()}`,
      ContentType: options.contentType || 'application/octet-stream'
    };

    const uploadId = (await this.r2Client.s3.createMultipartUpload(uploadParams).promise()).UploadId;

    // 并行上传部分
    const uploadPromises = [];
    const parts = [];

    for (let i = 0; i < numParts; i++) {
      const partNumber = i + 1;
      const start = i * partSize;
      const end = Math.min(start + partSize, file.length);
      const partData = file.slice(start, end);

      uploadPromises.push(
        this.r2Client.s3.uploadPart({
          Bucket: bucketName,
          Key: uploadParams.Key,
          UploadId: uploadId,
          PartNumber: partNumber,
          Body: partData
        }).promise().then(result => ({
          PartNumber: partNumber,
          ETag: result.ETag
        }))
      );
    }

    // 等待所有部分完成
    const partsResult = await Promise.all(uploadPromises);
    parts.push(...partsResult);

    // 完成上传
    return this.r2Client.s3.completeMultipartUpload({
      Bucket: bucketName,
      Key: uploadParams.Key,
      UploadId: uploadId,
      MultipartUpload: {
        Parts: parts.sort((a, b) => a.PartNumber - b.PartNumber)
      }
    }).promise();
  }
}
```

```javascript
// R2 缓存策略
class R2CacheOptimizer {
  constructor(r2Client) {
    this.r2Client = r2Client;
    this.cache = new Map();
  }

  async getWithCache(bucketName, key, cacheExpiry = 300) {
    // 检查内存缓存
    const cacheKey = `${bucketName}:${key}`;
    const cached = this.cache.get(cacheKey);

    if (cached && (Date.now() - cached.timestamp) < cacheExpiry * 1000) {
      return cached.data;
    }

    // 从 R2 获取
    const result = await this.r2Client.s3.getObject({
      Bucket: bucketName,
      Key: key
    }).promise();

    // 缓存结果
    this.cache.set(cacheKey, {
      data: result,
      timestamp: Date.now()
    });

    return result;
  }

  async batchGet(bucketName, keys) {
    // 并行获取多个对象
    const promises = keys.map(key =>
      this.getWithCache(bucketName, key).catch(error => {
        console.error(`Failed to get ${key}:`, error);
        return null;
      })
    );

    const results = await Promise.all(promises);
    return results.filter(result => result !== null);
  }

  // 预取热门内容
  async prefetchPopular(bucketName, popularKeys) {
    const prefetchPromises = popularKeys.map(key =>
      this.getWithCache(bucketName, key, 3600) // 1小时缓存
    );

    await Promise.all(prefetchPromises);
    console.log(`Prefetched ${popularKeys.length} popular objects`);
  }
}
```

```javascript
// R2 性能监控器
class R2PerformanceMonitor {
  constructor(r2Client) {
    this.r2Client = r2Client;
    this.metrics = {
      uploads: [],
      downloads: [],
      errors: []
    };
  }

  // 监控上传
  async monitoredUpload(bucketName, key, body, options = {}) {
    const startTime = Date.now();
    const size = body.length || 0;

    try {
      const result = await this.r2Client.s3.upload({
        Bucket: bucketName,
        Key: key,
        Body: body,
        ...options
      }).promise();

      const endTime = Date.now();
      const duration = endTime - startTime;
      const speedMBps = (size / (1024 * 1024)) / (duration / 1000);

      // 记录指标
      this.metrics.uploads.push({
        timestamp: new Date().toISOString(),
        key,
        size,
        duration,
        speedMBps,
        success: true
      });

      return result;
    } catch (error) {
      const endTime = Date.now();
      const duration = endTime - startTime;

      this.metrics.uploads.push({
        timestamp: new Date().toISOString(),
        key,
        size,
        duration,
        success: false,
        error: error.message
      });

      this.metrics.errors.push({
        timestamp: new Date().toISOString(),
        operation: 'upload',
        error: error.message,
        duration
      });

      throw error;
    }
  }

  // 生成性能报告
  generateReport() {
    const uploads = this.metrics.uploads;
    const successfulUploads = uploads.filter(u => u.success);
    const recentUploads = successfulUploads.slice(-100); // 最近100次

    return {
      total: uploads.length,
      success: successfulUploads.length,
      failure: uploads.length - successfulUploads.length,
      successRate: (successfulUploads.length / uploads.length * 100).toFixed(2) + '%',
      averageSpeed: recentUploads.length > 0
        ? (recentUploads.reduce((sum, u) => sum + u.speedMBps, 0) / recentUploads.length).toFixed(2) + ' MB/s'
        : 'N/A',
      averageDuration: recentUploads.length > 0
        ? (recentUploads.reduce((sum, u) => sum + u.duration, 0) / recentUploads.length).toFixed(0) + 'ms'
        : 'N/A'
    };
  }

  // 获取实时状态
  getRealtimeStats() {
    const now = Date.now();
    const last5Minutes = now - 5 * 60 * 1000;
    const recentUploads = this.metrics.uploads.filter(
      u => new Date(u.timestamp).getTime() > last5Minutes
    );

    return {
      last5Minutes: {
        uploads: recentUploads.length,
        avgSpeed: recentUploads.length > 0
          ? recentUploads.reduce((sum, u) => sum + u.speedMBps, 0) / recentUploads.length
          : 0
      }
    };
  }
}
```
