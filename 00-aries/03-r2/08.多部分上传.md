# R2 多部分上传

## 多部分上传概述

多部分上传（Multipart Upload）允许你上传大文件时将其分成多个部分独立上传，最后合并成完整的对象。这对于网络不稳定、大文件传输、可恢复上传等场景非常有用。

## 为什么使用多部分上传

### 优势

```javascript
// 多部分上传的优势
✅ 大文件支持 - 突破单次上传大小限制
✅ 网络容错 - 上传失败只需重新上传失败部分
✅ 并行上传 - 可同时上传多个部分，提高速度
✅ 可恢复性 - 暂停后可从断点继续
✅ 灵活控制 - 精确控制每个部分的传输
✅ 内存友好 - 不需要将整个文件加载到内存

// 适用场景
- 视频文件上传（GB级别）
- 数据库备份文件
- 软件包分发
- 大型媒体文件
- 网络不稳定环境下的上传
```

## 基础实现

### 1. 完整的上传流程

```javascript
class R2MultipartUploader {
  constructor(s3Client, bucketName) {
    this.s3 = s3Client;
    this.bucket = bucketName;
    this.defaultPartSize = 5 * 1024 * 1024; // 5MB
  }

  async uploadFile(file, options = {}) {
    const { partSize = this.defaultPartSize } = options;
    const key = options.key || `uploads/${Date.now()}-${file.name}`;
    const fileSize = file.length || file.size;
    const numParts = Math.ceil(fileSize / partSize);

    console.log(`Starting multipart upload for ${key}`);
    console.log(`File size: ${(fileSize / 1024 / 1024).toFixed(2)}MB`);
    console.log(`Part size: ${(partSize / 1024 / 1024).toFixed(2)}MB`);
    console.log(`Number of parts: ${numParts}`);

    // 1. 初始化多部分上传
    const uploadId = await this.initiateMultipartUpload(key, options);

    // 2. 上传各个部分
    const parts = await this.uploadParts(uploadId, key, file, partSize);

    // 3. 完成上传
    return await this.completeMultipartUpload(uploadId, key, parts);
  }

  // 初始化多部分上传
  async initiateMultipartUpload(key, options = {}) {
    const params = {
      Bucket: this.bucket,
      Key: key,
      ContentType: options.contentType || 'application/octet-stream',
      Metadata: options.metadata || {}
    };

    try {
      const result = await this.s3.createMultipartUpload(params).promise();
      console.log('Multipart upload initiated:', result.UploadId);
      return result.UploadId;
    } catch (error) {
      console.error('Failed to initiate multipart upload:', error);
      throw error;
    }
  }

  // 上传所有部分
  async uploadParts(uploadId, key, file, partSize) {
    const fileSize = file.length || file.size;
    const numParts = Math.ceil(fileSize / partSize);
    const parts = [];

    // 并发上传（限制并发数）
    const concurrency = options.concurrency || 5;
    const semaphore = new Semaphore(concurrency);

    for (let partNumber = 1; partNumber <= numParts; partNumber++) {
      const start = (partNumber - 1) * partSize;
      const end = Math.min(start + partSize, fileSize);

      const uploadPromise = semaphore.acquire().then(async () => {
        try {
          const part = await this.uploadPart(
            uploadId,
            key,
            partNumber,
            file.slice(start, end)
          );
          console.log(`Part ${partNumber}/${numParts} uploaded`);
          return part;
        } finally {
          semaphore.release();
        }
      });

      parts.push(uploadPromise);
    }

    const uploadedParts = await Promise.all(parts);

    // 按序排列
    return uploadedParts.sort((a, b) => a.PartNumber - b.PartNumber);
  }

  // 上传单个部分
  async uploadPart(uploadId, key, partNumber, partData) {
    const params = {
      Bucket: this.bucket,
      Key: key,
      UploadId: uploadId,
      PartNumber: partNumber,
      Body: partData
    };

    try {
      const result = await this.s3.uploadPart(params).promise();
      return {
        PartNumber: partNumber,
        ETag: result.ETag
      };
    } catch (error) {
      console.error(`Failed to upload part ${partNumber}:`, error);
      throw error;
    }
  }

  // 完成多部分上传
  async completeMultipartUpload(uploadId, key, parts) {
    const params = {
      Bucket: this.bucket,
      Key: key,
      UploadId: uploadId,
      MultipartUpload: {
        Parts: parts
      }
    };

    try {
      const result = await this.s3.completeMultipartUpload(params).promise();
      console.log('Multipart upload completed:', result.Location);
      return result;
    } catch (error) {
      console.error('Failed to complete multipart upload:', error);
      throw error;
    }
  }
}

// 信号量类，用于控制并发数
class Semaphore {
  constructor(maxConcurrency) {
    this.maxConcurrency = maxConcurrency;
    this.current = 0;
    this.queue = [];
  }

  acquire() {
    return new Promise((resolve) => {
      if (this.current < this.maxConcurrency) {
        this.current++;
        resolve();
      } else {
        this.queue.push(resolve);
      }
    });
  }

  release() {
    if (this.queue.length > 0) {
      const next = this.queue.shift();
      next();
    } else {
      this.current--;
    }
  }
}

// 使用示例
const uploader = new R2MultipartUploader(s3, 'my-bucket');

// 上传文件
const file = fs.readFileSync('large-file.zip');
await uploader.uploadFile(file, {
  partSize: 10 * 1024 * 1024, // 10MB parts
  concurrency: 10,           // 10 并发上传
  metadata: {
    original_name: 'large-file.zip',
    upload_date: new Date().toISOString()
  }
});
```

## 高级功能

### 1. 可恢复上传

```javascript
class ResumableMultipartUploader extends R2MultipartUploader {
  constructor(s3Client, bucketName) {
    super(s3Client, bucketName);
    this.uploadStates = new Map();
  }

  // 启动可恢复上传
  async startResumableUpload(file, options = {}) {
    const uploadId = options.uploadId || this.generateUploadId();
    const key = options.key || `uploads/${uploadId}-${file.name}`;

    // 检查是否已存在上传状态
    let uploadState = await this.loadUploadState(uploadId);

    if (!uploadState) {
      // 新上传
      uploadState = {
        uploadId,
        key,
        fileSize: file.length || file.size,
        partSize: options.partSize || this.defaultPartSize,
        uploadedParts: [],
        createdAt: Date.now()
      };
    }

    try {
      // 如果是新上传，初始化
      if (uploadState.uploadedParts.length === 0) {
        uploadState.uploadId = await this.initiateMultipartUpload(key, options);
      }

      // 获取已上传的部分
      const existingParts = await this.listUploadedParts(uploadState.uploadId);
      const uploadedParts = [...uploadState.uploadedParts, ...existingParts];

      // 计算需要上传的部分
      const totalParts = Math.ceil(uploadState.fileSize / uploadState.partSize);
      const uploadedPartNumbers = new Set(uploadedParts.map(p => p.PartNumber));
      const neededParts = [];

      for (let partNumber = 1; partNumber <= totalParts; partNumber++) {
        if (!uploadedPartNumbers.has(partNumber)) {
          neededParts.push(partNumber);
        }
      }

      console.log(`Resuming upload: ${uploadedParts.length}/${totalParts} parts already uploaded`);
      console.log(`Need to upload ${neededParts.length} parts`);

      // 上传缺失的部分
      const newParts = await this.uploadMissingParts(
        uploadState.uploadId,
        uploadState.key,
        file,
        neededParts,
        uploadState.partSize
      );

      const allParts = [...uploadedParts, ...newParts].sort((a, b) => a.PartNumber - b.PartNumber);

      // 完成上传
      const result = await this.completeMultipartUpload(uploadState.uploadId, uploadState.key, allParts);

      // 清理状态
      await this.clearUploadState(uploadId);

      return result;

    } catch (error) {
      // 保存当前状态以便恢复
      await this.saveUploadState(uploadId, uploadState);
      throw error;
    }
  }

  async uploadMissingParts(uploadId, key, file, partNumbers, partSize) {
    const parts = [];
    const concurrency = 5;
    const semaphore = new Semaphore(concurrency);

    for (const partNumber of partNumbers) {
      const start = (partNumber - 1) * partSize;
      const end = Math.min(start + partSize, file.length || file.size);

      const uploadPromise = semaphore.acquire().then(async () => {
        try {
          const part = await this.uploadPart(
            uploadId,
            key,
            partNumber,
            file.slice(start, end)
          );

          // 更新状态
          const uploadState = await this.loadUploadState(uploadId);
          uploadState.uploadedParts.push(part);
          await this.saveUploadState(uploadId, uploadState);

          console.log(`Part ${partNumber} uploaded (${uploadState.uploadedParts.length}/${Math.ceil(file.length / partSize)})`);

          return part;
        } finally {
          semaphore.release();
        }
      });

      parts.push(uploadPromise);
    }

    return await Promise.all(parts);
  }

  async listUploadedParts(uploadId) {
    try {
      const result = await this.s3.listParts({
        Bucket: this.bucket,
        Key: 'placeholder', // 不需要 key，需要 UploadId
        UploadId: uploadId
      }).promise();

      return result.Parts;
    } catch (error) {
      if (error.code === 'NoSuchUpload') {
        return [];
      }
      throw error;
    }
  }

  generateUploadId() {
    return `upload_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`;
  }

  async saveUploadState(uploadId, state) {
    // 应该保存到持久化存储（数据库、KV 等）
    // 这里简化示例
    this.uploadStates.set(uploadId, {
      ...state,
      lastUpdated: Date.now()
    });
    console.log(`Upload state saved for ${uploadId}`);
  }

  async loadUploadState(uploadId) {
    // 应该从持久化存储加载
    return this.uploadStates.get(uploadId) || null;
  }

  async clearUploadState(uploadId) {
    this.uploadStates.delete(uploadId);
    console.log(`Upload state cleared for ${uploadId}`);
  }
}
```

### 2. 带进度监控的上传

```javascript
class ProgressTrackingUploader extends R2MultipartUploader {
  constructor(s3Client, bucketName) {
    super(s3Client, bucketName);
    this.progressCallback = null;
  }

  setProgressCallback(callback) {
    this.progressCallback = callback;
  }

  async uploadFileWithProgress(file, options = {}) {
    const { partSize = this.defaultPartSize } = options;
    const key = options.key || `uploads/${Date.now()}-${file.name}`;
    const fileSize = file.length || file.size;
    const numParts = Math.ceil(fileSize / partSize);

    // 报告开始
    this.reportProgress({
      stage: 'initiation',
      loaded: 0,
      total: fileSize,
      percentage: 0,
      eta: null
    });

    // 初始化
    const uploadId = await this.initiateMultipartUpload(key, options);

    // 上传部分
    const parts = [];
    let totalLoaded = 0;

    for (let partNumber = 1; partNumber <= numParts; partNumber++) {
      const start = (partNumber - 1) * partSize;
      const end = Math.min(start + partSize, fileSize);
      const partData = file.slice(start, end);

      // 上传前报告进度
      this.reportProgress({
        stage: 'uploading',
        part: partNumber,
        totalParts: numParts,
        loaded: totalLoaded,
        total: fileSize,
        percentage: (totalLoaded / fileSize * 100).toFixed(2),
        eta: this.calculateETA(totalLoaded, fileSize, Date.now() - startTime)
      });

      const startTime = Date.now();
      const part = await this.uploadPart(uploadId, key, partNumber, partData);
      const duration = Date.now() - startTime;

      totalLoaded += partData.length;
      parts.push(part);

      // 上传后报告进度
      this.reportProgress({
        stage: 'uploaded',
        part: partNumber,
        totalParts: numParts,
        loaded: totalLoaded,
        total: fileSize,
        percentage: (totalLoaded / fileSize * 100).toFixed(2),
        eta: this.calculateETA(totalLoaded, fileSize, Date.now() - this.startTime),
        partDuration: duration
      });
    }

    // 完成上传
    this.reportProgress({
      stage: 'completing',
      loaded: fileSize,
      total: fileSize,
      percentage: 100,
      eta: null
    });

    const result = await this.completeMultipartUpload(uploadId, key, parts);

    this.reportProgress({
      stage: 'completed',
      loaded: fileSize,
      total: fileSize,
      percentage: 100,
      eta: null,
      url: result.Location
    });

    return result;
  }

  reportProgress(progress) {
    if (this.progressCallback) {
      this.progressCallback(progress);
    }

    // 也可以发送到事件发射器
    this.emit('progress', progress);
  }

  calculateETA(loaded, total, elapsedTime) {
    if (loaded === 0 || elapsedTime === 0) return null;

    const bytesPerMs = loaded / elapsedTime;
    const remaining = total - loaded;
    const remainingMs = remaining / bytesPerMs;

    return new Date(Date.now() + remainingMs);
  }

  // 事件发射器功能
  emit(eventName, data) {
    if (this.listeners && this.listeners[eventName]) {
      this.listeners[eventName].forEach(callback => {
        try {
          callback(data);
        } catch (error) {
          console.error(`Error in ${eventName} listener:`, error);
        }
      });
    }
  }

  on(eventName, callback) {
    if (!this.listeners) this.listeners = {};
    if (!this.listeners[eventName]) this.listeners[eventName] = [];
    this.listeners[eventName].push(callback);
  }

  off(eventName, callback) {
    if (this.listeners && this.listeners[eventName]) {
      const index = this.listeners[eventName].indexOf(callback);
      if (index > -1) {
        this.listeners[eventName].splice(index, 1);
      }
    }
  }
}

// 使用示例
const progressUploader = new ProgressTrackingUploader(s3, 'my-bucket');

progressUploader.on('progress', (progress) => {
  console.log(`Progress: ${progress.percentage}%`);
  console.log(`Stage: ${progress.stage}`);
  if (progress.eta) {
    console.log(`ETA: ${progress.eta.toLocaleTimeString()}`);
  }
});

progressUploader.setProgressCallback((progress) => {
  // 更新 UI 进度条
  updateProgressBar(progress.percentage);
  updateStatus(progress.stage);
});
```

### 3. 智能分块大小

```javascript
class SmartChunkUploader extends R2MultipartUploader {
  constructor(s3Client, bucketName) {
    super(s3Client, bucketName);
    this.minPartSize = 5 * 1024 * 1024;    // 5MB
    this.maxPartSize = 100 * 1024 * 1024; // 100MB
    this.defaultConcurrency = 5;
  }

  calculateOptimalPartSize(fileSize, networkSpeed, uploadTimeBudget) {
    // 根据文件大小自动计算最佳分块大小
    if (fileSize < this.minPartSize * 2) {
      return fileSize; // 小文件不分块
    }

    if (fileSize < 100 * 1024 * 1024) { // 100MB
      return this.minPartSize; // 小文件用最小分块
    }

    if (fileSize < 1 * 1024 * 1024 * 1024) { // 1GB
      return 10 * 1024 * 1024; // 中等文件用 10MB
    }

    if (fileSize < 10 * 1024 * 1024 * 1024) { // 10GB
      return 50 * 1024 * 1024; // 大文件用 50MB
    }

    // 超大文件根据网络条件调整
    return this.calculatePartSizeForNetwork(fileSize, networkSpeed, uploadTimeBudget);
  }

  calculatePartSizeForNetwork(fileSize, networkSpeed, uploadTimeBudget) {
    // 网络速度 (bytes per second)
    const bytesPerSecond = networkSpeed || 5 * 1024 * 1024; // 默认 5MB/s
    // 上传时间预算 (seconds)
    const maxUploadTime = uploadTimeBudget || 3600; // 默认 1小时

    // 计算可以完成的分块数
    const maxParts = maxUploadTime / 30; // 假设每个分块 30 秒上传时间
    const minPartSize = fileSize / maxParts;

    // 限制在合理范围内
    return Math.max(
      this.minPartSize,
      Math.min(minPartSize, this.maxPartSize)
    );
  }

  async smartUpload(file, options = {}) {
    const fileSize = file.length || file.size;
    const partSize = options.partSize || this.calculateOptimalPartSize(fileSize);
    const concurrency = options.concurrency || this.calculateOptimalConcurrency(fileSize, partSize);

    console.log('Smart Upload Configuration:');
    console.log(`  File size: ${(fileSize / 1024 / 1024).toFixed(2)} MB`);
    console.log(`  Part size: ${(partSize / 1024 / 1024).toFixed(2)} MB`);
    console.log(`  Concurrency: ${concurrency}`);
    console.log(`  Estimated parts: ${Math.ceil(fileSize / partSize)}`);

    return await this.uploadFile(file, {
      ...options,
      partSize,
      concurrency
    });
  }

  calculateOptimalConcurrency(fileSize, partSize) {
    const numParts = Math.ceil(fileSize / partSize);
    const availableMemory = 512 * 1024 * 1024; // 假设 512MB 内存
    const partMemory = partSize * 2; // 每个部分可能需要 2倍内存（读取 + 缓冲）

    // 基于内存限制
    const memoryConcurrency = Math.floor(availableMemory / partMemory);

    // 基于网络连接数
    const networkConcurrency = 10;

    // 基于分块数量（分块少时不需要太多并发）
    const sizeConcurrency = Math.min(numParts, 5);

    return Math.min(memoryConcurrency, networkConcurrency, sizeConcurrency, 10);
  }

  // 网络速度检测
  async measureNetworkSpeed() {
    const testDataSize = 1 * 1024 * 1024; // 1MB 测试数据
    const testKey = `speed-test-${Date.now()}`;

    const startTime = Date.now();

    try {
      await this.s3.upload({
        Bucket: this.bucket,
        Key: testKey,
        Body: Buffer.alloc(testDataSize, 'x')
      }).promise();

      const duration = Date.now() - startTime;
      const speed = testDataSize / (duration / 1000); // bytes per second

      // 清理测试数据
      await this.s3.deleteObject({ Bucket: this.bucket, Key: testKey }).promise();

      console.log(`Network speed: ${(speed / 1024 / 1024).toFixed(2)} MB/s`);
      return speed;
    } catch (error) {
      console.error('Failed to measure network speed:', error);
      return 5 * 1024 * 1024; // 默认 5MB/s
    }
  }
}

// 使用示例
const smartUploader = new SmartChunkUploader(s3, 'my-bucket');

// 测量网络速度
const networkSpeed = await smartUploader.measureNetworkSpeed();

// 智能上传（自动计算最佳配置）
const file = fs.readFileSync('large-file.zip');
await smartUploader.smartUpload(file, {
  networkSpeed: networkSpeed,
  uploadTimeBudget: 1800 // 30分钟预算
});
```

## 错误处理和恢复

### 1. 完整的错误处理

```javascript
class RobustMultipartUploader extends R2MultipartUploader {
  constructor(s3Client, bucketName) {
    super(s3Client, bucketName);
    this.maxRetries = 3;
    this.retryDelay = 1000; // 1秒
  }

  async uploadFileWithRetry(file, options = {}) {
    const { partSize = this.defaultPartSize } = options;
    const key = options.key || `uploads/${Date.now()}-${file.name}`;
    const fileSize = file.length || file.size;
    const numParts = Math.ceil(fileSize / partSize);

    let uploadId;
    const parts = [];
    const retryCount = {};

    try {
      // 初始化（带重试）
      uploadId = await this.withRetry('initiate', () =>
        this.initiateMultipartUpload(key, options)
      );

      // 上传每个部分（带重试）
      for (let partNumber = 1; partNumber <= numParts; partNumber++) {
        const start = (partNumber - 1) * partSize;
        const end = Math.min(start + partSize, fileSize);
        const partData = file.slice(start, end);

        const part = await this.uploadPartWithRetry(
          uploadId,
          key,
          partNumber,
          partData,
          retryCount[partNumber] || 0
        );

        parts.push(part);
        delete retryCount[partNumber]; // 成功后删除重试计数
      }

      // 完成上传（带重试）
      const result = await this.withRetry('complete', () =>
        this.completeMultipartUpload(uploadId, key, parts)
      );

      return result;

    } catch (error) {
      // 发生错误时清理
      if (uploadId) {
        console.log('Aborting multipart upload due to error...');
        try {
          await this.abortMultipartUpload(uploadId, key);
        } catch (abortError) {
          console.error('Failed to abort upload:', abortError);
        }
      }

      throw error;
    }
  }

  async uploadPartWithRetry(uploadId, key, partNumber, partData, retryCount) {
    while (true) {
      try {
        return await this.uploadPart(uploadId, key, partNumber, partData);
      } catch (error) {
        if (this.shouldRetry(error, retryCount)) {
          const delay = this.calculateRetryDelay(retryCount);
          console.log(`Part ${partNumber} failed, retrying in ${delay}ms... (${error.message})`);
          await this.delay(delay);
          retryCount++;
          continue;
        }
        throw error;
      }
    }
  }

  shouldRetry(error, retryCount) {
    // 达到最大重试次数
    if (retryCount >= this.maxRetries) {
      return false;
    }

    // 可重试的错误类型
    const retryableErrors = [
      'NetworkingError',
      'TimeoutError',
      'ServiceUnavailable',
      'SlowDown',
      'RequestTimeout',
      'RequestTimeoutException'
    ];

    return retryableErrors.some(errorType =>
      error.name === errorType || error.code === errorType
    );
  }

  calculateRetryDelay(retryCount) {
    // 指数退避
    return this.retryDelay * Math.pow(2, retryCount);
  }

  async delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  async withRetry(operation, fn, ...args) {
    let lastError;

    for (let i = 0; i <= this.maxRetries; i++) {
      try {
        return await fn(...args);
      } catch (error) {
        lastError = error;

        if (i < this.maxRetries && this.shouldRetry(error, i)) {
          const delay = this.calculateRetryDelay(i);
          await this.delay(delay);
          continue;
        }

        break;
      }
    }

    throw lastError;
  }

  async abortMultipartUpload(uploadId, key) {
    try {
      await this.s3.abortMultipartUpload({
        Bucket: this.bucket,
        Key: key,
        UploadId: uploadId
      }).promise();
    } catch (error) {
      console.error('Failed to abort multipart upload:', error);
    }
  }
}
```
