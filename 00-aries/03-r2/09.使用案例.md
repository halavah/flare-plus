# R2 使用案例

## 常见使用场景

Cloudflare R2 的零出口费用特性使其在各种场景中都极具优势。本章介绍 R2 的典型使用案例和实现方案。

## 1. 媒体存储和 CDN

### 图片托管服务

```javascript
// 图片托管服务示例
class ImageHostingService {
  constructor(r2Client, bucketName) {
    this.r2 = r2Client;
    this.bucket = bucketName;
    this.imageCache = new Map();
  }

  // 上传图片并生成多个尺寸
  async uploadImage(userId, file, metadata = {}) {
    const imageId = crypto.randomUUID();
    const originalKey = `users/${userId}/images/${imageId}/original`;
    const timestamp = Date.now();

    // 上传原图
    const uploadResult = await this.r2.upload({
      Bucket: this.bucket,
      Key: originalKey,
      Body: file.buffer,
      ContentType: file.mimetype,
      Metadata: {
        userId,
        imageId,
        originalName: file.originalname,
        uploadDate: new Date(timestamp).toISOString(),
        ...metadata
      }
    }).promise();

    // 生成缩略图
    const thumbnails = await this.generateThumbnails(originalKey, file.buffer);

    // 存储缩略图
    for (const [size, buffer] of Object.entries(thumbnails)) {
      await this.r2.upload({
        Bucket: this.bucket,
        Key: `users/${userId}/images/${imageId}/thumbs/${size}`,
        Body: buffer,
        ContentType: 'image/jpeg'
      }).promise();
    }

    // 存储图片信息
    const imageInfo = {
      imageId,
      originalKey,
      originalSize: file.buffer.length,
      thumbnails: Object.keys(thumbnails),
      url: `https://${this.bucket}.${this.r2.endpoint.hostname.split('.')[0]}.r2.cloudflarestorage.com/${originalKey}`,
      ...metadata
    };

    return imageInfo;
  }

  async generateThumbnails(key, imageBuffer) {
    const sharp = require('sharp');
    const sizes = {
      '150x150': { width: 150, height: 150 },
      '300x300': { width: 300, height: 300 },
      '600x600': { width: 600, height: 600 }
    };

    const thumbnails = {};

    for (const [sizeName, { width, height }] of Object.entries(sizes)) {
      thumbnails[sizeName] = await sharp(imageBuffer)
        .resize(width, height)
        .jpeg({ quality: 80 })
        .toBuffer();
    }

    return thumbnails;
  }

  // 获取图片 URL
  async getImageUrl(imageId, size = 'original') {
    const key = `users/${imageId}/images/${size}`;
    const url = await this.r2.getSignedUrl('getObject', {
      Bucket: this.bucket,
      Key: key,
      Expires: 3600 // 1小时有效
    });

    return url;
  }

  // 图片搜索
  async searchImages(userId, options = {}) {
    const params = {
      Bucket: this.bucket,
      Prefix: `users/${userId}/images/`,
      MaxKeys: options.limit || 100
    };

    if (options.continuesToken) {
      params.ContinuationToken = options.continuesToken;
    }

    const result = await this.r2.listObjectsV2(params).promise();
    const images = [];

    for (const object of result.Contents) {
      // 跳过缩略图
      if (object.Key.includes('/thumbs/')) {
        continue;
      }

      images.push({
        key: object.Key,
        lastModified: object.LastModified,
        size: object.Size,
        metadata: object.Metadata || {}
      });
    }

    return {
      images,
      continuationToken: result.NextContinuationToken,
      truncated: result.IsTruncated
    };
  }
}
```

```javascript
// 视频点播服务
class VideoStreamingService {
  constructor(r2Client, bucketName) {
    this.r2 = r2Client;
    this.bucket = bucketName;
  }

  // 上传视频
  async uploadVideo(userId, file) {
    const videoId = crypto.randomUUID();
    const videoKey = `videos/${userId}/${videoId}/original.${file.originalname.split('.').pop()}`;

    // 分块上传大视频文件
    if (file.buffer.length > 5 * 1024 * 1024) {
      return await this.multipartUpload(videoKey, file.buffer);
    }

    // 小文件直接上传
    await this.r2.upload({
      Bucket: this.bucket,
      Key: videoKey,
      Body: file.buffer,
      ContentType: file.mimetype,
      Metadata: {
        userId,
        videoId,
        originalName: file.originalname,
        uploadDate: new Date().toISOString()
      }
    }).promise();

    // 生成转码任务
    await this.transcodeVideo(videoId, videoKey);

    return {
      videoId,
      videoKey,
      originalName: file.originalname,
      size: file.buffer.length
    };
  }

  async multipartUpload(key, buffer) {
    const uploadId = await this.r2.createMultipartUpload({
      Bucket: this.bucket,
      Key: key
    }).promise().promise.UploadId;

    const partSize = 10 * 1024 * 1024; // 10MB
    const numParts = Math.ceil(buffer.length / partSize);
    const parts = [];

    for (let i = 0; i < numParts; i++) {
      const start = i * partSize;
      const end = Math.min((i + 1) * partSize, buffer.length);
      const partData = buffer.slice(start, end);

      const part = await this.r2.uploadPart({
        Bucket: this.bucket,
        Key: key,
        UploadId: uploadId,
        PartNumber: i + 1,
        Body: partData
      }).promise();

      parts.push({
        PartNumber: part.PartNumber,
        ETag: part.ETag
      });
    }

    await this.r2.completeMultipartUpload({
      Bucket: this.bucket,
      Key: key,
      UploadId: uploadId,
      MultipartUpload: { Parts: parts }
    }).promise();
  }

  // 转码视频
  async transcodeVideo(videoId, inputKey) {
    const outputKey = `videos/${videoId}/transcoded`;

    // 生成多种格式
    const formats = [
      { extension: 'mp4', format: 'h264', resolution: '720p' },
      { extension: 'mp4', format: 'h264', resolution: '480p' },
      { extension: 'webm', format: 'vp9', resolution: '720p' }
    ];

    for (const format of formats) {
      // 这里应该调用视频转码服务
      const outputKeyFormat = `${outputKey}/${format.resolution}.${format.extension}`;
      console.log(`Transcoding video ${videoId} to ${format.resolution} ${format.extension}`);

      // 模拟转码结果
      setTimeout(async () => {
        console.log(`Transcoding completed for ${outputKeyFormat}`);
      }, 10000);
    }
  }

  // 获取播放 URL
  async getPlaybackURL(videoId, format = '720p') {
    const videoKey = `videos/${videoId}/transcoded/${format}.mp4`;

    // 生成预签名 URL（有效期24小时）
    const url = await this.r2.getSignedUrl('getObject', {
      Bucket: this.bucket,
      Key: videoKey,
      Expires: 86400
    });

    return url;
  }

  // HLS 流媒体
  async generateHLS(videoId) {
    const segments = [];
    const duration = 10; // 每段10秒
    const segmentCount = 60; // 假设10分钟视频

    // 生成播放列表
    let playlist = '#EXTM3U\n#EXT-X-VERSION:3\n#EXT-X-TARGETDURATION:600\n\n';

    for (let i = 0; i < segmentCount; i++) {
      const segmentKey = `videos/${videoId}/hls/segment_${i}.ts`;
      const url = await this.r2.getSignedUrl('getObject', {
        Bucket: this.bucket,
        Key: segmentKey,
        Expires: 3600
      });

      segments.push({
        url: url,
        duration
      });

      playlist += `#EXTINF:${duration}\nsegment_${i}.ts?token=${url.split('?token=')[1]}\n`;
    }

    playlist += '#EXT-X-ENDLIST';

    return {
      playlist,
      segments
    };
  }
}
```

```javascript
// 数据库备份服务
class DatabaseBackupService {
  constructor(r2Client, bucketName) {
    this.r2 = r2Client;
    this.bucket = bucketName;
    this.compressionLevel = 6; // gzip 压缩级别
  }

  // 创建数据库备份
  async createBackup(database, options = {}) {
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const backupName = `${database}-${timestamp}.sql.gz`;
    const backupKey = `backups/${database}/${backupName}`;

    try {
      // 导出数据库
      const sqlDump = await this.exportDatabase(database, options);

      // 压缩备份
      const compressedBackup = await this.compressData(sqlDump);

      // 上传到 R2
      await this.r2.upload({
        Bucket: this.bucket,
        Key: backupKey,
        Body: compressedBackup,
        ContentType: 'application/gzip',
        Metadata: {
          database,
          timestamp,
          size: compressedBackup.length,
          originalSize: sqlDump.length,
          compressionRatio: ((compressedBackup.length / sqlDump.length) * 100).toFixed(2) + '%'
        }
      }).promise();

      // 记录备份信息
      await this.recordBackup({
        database,
        backupKey,
        name: backupName,
        timestamp,
        size: compressedBackup.length,
        originalSize: sqlDump.length
      });

      return {
        backupKey,
        name: backupName,
        size: compressedBackup.length,
        timestamp
      };
    } catch (error) {
      console.error('Backup failed:', error);
      throw error;
    }
  }

  async exportDatabase(database, options = {}) {
    // 这里应该连接实际的数据库
    // 示例：使用模拟数据
    switch (database) {
      case 'users':
        return await this.exportUsersTable();
      case 'products':
        return await this.exportProductsTable(options);
      default:
        return await this.exportFullDatabase(database);
    }
  }

  async exportUsersTable() {
    // 模拟导出用户表
    const users = [
      { id: 1, name: 'Alice', email: 'alice@example.com', created_at: '2024-01-01' },
      { id: 2, name: 'Bob', email: 'bob@example.com', created_at: '2024-01-02' }
    ];

    return users.map(user =>
      `INSERT INTO users VALUES (${user.id}, '${user.name}', '${user.email}', '${user.created_at}');`
    ).join('\n');
  }

  async compressData(data) {
    const zlib = require('zlib');
    return new Promise((resolve, reject) => {
      zlib.gzip(JSON.stringify(data), { level: this.compressionLevel }, (err, result) => {
        if (err) reject(err);
        else resolve(result);
      });
    });
  }

  // 增量备份
  async createIncrementalBackup(database, lastBackupDate) {
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const backupName = `${database}-${timestamp}-incremental.sql.gz`;
    const backupKey = `backups/${database}/${backupName}`;

    // 获取自上次备份以来的变更
    const changes = await this.getChangesSince(database, lastBackupDate);

    // 压缩并上传
    const compressedChanges = await this.compressData(changes);

    await this.r2.upload({
      Bucket: this.bucket,
      Key: backupKey,
      Body: compressedChanges,
      ContentType: 'application/gzip',
      Metadata: {
        database,
        type: 'incremental',
        fromTimestamp: lastBackupDate,
        toTimestamp: timestamp,
        changes: changes.split('\n').length
      }
    }).promise();

    return {
      backupKey,
      name: backupName,
      type: 'incremental',
      changes: changes.split('\n').length
    };
  }

  // 恢复数据库
  async restoreDatabase(backupKey, targetDatabase) {
    try {
      // 下载备份文件
      const backupData = await this.r2.getObject({
        Bucket: this.bucket,
        Key: backupKey
      }).promise();

      // 解压缩
      const sqlContent = await this.decompressData(backupData.Body);

      // 执行 SQL 语句
      await this.executeSQL(sqlContent, targetDatabase);

      return {
        backupKey,
        sqlCommands: sqlContent.split('\n').length,
        restored: true
      };
    } catch (error) {
      console.error('Restore failed:', error);
      throw error;
    }
  }

  async decompressData(compressedData) {
    const zlib = require('zlib');
    return new Promise((resolve, reject) => {
      zlib.gunzip(compressedData, (err, result) => {
        if (err) reject(err);
        else resolve(result);
      });
    });
  }

  // 自动备份调度
  async scheduleBackup(database, schedule = 'daily') {
    // 使用 Cron Triggers 或定时任务
    console.log(`Scheduling ${schedule} backup for ${database}`);

    switch (schedule) {
      case 'daily':
        // 每天 2AM 备份
        this.scheduleDailyBackup(database, 2, 0);
        break;
      case 'hourly':
        // 每小时备份
        this.scheduleHourlyBackup(database);
        break;
      case 'weekly':
        // 每周备份（周日 3AM）
        this.scheduleWeeklyBackup(database, 0, 3);
        break;
    }
  }

    const cron = `${minute} ${hour} * * *`;
    console.log(`Daily backup for ${database} scheduled at ${cron}`);

  }
}
```

```javascript
// 文件同步服务（类似 Dropbox）
class FileSyncService {
  constructor(r2Client, bucketName) {
    this.r2 = r2Client;
    this.bucket = bucketName;
    this.syncState = new Map();
  }

  // 同步本地文件夹到 R2
  async syncDirectory(localPath, userId) {
    const userPath = `users/${userId}`;

    // 扫描本地文件
    const localFiles = await this.scanLocalDirectory(localPath);

    // 扫描 R2 文件
    const r2Files = await this.scanR2Files(userPath);

    // 计算差异
    const diff = this.calculateDiff(localFiles, r2Files);

    // 上传新文件和更新文件
    for (const item of diff.uploads) {
      await this.uploadFile(`${userPath}/${item.path}`, item.localPath);
    }

    // 删除 R2 中多余的文件
    for (const item of diff.deletes) {
      await this.deleteFile(`${userPath}/${item.path}`);
    }

    // 创建同步状态记录
    const syncRecord = {
      timestamp: new Date().toISOString(),
      userId,
      localPath,
      uploaded: diff.uploads.length,
      deleted: diff.deletes.length,
      synced: true
    };

    await this.saveSyncRecord(syncRecord);
    return syncRecord;
  }

  async scanLocalDirectory(dirPath) {
    const fs = require('fs');
    const path = require('path');
    const files = [];

    async function scan(dir) {
      const items = fs.readdirSync(dir);

      for (const item of items) {
        const fullPath = path.join(dir, item);
        const stats = fs.statSync(fullPath);
        const relativePath = path.relative(dirPath, fullPath);

        if (stats.isDirectory()) {
          await scan(fullPath);
        } else {
          const hash = await this.calculateFileHash(fullPath);
          files.push({
            path: relativePath,
            size: stats.size,
            mtime: stats.mtime,
            hash
          });
        }
      }
    }

    await scan(dirPath);
    return files;
  }

  async scanR2Files(path) {
    const files = [];
    let continuationToken = null;

    do {
      const params = {
        Bucket: this.bucket,
        Prefix: path,
        ContinuationToken: continuationToken
      };

      if (continuationToken) {
        params.ContinuationToken = continuationToken;
      }

      const result = await this.r2.listObjectsV2(params).promise();

      for (const object of result.Contents) {
        const relativePath = object.Key.replace(path, '');
        files.push({
          path: relativePath,
          size: object.Size,
          mtime: new Date(object.LastModified),
          etag: object.ETag
        });
      }

      continuationToken = result.NextContinuationToken;
    } while (continuationToken);

    return files;
  }

  calculateDiff(localFiles, r2Files) {
    const localMap = new Map();
    const r2Map = new Map();

    localFiles.forEach(file => {
      localMap.set(file.path, file);
    });

    r2Files.forEach(file => {
      r2Map.set(file.path, file);
    });

    const uploads = [];
    const deletes = [];

    // 找出需要上传的文件（新增或修改的）
    for (const [path, localFile] of localMap) {
      const r2File = r2Map.get(path);

      if (!r2File) {
        // 新文件
        uploads.push({ path, localFile, type: 'new' });
      } else if (localFile.mtime > new Date(r2File.mtime) || localFile.hash !== r2File.etag.replace(/"/g, '')) {
        // 修改的文件
        uploads.push({ path, localFile, type: 'modified' });
      }
    }

    // 找出需要删除的文件
    for (const [path] of r2Map) {
      if (!localMap.has(path)) {
        deletes.push({ path });
      }
    }

    return { uploads, deletes };
  }

  calculateFileHash(filePath) {
    const crypto = require('crypto');
    const fs = require('fs');
    const data = fs.readFileSync(filePath);
    return crypto.createHash('md5').update(data).digest('hex');
  }

  // 通知功能
  async notifySyncEvent(event) {
    const message = {
      type: 'sync',
      event,
      timestamp: new Date().toISOString()
    };

    // 可以通过 WebSocket、SSE 或其他方式发送通知
    console.log('Sync event:', message);
  }
}
```

### 日志收集服务

```javascript
// 日志收集服务
class LogCollectionService {
  constructor(r2Client, bucketName) {
    this.r2 = r2Client;
    this.bucket = bucketName;
    this.batchSize = 1000; // 批量大小
    this.buffer = [];
  }

  // 收集日志
  async log(level, service, message, metadata = {}) {
    const logEntry = {
      timestamp: new Date().toISOString(),
      level, // info, warn, error, debug
      service,
      message,
      metadata
    };

    this.buffer.push(logEntry);

    // 达到批次大小时上传
    if (this.buffer.length >= this.batchSize) {
      await this.flushLogs();
    }

    // 或者定期上传（比如每5分钟）
    if (!this.flushTimer) {
      this.flushTimer = setTimeout(() => {
        this.flushLogs();
        this.flushTimer = null;
      }, 5 * 60 * 1000); // 5分钟
    }
  }

  // 刷新日志到 R2
  async flushLogs() {
    if (this.buffer.length === 0) {
      return;
    }

    const logsToFlush = [...this.buffer];
    this.buffer = [];

    // 按服务和级别分组
    const groupedLogs = this.groupLogs(logsToFlush);

    for (const [groupKey, logs] of Object.entries(groupedLogs)) {
      const logData = JSON.stringify(logs);
      const logKey = `logs/${groupKey}/${Date.now()}-${Math.random().toString(36).substring(2)}.json`;

      const compressedData = await this.compressLogData(logData);

      await this.r2.upload({
        Bucket: this.bucket,
        Key: logKey,
        Body: compressedData,
        ContentType: 'application/json',
        Metadata: {
          group: groupKey,
          count: logs.length,
          timestamp: new Date().toISOString(),
          compressed: true
        }
      }).promise();
    }
  }

  groupLogs(logs) {
    const grouped = {};

    for (const log of logs) {
      const groupKey = `${log.service}-${log.level}`;
      if (!grouped[groupKey]) {
        grouped[groupKey] = [];
      }
      grouped[groupKey].push(log);
    }

    return grouped;
  }

  compressLogData(data) {
    const zlib = require('zlib');
    return new Promise((resolve, reject) => {
      zlib.gzip(data, { level: 6 }, (err, result) => {
        if (err) reject(err);
        else resolve(result);
      });
    });
  }

  // 查询日志
  async queryLogs(queryOptions = {}) {
    const {
      service,
      level,
      startTime,
      endTime,
      limit = 1000,
      continuationToken
    } = queryOptions;

    let searchPath = 'logs/';

    // 构建搜索路径
    if (service) {
      searchPath += `${service}-`;
    }

    // 构建查询参数
    const params = {
      Bucket: this.bucket,
      Prefix: searchPath,
      MaxKeys: limit
    };

    // 读取日志文件
    const logs = [];
    let token = continuationToken;

    do {
      if (token) {
        params.ContinuationToken = token;
      }

      const result = await this.r2.listObjectsV2(params).promise();

      for (const object of result.Contents) {
        const logData = await this.parseLogFile(object.Key);

        if (this.matchesQuery(logData, queryOptions)) {
          logs.push(...logData);
        }
      }

      token = result.NextContinuationToken;
    } while (token && logs.length < limit);

    return {
      logs: logs.slice(0, limit),
      continuationToken: token
    };
  }

  async parseLogFile(logKey) {
    try {
      const result = await this.r2.getObject({
        Bucket: this.bucket,
        Key: logKey
      }).promise();

      let data = result.Body.toString();

      // 检查是否压缩
      if (result.Metadata && result.Metadata.compressed === 'true') {
        data = await this.decompressLogData(data);
      }

      return JSON.parse(data);
    } catch (error) {
      console.error(`Failed to parse log file ${logKey}:`, error);
      return [];
    }
  }

  decompressLogData(compressedData) {
    const zlib = require('zlib');
    return new Promise((resolve, reject) => {
      zlib.gunzip(compressedData, (err, result) => {
        if (err) reject(err);
        else resolve(result);
      });
    });
  }

  matchesQuery(log, queryOptions) {
    const { service, level, startTime, endTime } = queryOptions;

    if (service && log.service !== service) return false;
    if (level && log.level !== level) return false;

    const logTime = new Date(log.timestamp);

    if (startTime && logTime < startTime) return false;
    if (endTime && logTime > endTime) return false;

    return true;
  }

  // 日志分析
  async analyzeLogs(timeRange = '24h') {
    const endTime = new Date();
    const startTime = new Date(endTime.getTime() - this.parseTimeRange(timeRange));

    const logs = await this.queryLogs({
      startTime,
      endTime,
      limit: 10000
    });

    const analysis = {
      timeRange,
      totalLogs: logs.logs.length,
      byLevel: {},
      byService: {},
      topErrors: [],
      hourlyCounts: {}
    };

    for (const log of logs.logs) {
      // 按级别统计
      analysis.byLevel[log.level] = (analysis.byLevel[log.level] || 0) + 1;

      // 按服务统计
      analysis.byService[log.service] = (analysis.byService[log.service] || 0) + 1;

      // 收集错误信息
      if (log.level === 'error') {
        analysis.topErrors.push({
          message: log.message,
          service: log.service,
          timestamp: log.timestamp
        });
      }

      // 按小时统计
      const hour = new Date(log.timestamp).getHours();
      analysis.hourlyCounts[hour] = (analysis.hourlyCounts[hour] || 0) + 1;
    }

    // 排序错误
    analysis.topErrors.sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));
    analysis.topErrors = analysis.topErrors.slice(0, 10);

    return analysis;
  }

  parseTimeRange(timeRange) {
    const value = parseInt(timeRange);
    const unit = timeRange.slice(-1);

    switch (unit) {
      case 's': return value * 1000;
      case 'm': return value * 60 * 1000;
      case 'h': return value * 60 * 60 * 1000;
      case 'd': return value * 24 * 60 * 60 * 1000;
      default: return value;
    }
  }
}
```
