```javascript
// R2 访问控制管理器
class R2AccessController {
  constructor(s3Client, bucketName) {
    this.s3 = s3Client;
    this.bucket = bucketName;
    this.policies = new Map();
  }

  // 设置访问策略
  async setPolicy(policyId, rules) {
    this.policies.set(policyId, {
      rules,
      createdAt: new Date().toISOString()
    });
  }

  // 检查访问权限
  async checkAccess(userId, resource, action, context = {}) {
    // 收集所有适用的策略
    const applicablePolicies = this.getApplicablePolicies(userId, resource);

    for (const policy of applicablePolicies) {
      if (this.evaluatePolicy(policy, resource, action, context)) {
        return true;
      }
    }

    return false;
  }

  // 获取适用的策略
  getApplicablePolicies(userId, resource) {
    const policies = [];

    // 用户特定策略
    const userPolicies = await this.getUserPolicies(userId);
    policies.push(...userPolicies);

    // 资源特定策略
    const resourcePolicies = await this.getResourcePolicies(resource);
    policies.push(...resourcePolicies);

    // 默认策略
    const defaultPolicies = await this.getDefaultPolicies();
    policies.push(...defaultPolicies);

    return policies;
  }

  evaluatePolicy(policy, resource, action, context) {
    const { rules } = policy;

    for (const rule of rules) {
      if (this.matchesRule(rule, resource, action, context)) {
        return rule.effect === 'allow';
      }
    }

    return false;
  }

  matchesRule(rule, resource, action, context) {
    // 检查资源匹配
    if (rule.resource && !this.matchesResource(rule.resource, resource)) {
      return false;
    }

    // 检查动作匹配
    if (rule.action && rule.action !== '*' && !this.matchesAction(rule.action, action)) {
      return false;
    }

    // 检查条件
    if (rule.conditions) {
      return this.evaluateConditions(rule.conditions, context);
    }

    return true;
  }

  matchesResource(pattern, resource) {
    if (pattern === '*') return true;
    if (pattern.endsWith('*')) {
      return resource.startsWith(pattern.slice(0, -1));
    }
    return pattern === resource;
  }

  matchesAction(pattern, action) {
    return pattern === '*' || pattern === action || pattern.includes(action);
  }

  evaluateConditions(conditions, context) {
    if (!conditions || conditions.length === 0) return true;

    return conditions.every(condition => {
      switch (condition.operator) {
        case 'equals':
          return context[condition.attribute] === condition.value;
        case 'contains':
          return context[condition.attribute] &&
            context[condition.attribute].includes(condition.value);
        case 'startsWith':
          return context[condition.attribute] &&
            context[condition.attribute].startsWith(condition.value);
        case 'endsWith':
          return context[condition.attribute] &&
            context[condition.attribute].endsWith(condition.value);
        default:
          return true;
      }
    });
  }
}
```

```javascript
class R2EncryptionManager {
  constructor(s3Client, bucketName) {
    this.s3 = s3Client;
    this.bucket = bucketName;
    this.encryptionKeys = new Map();
  }

  // 客户端加密
  async uploadWithEncryption(key, data, options = {}) {
    const encryptionKey = options.encryptionKey || await this.getEncryptionKey(options.keyId);

    // 加密数据
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipher('aes-256-cbc', encryptionKey, iv);
    let encrypted = cipher.update(data, 'utf8', 'hex');
    encrypted += cipher.final('hex');

    // 上传加密数据和加密信息
    await this.s3.upload({
      Bucket: this.bucket,
      Key: key,
      Body: Buffer.from(encrypted, 'hex'),
      ContentType: 'application/octet-stream',
      Metadata: {
        encrypted: 'true',
        algorithm: 'AES-256-CBC',
        iv: iv.toString('hex'),
        keyId: options.keyId || 'default'
      }
    }).promise();

    return {
      encrypted: true,
      iv,
      keyId: options.keyId || 'default'
    };
  }

  // 下载时解密
  async downloadWithDecryption(key) {
    const result = await this.s3.getObject({
      Bucket: this.bucket,
      Key: key
    ).promise();

    if (result.Metadata && result.Metadata.encrypted === 'true') {
      const encryptionKey = await this.getEncryptionKey(result.Metadata.keyId);
      const iv = Buffer.from(result.Metadata.iv, 'hex');

      const decipher = crypto.createDecipher('aes-256-cbc', encryptionKey, iv);

const ENCRYPTION_KEYS = {
  'project-alpha': 'base64-encoded-key-1',
  'project-beta': 'base64-encoded-key-2'

// Worker 脚本
export default {
  async fetch(request, env) {
  const url = new URL(request.url);

  // 检查是否需要加密
  if (url.pathname === '/upload') {
      return this.handleEncryptedUpload(request, env);
    }

    if (url.pathname.startsWith('/download/')) {
      return this.handleEncryptedDownload(request, env);
    }

    return new Response('Not Found', { status: 404 });
  },

  async handleEncryptedUpload(request, env) {
    // 验证上传请求
    const formData = await request.formData();
    const file = formData.get('file');
    const encryptionKeyId = formData.get('keyId') || 'default';

    // 获取加密密钥
    const encryptionKey = ENCRYPTION_KEYS[encryptionKeyId];
    if (!encryptionKey) {
      return new Response('Invalid encryption key', { status: 400 });
    }

    // 加密文件
    const fileBuffer = await file.arrayBuffer();
    const iv = crypto.getRandomValues(new Uint8Array(16));
    const cipher = crypto.subtle.importKey(
      'raw',
      Buffer.from(encryptionKey, 'base64'),
      { name: 'AES-GCM', length: 256 }
    );

    const encryptedData = await crypto.subtle.encrypt(
      {
        name: 'AES-GCM',
        iv
      },
      cipher,
      fileBuffer
    );

    // 上传加密数据
    const uploadUrl = `https://${env.R2_BUCKET}.${env.R2_ACCOUNT_ID}.r2.cloudflarestorage.com/${file.name}`;

    await fetch(uploadUrl, {
      method: 'PUT',
      headers: {
        'Content-Type': file.type,
        'X-Amz-Meta-Encrypted': 'true',
        'X-Amz-Meta-Iv': Array.from(iv).join(','),
        'X-Amz-Meta-KeyId': encryptionKeyId
      },
      body: encryptedData
    });

    return new Response(JSON.stringify({
      success: true,
      keyId: encryptionKeyId
    }));
  }
}
```

### 1. 缓存策略

```javascript
// 多层缓存管理器
class R2CacheManager {
  constructor(r2Client, bucketName) {
    this.r2 = r2Client;
    this.bucket = bucketName;

    // L1: 内存缓存
    this.memoryCache = new Map();
    this.memoryCacheSize = 100;
    this.memoryCacheTTL = 5 * 60 * 1000; // 5分钟

    // L2: KV 缓存
    this.kvCache = null; // 应该从环境变量初始化
    this.kvCacheTTL = 10 * 60 * 1000; // 10分钟

    // L3: R2 缓存（通过 HTTP 头）
    this.httpCacheTTL = 3600; // 1小时
  }

  // 获取数据（带多层缓存）
  async get(key, options = {}) {
    // L1: 内存缓存
    const cached = this.getFromMemoryCache(key);
    if (cached) {
      return cached.data;
    }

    // L2: KV 缓存
    if (this.kvCache) {
      const kvCached = await this.getFromKVCache(key);
      if (kvCached) {
        // 写入内存缓存
        this.setToMemoryCache(key, kvCached.data);
        return kvCached.data;
      }
    }

    // L3: 从 R2 获取
    const r2Data = await this.getFromR2(key, options);

    // 写入缓存
    this.setToMemoryCache(key, r2Data);
    if (this.kvCache) {
      await this.setToKVCache(key, r2Data);
    }

    return r2Data;
  }

  async getFromMemoryCache(key) {
    const item = this.memoryCache.get(key);

    if (item && Date.now() < item.expiresAt) {
      // 更新访问时间
      item.accessedAt = Date.now();
      return item;
    }

    if (item && Date.now() >= item.expiresAt) {
      this.memoryCache.delete(key);
    }

    return null;
  }

  setToMemoryCache(key, data) {
    // 如果缓存已满，删除最久未访问的项目
    if (this.memoryCache.size >= this.memoryCacheSize) {
      this.evictLRU();
    }

    this.memoryCache.set(key, {
      data,
      createdAt: Date.now(),
      accessedAt: Date.now(),
      expiresAt: Date.now() + this.memoryCacheTTL
    });
  }

  evictLRU() {
    let oldestTime = Date.now();
    let oldestKey = null;

    for (const [key, item] of this.memoryCache) {
      if (item.accessedAt < oldestTime) {
        oldestTime = item.accessedAt;
        oldestKey = key;
      }
    }

    if (oldestKey) {
      this.memoryCache.delete(oldestKey);
    }
  }

  async getFromKVCache(key) {
    if (!this.kvCache) return null;

    try {
      const cached = await this.kvCache.get(key);

      if (cached) {
        const data = JSON.parse(cached);

        // 检查是否过期
        if (Date.now() > data.expiresAt) {
          await this.kvCache.delete(key);
          return null;
        }

        return data;
      }
    } catch (error) {
      console.error('KV cache error:', error);
    }

    return null;
  }

  async setToKVCache(key, data) {
    if (!this.kvCache) return;

    try {
      await this.kvCache.put(key, JSON.stringify({
        data,
        createdAt: Date.now(),
        expiresAt: Date.now() + this.kvCacheTTL
      }), {
        expirationTtl: this.kvCacheTTL / 1000
      });
    } catch (error) {
      console.error('KV cache error:', error);
    }
  }

  async getFromR2(key, options = {}) {
    const params = {
      Bucket: this.bucket,
      Key: key,
      ResponseCacheControl: `public, max-age=${this.httpCacheTTL}`
    };

    if (options.range) {
      params.Range = options.range;
    }

    return await this.r2.getObject(params).promise();
  }

  // 预热数据
  async preWarmData(keys) {
    for (const key of keys) {
      try {
        await this.get(key); // 触发缓存
      } catch (error) {
        console.error(`Failed to pre-warm ${key}:`, error);
      }
    }
  }

  // 清除缓存
  async clearCache(key) {
    // 清除内存缓存
    this.memoryCache.delete(key);

    // 清除 KV 缓存
    if (this.kvCache) {
      try {
        await this.kvCache.delete(key);
      } catch (error) {
        console.error('Failed to clear KV cache:', error);
      }
    }
  }
}
```

```javascript
// 智能上传优化器
class SmartUploadOptimizer {
  constructor() {
    this.networkSpeed = null;
    this.lastSpeedTest = 0;
    this.speedTestInterval = 5 * 60 * 1000; // 5分钟
  }

  // 优化上传参数
  async optimizeUploadParams(fileSize, options = {}) {
    // 测试网络速度（如果需要）
    if (!this.networkSpeed || Date.now() - this.lastSpeedTest > this.speedTestInterval) {
      this.networkSpeed = await this.testNetworkSpeed();
      this.lastSpeedTest = Date.now();
    }

    const params = {
      partSize: this.calculateOptimalPartSize(fileSize),
      concurrency: this.calculateOptimalConcurrency(fileSize, this.networkSpeed),
      useMultipart: fileSize > 5 * 1024 * 1024 // 5MB
    };

    return {
      ...params,
      ...options,
      ...params
    };
  }

  calculateOptimalPartSize(fileSize) {
    // 根据文件大小自动选择分块大小
    const MB = 1024 * 1024;

    if (fileSize < 10 * MB) {
      return fileSize; // 小文件不分块
    }

    if (fileSize < 100 * MB) {
      return 5 * MB; // 中文件 5MB
    }

    if (fileSize < 1 * 1024 * 1024 * 1024) { // 1GB
      return 10 * MB; // 大文件 10MB
    }

    if (fileSize < 10 * 1024 * 1024 * 1024) { // 10GB
      return 50 * MB; // 超大文件 50MB
    }

    // 超大文件根据网络条件调整
    return this.adjustForNetwork(fileSize);
  }

  adjustForNetwork(fileSize) {
    if (!this.networkSpeed) {
      return 100 * MB; // 默认值
    }

    // 计算最佳分块大小
    // 目标：每分块 30-60 秒上传时间
    const targetTime = 30000; // 30秒
    const bytesPerSecond = this.networkSpeed;
    const partSize = bytesPerSecond * targetTime / 2; // 考虑网络波动

    // 限制在合理范围内
    const minSize = 5 * MB;
    const maxSize = 100 * MB;

    return Math.max(minSize, Math.min(partSize, maxSize));
  }

  calculateOptimalConcurrency(fileSize, networkSpeed) {
    // 根据文件大小和网络速度计算并发数
    const partSize = this.calculateOptimalPartSize(fileSize);
    const numParts = Math.ceil(fileSize / partSize);

    // 基础并发数
    let concurrency = 5;

    // 根据网络速度调整
    if (networkSpeed > 10 * MB) {
      concurrency = 10;
    } else if (networkSpeed < 1 * MB) {
      concurrency = 3;
    }

    // 根据分块数量调整
    if (numParts < 10) {
      concurrency = Math.min(concurrency, numParts);
    } else if (numParts > 100) {
      concurrency = Math.min(concurrency, 20);
    }

    return concurrency;
  }

  async testNetworkSpeed() {
    const testDataSize = 1 * MB; // 1MB 测试数据
    const testKey = `speed-test-${Date.now()}`;

    const startTime = Date.now();

    try {
      const testData = Buffer.alloc(testDataSize, 'x');

      await this.s3.upload({
        Bucket: 'test-bucket',
        Key: testKey,
        Body: testData,
        ContentType: 'application/octet-stream'
      }).promise();

      const duration = Date.now() - startTime;
      this.networkSpeed = testDataSize / (duration / 1000);

      console.log(`Network speed: ${(this.networkSpeed / 1024 / 1024).toFixed(2)} MB/s`);
      return this.networkSpeed;
    } catch (error) {
      console.error('Network speed test failed:', error);
      return this.networkSpeed || 1 * MB;
    }
  }

  // 自适应上传策略
  async adaptiveUpload(file, options = {}) {
    const fileInfo = {
      size: file.length || file.size,
      type: file.type,
      name: file.name || 'unknown'
    };

    // 文件类型优化
    const optimizations = this.getOptimizationsForFileType(fileInfo);

    // 网络优化
    const networkOptimizations = await this.getNetworkOptimizations(fileInfo.size);

    // 组合优化参数
    const optimizedParams = {
      ...optimizations,
      ...networkOptimizations
    };

    console.log(`Optimized upload for ${fileInfo.name}:`, optimizedParams);

    return this.uploadWithParams(file, optimizedParams);
  }

  getOptimizationsForFileType(fileInfo) {
    const type = this.getFileCategory(fileInfo.type);

    switch (type) {
      case 'image':
        return {
          ContentType: 'image/jpeg',
          Metadata: {
            optimized_for: 'web',
            auto_optimize: 'true'
          }
        };

      case 'video':
        return {
          ContentType: 'video/mp4',
          Metadata: {
            optimized_for: 'streaming',
            auto_transcode: 'true'
          }
        };

      case 'text':
      case 'code':
        return {
          ContentType: 'text/plain',
          Metadata: {
            compressed: 'gzip'
          }
        };

      default:
        return {
          ContentType: 'application/octet-stream'
        };
    }
  }

  getFileCategory(mimeType) {
    if (mimeType.startsWith('image/')) return 'image';
    if (mimeType.startsWith('video/')) return 'video';
    if (mimeType.startsWith('audio/')) return 'audio';
    if (mimeType.startsWith('text/') || mimeType.includes('text') || mimeType.includes('json')) return 'text';
    if (mimeType.startsWith('application/')) return 'code';
    return 'binary';
  }

  async getNetworkOptimizations(fileSize) {
    // 使用优化的上传策略
    const strategy = this.selectUploadStrategy(fileSize);

    const params = strategy(fileSize);

    return {
      useMultipart: params.useMultipart,
      partSize: params.partSize,
      concurrency: params.concurrency,
      retryConfig: {
        maxRetries: params.maxRetries || 3,
        retryDelay: params.retryDelay || 1000,
        backoff: params.backoff || 'exponential'
      }
    };
  }

  selectUploadStrategy(fileSize) {
    // 根据文件大小选择最佳策略
    if (fileSize < 5 * 1024 * 1024) {
      // 小文件：直接上传
      return {
        useMultipart: false
      };
    } else if (fileSize < 100 * 1024 * 1024) {
      // 中等文件：多部分上传，中等并发
      return {
        useMultipart: true,
        partSize: 5 * 1024 * 1024,
        concurrency: 5,
        maxRetries: 3
      };
    } else {
      // 大文件：多部分上传，高并发，更多重试
      return {
        useMultipart: true,
        partSize: 10 * 1024 * 1024,
        concurrency: 10,
        maxRetries: 5
      };
    }
  }
}
```

```javascript
    }, 30000); // 30秒
  }

  async collectMetrics() {
    const timestamp = Date.now();
    const metrics = {
      timestamp,
      uploadSpeed: await this.measureUploadSpeed(),
      downloadSpeed: await this.measureDownloadSpeed(),
      errorRate: await this.calculateErrorRate()
    };

    this.metrics.set(timestamp, metrics);

    // 检查性能阈值
    this.checkPerformanceThresholds(metrics);
  }

  async measureUploadSpeed() {
    const testDataSize = 1 * 1024 * 1024; // 1MB
    const startTime = Date.now();

    try {
      await this.s3.upload({
        Bucket: 'performance-test',
        Key: `upload-test-${timestamp}`,
        Body: Buffer.alloc(testDataSize, 'x'),
        ContentType: 'application/octet-stream'
      }).promise();

      const duration = Date.now() - startTime;
      return (testDataSize / (duration / 1000)) * 8; // Mbps
    } catch (error) {
      console.error('Upload speed test failed:', error);
      return 0;
    }
  }

  async measureDownloadSpeed() {
    const startTime = Date.now();

    try {
      await this.s3.getObject({
        Bucket: 'performance-test',
        Key: 'download-test'
      }).promise();

      const duration = Date.now() - startTime;

      // 模拟下载 1MB 数据
      const testDataSize = 1 * 1024 * 1024;
      return (testDataSize / (duration / 1000)) * 8; // Mbps
    } catch (error) {
      console.error('Download speed test failed:', error);
      return 0;
    }
  }

  async calculateErrorRate() {
    // 模拟错误率计算
    // 实际应该从错误日志中统计
    const totalRequests = 1000; // 示例值
    const errorCount = Math.floor(Math.random() * 50); // 0-5%

    return errorCount / totalRequests * 100;
  }

  checkPerformanceThresholds(metrics) {
    const thresholds = {
      uploadSpeed: {
        warning: 5,   // 5 Mbps
        critical: 1   // 1 Mbps
      },
      downloadSpeed: {
        warning: 10,  // 10 Mbps
        critical: 5   // 5 Mbps
      },
      errorRate: {
        warning: 5,   // 5%
        critical: 10  // 10%
      }
    };

    for (const [metric, threshold] of Object.entries(thresholds)) {
      const value = metrics[metric];

      if (value >= threshold.critical) {
        this.sendAlert(`Critical: ${metric} is ${value} ${metric}`, 'critical');
      } else if (value >= threshold.warning) {
        this.sendAlert(`Warning: ${metric} is ${value} ${metric}`, 'warning');
      }
    }
  }

  sendAlert(message, severity) {
    console.log(`[${severity.toUpperCase()}] ${message}`);

    this.alerts.push({
      timestamp: new Date().toISOString(),
      message,
      severity
    });

    // 这里可以发送到监控系统
    this.sendToMonitoringSystem(message, severity);
  }

  async sendToMonitoringSystem(message, severity) {
    // 发送到外部监控系统
    // 示例：Slack, Datadog, Prometheus 等
    console.log(`Sending alert to monitoring: ${message} (${severity})`);
  }

  monitorMemoryUsage() {
    const memoryUsage = process.memoryUsage();
    const usagePercent = (memoryUsage.heapUsed / memoryUsage.heapTotal) * 100;

    if (usagePercent > 90) {
      this.sendAlert(`High memory usage: ${usagePercent.toFixed(1)}%`, 'warning');
    }
  }

  // 生成性能报告
  generateReport() {
    const metrics = Array.from(this.metrics.values()).sort((a, b) => a.timestamp - b.timestamp);

    if (metrics.length === 0) {
      return { message: 'No metrics available' };
    }

    const latest = metrics[metrics.length - 1];
    const hourAgo = new Date(Date.now() - 3600000);
    const hourAgoMetrics = metrics.filter(m => m.timestamp > hourAgo);

    return {
      current: latest,
      average: this.calculateAverage(hourAgoMetrics),
      trends: this.calculateTrends(hourAgoMetrics),
      alerts: this.alerts
    };
  }

  calculateAverage(metrics) {
    if (metrics.length === 0) return {};

    const sum = metrics.reduce((acc, m) => ({
      uploadSpeed: acc.uploadSpeed + m.uploadSpeed,
      downloadSpeed: acc.downloadSpeed + m.downloadSpeed,
      errorRate: acc.errorRate + m.errorRate
    }), {
      uploadSpeed: 0,
      downloadSpeed: 0,
      errorRate: 0
    });

    return {
      uploadSpeed: sum.uploadSpeed / metrics.length,
      downloadSpeed: sum.downloadSpeed / metrics.length,
      errorRate: sum.errorRate / metrics.length
    };
  }

  calculateTrends(metrics) {
    if (metrics.length < 2) return {};

    const first = metrics[0];
    const last = metrics[metrics.length - 1];
    const timeDiff = (last.timestamp - first.timestamp) / (1000 * 60); // 分钟

    return {
      uploadSpeedTrend: (last.uploadSpeed - first.uploadSpeed) / timeDiff,
      downloadSpeedTrend: (last.downloadSpeed - first.downloadSpeed) / timeDiff,
      errorRateTrend: (last.errorRate - first.errorRate) / timeDiff
    };
  }
}

// 使用示例
const monitor = new R2PerformanceMonitor(s3, 'my-bucket');

// 获取性能报告
const report = monitor.generateReport();
console.log('Performance Report:', report);

// 查看最近的警报
console.log('Recent alerts:', monitor.alerts);
```

```javascript
// 故障排除指南
class R2Troubleshooter {
  constructor(s3Client, bucketName) {
    this.s3 = s3;
    this.s3 = s3;
  }

  // 诊断连接问题
  async diagnoseConnection() {
    const issues = [];

    try {
      // 1. 检查凭证
      console.log('Checking credentials...');
      await this.testCredentials();

      // 2. 检查网络连接
      console.log('Testing network connection...');
      await this.testNetworkConnection();

      // 3. 检查存储桶访问
      console.log('Testing bucket access...');
      await this.testBucketAccess();

      console.log('All checks passed');
    } catch (error) {
      issues.push(error.message);
    }

    return {
      status: issues.length === 0 ? 'OK' : 'ERROR',
      issues
    };
  }

  async testCredentials() {
    try {
      await this.s3.headBucket({ Bucket: this.bucket }).promise();
    } catch (error) {
      throw new Error(`Credentials error: ${error.message}`);
    }
  }

  async testNetworkConnection() {
    try {
      // 创建一个测试对象
      const testKey = `test-connection-${Date.now()}`;
      await this.s3.putObject({
        Bucket: this.bucket,
        Key: testKey,
        Body: 'connection-test'
      }).promise();

      // 删除测试对象
      await this.s3.deleteObject({
        Bucket: this.bucket,
        Key: testKey
      }).promise();
    } catch (error) {
      throw new Error(`Network error: ${error.message}`);
    }
  }

  async testBucketAccess() {
    try {
      await this.s3.headBucket({ Bucket: this.bucket }).promise();
      await this.s3.listObjects({
        Bucket: this.bucket,
        MaxKeys: 1
      }).promise();
    } catch (error) {
      throw new Error(`Bucket access error: ${error.message}`);
    }
  }

  // 诊断上传问题
  async diagnoseUpload(issue) {
    const solutions = [];

    switch (issue.type) {
      case 'timeout':
        solutions.push({
          title: 'Upload Timeout',
          solution: [
            '使用多部分上传',
            '增加重试次数',
            '检查网络连接',
            '减小分块大小'
          ]
        });
        break;

      case 'access_denied':
        solutions.push({
          title: 'Access Denied',
          solution: [
            '检查访问密钥',
            '检查权限配置',
            '检查存储桶策略',
            '检查 CORS 配置'
          ]
        });
        };

      case 'file_too_large':
        solutions.push({
          'title: 'File Too Large',
          solution: [
            '使用多部分上传',
            '减小文件大小',
            '检查单次上传限制'
          ]
        });

      case 'storage_quota':
        {
          solutions.push({
            'title: 'Storage Quota Exceeded',
            solution: [
              '清理旧文件',
              '检查免费额度',
              '升级存储方案'
            ]
          });
        }

      default:
        solutions.push({
          title: 'Unknown Upload Error',
          solution: [
            '检查错误日志',
            '验证文件格式',
            '检查文件权限'
          ]
        });
    }

    return solutions;
  }

  // 常见错误解决方案
  getCommonErrors() {
    return {
      '403 Forbidden': {
        'description': '访问被拒绝',
        'solutions': [
          '检查 AWS 密钥是否正确配置',
          '检查 IP 白名单设置',
          '检查存储桶策略'
        ]
      },

      '404 Not Found': {
        '检查文件路径',
        'solutions': [
          '确认文件是否存在',
          '检查文件名拼写',
          '检查权限'
        ]
      }
    };
  }
}
```
