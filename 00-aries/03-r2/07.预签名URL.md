# R2 预签名 URL

## 预签名 URL 概述

预签名 URL 提供了一种安全的方式，让用户在不暴露凭证的情况下临时访问 R2 中的私有对象。这对于直接上传、下载和移动文件非常有用。

```javascript
    throw error;
  }
}
```

```javascript
// 生成带自定义查询参数的预签名 URL
async function getCustomPresignedUrl(bucketName, key, customParams = {}) {
  const params = {
    Bucket: bucketName,
    Key: key,
    Expires: customParams.expiresIn || 3600
  };

  // 添加查询参数
  const queryParams = new URLSearchParams();
  if (customParams.versionId) {
    params.VersionId = customParams.versionId;
    queryParams.set('versionId', customParams.versionId);
  }

  if (customParams.responseContentType) {
    params.ResponseContentType = customParams.responseContentType;
    queryParams.set('response-content-type', customParams.responseContentType);
  }

  if (customParams.responseContentDisposition) {
    params.ResponseContentDisposition = customParams.responseContentDisposition;
    queryParams.set('response-content-disposition', customParams.responseContentDisposition);
  }

  try {
    const baseUrl = await s3.getSignedUrl('getObject', params);
    const url = new URL(baseUrl);

    // 添加自定义查询参数
    queryParams.forEach((value, key) => {
      url.searchParams.set(key, value);
    });

    return url.toString();
  } catch (error) {
    console.error('Error generating custom URL:', error);
    throw error;
  }
}

// 使用示例
const downloadWithFilename = await getCustomPresignedUrl('my-bucket', 'report.pdf', {
  expiresIn: 1800,
  responseContentDisposition: 'attachment; filename="monthly-report.pdf"',

});
```

```javascript
class PresignedURLManager {
  constructor(s3Client) {
    this.s3 = s3Client;
  }

  // 批量生成下载 URL
  async generateDownloadURLs(bucketName, keys, expiresIn = 3600) {
    const urls = {};

    for (const key of keys) {
      try {
        urls[key] = await this.s3.getSignedUrl('getObject', {
          Bucket: bucketName,
          Key: key,
          Expires: expiresIn
        });
      } catch (error) {
        urls[key] = null;
        console.error(`Failed to generate URL for ${key}:`, error.message);
      }
    }

    return urls;
  }

  // 批量生成上传 URL
  async generateUploadURLs(bucketName, files, expiresIn = 3600) {
    const urls = {};

    for (const file of files) {
      try {
        urls[file.key] = await this.s3.getSignedUrl('putObject', {
          Bucket: bucketName,
          Key: file.key,
          Expires: expiresIn,
          ContentType: file.contentType
        });
      } catch (error) {
        urls[file.key] = null;
        console.error(`Failed to generate upload URL for ${file.key}:`, error.message);
      }
    }

    return urls;
  }

  // 生成带元数据的上传 URL
  async generateUploadURLWithMetadata(bucketName, key, metadata, expiresIn = 3600) {
    const params = {
      Bucket: bucketName,
      Key: key,
      Expires: expiresIn,
      Metadata: metadata
    };

    try {
      return await this.s3.getSignedUrl('putObject', params);
    } catch (error) {
      console.error('Error generating upload URL with metadata:', error);
      throw error;
    }
  }
}

// 使用示例
const manager = new PresignedURLManager(s3);
```

### 2. URL 签名验证

```javascript
// 验证预签名 URL（用于调试）
class PresignedURLValidator {
  constructor(accessKeyId, secretAccessKey, region = 'auto') {
    this.accessKeyId = accessKeyId;
    this.secretAccessKey = secretAccessKey;
    this.region = region;
  }

  validateURL(url) {
    const parsedUrl = new URL(url);
    const hostname = parsedUrl.hostname;
    const path = parsedUrl.pathname;
    const searchParams = parsedUrl.searchParams;

    // 提取 R2 组件
    const bucket = hostname.split('.')[0];
    const key = path.substring(1);

    // 提取签名参数
    const amzDate = searchParams.get('X-Amz-Date');
    const expires = searchParams.get('X-Amz-Expires');
    const signature = searchParams.get('X-Amz-Signature');

    // 检查 URL 是否过期
    if (this.isURLExpired(amzDate, expires)) {
      return { valid: false, reason: 'URL has expired' };
    }

    // 重新计算签名
    const expectedSignature = this.calculateSignature(
      bucket,
      key,
      amzDate,
      expires,
      parsedUrl
    );

    // 比较签名
    const isValid = this.compareSignatures(signature, expectedSignature);

    return {
      valid: isValid,
      bucket,
      key,
      expiresAt: this.getExpirationTime(amzDate, expires)
    };
  }

  isURLExpired(amzDate, expires) {
    const requestTime = new Date(amzDate);
    const expirationTime = new Date(requestTime.getTime() + expires * 1000);
    return Date.now() > expirationTime.getTime();
  }

  getExpirationTime(amzDate, expires) {
    const requestTime = new Date(amzDate);
    const expirationTime = new Date(requestTime.getTime() + expires * 1000);
    return expirationTime.toISOString();
  }

  calculateSignature(bucket, key, amzDate, expires, url) {
    // 这里应该实现 AWS4-HMAC-SHA256 签名算法
    // 简化版本，实际实现会更复杂
    const crypto = require('crypto');
    const stringToSign = this.createStringToSign(bucket, key, amzDate, expires, url);
    const signingKey = this.getSigningKey();
    const signature = crypto.createHmac('sha256', signingKey)
      .update(stringToSign)
      .digest('hex');

    return signature;
  }

  createStringToSign(bucket, key, amzDate, expires, url) {
    const host = url.hostname;
    const method = 'GET';
    const canonicalUri = url.pathname;
    const canonicalQuerystring = url.search
      .replace(/X-Amz-Signature=[^&]*/, '')
      .slice(1);
    const canonicalHeaders = `host:${host}\n`;
    const signedHeaders = 'host';

    return [
      method,
      canonicalUri,
      canonicalQuerystring,
      canonicalHeaders,
      signedHeaders,
      crypto.createHash('sha256').update('').digest('hex')
    ].join('\n');
  }

  getSigningKey() {
    const crypto = require('crypto');
    const date = new Date().toISOString().split('T')[0].replace(/-/g, '');
    const service = 's3';

    const kDate = crypto.createHmac('sha256', 'AWS4' + this.secretAccessKey)
      .update(date).digest();
    const kRegion = crypto.createHmac('sha256', kDate)
      .update(this.region).digest();
    const kService = crypto.createHmac('sha256', kRegion)
      .update(service).digest();
    const kSigning = crypto.createHmac('sha256', kService)
      .update('aws4_request').digest();

    return kSigning;
  }

  compareSignatures(sig1, sig2) {
    return sig1.toLowerCase() === sig2.toLowerCase();
  }
}

// 使用示例
const validator = new PresignedURLValidator(accessKey, secretKey);
const result = validator.validateURL(presignedUrl);
console.log('URL validation:', result);
```

```javascript
class SecurePresignedURLManager {
  constructor(s3Client, bucketName) {
    this.s3 = s3Client;
    this.bucketName = bucketName;
    this.accessLog = [];
  }

  // 带访问控制的 URL 生成
  async generateSecureURL(key, userId, permissions = 'read', expiresIn = 3600) {
    // 验证用户权限
    const hasPermission = await this.checkUserPermission(userId, key, permissions);
    if (!hasPermission) {
      throw new Error('Access denied');
    }

    // 记录访问请求
    await this.logAccessRequest(userId, key, permissions);

    // 生成令牌
    const token = this.generateAccessToken(userId, key, permissions, expiresIn);

    // 生成带令牌的 URL
    const operation = permissions === 'write' ? 'putObject' : 'getObject';
    const params = {
      Bucket: this.bucketName,
      Key: key,
      Expires: expiresIn
    };

    // 添加自定义元数据
    if (permissions === 'write') {
      params.Metadata = {
        uploaded_by: userId,
        access_token: token
      };
    }

    const url = await this.s3.getSignedUrl(operation, params);

    return {
      url,
      token,
      expiresAt: new Date(Date.now() + expiresIn * 1000),
      permissions
    };
  }

  async checkUserPermission(userId, key, permission) {
    // 实现实际的权限检查逻辑
    // 可以查询数据库、KV 或其他存储系统

    // 示例：检查用户是否拥有资源
    const resourceOwner = await this.getResourceOwner(key);
    const isAdmin = await this.isAdmin(userId);

    return isAdmin || resourceOwner === userId;
  }

  async logAccessRequest(userId, key, permission) {
    const logEntry = {
      timestamp: new Date().toISOString(),
      userId,
      key,
      permission,
      action: 'url_generated'
    };

    this.accessLog.push(logEntry);

    // 可以存储到数据库或发送到日志服务
    console.log('Access logged:', logEntry);
  }

  generateAccessToken(userId, key, permissions, expiresIn) {
    const payload = {
      sub: userId,
      key: key,
      permissions: permissions,
      iat: Math.floor(Date.now() / 1000),
      exp: Math.floor((Date.now() + expiresIn * 1000) / 1000)
    };

    // 这里应该使用 JWT 或其他安全令牌生成
    return Buffer.from(JSON.stringify(payload)).toString('base64');
  }

  // 验证访问令牌
  async validateAccess(token) {
    try {
      const payload = JSON.parse(Buffer.from(token, 'base64').toString());

      if (Date.now() / 1000 > payload.exp) {
        throw new Error('Token expired');
      }

      return payload;
    } catch (error) {
      throw new Error('Invalid token');
    }
  }

  // 获取访问日志
  getAccessLogs(userId = null) {
    if (userId) {
      return this.accessLog.filter(log => log.userId === userId);
    }
    return this.accessLog;
  }
}
```

```javascript
// 生成短期 URL，增加安全性
class ShortTermURLManager {
  constructor(s3Client, bucketName) {
    this.s3 = s3Client;
    this.bucketName = bucketName;
    this.urlCache = new Map();
  }

  // 生成短期 URL（15分钟内）
  async generateShortTermURL(key, options = {}) {
    const expiresIn = Math.min(options.expiresIn || 900, 900); // 最长15分钟

    // 检查缓存
    const cacheKey = `${key}:${expiresIn}`;
    if (this.urlCache.has(cacheKey)) {
      const cached = this.urlCache.get(cacheKey);
      if (Date.now() - cached.createdAt < 60000) { // 1分钟缓存
        return cached.url;
      }
    }

    const url = await this.s3.getSignedUrl('getObject', {
      Bucket: this.bucketName,
      Key: key,
      Expires: expiresIn
    });

    // 缓存 URL
    this.urlCache.set(cacheKey, {
      url,
      createdAt: Date.now()
    });

    return url;
  }

  // 生成一次性 URL
  async generateOneTimeURL(key) {
    const randomSuffix = crypto.randomBytes(8).toString('hex');
    const tempKey = `${key}?oneTime=${randomSuffix}`;

    const url = await this.s3.getSignedUrl('getObject', {
      Bucket: this.bucketName,
      Key: tempKey,
      Expires: 300 // 5分钟
    });

    // 存储一次性令牌
    await this.storeOneTimeToken(key, randomSuffix);

    return url;
  }

  // 验证一次性使用
  async validateOneTimeUse(key, token) {
    const storedToken = await this.getOneTimeToken(key);

    if (storedToken !== token) {
      throw new Error('Invalid one-time token');
    }

    // 删除令牌，确保只能使用一次
```

```javascript
// 媒体分享服务
class MediaShareService {
  constructor(s3Client, bucketName) {
    this.s3 = s3Client;
    this.bucketName = bucketName;
    this.shareLinks = new Map();
  }

  // 生成分享链接
  async generateShareLink(mediaId, expiresIn = 86400) { // 默认24小时
    const media = await this.getMediaInfo(mediaId);

    const shareToken = this.generateShareToken(mediaId);
    const url = await this.s3.getSignedUrl('getObject', {
      Bucket: this.bucketName,
      Key: media.fileKey,
      ResponseContentDisposition: `inline; filename="${media.fileName}"`,
      ResponseContentType: media.mimeType,
      Expires: expiresIn
    });

    const shareInfo = {
      mediaId,
      url,
      token: shareToken,
      expiresAt: new Date(Date.now() + expiresIn * 1000),
      downloadCount: 0
    };

    this.shareLinks.set(shareToken, shareInfo);

    return {
      shareUrl: url,
      token: shareToken,
      expiresAt: shareInfo.expiresAt
    };
  }

  // 跟踪下载
  async trackDownload(token) {
    const shareInfo = this.shareLinks.get(token);

    if (!shareInfo || Date.now() > shareInfo.expiresAt) {
      throw new Error('Invalid or expired share link');
    }

    shareInfo.downloadCount++;
    this.shareLinks.set(token, shareInfo);

    await this.logDownload(shareInfo.mediaId, shareInfo.downloadCount);
  }

  generateShareToken(mediaId) {
    const timestamp = Date.now();
    const random = Math.random().toString(36).substring(2);
    return `${mediaId}-${timestamp}-${random}`;
  }

  async logDownload(mediaId, downloadCount) {
    console.log(`Media ${mediaId} downloaded ${downloadCount} times`);
  }
}
```

## 性能优化

### 1. URL 缓存

```javascript
class PresignedURLCache {
  constructor(maxSize = 1000) {
    this.cache = new Map();
    this.maxSize = maxSize;
  }

  get(key) {
    const item = this.cache.get(key);
    if (item && Date.now() < item.expiresAt) {
      // 更新访问时间
      item.accessedAt = Date.now();
      return item.url;
    }
    return null;
  }

  set(key, url, ttl = 3600) {
    // 如果缓存已满，删除最久未访问的项目
    if (this.cache.size >= this.maxSize) {
      this.evictLRU();
    }

    this.cache.set(key, {
      url,
      createdAt: Date.now(),
      accessedAt: Date.now(),
      expiresAt: Date.now() + (ttl * 1000)
    });
  }

  evictLRU() {
    let oldestTime = Date.now();
    let oldestKey = null;

    for (const [key, item] of this.cache) {
      if (item.accessedAt < oldestTime) {
        oldestTime = item.accessedAt;
        oldestKey = key;
      }
    }

    if (oldestKey) {
      this.cache.delete(oldestKey);
    }
  }

  clear() {
    this.cache.clear();
  }

  stats() {
    return {
      size: this.cache.size,
      maxSize: this.maxSize
    };
  }
}

// 使用缓存的预签名 URL 管理器
class CachedPresignedURLManager extends PresignedURLManager {
  constructor(s3Client, bucketName, cacheSize = 1000) {
    super(s3Client, bucketName);
    this.cache = new PresignedURLCache(cacheSize);
  }

  async getDownloadURL(key, expiresIn = 3600) {
    const cacheKey = `${key}:${expiresIn}:download`;

    // 检查缓存
    const cachedURL = this.cache.get(cacheKey);
    if (cachedURL) {
      return cachedURL;
    }

    // 生成新 URL
    const url = await super.getDownloadURL(key, expiresIn);

    // 缓存 URL
    this.cache.set(cacheKey, url, Math.min(expiresIn / 2, 1800)); // 缓存一半时间

    return url;
  }
}
```
