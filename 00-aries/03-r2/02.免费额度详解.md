class SmartStorage {
  async store(file) {
    // 小文件直接存储
    if (file.size < 1 * 1024 * 1024) { // 1MB
      return r2.upload(file.name, file);
    }

    // 中等文件压缩
    if (file.size < 10 * 1024 * 1024) { // 10MB

      return r2.upload(file.name, compressed);

  async retrieve(key) {
    // 检查缓存

    if (cached) return cached;

    // 从 R2 获取
    const object = await r2.get(key);

    // 缓存热门文件
    await this.cacheIfNeeded(key, object);

    return object;
  }
}

## 成本估算工具

### 使用量计算器

```javascript
// R2 Cost Calculator
class R2CostCalculator {
  constructor() {
    this.rates = {
      storage: 0.015,      // $0.015 per GB/month
      class_a: 4.5,        // $4.50 per million
      class_b: 0.36        // $0.36 per million
    };

    this.freeTier = {
      storage: 10,         // 10 GB
      class_a: 1000000,    // 1 million
      class_b: 10000000    // 10 million
    };
  }

  calculate(usage) {
    const costs = {
      storage: this.calculateStorageCost(usage.storage_gb),
      class_a: this.calculateOperationCost(usage.class_a, 'class_a'),
      class_b: this.calculateOperationCost(usage.class_b, 'class_b')
    };

    costs.total = costs.storage + costs.class_a + costs.class_b;

    return costs;
  }

  calculateStorageCost(gb) {
    if (gb <= this.freeTier.storage) return 0;
    return (gb - this.freeTier.storage) * this.rates.storage;
  }

  calculateOperationCount(operations, type) {
    const freeOps = this.freeTier[type];
    const paidOps = Math.max(0, operations - freeOps);
    return (paidOps / 1000000) * this.rates[type];
  }
}

// 使用示例
const calculator = new R2CostCalculator();
const usage = {
  storage_gb: 15,      // 15 GB
  class_a: 2000000,    // 2 million
  class_b: 20000000    // 20 million
};

const cost = calculator.calculate(usage);
console.log(cost);
// { storage: 0.075, class_a: 4.5, class_b: 3.6, total: 8.175 }
```

## 最佳实践

### 1. 监控和告警

```javascript
// 设置使用量告警
const alertThresholds = {
  storage: 9,           // 90% of 10GB
  class_a: 900000,      // 90% of 1M
  class_b: 9000000      // 90% of 10M
};

function checkAndAlert(usage) {
  Object.entries(alertThresholds).forEach(([metric, threshold]) => {
    if (usage[metric] >= threshold) {
      sendAlert(`${metric} usage at ${((usage[metric] / threshold) * 100).toFixed(1)}%`);
    }
  });
}
```

### 2. 自动清理

```javascript
// 自动清理过期文件
class AutoCleanup {
  async cleanup() {
    const cutoffDate = new Date();
    cutoffDate.setMonth(cutoffDate.getMonth() - 6); // 6个月前

    const oldObjects = await this.listOldObjects(cutoffDate);

    for (const obj of oldObjects) {
      await this.archiveOrDelete(obj);
    }
  }

  async archiveOrDelete(obj) {
    // 移动到归档存储桶
    await r2.copyObject({
      SourceBucket: 'main-bucket',
      SourceKey: obj.key,
      DestinationBucket: 'archive-bucket',
      DestinationKey: obj.key,
      Metadata: { archived_at: new Date().toISOString() }
    });

    // 删除原文件
    await r2.deleteObject('main-bucket', obj.key);
  }
}
```
