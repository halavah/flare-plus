# R2 S3 兼容性

## S3 兼容性概述

Cloudflare R2 提供与 Amazon S3 完全兼容的 API，让你可以使用现有的 S3 工具、SDK 和应用程序无缝迁移到 R2，享受零出口费用的优势。

## 兼容性级别

### 完全兼容的操作

```javascript
// ✅ 完全支持的操作
核心对象操作:
- PUT Object (上传文件)
- GET Object (下载文件)
- DELETE Object (删除文件)
- HEAD Object (获取元数据)
- LIST Objects (列出对象)

多部分上传:
- CreateMultipartUpload
- UploadPart
- CompleteMultipartUpload
- AbortMultipartUpload
- ListParts

存储桶操作:
- CreateBucket (创建存储桶)
- DeleteBucket (删除存储桶)
- ListBuckets (列出存储桶)
- HeadBucket (检查存储桶)
```

### 部分兼容的操作

```javascript
// ⚠️ 部分支持或有限制
版本控制:
- 基础版本控制支持
- 无 MFA Delete 支持

生命周期管理:
- 基础转换规则
- 无智能分层存储

对象锁定:
- 基础支持
- 无 WORM 锁定

事件通知:
- 支持基础通知
- 无 SQS/SNS 集成
```

### 3. Go

```go
package main

import (
    "context"
    "fmt"
    "io"
    "net/http"
    "os"

    "github.com/aws/aws-sdk-go-v2/config"
    "github.com/aws/aws-sdk-go-v2/service/s3"
)

func main() {
    // 配置 AWS SDK for R2
    cfg, err := config.LoadDefaultConfig(context.TODO(),
        config.WithRegion("auto"),
        config.WithEndpointResolver(aws.EndpointResolverFunc(func(service, region string, options ...interface{}) (aws.Endpoint, error) {
            return aws.Endpoint{
                URL: "https://your-account-id.r2.cloudflarestorage.com",
            }, nil
        })),
    )
    if err != nil {
        panic(err)
    }

    client := s3.NewFromConfig(cfg)

    // 上传文件
    file, err := os.Open("example.txt")
    if err != nil {
        panic(err)
    }
    defer file.Close()

    _, err = client.PutObject(context.TODO(), &s3.PutObjectInput{
        Bucket: aws.String("my-bucket"),
        Key:    aws.String("example.txt"),
        Body:   file,
    })
    if err != nil {
        panic(err)
    }

    fmt.Println("Upload successful!")
}
```

## 测试兼容性

### 1. 兼容性测试脚本

```javascript
// compatibility-test.js
const AWS = require('aws-sdk');

class S3CompatibilityTester {
  constructor(s3Client) {
    this.s3 = s3Client;
    this.results = {
      passed: [],
      failed: [],
      warnings: []
    };
  }

  async runTests() {
    console.log('🧪 Starting S3 Compatibility Tests\n');

    // 基础操作测试
    await this.testBasicOperations();

    // 多部分上传测试
    await this.testMultipartUpload();

    // 元数据测试
    await this.testMetadata();

    // ACL 测试（预期失败）
    await this.testACL();

    // 生成报告
    this.generateReport();
  }

  async testBasicOperations() {
    const bucket = 'compatibility-test-' + Date.now();
    const key = 'test-file.txt';
    const content = 'Hello, R2 Compatibility Test!';

    try {
      // 创建存储桶
      await this.s3.createBucket({ Bucket: bucket }).promise();
      this.results.passed.push('✓ CreateBucket');

      // 上传对象
      await this.s3.putObject({
        Bucket: bucket,
        Key: key,
        Body: content,
        ContentType: 'text/plain'
      }).promise();
      this.results.passed.push('✓ PutObject');

      // 获取对象
      const getObject = await this.s3.getObject({
        Bucket: bucket,
        Key: key
      }).promise();

      if (getObject.Body.toString() === content) {
        this.results.passed.push('✓ GetObject');
      } else {
        this.results.failed.push('✗ GetObject - Content mismatch');
      }

      // 列出对象
      const listObjects = await this.s3.listObjects({
        Bucket: bucket
      }).promise();

      if (listObjects.Contents.length === 1) {
        this.results.passed.push('✓ ListObjects');
      } else {
        this.results.failed.push('✗ ListObjects - Wrong count');
      }

      // 删除对象
      await this.s3.deleteObject({
        Bucket: bucket,
        Key: key
      }).promise();
      this.results.passed.push('✓ DeleteObject');

      // 删除存储桶
      await this.s3.deleteBucket({ Bucket: bucket }).promise();
      this.results.passed.push('✓ DeleteBucket');

    } catch (error) {
      this.results.failed.push(`✗ Basic Operations: ${error.message}`);
    }
  }

  async testMultipartUpload() {
    const bucket = 'multipart-test-' + Date.now();
    const key = 'large-file.bin';
    const content = Buffer.alloc(20 * 1024 * 1024, 'x'); // 20MB

    try {
      await this.s3.createBucket({ Bucket: bucket }).promise();

      // 初始化多部分上传
      const uploadId = (await this.s3.createMultipartUpload({
        Bucket: bucket,
        Key: key
      }).promise()).UploadId;

      this.results.passed.push('✓ CreateMultipartUpload');

      // 上传部分
      const partSize = 5 * 1024 * 1024; // 5MB
      const parts = [];

      for (let i = 0; i < 4; i++) {
        const start = i * partSize;
        const end = Math.min(start + partSize, content.length);
        const partData = content.slice(start, end);

        const part = await this.s3.uploadPart({
          Bucket: bucket,
          Key: key,
          UploadId: uploadId,
          PartNumber: i + 1,
          Body: partData
        }).promise();

        parts.push({
          PartNumber: i + 1,
          ETag: part.ETag
        });
      }

      this.results.passed.push('✓ UploadPart');

      // 完成多部分上传
      await this.s3.completeMultipartUpload({
        Bucket: bucket,
        Key: key,
        UploadId: uploadId,
        MultipartUpload: { Parts: parts }
      }).promise();

      this.results.passed.push('✓ CompleteMultipartUpload');

      // 清理
      await this.s3.deleteObject({ Bucket: bucket, Key: key }).promise();
      await this.s3.deleteBucket({ Bucket: bucket }).promise();

    } catch (error) {
      this.results.failed.push(`✗ Multipart Upload: ${error.message}`);
    }
  }

  async testMetadata() {
    const bucket = 'metadata-test-' + Date.now();
    const key = 'metadata-test.txt';
    const metadata = {
      'custom-key': 'custom-value',
      'user-id': '12345'
    };

    try {
      await this.s3.createBucket({ Bucket: bucket }).promise();

      // 上传带元数据的对象
      await this.s3.putObject({
        Bucket: bucket,
        Key: key,
        Body: 'test',
        Metadata: metadata
      }).promise();

      // 获取对象元数据
      const headObject = await this.s3.headObject({
        Bucket: bucket,
        Key: key
      }).promise();

      // 检查元数据
      const customMetadata = headObject.Metadata;
      const allMetadataPresent = Object.keys(metadata).every(
        key => customMetadata[key] === metadata[key]
      );

      if (allMetadataPresent) {
        this.results.passed.push('✓ Object Metadata');
      } else {
        this.results.failed.push('✗ Object Metadata - Missing or incorrect metadata');
      }

      // 清理
      await this.s3.deleteObject({ Bucket: bucket, Key: key }).promise();
      await this.s3.deleteBucket({ Bucket: bucket }).promise();

    } catch (error) {
      this.results.failed.push(`✗ Metadata: ${error.message}`);
    }
  }

  async testACL() {
    // R2 不支持 ACL，这些操作预期失败
    try {
      await this.s3.putObjectAcl({
        Bucket: 'test',
        Key: 'test',
        ACL: 'public-read'
      }).promise();

      this.results.failed.push('✗ PutObjectAcl should have failed');
    } catch (error) {
      if (error.code === 'NotImplemented') {
        this.results.passed.push('✓ PutObjectAcl correctly not implemented');
      } else {
        this.results.warnings.push('⚠ PutObjectAcl error: ' + error.message);
      }
    }
  }

  generateReport() {
    console.log('\n📊 Test Results');
    console.log('================');

    console.log('\n✅ Passed:', this.results.passed.length);
    this.results.passed.forEach(test => console.log('  ' + test));

    if (this.results.failed.length > 0) {
      console.log('\n❌ Failed:', this.results.failed.length);
      this.results.failed.forEach(test => console.log('  ' + test));
    }

    if (this.results.warnings.length > 0) {
      console.log('\n⚠️  Warnings:', this.results.warnings.length);
      this.results.warnings.forEach(test => console.log('  ' + test));
    }

    const total = this.results.passed.length + this.results.failed.length;
    const successRate = (this.results.passed.length / total * 100).toFixed(1);
    console.log(`\n📈 Compatibility: ${successRate}%`);
  }
}
```

## 最佳实践

### 1. 渐进式迁移

```javascript
// 1. 双写策略
async function dualWrite(key, data) {
  // 同时写入 S3 和 R2
  const s3Promise = s3Client.putObject({
    Bucket: 's3-backup',
    Key: key,
    Body: data
  }).promise();

  const r2Promise = r2Client.putObject({
    Bucket: 'r2-primary',
    Key: key,
    Body: data
  }).promise();

  await Promise.all([s3Promise, r2Promise]);
}

// 2. 读取优先级
async function read(key) {
  // 优先从 R2 读取，失败时从 S3 读取
  try {
    return await r2Client.getObject({
      Bucket: 'r2-primary',
      Key: key
    }).promise();
  } catch (error) {
    return await s3Client.getObject({
      Bucket: 's3-backup',
      Key: key
    }).promise();
  }
}
```

### 2. 配置抽象

```javascript
// storage-config.js
class StorageConfig {
  constructor() {
    this.provider = process.env.STORAGE_PROVIDER || 's3';
    this.config = this.getConfig();
  }

  getConfig() {
    const baseConfig = {
      accessKeyId: process.env.AWS_ACCESS_KEY_ID,
      secretAccessKey: process.env.AWS_SECRET_ACCESS_KEY,
      region: process.env.AWS_DEFAULT_REGION || 'auto'
    };

    if (this.provider === 'r2') {
      return {
        ...baseConfig,
        endpoint: process.env.AWS_ENDPOINT_URL
      };
    }

    return baseConfig;
  }

  getClient() {
    return new AWS.S3(this.config);
  }
}
```
