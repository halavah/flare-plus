```javascript
// 优化订阅配置以节省配额
export class SubscriptionOptimizer {
  constructor(env) {
    this.env = env;
  }

  async optimizeSubscriptions() {
    // 分析事件频率
    const eventFrequency = await this.analyzeEventFrequency();

    // 合并相似事件
    const optimizedSubscriptions = this.mergeSimilarEvents(eventFrequency);

    // 移除低价值订阅
    const valuableSubscriptions = await this.filterValuableSubscriptions(
      optimizedSubscriptions
    );

    return valuableSubscriptions;
  }

  async analyzeEventFrequency() {
    const thirtyDaysAgo = new Date();
    thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);

    return await this.env.DB.prepare(`
      SELECT
        event_type,
        COUNT(*) as frequency,
        AVG(processing_time) as avg_processing_time
      FROM webhook_events
      WHERE created_at > ?
      GROUP BY event_type
      ORDER BY frequency DESC
    `).bind(thirtyDaysAgo.toISOString()).all();
  }

  mergeSimilarEvents(eventFrequency) {
    // 合理合并策略示例
    return eventFrequency.results.reduce((merged, event) => {
      if (event.event_type.startsWith('r2.')) {
        const r2Index = merged.findIndex(e => e.event_type === 'r2.all_events');
        if (r2Index >= 0) {
          merged[r2Index].frequency += event.frequency;
        } else {
          merged.push({
            event_type: 'r2.all_events',
            frequency: event.frequency
          });
        }
      } else {
        merged.push(event);
      }
      return merged;
    }, []);
  }
}
```

```javascript
  await archiveImportantEvents(sevenDaysAgo, env);

async function archiveImportantEvents(cutoffDate, env) {
  const importantEvents = await env.DB.prepare(`
    SELECT * FROM webhook_events
    WHERE created_at < ?
    AND (event_type LIKE '%deploy%' OR event_type LIKE '%error%')
  `).bind(cutoffDate.toISOString()).all();

  for (const event of importantEvents.results) {
    // 存储到 R2 进行长期归档
    await env.ARCHIVE_BUCKET.put(
      `webhook-events/${event.id}.json`,
      JSON.stringify(event),
      {
        customMetadata: {
          event_type: event.event_type,
          archived_at: new Date().toISOString()
        }
      }
    );
  }
}
```

### 4. 重试策略

#### 免费层重试配置
```javascript
// Webhook 重试管理
class WebhookRetryManager {
  constructor() {
    this.maxRetries = 3; // 免费层限制
    this.retryDelays = [5000, 15000, 60000]; // 5s, 15s, 60s
  }

  async retry webhook(url, payload, attempt = 0) {
    if (attempt >= this.maxRetries) {
      throw new Error(`Max retries exceeded for ${url}`);
    }

    try {
      const response = await fetch(url, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Cloudflare-Webhook-Retry-Count': attempt.toString()
        },
        body: JSON.stringify(payload)
      });

      if (!response.ok) {
        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
      }

      return response;
    } catch (error) {
      console.error(`Webhook attempt ${attempt + 1} failed:`, error);

      // 指数退避
      const delay = this.retryDelays[attempt] || this.retryDelays[this.retryDelays.length - 1];
      await new Promise(resolve => setTimeout(resolve, delay));

      return this.retryWebhook(url, payload, attempt + 1);
    }
  }

  // 计算成功交付率
  calculateSuccessRate(deliveryHistory) {
    const successful = deliveryHistory.filter(d => d.status === 'success').length;
    return (successful / deliveryHistory.length * 100).toFixed(2);
  }
}
```

```javascript
async function getRealTimeUsage(env) {
  const today = new Date().toISOString().split('T')[0];
  const monthStart = new Date().toISOString().substring(0, 8) + '01';

  // 今日使用量
  const todayUsage = await env.WEBHOOK_STATS_KV.get(`usage:${today}`) || '0';

  // 本月累计使用量
  const monthlyUsage = await env.WEBHOOK_STATS_KV.get(`usage:${monthStart}`) || '0';

  return {
    today: parseInt(todayUsage),
    month: parseInt(monthlyUsage),
    percentage: (parseInt(monthlyUsage) / 100000 * 100).toFixed(2)
  };
}

async function getUsageHistory(env, days = 30) {
  const history = [];
  const endDate = new Date();

  for (let i = 0; i < days; i++) {
    const date = new Date(endDate);
    date.setDate(date.getDate() - i);
    const dateStr = date.toISOString().split('T')[0];

    const usage = await env.WEBHOOK_STATS_KV.get(`usage:${dateStr}`) || '0';

    history.push({
      date: dateStr,
      usage: parseInt(usage)
    });
  }

  return history.reverse(); // 按时间顺序
}

async function predictUsage(env) {
  // 获取过去7天的使用量
  const recentUsage = await getUsageHistory(env, 7);
  const avgDaily = recentUsage.reduce((sum, day) => sum + day.usage, 0) / 7;

  // 预测本月剩余天数
  const daysInMonth = new Date(new Date().getFullYear(), new Date().getMonth() + 1, 0).getDate();
  const daysRemaining = daysInMonth - new Date().getDate();
  const predictedRemaining = Math.round(avgDaily * daysRemaining);

  const currentUsage = await getCurrentMonthUsage(env);

  return {
    average_daily: Math.round(avgDaily),
    predicted_monthly: currentUsage + predictedRemaining,
    will_exceed_limit: (currentUsage + predictedRemaining) > 100000
  };
}
```

### 1. 事件聚合策略

```javascript
// 事件聚合减少调用次数
export class EventAggregator {
  constructor(env) {
    this.env = env;
    this.batchSize = 50;
    this.flushInterval = 30000; // 30秒
    this.pendingEvents = [];
  }

  async addEvent(event) {
    this.pendingEvents.push({
      ...event,
      timestamp: Date.now()
    });

    // 达到批量大小或超时触发处理
    if (this.pendingEvents.length >= this.batchSize) {
      await this.flush();
    } else if (!this.flushTimer) {
      this.flushTimer = setTimeout(() => this.flush(), this.flushInterval);
    }
  }

  async flush() {
    if (this.pendingEvents.length === 0) return;

    const events = [...this.pendingEvents];
    this.pendingEvents = [];

    if (this.flushTimer) {
      clearTimeout(this.flushTimer);
      this.flushTimer = null;
    }

    // 聚合同类型事件
    const aggregated = this.aggregateEvents(events);

    // 发送聚合后的事件
    await Promise.all(
      aggregated.map(batch => this.sendBatch(batch))
    );
  }

  aggregateEvents(events) {
    const grouped = events.reduce((acc, event) => {
      const key = `${event.type}:${event.source}`;
      if (!acc[key]) acc[key] = [];
      acc[key].push(event);
      return acc;
    }, {});

    return Object.values(grouped);
  }

  async sendBatch(events) {
    const payload = {
      type: 'batch',
      count: events.length,
      events: events.map(e => ({
        id: e.id,
        data: e.data,
        timestamp: e.timestamp
      }))
    };

    // 单次调用发送多个事件
    await this.sendWebhook(payload);
  }
}
```

```javascript
// 基于重要性的智能过滤
export class SmartEventFilter {
  constructor(env) {
    this.env = env;
    this.importanceRules = await this.loadImportanceRules();
  }

  async shouldProcessEvent(event) {
    // 检查事件重要性
    const importance = await this.calculateImportance(event);

    // 如果事件不重要，跳过处理
    if (importance < 0.3) {
      await this.logFilteredEvent(event, 'low_importance');
      return false;
    }

    // 检查频率限制
    if (await this.isRateLimited(event)) {
      await this.logFilteredEvent(event, 'rate_limited');
      return false;
    }

    return true;
  }

  async calculateImportance(event) {
    let score = 0;

    // 基于事件类型的分数
    score += this.getEventTypeScore(event.type);

    // 基于来源的分数
    score += this.getSourceScore(event.source);

    // 基于时间的分数（工作时间事件更重要）
    const hour = new Date(event.timestamp).getHours();
    if (hour >= 9 && hour <= 17) score += 0.2;

    // 基于历史数据的分数
    const historicalScore = await this.getHistoricalScore(event);
    score += historicalScore;

    return Math.min(1, score);
  }

  getEventTypeScore(type) {
    const scores = {
      'worker.deploy': 0.9,
      'pages.build': 0.8,
      'r2.object_deleted': 0.7,
      'd1.query': 0.4,
      'kv.write': 0.3
    };

    return scores[type] || 0.5;
  }

  async isRateLimited(event) {
    const key = `rate_limit:${event.type}:${event.source}`;
    const window = 300000; // 5分钟
    const limit = 10; // 每5分钟最多10个

    const current = await this.env.RATE_LIMIT_KV.get(key) || '0';
    const count = parseInt(current);

    if (count >= limit) {
      return true;
    }

    await this.env.RATE_LIMIT_KV.put(key, (count + 1).toString(), {
      expirationTtl: window / 1000
    });

    return false;
  }
}
```

### 3. 缓存和去重

```javascript
// 事件去重减少无效调用
export class EventDeduplicator {
  constructor(env) {
    this.env = env;
    this.dedupWindow = 60000; // 1分钟去重窗口
  }

  async isDuplicate(event) {
    // 生成事件指纹
    const fingerprint = this.generateFingerprint(event);
    const key = `dedup:${fingerprint}`;

    // 检查是否在去重窗口内
    const exists = await this.env.DEDUP_KV.get(key);
    if (exists) {
      return true;
    }

    // 记录事件指纹
    await this.env.DEDUP_KV.put(key, '1', {
      expirationTtl: this.dedupWindow / 1000
    });

    return false;
  }

  generateFingerprint(event) {
    const key = `${event.type}:${event.source}:${JSON.stringify(event.data)}`;
    return crypto.subtle.digest('SHA-256', new TextEncoder().encode(key));
  }

  // 智能缓存相似事件
  async cacheSimilarEvent(event, result) {
    const cacheKey = this.generateCacheKey(event);
    await this.env.EVENT_CACHE_KV.put(cacheKey, JSON.stringify(result), {
      expirationTtl: 300 // 5分钟缓存
    });
  }

  async getCachedResult(event) {
    const cacheKey = this.generateCacheKey(event);
    const cached = await this.env.EVENT_CACHE_KV.get(cacheKey);
    return cached ? JSON.parse(cached) : null;
  }

  generateCacheKey(event) {
    // 生成缓存键，忽略时间戳等变化字段
    const stableData = {
      type: event.type,
      source: event.source,
      // 只包含稳定的数据字段
      ...Object.keys(event.data)
        .filter(key => !['timestamp', 'request_id'].includes(key))
        .reduce((obj, key) => {
          obj[key] = event.data[key];
          return obj;
        }, {})
    };

    return `cache:${event.type}:${crypto.subtle.digest(
      'SHA-256',
      new TextEncoder().encode(JSON.stringify(stableData))
    )}`;
  }
}
```
