## ⚡ 性能优化概述

### 1. 批量处理

```javascript
// Webhook 批量处理器
export class WebhookBatchProcessor {
  constructor(options = {}) {
    this.batchSize = options.batchSize || 100;
    this.flushInterval = options.flushInterval || 5000; // 5秒
    this.maxWaitTime = options.maxWaitTime || 30000; // 30秒
    this.processors = new Map();
    this.metrics = {
      batchesProcessed: 0,
      eventsProcessed: 0,
      averageBatchSize: 0,
      totalProcessingTime: 0
    };
  }

  // 添加事件到批次
  async addEvent(eventType, eventData) {
    if (!this.processors.has(eventType)) {
      this.processors.set(eventType, new EventBatch(
        eventType,
        this.batchSize,
        this.flushInterval,
        this.maxWaitTime
      ));
    }

    const processor = this.processors.get(eventType);
    return await processor.add(eventData);
  }

  // 处理批次
  async processBatch(eventType, batch) {
    const startTime = Date.now();

    try {
      // 预处理批次
      const processedBatch = await this.preprocessBatch(batch);

      // 并行处理事件
      const results = await Promise.allSettled(
        processedBatch.map(event => this.processEvent(event))
      );

      // 后处理
      await this.postprocessBatch(batch, results);

      // 更新指标
      this.updateMetrics(batch.length, Date.now() - startTime);

      return {
        success: true,
        processed: batch.length,
        results: results
      };
    } catch (error) {
      console.error(`Batch processing failed for ${eventType}:`, error);
      throw error;
    }
  }

  // 预处理批次
  async preprocessBatch(batch) {
    // 去重
    const deduplicated = await this.deduplicateBatch(batch);

    // 聚合相似事件
    const aggregated = await this.aggregateBatch(deduplicated);

    // 优先级排序
    const sorted = this.sortByPriority(aggregated);

    return sorted;
  }

  // 事件去重
  async deduplicateBatch(batch) {
    const seen = new Set();
    const deduplicated = [];

    for (const event of batch) {
      const fingerprint = this.generateFingerprint(event);
      if (!seen.has(fingerprint)) {
        seen.add(fingerprint);
        deduplicated.push(event);
      }
    }

    return deduplicated;
  }

  // 事件聚合
  async aggregateBatch(batch) {
    const aggregated = new Map();

    for (const event of batch) {
      const key = this.getAggregationKey(event);

      if (!aggregated.has(key)) {
        aggregated.set(key, {
          type: 'aggregated',
          key: key,
          events: [],
          count: 0,
          firstTimestamp: event.timestamp,
          lastTimestamp: event.timestamp
        });
      }

      const agg = aggregated.get(key);
      agg.events.push(event);
      agg.count++;
      agg.lastTimestamp = event.timestamp;
    }

    return Array.from(aggregated.values());
  }

  // 生成事件指纹
  generateFingerprint(event) {
    const keyFields = ['id', 'type', 'source'];
    return keyFields
      .map(field => event[field] || '')
      .join('|');
  }

  // 获取聚合键
  getAggregationKey(event) {
    return `${event.type}:${event.source}:${event.data?.category || 'default'}`;
  }

  // 按优先级排序
  sortByPriority(batch) {
    const priorityMap = {
      'critical': 1,
      'high': 2,
      'medium': 3,
      'low': 4
    };

    return batch.sort((a, b) => {
      const priorityA = priorityMap[a.priority] || 3;
      const priorityB = priorityMap[b.priority] || 3;
      return priorityA - priorityB;
    });
  }

  // 更新性能指标
  updateMetrics(batchSize, processingTime) {
    this.metrics.batchesProcessed++;
    this.metrics.eventsProcessed += batchSize;
    this.metrics.totalProcessingTime += processingTime;
    this.metrics.averageBatchSize =
      this.metrics.eventsProcessed / this.metrics.batchesProcessed;
  }

  // 获取性能报告
  getPerformanceReport() {
    const avgProcessingTime = this.metrics.totalProcessingTime / this.metrics.batchesProcessed;

    return {
      ...this.metrics,
      averageProcessingTime: avgProcessingTime,
      throughput: this.metrics.eventsProcessed / (Date.now() / 1000), // 每秒事件数
      efficiency: this.metrics.averageBatchSize / this.batchSize
    };
  }
}

// 单个事件批处理器
class EventBatch {
  constructor(eventType, batchSize, flushInterval, maxWaitTime) {
    this.eventType = eventType;
    this.batchSize = batchSize;
    this.flushInterval = flushInterval;
    this.maxWaitTime = maxWaitTime;
    this.events = [];
    this.flushTimer = null;
    this.lastFlush = Date.now();
  }

  async add(event) {
    this.events.push(event);

    // 检查是否需要立即刷新
    if (this.events.length >= this.batchSize) {
      await this.flush();
    } else if (!this.flushTimer) {
      // 设置定时刷新
      this.flushTimer = setTimeout(() => this.flush(), this.flushInterval);
    }

    // 检查最大等待时间
    if (Date.now() - this.lastFlush > this.maxWaitTime) {
      await this.flush();
    }
  }

  async flush() {
    if (this.events.length === 0) return;

    const batch = [...this.events];
    this.events = [];
    this.lastFlush = Date.now();

    if (this.flushTimer) {
      clearTimeout(this.flushTimer);
      this.flushTimer = null;
    }

    return batch;
  }
}
```

### 2. 异步处理管道

```javascript
// Webhook 异步处理管道
export class WebhookPipeline {
  constructor(options = {}) {
    this.stages = [];
    this.workers = options.workers || 4;
    this.queue = [];
    this.activeJobs = 0;
    this.metrics = {
      processed: 0,
      failed: 0,
      averageLatency: 0,
      throughput: 0
    };
  }

  // 添加处理阶段
  addStage(name, processor, options = {}) {
    this.stages.push({
      name,
      processor,
      concurrency: options.concurrency || 1,
      timeout: options.timeout || 30000,
      retryCount: options.retryCount || 3
    });
  }

  // 处理事件
  async process(event) {
    const startTime = Date.now();

    try {
      // 创建处理上下文
      const context = {
        event,
        stage: 0,
        attempts: 0,
        startTime,
        metadata: {}
      };

      // 添加到队列
      const result = await this.addToQueue(context);

      // 更新指标
      this.updateMetrics(startTime, true);

      return result;
    } catch (error) {
      this.updateMetrics(startTime, false);
      throw error;
    }
  }

  // 添加到队列
  async addToQueue(context) {
    return new Promise((resolve, reject) => {
      this.queue.push({
        context,
        resolve,
        reject
      });

      this.processQueue();
    });
  }

  // 处理队列
  async processQueue() {
    if (this.activeJobs >= this.workers || this.queue.length === 0) {
      return;
    }

    const job = this.queue.shift();
    if (!job) return;

    this.activeJobs++;

    try {
      const result = await this.executePipeline(job.context);
      job.resolve(result);
    } catch (error) {
      job.reject(error);
    } finally {
      this.activeJobs--;
      // 继续处理队列
      this.processQueue();
    }
  }

  // 执行管道
  async executePipeline(context) {
    let result = context.event;

    for (let i = context.stage; i < this.stages.length; i++) {
      const stage = this.stages[i];
      context.stage = i;

      try {
        // 执行阶段处理器
        result = await this.executeStage(stage, result, context);
        context.metadata[stage.name] = {
          processed: true,
          timestamp: Date.now()
        };
      } catch (error) {
        // 重试逻辑
        if (context.attempts < stage.retryCount) {
          context.attempts++;
          await this.delay(Math.pow(2, context.attempts) * 1000);
          i--; // 重试当前阶段
          continue;
        } else {
          throw error;
        }
      }
    }

    return result;
  }

  // 执行单个阶段
  async executeStage(stage, data, context) {
    return new Promise((resolve, reject) => {
      const timer = setTimeout(() => {
        reject(new Error(`Stage ${stage.name} timeout`));
      }, stage.timeout);

      stage.processor(data, context)
        .then(result => {
          clearTimeout(timer);
          resolve(result);
        })
        .catch(error => {
          clearTimeout(timer);
          reject(error);
        });
    });
  }

  // 延迟函数
  delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  // 更新指标
  updateMetrics(startTime, success) {
    const latency = Date.now() - startTime;

    if (success) {
      this.metrics.processed++;
    } else {
      this.metrics.failed++;
    }

    // 计算平均延迟
    const totalEvents = this.metrics.processed + this.metrics.failed;
    this.metrics.averageLatency =
      (this.metrics.averageLatency * (totalEvents - 1) + latency) / totalEvents;

    // 计算吞吐量
    this.metrics.throughput = this.metrics.processed / (Date.now() / 1000);
  }

  // 获取管道状态
  getStatus() {
    return {
      activeJobs: this.activeJobs,
      queueLength: this.queue.length,
      metrics: this.metrics,
      stages: this.stages.map(stage => ({
        name: stage.name,
        concurrency: stage.concurrency
      }))
    };
  }
}

// 使用示例
const pipeline = new WebhookPipeline({ workers: 8 });

// 添加处理阶段
pipeline.addStage('validation', async (event, context) => {
  return await validateEvent(event);
});

pipeline.addStage('enrichment', async (event, context) => {
  return await enrichEventData(event);
}, { concurrency: 4 });

pipeline.addStage('persistence', async (event, context) => {
  return await persistEvent(event);
}, { concurrency: 2 });

pipeline.addStage('notification', async (event, context) => {
  return await sendNotification(event);
}, { retryCount: 5 });
```

### 1. 多层缓存策略

```javascript
// Webhook 缓存管理器
export class WebhookCacheManager {
  constructor(options = {}) {
    // 多层缓存：L1(内存) -> L2(Redis) -> L3(数据库)
    this.l1Cache = new Map(); // 内存缓存
    this.l2Cache = options.redis; // Redis 缓存
    this.l3Storage = options.storage; // 持久化存储

    this.config = {
      l1: {
        maxSize: options.l1MaxSize || 1000,
        ttl: options.l1TTL || 60000 // 1分钟
      },
      l2: {
        ttl: options.l2TTL || 300000 // 5分钟
      },
      l3: {
        ttl: options.l3TTL || 86400000 // 24小时
      }
    };

    this.metrics = {
      hits: { l1: 0, l2: 0, l3: 0 },
      misses: { l1: 0, l2: 0, l3: 0 },
      writes: { l1: 0, l2: 0, l3: 0 }
    };

    // L1 缓存清理
    this.startL1CacheCleanup();
  }

  // 获取缓存数据
  async get(key) {
    // L1 缓存检查
    if (this.l1Cache.has(key)) {
      this.metrics.hits.l1++;
      return this.l1Cache.get(key);
    }
    this.metrics.misses.l1++;

    // L2 缓存检查
    try {
      const l2Data = await this.l2Cache.get(key);
      if (l2Data) {
        const parsed = JSON.parse(l2Data);
        this.metrics.hits.l2++;

        // 回填 L1 缓存
        this.setL1(key, parsed);
        return parsed;
      }
    } catch (error) {
      console.error('L2 cache error:', error);
    }
    this.metrics.misses.l2++;

    // L3 存储检查
    try {
      const l3Data = await this.l3Storage.get(key);
      if (l3Data) {
        this.metrics.hits.l3++;
        const data = JSON.parse(l3Data);

        // 回填 L2 和 L1 缓存
        await this.setL2(key, data);
        this.setL1(key, data);

        return data;
      }
    } catch (error) {
      console.error('L3 storage error:', error);
    }
    this.metrics.misses.l3++;

    return null;
  }

  // 设置缓存数据
  async set(key, value) {
    // 写入所有层
    this.setL1(key, value);
    await this.setL2(key, value);
    await this.setL3(key, value);
  }

  // L1 缓存设置
  setL1(key, value) {
    // 检查容量限制
    if (this.l1Cache.size >= this.config.l1.maxSize) {
      // LRU 淘汰
      const firstKey = this.l1Cache.keys().next().value;
      this.l1Cache.delete(firstKey);
    }

    this.l1Cache.set(key, {
      value,
      timestamp: Date.now()
    });
    this.metrics.writes.l1++;
  }

  // L2 缓存设置
  async setL2(key, value) {
    try {
      await this.l2Cache.setex(
        key,
        Math.ceil(this.config.l2.ttl / 1000),
        JSON.stringify(value)
      );
      this.metrics.writes.l2++;
    } catch (error) {
      console.error('L2 cache write error:', error);
    }
  }

  // L3 存储设置
  async setL3(key, value) {
    try {
      await this.l3Storage.set(key, JSON.stringify(value), {
        expirationTtl: Math.ceil(this.config.l3.ttl / 1000)
      });
      this.metrics.writes.l3++;
    } catch (error) {
      console.error('L3 storage write error:', error);
    }
  }

  // L1 缓存清理
  startL1CacheCleanup() {
    setInterval(() => {
      const now = Date.now();
      const expired = [];

      for (const [key, entry] of this.l1Cache.entries()) {
        if (now - entry.timestamp > this.config.l1.ttl) {
          expired.push(key);
        }
      }

      expired.forEach(key => this.l1Cache.delete(key));
    }, 30000); // 每30秒清理一次
  }

  // 获取缓存统计
  getStats() {
    const totalHits = Object.values(this.metrics.hits).reduce((a, b) => a + b, 0);
    const totalMisses = Object.values(this.metrics.misses).reduce((a, b) => a + b, 0);
    const totalRequests = totalHits + totalMisses;

    return {
      ...this.metrics,
      hitRate: totalRequests > 0 ? totalHits / totalRequests : 0,
      l1Size: this.l1Cache.size,
      l1HitRate: this.metrics.hits.l1 / (this.metrics.hits.l1 + this.metrics.misses.l1) || 0,
      l2HitRate: this.metrics.hits.l2 / (this.metrics.hits.l2 + this.metrics.misses.l2) || 0,
      l3HitRate: this.metrics.hits.l3 / (this.metrics.hits.l3 + this.metrics.misses.l3) || 0
    };
  }

  // 预热缓存
  async warmup(keys) {
    const promises = keys.map(key => this.get(key));
    await Promise.allSettled(promises);
  }
}
```

### 2. 数据库连接池优化

```javascript
// Webhook 数据库连接池
export class WebhookDBPool {
  constructor(config) {
    this.config = {
      min: config.min || 5,
      max: config.max || 20,
      acquireTimeoutMillis: config.acquireTimeoutMillis || 30000,
      idleTimeoutMillis: config.idleTimeoutMillis || 30000,
      createTimeoutMillis: config.createTimeoutMillis || 30000,
      ...config
    };

    this.pool = [];
    this.waitingQueue = [];
    this.activeConnections = 0;
    this.totalConnections = 0;
    this.metrics = {
      acquired: 0,
      released: 0,
      created: 0,
      destroyed: 0,
      timeouts: 0
    };
  }

  // 获取连接
  async acquire() {
    // 检查是否有可用连接
    if (this.pool.length > 0) {
      const connection = this.pool.pop();
      this.metrics.acquired++;
      return connection;
    }

    // 检查是否可以创建新连接
    if (this.totalConnections < this.config.max) {
      const connection = await this.createConnection();
      this.metrics.acquired++;
      return connection;
    }

    // 等待连接释放
    return await this.waitForConnection();
  }

  // 释放连接
  async release(connection) {
    // 检查连接是否仍然有效
    if (await this.isConnectionValid(connection)) {
      // 归还到池中
      this.pool.push(connection);
      this.metrics.released++;

      // 检查等待队列
      if (this.waitingQueue.length > 0) {
        const waiter = this.waitingQueue.shift();
        const conn = this.pool.pop();
        waiter.resolve(conn);
      }
    } else {
      // 连接无效，销毁
      await this.destroyConnection(connection);
    }
  }

  // 创建连接
  async createConnection() {
    const startTime = Date.now();

    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => {
        reject(new Error('Connection creation timeout'));
      }, this.config.createTimeoutMillis);

      this.createActualConnection()
        .then(connection => {
          clearTimeout(timeout);
          this.totalConnections++;
          this.activeConnections++;
          this.metrics.created++;
          resolve(connection);
        })
        .catch(error => {
          clearTimeout(timeout);
          reject(error);
        });
    });
  }

  // 创建实际连接（示例）
  async createActualConnection() {
    // 这里应该是实际的数据库连接创建逻辑
    // 例如：pg, mysql, mongodb 等
    return {
      id: crypto.randomUUID(),
      created: Date.now(),
      lastUsed: Date.now(),
      query: async (sql, params) => {
        // 执行查询
        return { rows: [] };
      }
    };
  }

  // 检查连接有效性
  async isConnectionValid(connection) {
    // 检查空闲时间
    const idleTime = Date.now() - connection.lastUsed;
    if (idleTime > this.config.idleTimeoutMillis) {
      return false;
    }

    // 执行健康检查
    try {
      await connection.query('SELECT 1');
      connection.lastUsed = Date.now();
      return true;
    } catch (error) {
      return false;
    }
  }

  // 销毁连接
  async destroyConnection(connection) {
    try {
      // 关闭连接
      if (connection.close) {
        await connection.close();
      }
    } catch (error) {
      console.error('Error destroying connection:', error);
    } finally {
      this.totalConnections--;
      this.activeConnections--;
      this.metrics.destroyed++;
    }
  }

  // 等待连接
  async waitForConnection() {
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => {
        const index = this.waitingQueue.findIndex(w => w.resolve === resolve);
        if (index >= 0) {
          this.waitingQueue.splice(index, 1);
        }
        this.metrics.timeouts++;
        reject(new Error('Connection acquire timeout'));
      }, this.config.acquireTimeoutMillis);

      this.waitingQueue.push({
        resolve: (connection) => {
          clearTimeout(timeout);
          this.metrics.acquired++;
          resolve(connection);
        },
        reject: (error) => {
          clearTimeout(timeout);
          reject(error);
        }
      });
    });
  }

  // 获取池状态
  getStatus() {
    return {
      totalConnections: this.totalConnections,
      activeConnections: this.activeConnections,
      idleConnections: this.pool.length,
      waitingQueue: this.waitingQueue.length,
      metrics: this.metrics
    };
  }
}

// 连接池管理器
export class ConnectionPoolManager {
  constructor() {
    this.pools = new Map();
  }

  // 获取连接池
  getPool(name, config) {
    if (!this.pools.has(name)) {
      this.pools.set(name, new WebhookDBPool(config));
    }
    return this.pools.get(name);
  }

  // 关闭所有连接池
  async closeAll() {
    const promises = Array.from(this.pools.values()).map(pool => {
      return pool.close();
    });

    await Promise.all(promises);
    this.pools.clear();
  }
}
```

```javascript
// Webhook 智能限流器
export class IntelligentRateLimiter {
  constructor(options = {}) {
    this.config = {
      // 基础限制
      globalRPS: options.globalRPS || 1000,
      perIPRPS: options.perIPRPS || 10,
      perEventTypeRPS: options.perEventTypeRPS || 100,

      // 突发处理
      burstSize: options.burstSize || 50,
      burstWindow: options.burstWindow || 1000,

      // 自适应调整
      enableAdaptive: options.enableAdaptive !== false,
      adjustmentWindow: options.adjustmentWindow || 60000,
      targetLatency: options.targetLatency || 100,

      // 优先级
      priorityLevels: options.priorityLevels || 3
    };

    this.counters = {
      global: new SlidingWindow(this.config.burstWindow),
      perIP: new Map(),
      perEventType: new Map(),
      perPriority: Array(this.config.priorityLevels).fill(0).map(() =>
        new SlidingWindow(this.config.burstWindow)
      )
    };

    this.metrics = {
      totalRequests: 0,
      allowedRequests: 0,
      blockedRequests: 0,
      averageLatency: 0
    };

    // 自适应调整
    if (this.config.enableAdaptive) {
      this.startAdaptiveAdjustment();
    }
  }

  // 检查是否允许请求
  async checkLimit(identifier, eventType, priority = 1) {
    const startTime = Date.now();

    // 优先级调整
    const adjustedRPS = this.getPriorityAdjustedRPS(priority);

    // 全局限制检查
    if (!this.counters.global.check(this.config.globalRPS)) {
      this.metrics.blockedRequests++;
      return {
        allowed: false,
        reason: 'global_limit',
        retryAfter: this.calculateRetryAfter()
      };
    }

    // IP 限制检查
    const ipKey = identifier.ip || identifier;
    if (!this.getIPCounter(ipKey).check(this.config.perIPRPS)) {
      this.metrics.blockedRequests++;
      return {
        allowed: false,
        reason: 'ip_limit',
        retryAfter: this.calculateRetryAfter()
      };
    }

    // 事件类型限制检查
    if (eventType && !this.getEventTypeCounter(eventType).check(adjustedRPS)) {
      this.metrics.blockedRequests++;
      return {
        allowed: false,
        reason: 'event_type_limit',
        retryAfter: this.calculateRetryAfter()
      };
    }

    // 更新计数器
    this.counters.global.increment();
    this.getIPCounter(ipKey).increment();
    if (eventType) {
      this.getEventTypeCounter(eventType).increment();
    }

    // 记录延迟
    this.recordLatency(Date.now() - startTime);

    this.metrics.allowedRequests++;
    this.metrics.totalRequests++;

    return {
      allowed: true
    };
  }

  // 获取 IP 计数器
  getIPCounter(ip) {
    if (!this.counters.perIP.has(ip)) {
      this.counters.perIP.set(ip, new SlidingWindow(this.config.burstWindow));
    }
    return this.counters.perIP.get(ip);
  }

  // 获取事件类型计数器
  getEventTypeCounter(eventType) {
    if (!this.counters.perEventType.has(eventType)) {
      this.counters.perEventType.set(eventType, new SlidingWindow(this.config.burstWindow));
    }
    return this.counters.perEventType.get(eventType);
  }

  // 获取优先级调整后的 RPS
  getPriorityAdjustedRPS(priority) {
    const priorityMultiplier = 1 + (this.config.priorityLevels - priority) * 0.5;
    return Math.floor(this.config.perEventTypeRPS * priorityMultiplier);
  }

  // 计算重试时间
  calculateRetryAfter() {
    return Math.ceil(this.config.burstWindow / 1000);
  }

  // 记录延迟
  recordLatency(latency) {
    const totalLatency = this.metrics.averageLatency * (this.metrics.totalRequests - 1) + latency;
    this.metrics.averageLatency = totalLatency / this.metrics.totalRequests;
  }

  // 自适应调整
  startAdaptiveAdjustment() {
    setInterval(() => {
      this.adjustLimits();
    }, this.config.adjustmentWindow);
  }

  // 动态调整限制
  adjustLimits() {
    const currentLatency = this.metrics.averageLatency;
    const targetLatency = this.config.targetLatency;

    if (currentLatency > targetLatency * 1.5) {
      // 延迟过高，降低限制
      this.config.globalRPS = Math.max(
        this.config.globalRPS * 0.9,
        100
      );
    } else if (currentLatency < targetLatency * 0.8) {
      // 延迟较低，可以提高限制
      this.config.globalRPS = Math.min(
        this.config.globalRPS * 1.1,
        2000
      );
    }
  }

  // 获取状态
  getStatus() {
    return {
      config: this.config,
      metrics: this.metrics,
      counters: {
        global: this.counters.global.getCount(),
        perIP: this.counters.perIP.size,
        perEventType: this.counters.perEventType.size
      }
    };
  }
}
```

```javascript
// Webhook 负载均衡器
export class WebhookLoadBalancer {
  constructor(options = {}) {
    this.workers = [];
    this.currentWorker = 0;
    this.strategy = options.strategy || 'round-robin';
    this.healthCheckInterval = options.healthCheckInterval || 30000;
    this.retryAttempts = options.retryAttempts || 3;

    this.metrics = {
      requests: 0,
      failures: 0,
      activeWorkers: 0,
      averageResponseTime: 0
    };

    // 启动健康检查
    this.startHealthCheck();
  }

  // 添加工作节点
  addWorker(worker) {
    worker.id = worker.id || crypto.randomUUID();
    worker.metrics = {
      requests: 0,
      failures: 0,
      activeRequests: 0,
      averageResponseTime: 0,
      lastHealthCheck: Date.now(),
      healthy: true
    };

    this.workers.push(worker);
  }

  // 分发请求
  async distribute(request) {
    let attempts = 0;
    let lastError = null;

    while (attempts < this.retryAttempts) {
      const worker = this.selectWorker();

      if (!worker) {
        throw new Error('No healthy workers available');
      }

      try {
        const startTime = Date.now();
        worker.metrics.activeRequests++;
        worker.metrics.requests++;

        const result = await worker.process(request);

        const responseTime = Date.now() - startTime;
        this.updateWorkerMetrics(worker, responseTime, true);

        return result;
      } catch (error) {
        worker.metrics.activeRequests--;
        worker.metrics.failures++;
        lastError = error;

        attempts++;

        // 标记工作节点为不健康
        if (attempts >= this.retryAttempts) {
          worker.metrics.healthy = false;
        }
      }
    }

    this.metrics.failures++;
    throw lastError || new Error('All workers failed');
  }

  // 选择工作节点
  selectWorker() {
    const healthyWorkers = this.workers.filter(w => w.metrics.healthy);

    if (healthyWorkers.length === 0) {
      return null;
    }

    switch (this.strategy) {
      case 'round-robin':
        return this.roundRobinSelect(healthyWorkers);
      case 'least-connections':
        return this.leastConnectionsSelect(healthyWorkers);
      case 'weighted':
        return this.weightedSelect(healthyWorkers);
      case 'response-time':
        return this.responseTimeSelect(healthyWorkers);
      default:
        return healthyWorkers[0];
    }
  }

  // 轮询选择
  roundRobinSelect(workers) {
    const worker = workers[this.currentWorker % workers.length];
    this.currentWorker++;
    return worker;
  }

  // 最少连接选择
  leastConnectionsSelect(workers) {
    return workers.reduce((min, current) => {
      return current.metrics.activeRequests < min.metrics.activeRequests
        ? current : min;
    });
  }

  // 权重选择
  weightedSelect(workers) {
    const totalWeight = workers.reduce((sum, w) => sum + (w.weight || 1), 0);
    let random = Math.random() * totalWeight;

    for (const worker of workers) {
      random -= worker.weight || 1;
      if (random <= 0) {
        return worker;
      }
    }

    return workers[0];
  }

  // 响应时间选择
  responseTimeSelect(workers) {
    return workers.reduce((best, current) => {
      const bestScore = best.metrics.averageResponseTime || Infinity;
      const currentScore = current.metrics.averageResponseTime || Infinity;

      return currentScore < bestScore ? current : best;
    });
  }

  // 更新工作节点指标
  updateWorkerMetrics(worker, responseTime, success) {
    if (success) {
      // 更新平均响应时间
      const total = worker.metrics.averageResponseTime * (worker.metrics.requests - 1) + responseTime;
      worker.metrics.averageResponseTime = total / worker.metrics.requests;

      this.metrics.averageResponseTime =
        (this.metrics.averageResponseTime * (this.metrics.requests - 1) + responseTime) / this.metrics.requests;
    }

    worker.metrics.activeRequests--;
    this.metrics.requests++;
  }

  // 健康检查
  startHealthCheck() {
    setInterval(() => {
      this.performHealthCheck();
    }, this.healthCheckInterval);
  }

  async performHealthCheck() {
    const healthCheckPromises = this.workers.map(worker =>
      this.checkWorkerHealth(worker)
    );

    await Promise.allSettled(healthCheckPromises);
  }

  async checkWorkerHealth(worker) {
    try {
      const startTime = Date.now();
      await worker.healthCheck?.();
      const responseTime = Date.now() - startTime;

      // 更新健康状态
      worker.metrics.healthy = true;
      worker.metrics.lastHealthCheck = Date.now();

      // 如果响应时间过长，标记为不健康
      if (responseTime > 5000) {
        worker.metrics.healthy = false;
      }
    } catch (error) {
      worker.metrics.healthy = false;
      worker.metrics.lastHealthCheck = Date.now();
    }
  }

  // 获取负载均衡状态
  getStatus() {
    const healthyWorkers = this.workers.filter(w => w.metrics.healthy);

    return {
      totalWorkers: this.workers.length,
      healthyWorkers: healthyWorkers.length,
      strategy: this.strategy,
      metrics: this.metrics,
      workers: this.workers.map(w => ({
        id: w.id,
        healthy: w.metrics.healthy,
        requests: w.metrics.requests,
        failures: w.metrics.failures,
        activeRequests: w.metrics.activeRequests,
        averageResponseTime: w.metrics.averageResponseTime
      }))
    };
  }
}
```

```javascript
// 指标收集器
class Histogram {
  constructor(buckets) {
    this.buckets = buckets.sort((a, b) => a - b);
    this.counts = new Array(buckets.length + 1).fill(0);
    this.sum = 0;
    this.count = 0;
  }

  observe(value) {
    this.sum += value;
    this.count++;

    // 找到合适的桶
    let bucketIndex = this.buckets.findIndex(b => value <= b);
    if (bucketIndex === -1) {
      bucketIndex = this.buckets.length;
    }

    this.counts[bucketIndex]++;
  }

  getPercentiles() {
    const percentiles = {};
    const pValues = [0.5, 0.9, 0.95, 0.99];

    for (const p of pValues) {
      percentiles[`p${Math.floor(p * 100)}`] = this.getPercentile(p);
    }

    return percentiles;
  }

  getPercentile(p) {
    const threshold = this.count * p;
    let cumulative = 0;

    for (let i = 0; i < this.buckets.length; i++) {
      cumulative += this.counts[i];
      if (cumulative >= threshold) {
        return this.buckets[i];
      }
    }

    return Infinity;
  }
}

class Rate {
  constructor(windowSize = 60000) {
    this.windowSize = windowSize;
    this.samples = [];
  }

  mark(value = 1) {
    this.samples.push({
      value,
      timestamp: Date.now()
    });

    // 清理过期样本
    const cutoff = Date.now() - this.windowSize;
    this.samples = this.samples.filter(s => s.timestamp > cutoff);
  }

  rate() {
    const cutoff = Date.now() - this.windowSize;
    const validSamples = this.samples.filter(s => s.timestamp > cutoff);
    const sum = validSamples.reduce((acc, s) => acc + s.value, 0);
    return sum / (this.windowSize / 1000);
  }
}

class Gauge {
  constructor() {
    this.value = 0;
    this.labels = new Map();
  }

  set(value, labels = {}) {

      const key = JSON.stringify(labels);
      this.labels.set(key, value);
    } else {
      this.value = value;
    }
  }

  get(labels = {}) {
    if (Object.keys(labels).length > 0) {
      const key = JSON.stringify(labels);
      return this.labels.get(key) || 0;
```

```javascript
    for (const rec of highPriorityRecs) {
      await this.optimizer.autoApplyOptimizations(rec);
    }
  }
}
```
