- **性能优化**：高效的请求处理和资源利用
- **可观测性**：全面的监控、日志和告警
- **自动化**：尽可能自动化部署、测试和运维

    ];
  }

      ip: req.ip,

      contentType: req.headers['content-type'],

      timestamp: Date.now(),
      headers: req.headers
    });

        ip: req.ip,
        userAgent: req.headers['user-agent'],
        risk: analysis.risk,
        reasons: analysis.reasons
      });

      if (analysis.risk > 0.9) {
        return res.status(403).json({
          error: 'Request flagged as suspicious',
          code: 'HIGH_RISK'
        });
      }
    }

    req.securityAnalysis = analysis;
    next();
  }
}

class SensitiveDataHandler {
  constructor(options = {}) {
    this.encryption = new AESEncryption(options.encryptionKey);
    this.masking = new DataMasking();
    this.piiDetector = new PIIDetector();
    this.dataRetention = options.dataRetention || 86400 * 30; // 30天
  }

  // 处理前脱敏
  async beforeStorage(event) {
    const sanitized = { ...event };

    // 检测和处理 PII
    const piiFields = await this.piiDetector.detect(event);
    if (piiFields.length > 0) {
      console.log(`PII detected in fields: ${piiFields.join(', ')}`);

      // 加密敏感字段
      for (const field of piiFields) {
        if (sanitized.data[field]) {
          sanitized.data[field] = await this.encryption.encrypt(sanitized.data[field]);
        }
      }
    }

    // 记录审计日志（不包含敏感数据）
    await this.logSecureAccess(event, piiFields);

    return sanitized;
  }

  // 查询后解密
  async afterRetrieval(event, userPermissions) {
    const decrypted = { ...event };

    // 检查权限
    if (!this.hasPermissionToViewPII(userPermissions)) {
      // 返回掩码数据
      return this.maskEvent(event);
    }

    // 解密字段
    for (const [key, value] of Object.entries(event.data)) {
      if (typeof value === 'string' && value.startsWith('encrypted:')) {
        try {
          const encrypted = value.substring(10); // 移除 'encrypted:' 前缀
          decrypted.data[key] = await this.encryption.decrypt(encrypted);
        } catch (error) {
          console.error(`Failed to decrypt field ${key}:`, error);
        }
      }
    }

    return decrypted;
  }

  // 数据保留策略
  async applyRetentionPolicy(data) {
    const age = Date.now() - new Date(data.createdAt).getTime();

    if (age > this.dataRetention) {
      // 加密归档
      const archived = {
        ...data,
        archivedAt: new Date().toISOString(),
        archived: true
      };

      // 移动到长期存储
      await this.archiveData(archived);

      // 删除活跃数据
      return null;
    }

    return data;
  }
}

class WebhookAccessControl {

    this.roles = new Map();
    this.permissions = new Map();
    this.userSessions = new Map();
    this.jwtSecret = options.jwtSecret;
  }

  // 定义角色和权限
  defineRoles() {
    this.roles.set('viewer', ['webhooks.read', 'metrics.read']);
    this.roles.set('editor', [
      'webhooks.read',
      'webhooks.create',
      'webhooks.update',
      'metrics.read'
    ]);
    this.roles.set('admin', ['*']); // 所有权限
  }

  // JWT 生成
  generateToken(userId, role) {
    return jwt.sign(
      {
        userId,
        role,
        exp: Math.floor(Date.now() / 1000) + (60 * 60) // 1小时过期
      },
      this.jwtSecret
    );
  }

  // 权限检查中间件
  requirePermission(permission) {
    return (req, res, next) => {
      const token = req.headers.authorization?.replace('Bearer ', '');

      if (!token) {
        return res.status(401).json({ error: 'No token provided' });
      }

      try {
        const decoded = jwt.verify(token, this.jwtSecret);
        const userRole = decoded.role;
        const userPermissions = this.roles.get(userRole) || [];

        if (!this.hasPermission(userPermissions, permission)) {
          return res.status(403).json({
            error: 'Insufficient permissions',
            required: permission
          });
        }

        req.user = decoded;
        next();
      } catch (error) {
        return res.status(401).json({ error: 'Invalid token' });
      }
    };
  }

  // 权限检查
  hasPermission(permissions, required) {
    if (permissions.includes('*')) {
      return true;
    }
    return permissions.includes(required);
  }
}

// 高性能 Webhook 处理器
class HighPerformanceWebhookProcessor {
  constructor(options = {}) {
    this.batchSize = options.batchSize || 100;
    this.flushInterval = options.flushInterval || 5000;
    this.concurrency = options.concurrency || 10;
    this.processors = new Array(this.concurrency).fill(null);
    this.queue = new PriorityQueue();
    this.metrics = new PerformanceMetrics();

    // 启动处理器
    this.startProcessors();
  }

  // 处理 Webhook
  async handle(event) {
    // 添加优先级
    const priority = this.calculatePriority(event);

    // 添加到队列
    this.queue.enqueue({ event, priority });

    // 记录指标
    this.metrics.increment('queued');
  }

  // 计算优先级
  calculatePriority(event) {
    let priority = 5; // 默认优先级

    // 基于事件类型
    const eventPriorities = {
      'critical': 1,
      'high': 2,
      'medium': 3,
      'low': 4
    };

    priority = eventPriorities[event.priority] || priority;

    // 基于来源
    if (event.source === 'production') {
      priority -= 1;
    }

    return Math.max(1, priority);
  }

  // 启动处理器
  startProcessors() {
    for (let i = 0; i < this.concurrency; i++) {
      this.processors[i] = new WorkerProcessor(i, this);
      this.processors[i].start();
    }
  }

  // Worker 处理器
  async processBatch() {
    const batch = [];
    const deadline = Date.now() + this.flushInterval;

    // 收集批次
    while (batch.length < this.batchSize && Date.now() < deadline) {
      const item = this.queue.dequeue();
      if (!item) break;
      batch.push(item);
    }

    if (batch.length === 0) return;

    const startTime = Date.now();

    try {
      // 并行处理批次
      const results = await Promise.allSettled(
        batch.map(item => this.processItem(item.event))
      );

      // 统计结果
      const successful = results.filter(r => r.status === 'fulfilled').length;
      const failed = results.length - successful;

      this.metrics.increment('processed', successful);
      this.metrics.increment('failed', failed);

      // 记录处理时间
      const duration = Date.now() - startTime;
      this.metrics.recordLatency(duration);

    } catch (error) {
      console.error('Batch processing error:', error);
      this.metrics.increment('batch_errors');

      // 将失败的事件重新入队
      batch.forEach(item => this.queue.enqueue(item));
    }
  }

  // 异步处理循环
  async processLoop() {
    while (true) {
      await this.processBatch();
      // 短暂休避免 CPU 占用过高
      await new Promise(resolve => setTimeout(resolve, 10));
    }
  }

  // 性能指标收集
  getMetrics() {
    return {
      ...this.metrics.getSnapshot(),
      queueSize: this.queue.size,
      processorUtilization: this.getProcessorUtilization()
    };
  }
}

// 性能指标收集器
class PerformanceMetrics {
  constructor() {
    this.counters = new Map();
    this.histograms = new Map();
    this.timers = new Map();
  }

  increment(name, value = 1) {
    this.counters.set(name, (this.counters.get(name) || 0) + value);
  }

  recordLatency(duration) {
    const histogram = this.histograms.get('latency') ||
                     new Histogram([10, 50, 100, 500, 1000, 5000]);
    histogram.observe(duration);
    this.histograms.set('latency', histogram);
  }

  getSnapshot() {
    const snapshot = {};

    // 计数器
    for (const [key, value] of this.counters) {
      snapshot[key] = value;
    }

    // 直方图
    for (const [key, histogram] of this.histograms) {
      snapshot[key] = histogram.getPercentiles();
    }

    return snapshot;
  }

  // 定期重置指标
  reset() {
    const counters = Object.fromEntries(this.counters);
    const histograms = Object.fromEntries(this.histograms);

    // 发送到监控系统
    this.sendMetrics({ counters, histograms });

    // 重置
    this.counters.clear();
    this.histograms.clear();
  }
}

// 多层缓存系统
class MultiTierCacheSystem {
  constructor(options = {}) {
    // L1: 内存缓存
    this.l1Cache = new LRUCache({
      maxSize: options.l1MaxSize || 1000,
      ttl: options.l1TTL || 60000 // 1分钟
    });

    // L2: Redis
    this.l2Cache = options.redis;
    this.l2TTL = options.l2TTL || 300000; // 5分钟

    // L3: 数据库/文件存储
    this.l3Storage = options.storage;
    this.l3TTL = options.l3TTL || 86400000; // 24小时

    this.warmupData = options.warmupData || [];
  }

  // 获取数据
  async get(key) {
    // L1 缓存
    let value = this.l1Cache.get(key);
    if (value !== undefined) {
      return value;
    }

    // L2 缓存
    try {
      value = await this.l2Cache.get(key);
      if (value) {
        value = JSON.parse(value);
        // 回填 L1
        this.l1Cache.set(key, value);
        return value;
      }
    } catch (error) {
      console.error('L2 cache error:', error);
    }

    // L3 存储
    try {
      value = await this.l3Storage.get(key);
      if (value) {
        value = JSON.parse(value);
        // 回填 L2 和 L1
        await this.l2Cache.setex(key, Math.ceil(this.l2TTL / 1000), JSON.stringify(value));
        this.l1Cache.set(key, value);
        return value;
      }
    } catch (error) {
      console.error('L3 storage error:', error);
    }

    return null;
  }

  // 设置数据
  async set(key, value, options = {}) {
    const ttl = options.ttl || this.l1TTL;

    // 设置 L1
    this.l1Cache.set(key, value, ttl);

    // 设置 L2
    try {
      await this.l2Cache.setex(
        key,
        Math.ceil((options.l2TTL || this.l2TTL) / 1000),
        JSON.stringify(value)
      );
    } catch (error) {
      console.error('L2 cache set error:', error);
    }

    // 设置 L3
    try {
      await this.l3Storage.put(key, JSON.stringify(value), {
        expirationTtl: Math.ceil((options.l3TTL || this.l3TTL) / 1000)
      });
    } catch (error) {
      console.error('L3 storage set error:', error);
    }
  }

  // 缓存预热
  async warmup() {
    console.log('Starting cache warmup...');

    const warmupPromises = this.warmupData.map(async (item) => {
      const value = await this.get(item.key);
      if (!value) {
        await this.set(item.key, item.value);
      }
    });

    await Promise.all(warmupPromises);

    console.log(`Cache warmup completed. Loaded ${this.warmupData.length} items.`);
  }

  // 缓存统计
  getStats() {
    return {
      l1: {
        size: this.l1Cache.size,
        maxSize: this.l1Cache.maxSize
      },
      hitRate: this.l1Cache.hitRate
    };
  }
}

      }
    }

    results.duration = Date.now() - startTime;
    results.successRate = (results.passed / results.totalTests) * 100;

    // 输出摘要
    console.log(`\n📊 Test Results for ${suite.name}:`);

    return results;

    let retries = 0;

    while (retries <= test.retries) {

          setTimeout(() => reject(new Error('Test timeout')), test.timeout)

          name: test.name,

        }
      }
    }

    return {

      status: 'failed',
      error: lastError.message,
      duration: Date.now(),

// 智能错误处理
class IntelligentErrorHandler {
  constructor(options = {}) {
    this.errorClassifier = new ErrorClassifier();
    this.retryStrategy = new RetryStrategy(options.retry);
    this.circuitBreaker = new CircuitBreaker(options.circuit);
    this.fallbackHandlers = new Map();
    this.errorReporters = options.errorReporters || [];
  }

  // 处理错误
  async handle(error, context) {
    // 分类错误
    const classification = this.errorClassifier.classify(error);

    // 记录错误
    await this.logError(error, context, classification);

    // 根据分类处理
    switch (classification.type) {
      case 'temporary':
        return await this.handleTemporaryError(error, context, classification);

      case 'permanent':
        return await this.handlePermanentError(error, context, classification);

      case 'timeout':
        return await this.handleTimeoutError(error, context, classification);

      case 'rate_limit':
        return await this.handleRateLimitError(error, context, classification);

      case 'circuit_open':
        return await this.handleCircuitOpenError(error, context, classification);

      default:
        return await this.handleUnknownError(error, context, classification);
    }
  }

  // 临时错误处理
  async handleTemporaryError(error, context, classification) {
    // 指数退避重试
    const retryDelay = this.retryStrategy.calculateDelay(
      context.attempt || 0,
      classification
    );

    // 等待后重试
    if (retryDelay > 0) {
      await this.sleep(retryDelay);
    }

    // 检查熔断器
    if (this.circuitBreaker.isOpen(context.service)) {
      throw new Error('Circuit breaker is open');
    }

    // 返回重试指示
    return {
      action: 'retry',
      delay: retryDelay,
      maxRetries: classification.maxRetries || 5,
      nextAttempt: (context.attempt || 0) + 1
    };
  }

  // 永久错误处理
  async handlePermanentError(error, context, classification) {
    // 触发熔断器
    this.circuitBreaker.recordFailure(context.service);

    // 查找降级处理器
    const fallbackHandler = this.fallbackHandlers.get(context.operation);
    if (fallbackHandler) {
      try {
        return await fallbackHandler(error, context);
      } catch (fallbackError) {
        // 降级处理器也失败了
        await this.reportCriticalError(error, context, fallbackError);
      }
    }

    // 返回错误响应
    return {
      action: 'fail',
      error: this.formatError(error, classification),
      code: classification.code || 'INTERNAL_ERROR',
      userMessage: this.getUserMessage(classification)
    };
  }

  // 用户友好的错误消息
  getUserMessage(classification) {
    const messages = {
      'connection_error': 'Service temporarily unavailable. Please try again later.',
      'permission_denied': 'You do not have permission to perform this action.',
      'resource_not_found': 'The requested resource was not found.',
      'rate_limit': 'Too many requests. Please wait and try again.',
      'timeout': 'Request timed out. Please try again.',
      'internal': 'An internal error occurred. Please contact support.'
    };

    return messages[classification.category] || messages.internal;
  }

  // 格式化错误
  formatError(error, classification) {
    return {
      type: classification.type,
      category: classification.category,
      code: classification.code,
      message: error.message,
      stack: process.env.NODE_ENV === 'development' ? error.stack : undefined,
      timestamp: new Date().toISOString()
    };
  }

  // 错误日志记录
  async logError(error, context, classification) {
    const logEntry = {
      timestamp: new Date().toISOString(),
      error: {
        name: error.name,
        message: error.message,
        stack: error.stack
      },
      context: {
        operation: context.operation,
        userId: context.userId,
        requestId: context.requestId,
        attempt: context.attempt || 0
      },
      classification,
      severity: classification.severity
    };

    // 根据严重程度选择日志级别
    const logLevel = classification.severity === 'critical' ? 'error' :
                    classification.severity === 'high' ? 'warn' : 'info';

    console[logLevel](`[WEBHOOK_ERROR]`, logEntry);

    // 发送到错误报告系统
    for (const reporter of this.errorReporters) {
      await reporter.report(logEntry);
    }
  }
}

    };

      parentSpanId: options.parentSpanId,
      operationName: type,
      startTime: Date.now(),
      tags: options.tags || {},
      data
    };

    return this.tracing.start(span);
  }

  // 结束追踪

    this.tracing.end(traceId, result, error);
  }

  // 发送告警

    const alert = {
      id: crypto.randomUUID(),
      level,
      message,
      data,
      timestamp: new Date().toISOString()
    };

    await this.alerting.send(alert);

      this.checkDatabaseHealth(),
      this.checkCacheHealth(),
      this.checkExternalServicesHealth()

    }

    return {
      status: allHealthy ? 'healthy' : 'unhealthy',
      checks: checks.map(check => ({
        name: check.value?.name || 'unknown',
        status: check.status,
        value: check.value
      }))
    };

// 自动化运维系统
class AutomationOps {
  constructor(options = {}) {
    this.autoScaler = new AutoScaler(options.scaling);
    self.deployer = new AutoDeployer(options.deployment);
    this.maintenance = new MaintenanceScheduler(options.maintenance);
    this.monitor = new SystemMonitor(options.monitor);
    this.actions = new Map();
  }

  // 注册自动化动作
  registerAction(trigger, action) {
    this.actions.set(trigger, action);
  }

  // 启动自动化
  startAutomation() {
    // 监控触发条件
    setInterval(async () => {
      const metrics = await this.monitor.collectMetrics();
      await this.evaluateAndExecute(metrics);
    }, 60000); // 每分钟

    // 定期维护
    this.maintenance.schedulePeriodicMaintenance();
  }

  // 评估和执行自动化动作
  async evaluateAndExecute(metrics) {
    const evaluations = [
      this.evaluateAutoScale(metrics),
      this.evaluateDeployRollback(metrics),
      this.evaluateCleanup(metrics),
      this.evaluateOptimization(metrics)
    ];

    for (const evaluation of evaluations) {
      if (evaluation.shouldExecute) {
        await this.executeAction(evaluation.trigger, evaluation.data);
      }
    }
  }

  // 评估自动扩缩
  evaluateAutoScale(metrics) {
    const {
      cpu = metrics.cpu,
      memory = metrics.memory,
      queueSize = metrics.queueSize,
      responseTime = metrics.averageResponseTime
    } = metrics;

    // 扩容条件
    const shouldScaleUp =
      cpu > 0.8 ||
      memory > 0.8 ||
      queueSize > 100 ||
      responseTime > 5000;

    // 缩容条件
    const shouldScaleDown =
      cpu < 0.3 &&
      memory < 0.3 &&
      queueSize < 10 &&
      responseTime < 1000;

    if (shouldScaleUp) {
      return {
        shouldExecute: true,
        trigger: 'scale_up',
        data: {
          currentMetrics: metrics,
          reason: 'High resource usage',
          targetSize: Math.ceil(queueSize * 1.5)
        }
      };
    }

    if (shouldScaleDown) {
      return {
        shouldExecute: true,
        trigger: 'scale_down',
        data: {
          currentMetrics: metrics,
          reason: 'Low resource usage',
          targetSize: Math.max(1, Math.floor(queueSize * 0.7))
        }
      };
    }

    return { shouldExecute: false };
  }

  // 执行动作
  async executeAction(trigger, data) {
    const action = this.actions.get(trigger);
    if (!action) {
      console.error(`No action found for trigger: ${trigger}`);
      return;
    }

    try {
      console.log(`Executing action: ${trigger}`);
      await action.execute(data);
      console.log(`Action completed: ${trigger}`);
    } catch (error) {
      console.error(`Action failed: ${trigger}`, error);
      await this.handleActionFailure(trigger, error, data);
    }
  }
}
