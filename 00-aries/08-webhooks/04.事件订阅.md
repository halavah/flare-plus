# Webhooks 事件订阅

## 🎯 事件订阅概述

事件订阅是 Webhooks 系统的核心功能，允许你选择性地接收特定类型的事件通知。通过配置事件过滤器，你可以精确控制需要接收哪些事件，避免不必要的数据传输和处理开销。

### 订阅架构
```
Cloudflare 服务 → 事件生成 → 过滤器匹配 → Webhook 触发 → 接收端处理
      ↓                ↓              ↓              ↓              ↓
   Workers/Pages   → Event Bus  → Filter Engine → Webhook → Your App
```

```javascript
// 自定义过滤器处理器
export class CustomFilterProcessor {
  constructor(filterRules) {
    this.filterRules = filterRules;
  }

  // 处理自定义过滤逻辑
  async processEvent(event) {
    for (const rule of this.filterRules) {
      if (await this.evaluateRule(rule, event)) {
        return rule.action;
      }
    }

    return 'pass';  // 默认通过
  }

  // 评估单个规则
  async evaluateRule(rule, event) {
    switch (rule.type) {
      case 'rate_limit':
        return await this.checkRateLimit(rule.config, event);

      case 'content_filter':
        return this.checkContentFilter(rule.config, event);

      case 'business_rule':
        return await this.checkBusinessRule(rule.config, event);

      case 'machine_learning':
        return await this.checkMLRule(rule.config, event);

      default:
        return true;
    }
  }

  // 速率限制检查
  async checkRateLimit(config, event) {
    const key = this.generateRateLimitKey(config, event);
    const current = await this.getRateLimitCount(key);

    return current < config.limit;
  }

  // 内容过滤检查
  checkContentFilter(config, event) {
    const content = this.extractContent(event);

    return config.patterns.some(pattern => {
      const regex = new RegExp(pattern, 'i');
      return regex.test(content);
    });
  }

  // 业务规则检查
  async checkBusinessRule(config, event) {
    // 实现自定义业务逻辑
    switch (config.rule) {
      case 'user_tier_check':
        const userTier = await getUserTier(event.user_id);
        return config.allowed_tiers.includes(userTier);

      case 'quota_check':
        const usage = await getCurrentUsage(event.account_id);
        return usage < config.quota_limit;

      default:
        return true;
    }
  }

  // 机器学习检查
  async checkMLRule(config, event) {
    // 调用 ML 模型进行判断
    const features = this.extractFeatures(event);
    const prediction = await this.mlModel.predict(features);

    return prediction.probability > config.threshold;
  }
}
```

```javascript
// Webhook 订阅管理器
export class SubscriptionManager {
  constructor(webhookManager, storage) {
    this.webhookManager = webhookManager;
    this.storage = storage;
  }

  // 创建带生命周期的订阅
  async createTimedSubscription(config, duration) {
    const subscription = await this.webhookManager.createWebhook(config);

    // 设置过期时间
    const expiresAt = new Date(Date.now() + duration);

    // 存储过期信息
    await this.storage.put(`subscription_expiry:${subscription.id}`, {
      expiresAt: expiresAt.toISOString(),
      webhookId: subscription.id
    });

    // 设置清理定时器
    setTimeout(() => {
      this.expireSubscription(subscription.id);
    }, duration);

    return {
      ...subscription,
      expiresAt: expiresAt
    };
  }

  // 过期订阅
  async expireSubscription(webhookId) {
    console.log(`Expiring subscription: ${webhookId}`);

    try {
      await this.webhookManager.updateWebhook(webhookId, {
        enabled: false
      });

      // 记录过期原因
      await this.logSubscriptionEvent(webhookId, 'expired');
    } catch (error) {
      console.error('Failed to expire subscription:', error);
    }
  }

  // 自动续期
  async autoRenewSubscription(webhookId, duration) {
    const subscription = await this.webhookManager.getWebhook(webhookId);

    if (subscription && subscription.auto_renew) {
      const newExpiresAt = new Date(Date.now() + duration);

      await this.storage.put(`subscription_expiry:${webhookId}`, {
        expiresAt: newExpiresAt.toISOString(),
        webhookId: webhookId
      });

      console.log(`Subscription ${webhookId} renewed until ${newExpiresAt}`);
    }
  }
}
```

```javascript
      this.alertLowSuccessRate(webhookId, metrics);

      this.alertHighLatency(webhookId, metrics);
    }

    return metrics;
  }

  // 订阅健康检查
  async healthCheck(webhookId) {
    const testPayload = {
      type: 'health_check',
      timestamp: new Date().toISOString(),
      test_id: crypto.randomUUID()
    };

    try {
      const startTime = Date.now();
      const result = await this.webhookManager.testWebhook(webhookId, testPayload);
      const duration = Date.now() - startTime;

      return {
        healthy: result.success,
        latency: duration,
        timestamp: new Date().toISOString()
      };
    } catch (error) {
      return {
        healthy: false,
        error: error.message,
        timestamp: new Date().toISOString()
      };
    }
  }

  // 订阅优化建议
  async getOptimizationSuggestions(webhookId) {
    const stats = await this.getSubscriptionStats(webhookId, '7d');
    const suggestions = [];

    // 频率分析
    if (stats.total_calls > 10000) {
      suggestions.push({
        type: 'frequency',
        message: 'High call volume detected. Consider event aggregation.',
        impact: 'high'
      });
    }

    // 错误率分析
    const errorRate = stats.failed_calls / stats.total_calls;

      suggestions.push({
        type: 'reliability',
        message: 'High error rate detected. Check endpoint availability.',
        impact: 'critical'
      });
    }

    // 延迟分析
    if (stats.average_latency > 3000) {
      suggestions.push({
        type: 'performance',
        message: 'High latency detected. Optimize endpoint performance.',
        impact: 'medium'
      });
    }

    return suggestions;
  }
}
```

```javascript
// 过滤器优化指南
class FilterOptimizer {
  // 优化过滤器顺序（将高选择性的过滤器放在前面）
  static optimizeFilterOrder(filters) {
    const selectivity = {
      'exact_match': 0.95,
      'regex': 0.8,
      'range': 0.6,
      'existence': 0.4
    };

    return filters.sort((a, b) => {
      const scoreA = this.calculateFilterScore(a, selectivity);
      const scoreB = this.calculateFilterScore(b, selectivity);
      return scoreB - scoreA;
    });
  }

  // 合并相似的过滤器
  static mergeSimilarFilters(filters) {
    const merged = [];
    const groups = this.groupSimilarFilters(filters);

    for (const group of groups) {
      if (group.length > 1) {
        merged.push(this.mergeFilterGroup(group));
      } else {
        merged.push(group[0]);
      }
    }

    return merged;
  }

  // 移除冗余过滤器
  static removeRedundantFilters(filters) {
    return filters.filter(filter => {
      return !this.isRedundant(filter, filters);
    });
  }
}
```

### 2. 事件去重

```javascript
// 事件去重处理器
export class EventDeduplicator {
  constructor(cache, dedupWindow = 60000) {
    this.cache = cache;
    this.dedupWindow = dedupWindow;
  }

  // 生成事件指纹
  generateEventFingerprint(event) {
    const keyFields = ['type', 'source', 'data.id'];
    const fingerprint = keyFields
      .map(field => this.getNestedValue(event, field))
      .join('|');

    return crypto.subtle.digest('SHA-256',
      new TextEncoder().encode(fingerprint)
    );
  }

  // 检查是否重复
  async isDuplicate(event) {
    const fingerprint = await this.generateEventFingerprint(event);
    const key = `dedup:${fingerprint}`;

    const exists = await this.cache.get(key);
    if (exists) {
      return true;
    }

    // 记录事件
    await this.cache.put(key, '1', {
      expirationTtl: this.dedupWindow / 1000
    });

    return false;
  }

  // 批量去重
  async deduplicateBatch(events) {
    const uniqueEvents = [];
    const fingerprints = new Set();

    for (const event of events) {
      const fingerprint = await this.generateEventFingerprint(event);

      if (!fingerprints.has(fingerprint)) {
        fingerprints.add(fingerprint);
        uniqueEvents.push(event);
      }
    }

    return uniqueEvents;
  }
}
```

通过合理配置事件订阅，你可以精确控制需要接收的事件类型，优化系统性能，并确保只处理相关的事件数据。记住定期审查和优化订阅配置，以适应业务需求的变化。
