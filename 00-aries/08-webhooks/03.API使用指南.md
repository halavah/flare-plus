# Webhooks API 使用指南

## 🚀 API 概览

Cloudflare Webhooks API 提供了完整的 Webhook 管理功能，包括创建、配置、监控和管理 Webhook 订阅。通过 RESTful API 接口，你可以程序化管理所有 Webhook 相关操作。

### API 基础信息
```
基础URL: https://api.cloudflare.com/client/v4
认证方式: Bearer Token 或 API Key
数据格式: JSON
API版本: v4
```

### 认证设置


```javascript
// 创建新 Webhook
async function createWebhook(config) {
  const webhookData = {
    name: config.name,
    url: config.url,
    secret: config.secret || generateSecret(),
    event_filters: config.event_filters || [],
    retry_config: config.retry_config || {
      max_retries: 3,
      retry_delay: 5000
    },
    enabled: config.enabled !== false
  };

  const response = await fetch(
    `${API_BASE_URL}/accounts/${ACCOUNT_ID}/webhooks`,
    {
      method: 'POST',
      headers: authHeaders,
      body: JSON.stringify(webhookData)
    }
  );

  const data = await response.json();

  if (!data.success) {
    throw new Error(data.errors[0].message);
  }

  return data.result;
}

// 生成安全的密钥
function generateSecret() {
  const array = new Uint8Array(32);
  crypto.getRandomValues(array);
  return Array.from(array, byte => byte.toString(16).padStart(2, '0')).join('');
}

// 使用示例
const newWebhook = await createWebhook({
  name: 'My Production Webhook',
  url: 'https://my-server.com/webhook/cloudflare',
  secret: 'my-secret-key',
  event_filters: [
    {
      type: 'worker.deploy',
      filters: {
        script_names: ['my-worker']
      }
    },
    {
      type: 'pages.build',
      filters: {
        project_names: ['my-website']
      }
    }
  ]
});

console.log('Created webhook:', newWebhook);
```

### 3. 实时监控

```javascript
// 实时 Webhook 监控
export class WebhookMonitor {
  constructor(accountId, apiToken) {
    this.accountId = accountId;
    this.apiToken = apiToken;
    this.isConnected = false;
    this.eventHandlers = new Map();
  }

  // 连接到 WebSocket 进行实时监控
  async connect() {
    const wsUrl = `wss://api.cloudflare.com/client/v4/accounts/${this.accountId}/webhooks/monitor`;

    this.ws = new WebSocket(wsUrl, [], {
      headers: {
        'Authorization': `Bearer ${this.apiToken}`
      }
    });

    this.ws.onopen = () => {
      console.log('Connected to webhook monitor');
      this.isConnected = true;
      this.emit('connected');
    };

    this.ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      this.handleWebhookEvent(data);
    };

    this.ws.onerror = (error) => {
      console.error('WebSocket error:', error);
      this.emit('error', error);
    };

    this.ws.onclose = () => {
      console.log('Disconnected from webhook monitor');
      this.isConnected = false;
      this.emit('disconnected');

      // 自动重连
      setTimeout(() => this.connect(), 5000);
    };
  }

  handleWebhookEvent(event) {
    switch (event.type) {
      case 'webhook_call':
        this.emit('call', event.data);
        break;
      case 'webhook_failure':
        this.emit('failure', event.data);
        break;
      case 'webhook_success':
        this.emit('success', event.data);
        break;
    }
  }

  on(event, handler) {
    if (!this.eventHandlers.has(event)) {
      this.eventHandlers.set(event, []);
    }
    this.eventHandlers.get(event).push(handler);
  }

  emit(event, data) {
    const handlers = this.eventHandlers.get(event) || [];
    handlers.forEach(handler => handler(data));
  }

  disconnect() {
    if (this.ws) {
      this.ws.close();
    }
  }
}

// 使用示例
const monitor = new WebhookMonitor(ACCOUNT_ID, API_TOKEN);

monitor.on('call', (data) => {
  console.log('Webhook called:', data);
});

monitor.on('failure', (data) => {
  console.error('Webhook failed:', data);
  // 发送告警
  sendAlert(data);
});

await monitor.connect();
```

```javascript
async function createMultipleWebhooks(webhookConfigs) {
```

```javascript
// Webhook 管理器
export class CloudflareWebhookManager {
  constructor(accountId, apiToken) {
    this.accountId = accountId;
    this.apiToken = apiToken;
    this.baseUrl = 'https://api.cloudflare.com/client/v4';
    this.headers = {
      'Authorization': `Bearer ${apiToken}`,
      'Content-Type': 'application/json'
    };
  }

  // 通用 API 请求方法
  async request(method, endpoint, data = null) {
    const url = `${this.baseUrl}/accounts/${this.accountId}${endpoint}`;
    const options = {
      method,
      headers: this.headers
    };

    if (data) {
      options.body = JSON.stringify(data);
    }

    const response = await fetch(url, options);
    const result = await response.json();

    if (!result.success) {
      throw new Error(result.errors?.[0]?.message || 'API request failed');
    }

    return result.result;
  }

  // 获取所有 Webhooks
  async getWebhooks() {
    return await this.request('GET', '/webhooks');
  }

  // 创建 Webhook
  async createWebhook(config) {
    return await this.request('POST', '/webhooks', config);
  }

  // 更新 Webhook
  async updateWebhook(id, updates) {
    return await this.request('PUT', `/webhooks/${id}`, updates);
  }

  // 删除 Webhook
  async deleteWebhook(id) {
    return await this.request('DELETE', `/webhooks/${id}`);
  }

  // 获取统计信息
  async getStats(id, period = '24h') {
    return await this.request('GET', `/webhooks/${id}/stats?period=${period}`);
  }

  // 测试 Webhook
  async testWebhook(id, testData) {
    return await this.request('POST', `/webhooks/${id}/test`, testData);
  }

  // 暂停/恢复 Webhook
  async toggleWebhook(id, enabled) {
    return await this.updateWebhook(id, { enabled });
  }

  // 获取使用配额
  async getQuota() {
    return await this.request('GET', '/webhooks/quota');
  }
}

// 使用示例
```

### 验证工具

```javascript
// Webhook 签名验证工具
export class WebhookValidator {
  constructor(secret) {
    this.secret = secret;
  }

  // 验证请求签名
  verifyRequest(headers, body) {
    const signature = headers['cf-webhook-signature'];
    const timestamp = headers['cf-webhook-timestamp'];

    if (!signature || !timestamp) {
      return { valid: false, error: 'Missing signature headers' };
    }

    // 验证时间戳（防止重放攻击）
    const now = Math.floor(Date.now() / 1000);
    if (Math.abs(now - parseInt(timestamp)) > 300) {
      return { valid: false, error: 'Timestamp too old' };
    }

    // 构造待验证字符串
    const message = `${timestamp}.${body}`;

    // 计算预期签名
    const expectedSignature = crypto
      .createHmac('sha256', this.secret)
      .update(message)
      .digest('hex');

    // 验证签名
    const isValid = crypto.timingSafeEqual(
      Buffer.from(signature, 'hex'),
      Buffer.from(expectedSignature, 'hex')
    );

    return {
      valid: isValid,
      error: isValid ? null : 'Invalid signature'
    };
  }

  // 验证事件数据
  validateEvent(eventData) {
    const required = ['id', 'type', 'timestamp', 'account_id', 'data'];
    const missing = required.filter(field => !eventData[field]);

    if (missing.length > 0) {
      return {
        valid: false,
        error: `Missing required fields: ${missing.join(', ')}`
      };
    }

    // 验证时间戳格式
    const timestamp = new Date(eventData.timestamp);
    if (isNaN(timestamp.getTime())) {
      return {
        valid: false,
        error: 'Invalid timestamp format'
      };
    }

    return { valid: true };
  }
}

// 使用示例
const validator = new WebhookValidator('my-webhook-secret');

// 在 Express 中使用
app.post('/webhook', (req, res) => {
  const verification = validator.verifyRequest(req.headers, req.body);

  if (!verification.valid) {
    return res.status(401).json({ error: verification.error });
  }

  const event = JSON.parse(req.body);
  const eventValidation = validator.validateEvent(event);

  if (!eventValidation.valid) {
    return res.status(400).json({ error: eventValidation.error });
  }

  // 处理事件
  handleEvent(event);
  res.status(200).json({ status: 'received' });
});
```

## 📝 错误处理

### 常见错误码和处理

```javascript
// 错误处理工具
export class WebhookAPIError extends Error {
  constructor(message, code, details) {
    super(message);
    this.name = 'WebhookAPIError';
    this.code = code;
    this.details = details;
  }
}

// API 错误码映射
const ERROR_CODES = {
  1000: 'Authentication failed',
  1001: 'Invalid account ID',
  1002: 'Webhook not found',
  1003: 'Invalid webhook configuration',
  1004: 'Rate limit exceeded',
  1005: 'Quota exceeded',
  1006: 'Invalid event filter',
  1007: 'Webhook URL unreachable',
  1008: 'Invalid secret format'
};

// 带重试的 API 调用
async function callAPIWithRetry(apiCall, maxRetries = 3) {
  let lastError;

  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await apiCall();
    } catch (error) {
      lastError = error;

      // 不重试的错误
      if ([1000, 1001, 1002, 1003, 1006, 1008].includes(error.code)) {
        throw error;
      }

      // 速率限制 - 等待后重试
      if (error.code === 1004) {
        const waitTime = Math.pow(2, attempt) * 1000;
        await new Promise(resolve => setTimeout(resolve, waitTime));
        continue;
      }

      // 最后一次尝试失败
      if (attempt === maxRetries) {
        throw error;
      }
    }
  }

  throw lastError;
}

// 使用示例
try {
  const webhook = await callAPIWithRetry(() =>
    webhookManager.createWebhook(config)
  );
  console.log('Webhook created:', webhook);
} catch (error) {
  if (error instanceof WebhookAPIError) {
    console.error(`API Error ${error.code}: ${error.message}`);

    // 根据错误码处理
    switch (error.code) {
      case 1004:
        console.log('Rate limited. Please try again later.');
        break;
      case 1005:
        console.log('Quota exceeded. Consider upgrading your plan.');
        break;
      case 1007:
        console.log('Webhook URL is not reachable. Please check the URL.');
        break;
      default:
        console.log('Please check your configuration and try again.');
    }
  }
}
```
