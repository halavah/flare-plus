# Webhooks 常见问题

## ❓ 基础问题

### Q1: Webhook 和传统 API 轮询有什么区别？

**A:** Webhook 和轮询是两种不同的数据获取方式：

| 特性 | Webhook | API 轮询 |
|------|---------|----------|
| **数据流向** | 推送（Push） | 拉取（Pull） |
| **实时性** | 近实时（秒级） | 延迟（取决于轮询间隔） |
| **资源消耗** | 事件触发时消耗 | 持续消耗 |
| **服务器负载** | 按需处理 | 固定负载 |
| **实现复杂度** | 需要接收端点 | 实现简单 |
| **适用场景** | 实时通知、事件驱动 | 定期数据检查 |

```javascript
// 轮询 vs Webhook 对比
// 轮询方式 - 需要定期请求
setInterval(async () => {
  const response = await fetch('https://api.example.com/orders');
  const orders = await response.json();

  orders.forEach(order => {
    if (order.status === 'new') {
      processNewOrder(order);
    }
  });
}, 30000); // 每30秒检查一次

// Webhook 方向 - 事件驱动
app.post('/webhook/orders', (req, res) => {
  const event = req.body;

  if (event.type === 'order.created') {
    processNewOrder(event.data);
  }

  res.status(200).send('OK');
});
```

### Q2: 如何确保 Webhook 的高可用性？

**A:** 确保高可用性需要多层次的保障：

```javascript
// 高可用 Webhook 接收器
class HighAvailabilityWebhookReceiver {
  constructor(options = {}) {
    this.primaryEndpoint = options.primaryEndpoint;
    this.secondaryEndpoint = options.secondaryEndpoint;
    this.loadBalancer = options.loadBalancer;
    this.circuitBreaker = options.circuitBreaker;
    this.retryQueue = new Map();
    this.healthyEndpoints = new Set([this.primaryEndpoint, this.secondaryEndpoint]);
  }

  // 智能负载均衡
  async processWebhook(event) {
    const endpoints = Array.from(this.healthyEndpoints);

    // 随机选择健康端点
    const endpoint = endpoints[Math.floor(Math.random() * endpoints.length)];

    try {
      const result = await this.sendToEndpoint(endpoint, event);

      // 成功后清除失败重试
      this.retryQueue.delete(event.id);
      return result;

    } catch (error) {
      console.error(`Endpoint ${endpoint} failed:`, error);

      // 标记端点为不健康
      this.markEndpointUnhealthy(endpoint);

      // 尝试其他端点
      if (this.healthyEndpoints.size > 0) {
        return await this.processWebhook(event);
      } else {
        // 所有端点都失败，加入重试队列
        this.addToRetryQueue(event);
        throw error;
      }
    }
  }

  // 端点健康检查
  async healthCheck() {
    const endpoints = [this.primaryEndpoint, this.secondaryEndpoint];

    for (const endpoint of endpoints) {
      try {
        const response = await fetch(`${endpoint}/health`, {
          method: 'GET',
          timeout: 5000
        });

        if (response.ok) {
          this.healthyEndpoints.add(endpoint);
        } else {
          this.healthyEndpoints.delete(endpoint);
        }
      } catch (error) {
        this.healthyEndpoints.delete(endpoint);
      }
    }
  }

  // 重试队列处理
  async processRetryQueue() {
    const now = Date.now();

    for (const [eventId, retryData] of this.retryQueue) {
      if (now >= retryData.nextRetry) {
        try {
          await this.processWebhook(retryData.event);
        } catch (error) {
          // 指数退避重试
          retryData.retryCount++;
          retryData.nextRetry = now + Math.pow(2, retryData.retryCount) * 1000;

          // 最多重试5次
          if (retryData.retryCount >= 5) {
            this.retryQueue.delete(eventId);
            await this.handlePermanentFailure(retryData);
          }
        }
      }
    }
  }

  // 定期健康检查
  startHealthCheck() {
    setInterval(() => {
      this.healthCheck();
    }, 30000); // 每30秒检查一次

    setInterval(() => {
      this.processRetryQueue();
    }, 60000); // 每分钟处理重试队列
  }
}
```

### Q4: 如何处理大量并发 Webhook 请求？

**A:** 使用队列和批处理来处理高并发：

```javascript
// 高并发 Webhook 处理器
class ConcurrentWebhookProcessor {
  constructor(options = {}) {
    this.queue = [];
    this.batchSize = options.batchSize || 100;
    this.batchTimeout = options.batchTimeout || 5000;
    this.workers = options.workers || 5;
    this.processing = false;

    this.metrics = {
      received: 0,
      processed: 0,
      failed: 0,
      averageBatchSize: 0,
      averageProcessingTime: 0
    };
  }

  // 接收 Webhook
  async receive(event) {
    this.metrics.received++;

    // 添加到队列
    this.queue.push({
      ...event,
      receivedAt: Date.now()
    });

    // 如果队列满了，使用背压策略
    if (this.queue.length > this.batchSize * 10) {
      await this.applyBackpressure();
    }

    // 如果没有在处理，启动处理
    if (!this.processing) {
      this.startProcessing();
    }
  }

  // 批量处理
  async startProcessing() {
    if (this.processing) return;

    this.processing = true;

    while (this.queue.length > 0) {
      const startTime = Date.now();

      // 获取批次
      const batch = this.queue.splice(0, this.batchSize);

      try {
        // 并行处理批次中的事件
        const results = await Promise.allSettled(
          batch.map(event => this.processEvent(event))
        );

        // 统计结果
        const successful = results.filter(r => r.status === 'fulfilled').length;
        const failed = results.filter(r => r.status === 'rejected').length;

        this.metrics.processed += successful;
        this.metrics.failed += failed;

        // 更新平均指标
        this.updateMetrics(batch.length, Date.now() - startTime);

        // 处理失败的事件
        await this.handleFailedEvents(batch, results);

      } catch (error) {
        console.error('Batch processing error:', error);
        // 将事件重新加入队列
        this.queue.unshift(...batch);
      }
    }

    this.processing = false;
  }

  // 背压策略
  async applyBackpressure() {
    // 1. 拒绝新请求
    this.rejectNewRequests = true;

    // 2. 增加批处理大小
    this.batchSize = Math.min(this.batchSize * 1.5, 500);

    // 3. 启动更多 Worker（如果可用）
    if (this.workers < 10) {
      this.workers++;
    }

    // 4. 发送告警
    await this.sendBackpressureAlert();
  }

  // 处理失败事件
  async handleFailedEvents(batch, results) {
    const failedEvents = [];

    batch.forEach((event, index) => {
      if (results[index].status === 'rejected') {
        failedEvents.push({
          event,
          error: results[index].reason
        });
      }
    });

    if (failedEvents.length > 0) {
      // 重试策略
      await this.retryFailedEvents(failedEvents);
    }
  }

  // 使用 Worker 并行处理
  async createWorkerPool() {
    const workers = [];

    for (let i = 0; i < this.workers; i++) {
      workers.push(new Promise((resolve) => {
        this.workerLoop(i, resolve);
      }));
    }

    return Promise.race(workers);
  }

  async workerLoop(workerId, resolve) {
    while (true) {
      const event = await this.getNextEvent();
      if (!event) {
        await new Promise(r => setTimeout(r, 100));
        continue;
      }

      try {
        await this.processEvent(event);
      } catch (error) {
        console.error(`Worker ${workerId} error:`, error);
      }
    }
  }
}
```

### Q6: 如何处理 Webhook 的幂等性？

**A:** 实现幂等性确保重复请求不会产生重复效果：

```javascript
// 幂等性处理
class IdempotentWebhookHandler {
  constructor(options = {}) {
    this.cache = options.cache || new Map();
    this.idempotencyWindow = options.idempotencyWindow || 86400000; // 24小时
    this.keyGenerator = options.keyGenerator || this.defaultKeyGenerator;
  }

  // 默认密钥生成器
  defaultKeyGenerator(event) {
    // 使用事件 ID 作为唯一标识
    return event.id || `${event.type}-${event.timestamp}-${JSON.stringify(event.data)}`;
  }

  // 处理 Webhook
  async handle(event) {
    const idempotencyKey = this.keyGenerator(event);

    // 检查是否已处理过
    const existing = await this.cache.get(idempotencyKey);
    if (existing) {
      console.log(`Duplicate event detected: ${idempotencyKey}`);
      return existing.response; // 返回之前的响应
    }

    // 处理新事件
    try {
      const response = await this.processEvent(event);

      // 缓存响应
      await this.cache.set(idempotencyKey, {
        response,
        timestamp: Date.now()
      }, {
        expirationTtl: this.idempotencyWindow / 1000
      });

      return response;
    } catch (error) {
      // 不缓存错误响应，允许重试
      throw error;
    }
  }

  // 处理事件（需要实现具体逻辑）
  async processEvent(event) {
    console.log(`Processing new event: ${event.type}`);

    switch (event.type) {
      case 'user.created':
        return await this.createUser(event.data);
      case 'order.created':
        return await this.createOrder(event.data);
      default:
        return { status: 'ok', message: 'Event processed' };
    }
  }

  // 使用 Redis 缓存
  async setupRedisCache() {
    this.cache = {
      get: async (key) => {
        const value = await redis.get(key);
        return value ? JSON.parse(value) : null;
      },

      set: async (key, value, options = {}) => {
        if (options.expirationTtl) {
          await redis.setex(key, options.expirationTtl, JSON.stringify(value));
        } else {
          await redis.set(key, JSON.stringify(value));
        }
      }
    };
  }

  // 清理过期的缓存条目
  async cleanupExpiredCache() {
    const cutoff = Date.now() - this.idempotencyWindow;

    // 需要根据具体的缓存实现
    for (const [key, value] of this.cache.entries()) {
      if (value.timestamp < cutoff) {
        this.cache.delete(key);
      }
    }
  }

  // Express 中间件
  middleware() {
    return async (req, res, next) => {
      // 从请求中获取幂等性键
      const idempotencyKey = req.headers['idempotency-key'] ||
                              req.body?.id ||
                              req.headers['x-request-id'];

      if (!idempotencyKey) {
        // 没有幂等性键，正常处理
        return next();
      }

      // 检查是否已处理
      const existing = await this.cache.get(idempotencyKey);
      if (existing) {
        // 返回缓存的响应
        res.status(existing.response.statusCode || 200)
           .json(existing.response.data);
        return;
      }

      // 在请求对象中存储幂等性信息
      req.idempotencyKey = idempotencyKey;
      req.cacheResponse = (statusCode, data) => {
        // 缓存响应
        this.cache.set(idempotencyKey, {
          response: { statusCode, data },
          timestamp: Date.now()
        }, { expirationTtl: this.idempotencyWindow / 1000 });
      };

      next();
    };
  }
}

// 使用示例
const idempotentHandler = new IdempotentWebhookHandler({
  cache: redisClient,
  idempotencyWindow: 3600000 // 1小时
});

app.post('/webhook', idempotentHandler.middleware(), async (req, res) => {
  try {
    const response = await idempotentHandler.handle(req.body);

    // 缓存成功响应
    if (req.cacheResponse) {
      req.cacheResponse(200, response);
    }

    res.json(response);
  } catch (error) {
    // 不缓存错误响应
    res.status(500).json({ error: error.message });
  }
});
```

### Q7: Webhook 送达失败怎么办？

**A:** 实施多层重试和通知机制：

```javascript
// Webhook 重试管理器
class WebhookRetryManager {
  constructor(options = {}) {
    this.maxRetries = options.maxRetries || 5;
    this.retryDelays = options.retryDelays || [1000, 2000, 4000, 8000, 16000]; // 指数退避
    this.deadLetterQueue = options.deadLetterQueue || [];
    this.retryStore = options.retryStore || new Map();
    this.notifier = options.notifier;
  }

  // 发送带重试的 Webhook
  async sendWithRetry(url, payload, options = {}) {
    const attemptId = crypto.randomUUID();
    let lastError = null;

    for (let attempt = 0; attempt <= this.maxRetries; attempt++) {
      try {
        const startTime = Date.now();

        const response = await fetch(url, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'X-Attempt': attempt.toString(),
            'X-Max-Retries': this.maxRetries.toString()
          },
          body: JSON.stringify(payload),
          signal: AbortSignal.timeout(options.timeout || 30000)
        });

        const duration = Date.now() - startTime;

        if (response.ok) {
          // 发送成功，记录成功
          await this.recordSuccess(attemptId, {
            url,
            attempt,
            duration,
            responseStatus: response.status
          });

          return {
            success: true,
            attempt,
            response: await response.json()
          };
        } else {
          throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }

      } catch (error) {
        lastError = error;

        // 记录失败
        await this.recordFailure(attemptId, {
          url,
          attempt,
          error: error.message,
          timestamp: new Date().toISOString()
        });

        // 如果不是最后一次尝试，等待后重试
        if (attempt < this.maxRetries) {
          const delay = this.getRetryDelay(attempt);
          await this.sleep(delay);
        }
      }
    }

    // 所有重试都失败
    await this.handlePermanentFailure(attemptId, url, payload, lastError);

    return {
      success: false,
      attempts: this.maxRetries + 1,
      error: lastError
    };
  }

  // 获取重试延迟
  getRetryDelay(attempt) {
    if (attempt < this.retryDelays.length) {
      return this.retryDelays[attempt];
    }

    // 使用指数退避
    return Math.pow(2, attempt) * 1000;
  }

  // 处理永久失败
  async handlePermanentFailure(attemptId, url, payload, error) {
    // 添加到死信队列
    const deadLetter = {
      attemptId,
      url,
      payload,
      error: error.message,
      timestamp: new Date().toISOString(),
      attempts: this.maxRetries + 1
    };

    this.deadLetterQueue.push(deadLetter);

    // 通知管理员
    if (this.notifier) {
      await this.notifier.notify('webhook_permanent_failure', deadLetter);
    }

    // 记录到数据库
    await this.storeDeadLetter(deadLetter);

    console.error(`Webhook permanently failed: ${attemptId}`, deadLetter);
  }

  // 批量重试死信队列
  async retryDeadLetterQueue(maxRetries = 3) {
    const failed = [...this.deadLetterQueue];
    this.deadLetterQueue = [];

    for (const deadLetter of failed) {
      try {
        const result = await this.sendWithRetry(
          deadLetter.url,
          deadLetter.payload,
          { maxRetries: maxRetries }
        );

        if (result.success) {
          console.log(`Successfully retried dead letter: ${deadLetter.attemptId}`);
          await this.markDeadLetterResolved(deadLetter);
        } else {
          // 仍然失败，重新加入队列
          this.deadLetterQueue.push(deadLetter);
        }
      } catch (error) {
        console.error(`Failed to retry dead letter: ${deadLetter.attemptId}`, error);
        this.deadLetterQueue.push(deadLetter);
      }
    }
  }

  // 获取重试统计
  async getRetryStatistics(timeRange = '24h') {
    const stats = {
      total: 0,
      successful: 0,
      failed: 0,
      averageRetries: 0,
      commonErrors: {},
      retryDistribution: new Array(this.maxRetries + 1).fill(0)
    };

    // 从存储获取统计数据
    const records = await this.getRetryRecords(timeRange);

    records.forEach(record => {
      stats.total++;

      if (record.success) {
        stats.successful++;
      } else {
        stats.failed++;
      }

      stats.averageRetries += record.attempts;
      stats.retryDistribution[record.attempts]++;

      const errorKey = record.error.split(':')[0];
      stats.commonErrors[errorKey] = (stats.commonErrors[errorKey] || 0) + 1;
    });

    stats.averageRetries = stats.total > 0 ? stats.averageRetries / stats.total : 0;
    stats.successRate = stats.total > 0 ? stats.successful / stats.total : 0;

    return stats;
  }
}
```

```javascript
// Webhook 健康监控
class WebhookHealthMonitor {
  constructor(options = {}) {
    this.metrics = new Map();
    this.alerts = new Map();
    this.healthChecks = new Map();
    this.dashboard = options.dashboard;
    this.alerter = options.alerter;

    // 初始化默认健康检查
    this.setupDefaultHealthChecks();
  }

  // 添加健康检查
  addHealthCheck(name, checkFn, options = {}) {
    this.healthChecks.set(name, {
      name,
      checkFn,
      interval: options.interval || 60000,
      timeout: options.timeout || 5000,
      critical: options.critical || false,
      enabled: options.enabled !== false
    });

    // 立即执行一次
    if (this.healthChecks.get(name).enabled) {
      this.runHealthCheck(name);
    }

    // 设置定期检查
    setInterval(() => {
      this.runHealthCheck(name);
    }, this.healthChecks.get(name).interval);
  }

  // 运行健康检查
  async runHealthCheck(name) {
    const check = this.healthChecks.get(name);
    if (!check.enabled) return;

    const startTime = Date.now();
    let result = {
      name,
      status: 'unknown',
      message: 'Checking...',
      timestamp: new Date().toISOString(),
      duration: 0,
      metadata: {}
    };

    try {
      // 设置超时
      const timeoutPromise = new Promise((_, reject) =>
        setTimeout(() => reject(new Error('Health check timeout')), check.timeout)
      );

      // 执行检查
      const checkPromise = check.checkFn();

      const response = await Promise.race([checkPromise, timeoutPromise]);

      result = {
        name,
        status: response.healthy ? 'healthy' : 'unhealthy',
        message: response.message,
        timestamp: new Date().toISOString(),
        duration: Date.now() - startTime,
        metadata: response.metadata || {}
      };

    } catch (error) {
      result = {
        name,
        status: 'error',
        message: error.message,
        timestamp: new Date().toISOString(),
        duration: Date.now() - startTime,
        metadata: { error: error.stack }
      };
    }

    // 更新状态
    this.metrics.set(name, result);

    // 检查是否需要告警
    await this.checkForAlerts(name, result);

    // 更新仪表板
    if (this.dashboard) {
      await this.updateDashboard(name, result);
    }

    return result;
  }

  // 设置默认健康检查
  setupDefaultHealthChecks() {
    // Webhook 端点检查
    this.addHealthCheck('webhook_endpoints', async () => {
      const endpoints = await this.getMonitoredEndpoints();
      const results = await Promise.allSettled(
        endpoints.map(endpoint => this.pingEndpoint(endpoint))
      );

      const healthy = results.filter(r => r.status === 'fulfilled').length;
      const total = results.length;

      return {
        healthy: healthy >= total * 0.8, // 80% 端点健康
        message: `${healthy}/${total} endpoints healthy`,
        metadata: {
          healthy,
          total,
          unhealthy: total - healthy
        }
      };
    }, {
      interval: 30000, // 30秒
      critical: true
    });

    // 响应时间检查
    this.addHealthCheck('response_time', async () => {
      const metrics = await this.getResponseTimeMetrics();
      const avgResponseTime = metrics.average;

      return {
        healthy: avgResponseTime < 5000, // 5秒
        message: `Average response time: ${avgResponseTime}ms`,
        metadata: metrics
      };
    }, {
      interval: 60000 // 1分钟
    });

    // 错误率检查
    this.addHealthCheck('error_rate', async () => {
      const metrics = await this.getErrorRateMetrics();
      const errorRate = metrics.errorRate;

      return {
        healthy: errorRate < 0.05, // 5%
        message: `Error rate: ${(errorRate * 100).toFixed(2)}%`,
        metadata: metrics
      };
    }, {
      interval: 60000,
      critical: true
    });
  }

  // 检查告警条件
  async checkForAlerts(checkName, result) {
    const check = this.healthChecks.get(checkName);

    // 如果状态从健康变为不健康
    const previousResult = this.metrics.get(`${checkName}_previous`);
    const wasHealthy = previousResult && previousResult.status === 'healthy';
    const isHealthy = result.status === 'healthy';

    if (wasHealthy && !isHealthy) {
      await this.triggerAlert('health_degradation', {
        check: checkName,
        status: result.status,
        message: result.message,
        critical: check.critical
      });
    }

    // 如果关键检查失败
    if (check.critical && result.status !== 'healthy') {
      await this.triggerAlert('critical_health_failure', {
        check: checkName,
        status: result.status,
        message: result.message
      });
    }

    // 保存当前状态用于下次比较
    this.metrics.set(`${checkName}_previous`, result);
  }

  // 触发告警
  async triggerAlert(alertType, details) {
    const alert = {
      id: crypto.randomUUID(),
      type: alertType,
      timestamp: new Date().toISOString(),
      details,
      severity: details.critical ? 'critical' : 'warning'
    };

    // 记录告警
    this.alerts.set(alert.id, alert);

    // 发送通知
    if (this.alerter) {
      await this.alerter.send(alert);
    }

    // 记录到日志
    console.warn(`[WEBHOOK_ALERT] ${alertType}:`, details);
  }

  // 获取健康状态总览
  async getHealthOverview() {
    const overview = {
      overall: 'healthy',
      checks: [],
      lastUpdate: new Date().toISOString(),
      issues: []
    };

    const checks = Array.from(this.metrics.values());
    const criticalChecks = checks.filter(c =>
      this.healthChecks.get(c.name)?.critical
    );

    for (const check of checks) {
      overview.checks.push({
        name: check.name,
        status: check.status,
        message: check.message,
        lastCheck: check.timestamp
      });

      if (check.status !== 'healthy') {
        overview.issues.push({
          check: check.name,
          status: check.status,
          message: check.message
        });
      }
    }

    // 确定整体状态
    const criticalUnhealthy = criticalChecks.filter(c => c.status !== 'healthy').length;
    if (criticalUnhealthy > 0) {
      overview.overall = 'critical';
    } else if (overview.issues.length > 0) {
      overview.overall = 'warning';
    }

    return overview;
  }

  // 更新仪表板
  async updateDashboard(checkName, result) {
    const dashboardData = {
      check: checkName,
      status: result.status,
      message: result.message,
      timestamp: result.timestamp,
      duration: result.duration
    };

    // 发送到仪表板服务
    if (this.dashboard) {
      await this.dashboard.update(dashboardData);
    }
  }
}
```

```javascript
// Webhook 最佳实践实现
class WebhookBestPractices {
  constructor() {
    this.validator = new EventValidator();
    this.idempotency = new IdempotentWebhookHandler();
    this.monitor = new WebhookHealthMonitor();
    this.retry = new WebhookRetryManager();

  // 完整的 Webhook 处理流程
  async handleWebhook(req, res) {
    const startTime = Date.now();

    try {
      // 1. 验证事件
      const event = req.body;
      const validation = this.validator.validate(event);
      if (!validation.valid) {
        return res.status(400).json({
          error: 'Invalid event',
          details: validation.errors
        });
      }

      // 2. 幂等性检查
      const response = await this.idempotency.handle(event);

      // 3. 记录指标
      this.recordMetrics(event, response, Date.now() - startTime);

      // 4. 返回响应
      res.status(200).json(response);

    } catch (error) {
      // 5. 错误处理
      console.error('Webhook processing error:', error);
      res.status(500).json({
        error: 'Internal server error',
        requestId: req.headers['x-request-id']
      });

      // 记录错误指标
      this.recordError(req, error);
    }
  }

  // 记录指标
  recordMetrics(event, response, duration) {
    const metrics = {
      eventType: event.type,
      status: response.status || 'processed',
      duration,
      timestamp: Date.now()
    };

    // 发送到监控系统
    this.monitor.record(metrics);
  }
}
```
