## 🔐 签名验证

### HMAC-SHA256 签名机制

```javascript
// Cloudflare Webhook 签名验证
export class CloudflareWebhookValidator {
  constructor(secret, options = {}) {
    this.secret = secret;
    this.clockSkewTolerance = options.clockSkewTolerance || 300; // 5分钟
    this.cache = options.cache || new Map(); // 用于防重放
  }

  // 验证 Webhook 请求
  async validateRequest(headers, body) {
    const validation = {
      signature: this.verifySignature(headers, body),
      timestamp: this.verifyTimestamp(headers),
      replay: await this.checkReplayAttack(headers),
      ip: await this.verifySourceIP(headers)
    };

    return {
      valid: Object.values(validation).every(v => v.valid),
      details: validation
    };
  }

  // 验证签名
  verifySignature(headers, body) {
    const signature = headers['cf-webhook-signature'];
    const timestamp = headers['cf-webhook-timestamp'];

    if (!signature || !timestamp) {
      return {
        valid: false,
        error: 'Missing signature headers'
      };
    }

    try {
      // 构造待签名字符串
      const message = `${timestamp}.${body}`;

      // 计算 HMAC-SHA256
      const expectedSignature = crypto
        .createHmac('sha256', this.secret)
        .update(message)
        .digest('hex');

      // 安全比较（防止时序攻击）
      const isValid = this.timingSafeEqual(
        Buffer.from(signature, 'hex'),
        Buffer.from(expectedSignature, 'hex')
      );

      return {
        valid: isValid,
        error: isValid ? null : 'Invalid signature'
      };

    } catch (error) {
      return {
        valid: false,
        error: `Signature verification failed: ${error.message}`
      };
    }
  }

  // 验证时间戳
  verifyTimestamp(headers) {
    const timestamp = headers['cf-webhook-timestamp'];

    if (!timestamp) {
      return {
        valid: false,
        error: 'Missing timestamp'
      };
    }

    const now = Math.floor(Date.now() / 1000);
    const requestTime = parseInt(timestamp);
    const timeDiff = Math.abs(now - requestTime);

    if (timeDiff > this.clockSkewTolerance) {
      return {
        valid: false,
        error: `Timestamp too old or too far in future: ${timeDiff}s`
      };
    }

    return {
      valid: true,
      timestamp: requestTime
    };
  }

  // 防重放攻击
  async checkReplayAttack(headers) {
    const signature = headers['cf-webhook-signature'];
    const timestamp = headers['cf-webhook-timestamp'];

    if (!signature || !timestamp) {
      return {
        valid: false,
        error: 'Missing headers for replay check'
      };
    }

    const key = `${timestamp}:${signature}`;

    // 检查是否已处理过
    if (this.cache.has(key)) {
      return {
        valid: false,
        error: 'Duplicate request detected'
      };
    }

    // 记录已处理的请求
    this.cache.set(key, true);

    // 设置过期时间（略大于时间戳容忍度）
    setTimeout(() => {
      this.cache.delete(key);
    }, (this.clockSkewTolerance + 60) * 1000);

    return { valid: true };
  }

  // 安全的字符串比较（防止时序攻击）
  timingSafeEqual(a, b) {
    if (a.length !== b.length) {
      return false;
    }

    let result = 0;
    for (let i = 0; i < a.length; i++) {
      result |= a[i] ^ b[i];
    }

    return result === 0;
  }
}

// Express 中间件示例
function createWebhookMiddleware(secret) {
  const validator = new CloudflareWebhookValidator(secret);

  return async (req, res, next) => {
    try {
      const validation = await validator.validateRequest(req.headers, req.body);

      if (!validation.valid) {
        console.error('Webhook validation failed:', validation.details);
        return res.status(401).json({
          error: 'Unauthorized webhook request',
          details: validation.details
        });
      }

      // 将验证结果添加到请求对象
      req.webhookValidated = true;
      req.webhookDetails = validation.details;

      next();
    } catch (error) {
      console.error('Webhook middleware error:', error);
      return res.status(500).json({
        error: 'Webhook validation failed'
      });
    }
  };
}
```

```javascript
// 支持多个密钥的验证器
export class MultiKeyWebhookValidator {
  constructor(keyStore, options = {}) {
    this.keyStore = keyStore; // 密钥存储接口
    this.keyRotationWindow = options.keyRotationWindow || 86400000; // 24小时
  }

  // 验证请求（尝试所有密钥）
  async validateRequest(headers, body) {
    const timestamp = headers['cf-webhook-timestamp'];
    const signature = headers['cf-webhook-signature'];

    if (!timestamp || !signature) {
      return { valid: false, error: 'Missing required headers' };
    }

    // 获取所有有效密钥
    const keys = await this.keyStore.getValidKeys(timestamp);

    // 尝试每个密钥
    for (const key of keys) {
      const isValid = await this.tryValidateWithKey(
        headers,
        body,
        key.secret
      );

      if (isValid) {
        return {
          valid: true,
          keyId: key.id,
          keyVersion: key.version
        };
      }
    }

    return { valid: false, error: 'No valid key found' };
  }

  // 尝试使用特定密钥验证
  async tryValidateWithKey(headers, body, secret) {
    const timestamp = headers['cf-webhook-timestamp'];
    const signature = headers['cf-webhook-signature'];

    const message = `${timestamp}.${body}`;
    const expectedSignature = crypto
      .createHmac('sha256', secret)
      .update(message)
      .digest('hex');

    return this.timingSafeEqual(
      Buffer.from(signature, 'hex'),
      Buffer.from(expectedSignature, 'hex')
    );
  }
}

// 密钥存储接口实现
class RedisKeyStore {
  constructor(redisClient) {
    this.redis = redisClient;
  }

  // 获取指定时间的有效密钥
  async getValidKeys(timestamp) {
    const timestampMs = parseInt(timestamp) * 1000;
    const windowStart = timestampMs - this.keyRotationWindow;
    const windowEnd = timestampMs + this.keyRotationWindow;

    // 获取时间窗口内的所有密钥
    const keys = await this.redis.zrangebyscore(
      'webhook_keys',
      windowStart,
      windowEnd
    );

    return Promise.all(keys.map(keyId => this.getKeyInfo(keyId)));
  }

  // 获取密钥信息
  async getKeyInfo(keyId) {
    const keyData = await this.redis.hgetall(`webhook_key:${keyId}`);
    return {
      id: keyId,
      version: keyData.version,
      secret: keyData.secret,
      createdAt: parseInt(keyData.created_at),
      expiresAt: parseInt(keyData.expires_at)
    };
  }
}
```

```javascript
export class CloudflareIPValidator {
  constructor() {

    this.ipRanges = {

        '172.64.0.0/13',
        '173.245.48.0/20',
        '188.114.96.0/20',
        '190.93.240.0/20',
        '197.234.240.0/22',
        '198.41.128.0/17'
      ],
      ipv6: [
        '2400:cb00::/32',
        '2606:4700::/32',
        '2803:f800::/32',
        '2405:b500::/32',
        '2405:8100::/32',
        '2c0f:f248::/32',
        '2a06:98c0::/29'
      ]
    };

    // 预编译 IP 范围检查器
    this.rangeCheckers = {
      ipv4: this.ipRanges.ipv4.map(range => this.createIPv4Checker(range)),
      ipv6: this.ipRanges.ipv6.map(range => this.createIPv6Checker(range))
    };
  }

  // 验证 IP 地址
  isValidIP(ip) {
    if (ip.includes(':')) {
      return this.isValidIPv6(ip);
    } else {
      return this.isValidIPv4(ip);
    }
  }

  // 验证 IPv4
  isValidIPv4(ip) {
    const ipInt = this.ipv4ToInt(ip);

    return this.rangeCheckers.ipv4.some(checker => checker(ipInt));
  }

  // 验证 IPv6
  isValidIPv6(ip) {
    const ipBuf = this.ipv6ToBuffer(ip);

    return this.rangeCheckers.ipv6.some(checker => checker(ipBuf));
  }

  // IPv4 转整数
  ipv4ToInt(ip) {
    const parts = ip.split('.');
    return parts.reduce((acc, part, index) => {
      return acc + (parseInt(part) << (24 - index * 8));
    }, 0) >>> 0;
  }

  // IPv6 转 Buffer
  ipv6ToBuffer(ip) {
    const parts = ip.split(':');
    const buffer = new Uint8Array(16);
    let partIndex = 0;
    let bufferIndex = 0;

    for (const part of parts) {
      if (part === '') {
        // 处理 :: 压缩
        const remainingParts = 8 - parts.length + 1;
        partIndex += remainingParts;
      } else {
        const value = parseInt(part, 16);
        buffer[bufferIndex] = (value >> 8) & 0xff;
        buffer[bufferIndex + 1] = value & 0xff;
        bufferIndex += 2;
      }
    }

    return buffer;
  }

  // 创建 IPv4 范围检查器
  createIPv4Checker(range) {
    const [network, prefixLength] = range.split('/');
    const networkInt = this.ipv4ToInt(network);
    const mask = (0xFFFFFFFF << (32 - parseInt(prefixLength))) >>> 0;

    return (ipInt) => (ipInt & mask) === (networkInt & mask);
  }

  // 创建 IPv6 范围检查器
  createIPv6Checker(range) {
    const [network, prefixLength] = range.split('/');
    const networkBuf = this.ipv6ToBuffer(network);
    const bits = parseInt(prefixLength);

    return (ipBuf) => {
      const fullBytes = Math.floor(bits / 8);
      const remainingBits = bits % 8;

      // 检查完整字节
      for (let i = 0; i < fullBytes; i++) {
        if (ipBuf[i] !== networkBuf[i]) return false;
      }

      // 检查剩余位
      if (remainingBits > 0) {
        const mask = (0xff << (8 - remainingBits)) & 0xff;
        if ((ipBuf[fullBytes] & mask) !== (networkBuf[fullBytes] & mask)) {
          return false;
        }
      }

      return true;
    };
  }

  // 更新 IP 范围
  async updateIPRanges() {
    try {
      // 从 Cloudflare API 获取最新 IP 范围
      const response = await fetch('https://api.cloudflare.com/client/v4/ips');
      const data = await response.json();

      if (data.success) {
        this.ipRanges = {
          ipv4: data.result.ipv4_cidrs,
          ipv6: data.result.ipv6_cidrs
        };

        // 重新编译检查器
        this.rangeCheckers = {
          ipv4: this.ipRanges.ipv4.map(range => this.createIPv4Checker(range)),
          ipv6: this.ipRanges.ipv6.map(range => this.createIPv6Checker(range))
        };

        console.log('Updated Cloudflare IP ranges');
      }
    } catch (error) {
      console.error('Failed to update IP ranges:', error);
    }
  }
}

// Express IP 验证中间件
function createIPValidatorMiddleware() {
  const ipValidator = new CloudflareIPValidator();

  // 定期更新 IP 范围
  setInterval(() => {
    ipValidator.updateIPRanges();
  }, 24 * 60 * 60 * 1000); // 每天更新

  return (req, res, next) => {
    const clientIP = req.ip ||
                     req.connection.remoteAddress ||
                     req.headers['x-forwarded-for']?.split(',')[0]?.trim();

    if (!clientIP) {
      return res.status(400).json({ error: 'Unable to determine client IP' });
    }

    if (!ipValidator.isValidIP(clientIP)) {
      console.warn(`Unauthorized IP attempt: ${clientIP}`);
      return res.status(403).json({ error: 'Unauthorized source IP' });
    }

    req.clientIP = clientIP;
    next();
  };
}

  // 处理 Webhook
});
```

```javascript
// Nonce 防重放攻击
export class NonceManager {
  constructor(store, windowSize = 300000) { // 5分钟窗口
    this.store = store;
    this.windowSize = windowSize;
  }

  // 生成 Nonce
  generateNonce() {
    const timestamp = Date.now();
    const random = crypto.randomBytes(16).toString('hex');
    return `${timestamp}_${random}`;
  }

  // 验证 Nonce
  async verifyNonce(nonce) {
    const [timestamp, random] = nonce.split('_');

    if (!timestamp || !random) {
      return { valid: false, error: 'Invalid nonce format' };
    }

    const nonceTime = parseInt(timestamp);
    const now = Date.now();

    // 检查时间窗口
    if (Math.abs(now - nonceTime) > this.windowSize) {
      return { valid: false, error: 'Nonce expired' };
    }

    // 检查是否已使用
    const key = `nonce:${nonce}`;
    const exists = await this.store.get(key);

    if (exists) {
      return { valid: false, error: 'Nonce already used' };
    }

    // 标记为已使用
    await this.store.set(key, '1', {
      expiration: this.windowSize / 1000 + 60 // 额外60秒缓冲
    });

    // 清理过期的 Nonce
    this.cleanupExpiredNonces();

    return { valid: true };
  }

  // 定期清理过期 Nonce
  cleanupExpiredNonces() {
    // 使用后台任务清理
    setTimeout(() => {
      this.performCleanup();
    }, 60000); // 每分钟清理一次
  }

  async performCleanup() {
    const cutoff = Date.now() - this.windowSize;
    const pattern = 'nonce:';

    // 使用存储系统的扫描功能清理
    await this.store.deleteExpired(pattern, cutoff);
  }
}
```

```javascript
// Webhook 请求限流
export class WebhookRateLimiter {
  constructor(store, options = {}) {
    this.store = store;
    this.windowMs = options.windowMs || 60000; // 1分钟
    this.maxRequests = options.maxRequests || 1000;
  }

  // 检查速率限制
  async checkLimit(identifier) {
    const now = Date.now();
    const window = Math.floor(now / this.windowMs);
    const key = `rate_limit:${identifier}:${window}`;

    // 获取当前计数
    const current = await this.store.get(key) || '0';
    const count = parseInt(current);

    if (count >= this.maxRequests) {
      const resetTime = (window + 1) * this.windowMs;
      const retryAfter = Math.ceil((resetTime - now) / 1000);

      return {
        allowed: false,
        count: count,
        limit: this.maxRequests,
        resetTime: resetTime,
        retryAfter: retryAfter
      };
    }

    // 增加计数
    await this.store.set(key, (count + 1).toString(), {
      expiration: this.windowMs / 1000 + 10
    });

    return {
      allowed: true,
      count: count + 1,
      limit: this.maxRequests,
      remaining: this.maxRequests - count - 1
    };
  }

  // Express 中间件
  middleware() {
    return async (req, res, next) => {
      const identifier = this.getIdentifier(req);
      const result = await this.checkLimit(identifier);

      // 设置响应头
      res.set({
        'X-RateLimit-Limit': result.limit,
        'X-RateLimit-Remaining': result.remaining || 0,
        'X-RateLimit-Reset': new Date(result.resetTime || 0).toISOString()
      });

      if (!result.allowed) {
        res.set('Retry-After', result.retryAfter);
        return res.status(429).json({
          error: 'Rate limit exceeded',
          retryAfter: result.retryAfter
        });
      }

      next();
    };
  }

  // 获取限流标识符
  getIdentifier(req) {
    // 可以基于 IP、API Key 或其他标识
    return req.ip || req.headers['cf-connecting-ip'];
  }
}
```

```javascript
// Webhook 内容验证
export class WebhookContentValidator {
  constructor() {
    this.schemaRegistry = new Map();
    this.contentFilters = [];
  }

  // 注册事件模式
  registerSchema(eventType, schema) {
    this.schemaRegistry.set(eventType, schema);
  }

  // 添加内容过滤器
  addFilter(filter) {
    this.contentFilters.push(filter);
  }

  // 验证事件内容
  async validateEvent(event) {
    const results = {
      schema: this.validateSchema(event),
      content: await this.validateContent(event),
      business: await this.validateBusinessRules(event)
    };

    return {
      valid: Object.values(results).every(r => r.valid),
      details: results
    };
  }

  // 模式验证
  validateSchema(event) {
    const schema = this.schemaRegistry.get(event.type);

    if (!schema) {
      return {
        valid: false,
        error: `No schema registered for event type: ${event.type}`
      };
    }

    try {
      // 使用 JSON Schema 验证
      const ajv = new Ajv();
      const validate = ajv.compile(schema);
      const valid = validate(event);

      return {
        valid,
        error: valid ? null : validate.errors
      };
    } catch (error) {
      return {
        valid: false,
        error: `Schema validation error: ${error.message}`
      };
    }
  }

  // 内容过滤
  async validateContent(event) {
    const content = JSON.stringify(event);

    for (const filter of this.contentFilters) {
      const result = await filter.check(content, event);
      if (!result.valid) {
        return {
          valid: false,
          error: result.error,
          filter: filter.name
        };
      }
    }

    return { valid: true };
  }

  // 业务规则验证
  async validateBusinessRules(event) {
    // 实现自定义业务逻辑验证
    switch (event.type) {
      case 'worker.deploy':
        return this.validateWorkerDeploy(event);
      case 'pages.build':
        return this.validatePagesBuild(event);
      case 'r2.object_created':
        return this.validateR2ObjectCreated(event);
      default:
        return { valid: true };
    }
  }
}

class MaliciousContentFilter {
  constructor() {

    this.patterns = [
      /<script/i,
      /javascript:/i,
      /eval\(/i,
      /exec\(/i

      if (pattern.test(content)) {
        return {
          valid: false,
          error: `Suspicious content detected: ${pattern.source}`
        };
      }
    }

    return { valid: true };
  }
}
```

```javascript
export class WebhookSecurityMonitor {
  constructor(logger, alertManager) {

      'signature_invalid': 'high',
      'timestamp_invalid': 'medium',
      'replay_attack': 'critical',
      'ip_unauthorized': 'high',
      'rate_limit_exceeded': 'medium',
      'content_malicious': 'critical',
      'schema_invalid': 'low'
    };

    return severityMap[event] || 'low';
  }

  // 检查告警条件
  async checkAlertConditions(event) {
    const recentEvents = this.getRecentEvents(300000); // 5分钟内

    // 失败尝试过多
    const failedAttempts = recentEvents.filter(e =>
      ['signature_invalid', 'timestamp_invalid', 'replay_attack'].includes(e.type)
    ).length;

    if (failedAttempts > this.thresholds.failedAttempts) {
      await this.alertManager.sendAlert({
        type: 'high_failure_rate',
        count: failedAttempts,
        events: recentEvents.slice(-10)
      });
    }

    // 可疑IP活动
    const suspiciousIPs = this.getSuspiciousIPs(recentEvents);
    if (suspiciousIPs.length > this.thresholds.suspiciousIP) {
      await this.alertManager.sendAlert({
        type: 'suspicious_ips',
        ips: suspiciousIPs,
        count: suspiciousIPs.length
      });
    }

    // 快速连续请求
    const rapidEvents = this.getRapidEvents(recentEvents, 20000); // 20秒内
    if (rapidEvents.length > this.thresholds.rapidFire) {
      await this.alertManager.sendAlert({
        type: 'rapid_fire',
        count: rapidEvents.length,
        duration: '20 seconds'
      });
    }
  }

  // 获取最近事件
  getRecentEvents(timeWindow) {
    const cutoff = Date.now() - timeWindow;
    return this.events.filter(e => new Date(e.timestamp).getTime() > cutoff);
  }

  // 获取可疑IP
  getSuspiciousIPs(events) {
    const ipCounts = {};

    events.forEach(event => {
      const ip = event.details.ip;
      if (ip && ['ip_unauthorized', 'signature_invalid'].includes(event.type)) {
        ipCounts[ip] = (ipCounts[ip] || 0) + 1;
      }
    });

    return Object.entries(ipCounts)
      .filter(([ip, count]) => count > 3)
      .map(([ip]) => ip);
  }

  // 获取快速连续事件
  getRapidEvents(events, timeWindow) {
    const eventsByTime = events
      .sort((a, b) => new Date(a.timestamp) - new Date(b.timestamp))
      .map(e => ({ ...e, time: new Date(e.timestamp).getTime() }));

    let rapidEvents = [];
    let windowStart = null;

    for (const event of eventsByTime) {
      if (!windowStart) {
        windowStart = event.time;
        rapidEvents = [event];
      } else if (event.time - windowStart <= timeWindow) {
        rapidEvents.push(event);
      } else {
        if (rapidEvents.length > this.thresholds.rapidFire) {
          return rapidEvents;
        }
        windowStart = event.time;
        rapidEvents = [event];
      }
    }

    return rapidEvents.length > this.thresholds.rapidFire ? rapidEvents : [];
  }
}
```

```javascript
// 完整的 Webhook 安全中间件
export class WebhookSecurityMiddleware {
  constructor(options = {}) {
    this.validator = new CloudflareWebhookValidator(
      options.secret,
      options.validatorOptions
    );

    this.ipValidator = new CloudflareIPValidator();
    this.rateLimiter = new WebhookRateLimiter(options.store, options.rateLimit);
    this.contentValidator = new WebhookContentValidator();
    this.securityMonitor = new WebhookSecurityMonitor(
      options.logger,
      options.alertManager
    );

    // 自动更新 IP 范围
    setInterval(() => {
      this.ipValidator.updateIPRanges();
    }, 24 * 60 * 60 * 1000);
  }

  // 创建 Express 中间件
  middleware() {
    return [
      // 1. IP 验证
      this.ipMiddleware(),
      // 2. 速率限制
      this.rateLimiter.middleware(),
      // 3. 签名验证
      this.signatureMiddleware(),
      // 4. 内容验证
      this.contentMiddleware()
    ];
  }

  // IP 验证中间件
  ipMiddleware() {
    return (req, res, next) => {
      const clientIP = req.ip || req.headers['cf-connecting-ip'];

      if (!this.ipValidator.isValidIP(clientIP)) {
        this.securityMonitor.recordSecurityEvent('ip_unauthorized', {
          ip: clientIP,
          userAgent: req.headers['user-agent']
        });

        return res.status(403).json({ error: 'Unauthorized source IP' });
      }

      req.clientIP = clientIP;
      next();
    };
  }

  // 签名验证中间件
  signatureMiddleware() {
    return async (req, res, next) => {
      try {
        const validation = await this.validator.validateRequest(
          req.headers,
          req.body.toString()
        );

        if (!validation.valid) {
          await this.securityMonitor.recordSecurityEvent(
            'signature_invalid',
            {
              ip: req.clientIP,
              error: validation.details.signature.error
            }
          );

          return res.status(401).json({
            error: 'Invalid webhook signature',
            details: validation.details
          });
        }

        req.webhookValidated = true;
        next();
      } catch (error) {
        await this.securityMonitor.recordSecurityEvent(
          'validation_error',
          { ip: req.clientIP, error: error.message }
        );

        return res.status(500).json({ error: 'Webhook validation failed' });
      }
    };
  }

  // 内容验证中间件
  contentMiddleware() {
    return async (req, res, next) => {
      try {
        const event = JSON.parse(req.body.toString());
        const validation = await this.contentValidator.validateEvent(event);

        if (!validation.valid) {
          await this.securityMonitor.recordSecurityEvent(
            'content_invalid',
            {
              ip: req.clientIP,
              error: validation.details
            }
          );

          return res.status(400).json({
            error: 'Invalid webhook content',
            details: validation.details
          });
        }

        req.event = event;
        next();
      } catch (error) {
        await this.securityMonitor.recordSecurityEvent(
          'content_parse_error',
          { ip: req.clientIP, error: error.message }
        );

        return res.status(400).json({ error: 'Invalid JSON content' });
      }
    };
  }
}

// 使用示例
const app = express();

  secret: process.env.CLOUDFLARE_WEBHOOK_SECRET,
  store: redisClient,
  rateLimit: {
    windowMs: 60000,
    maxRequests: 1000
  },
  logger: logger,
  alertManager: alertManager
});

// 应用所有安全中间件
app.post('/webhook/cloudflare', ...securityMiddleware.middleware(), async (req, res) => {
  try {
    // 处理验证过的事件
    await handleWebhookEvent(req.event);

    // 记录成功事件
    await securityMiddleware.securityMonitor.recordSecurityEvent(
      'webhook_processed',
      { ip: req.clientIP, eventType: req.event.type }
    );

    res.status(200).json({ status: 'received' });
  } catch (error) {
    console.error('Webhook processing error:', error);
    res.status(500).json({ error: 'Processing failed' });
  }
});
```
