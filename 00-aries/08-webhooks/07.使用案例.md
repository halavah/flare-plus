# Webhooks 使用案例

## 🎯 实际应用场景

Webhooks 在各种业务场景中都能发挥重要作用，通过事件驱动的方式实现系统间的实时通信和自动化处理。以下是一些常见且实用的应用案例。

## 📦 CI/CD 自动化

### 案例1：智能 CI/CD 管道

```javascript
// 智能构建触发器
export class SmartBuildTrigger {
  constructor(github, dockerHub, slack) {
    this.github = github;
    this.dockerHub = dockerHub;
    this.slack = slack;
    this.buildQueue = [];
    this.buildHistory = new Map();
  }

  // 处理 GitHub 推送事件
  async handleGitHubPush(event) {
    const {
      repository,
      ref,
      commits,
      pusher
    } = event.data;

    console.log(`Push to ${repository.full_name}: ${ref}`);

    // 分析推送特征
    const pushAnalysis = await this.analyzePush(commits);

    // 决定构建策略
    const buildStrategy = this.determineBuildStrategy(
      repository.name,
      ref,
      pushAnalysis
    );

    if (buildStrategy.shouldBuild) {
      // 创建构建任务
      const buildTask = await this.createBuildTask({
        repository: repository.full_name,
        ref: ref,
        commits: commits,
        strategy: buildStrategy,
        triggeredBy: pusher.name
      });

      // 加入构建队列
      await this.addToQueue(buildTask);

      // 通知团队
      await this.notifyBuildStarted(buildTask);
    }
  }

  // 分析推送内容
  async analyzePush(commits) {
    const analysis = {
      fileTypes: new Set(),
      directories: new Set(),
      hasTests: false,
      hasDocs: false,
      hasConfig: false,
      severity: 'low'
    };

    for (const commit of commits) {
      for (const file of commit.modified || commit.added || []) {
        const ext = file.split('.').pop().toLowerCase();
        analysis.fileTypes.add(ext);

        const dir = file.split('/')[0];
        analysis.directories.add(dir);

        if (file.includes('test') || file.includes('spec')) {
          analysis.hasTests = true;
        }
        if (file.includes('doc') || file.includes('readme')) {
          analysis.hasDocs = true;
        }
        if (file.includes('config') || file.includes('.yml') || file.includes('.json')) {
          analysis.hasConfig = true;
        }
      }
    }

    // 评估变更严重性
    if (analysis.fileTypes.has('sql') || analysis.fileTypes.has('migration')) {
      analysis.severity = 'high';
    } else if (analysis.fileTypes.has('js') || analysis.fileTypes.has('py')) {
      analysis.severity = 'medium';
    }

    return analysis;
  }

  // 确定构建策略
  determineBuildStrategy(repoName, ref, analysis) {
    const isMainBranch = ref === 'refs/heads/main' || ref === 'refs/heads/master';
    const isReleaseBranch = ref.includes('release');
    const isHotfix = ref.includes('hotfix');

    let strategy = {
      shouldBuild: false,
      priority: 'normal',
      runTests: true,
      runSecurity: false,
      deployToProduction: false,
      deployToStaging: false
    };

    // 主分支策略
    if (isMainBranch) {
      strategy.shouldBuild = true;
      strategy.priority = analysis.severity === 'high' ? 'high' : 'normal';
      strategy.runTests = true;
      strategy.runSecurity = true;
      strategy.deployToStaging = true;

      // 如果只有文档更新，可以跳过测试
      if (analysis.directories.size === 1 && analysis.hasDocs && !analysis.hasTests) {
        strategy.runTests = false;
        strategy.priority = 'low';
      }
    }

    // 发布分支策略
    if (isReleaseBranch) {
      strategy.shouldBuild = true;
      strategy.priority = 'high';
      strategy.runTests = true;
      strategy.runSecurity = true;
      strategy.deployToProduction = true;
    }

    // 热修复策略
    if (isHotfix) {
      strategy.shouldBuild = true;
      strategy.priority = 'critical';
      strategy.runTests = true;
      strategy.runSecurity = true;
      strategy.deployToProduction = true;
      strategy.deployToStaging = true;
    }

    // 功能分支策略
    if (!isMainBranch && !isReleaseBranch && !isHotfix) {
      // 只在特定条件下构建
      if (analysis.hasConfig || analysis.severity !== 'low') {
        strategy.shouldBuild = true;
        strategy.priority = 'low';
        strategy.runTests = true;
        strategy.deployToStaging = false;
      }
    }

    return strategy;
  }

  // 创建构建任务
  async createBuildTask(config) {
    const task = {
      id: crypto.randomUUID(),
      repository: config.repository,
      ref: config.ref,
      strategy: config.strategy,
      status: 'pending',
      createdAt: new Date().toISOString(),
      steps: this.generateBuildSteps(config.strategy)
    };

    return task;
  }

  // 生成构建步骤
  generateBuildSteps(strategy) {
    const steps = [
      {
        name: 'checkout',
        command: 'git checkout $REF',
        timeout: 30000
      },
      {
        name: 'setup',
        command: 'npm ci',
        timeout: 120000
      }
    ];

    if (strategy.runTests) {
      steps.push(
        {
          name: 'lint',
          command: 'npm run lint',
          timeout: 60000
        },
        {
          name: 'test',
          command: 'npm run test:coverage',
          timeout: 300000
        }
      );
    }

    if (strategy.runSecurity) {
      steps.push(
        {
          name: 'security-audit',
          command: 'npm audit --audit-level=moderate',
          timeout: 60000
        },
        {
          name: 'dependency-check',
          command: 'npm run security-check',
          timeout: 120000
        }
      );
    }

    steps.push({
      name: 'build',
      command: 'npm run build',
      timeout: 300000
    });

    if (strategy.deployToStaging) {
      steps.push({
        name: 'deploy-staging',
        command: 'npm run deploy:staging',
        timeout: 180000
      });
    }

    if (strategy.deployToProduction) {
      steps.push({
        name: 'deploy-production',
        command: 'npm run deploy:production',
        timeout: 300000
      });
    }

    return steps;
  }

  // 处理构建状态更新
  async handleBuildStatus(event) {
    const {
      build_id,
      status,
      step,
      output_url,
      coverage
    } = event.data;

    console.log(`Build ${build_id} status: ${status}`);

    // 更新构建历史
    this.buildHistory.set(build_id, {
      ...this.buildHistory.get(build_id),
      status,
      step,
      outputUrl: output_url,
      coverage
    });

    // 发送状态通知
    await this.notifyBuildStatus(build_id, status);

    // 处理构建完成
    if (status === 'success' || status === 'failure') {
      await this.handleBuildCompletion(build_id, status);
    }
  }

  // 处理构建完成
  async handleBuildCompletion(buildId, status) {
    const build = this.buildHistory.get(buildId);

    if (status === 'success' && build.strategy.deployToProduction) {
      // 触发 Docker 镜像构建
      await this.triggerDockerBuild(build);

      // 更新部署状态
      await this.updateDeploymentStatus(build);
    }

    if (status === 'failure') {
      // 创建 issue 跟踪
      await this.createBuildIssue(build);
    }

    // 清理队列中的任务
    this.removeFromQueue(buildId);
  }

  // 触发 Docker 构建
  async triggerDockerBuild(build) {
    const dockerBuild = {
      repository: build.repository,
      ref: build.ref,
      tag: this.generateDockerTag(build.ref),
      buildId: build.id
    };

    await this.dockerHub.triggerBuild(dockerBuild);
  }

  // 通知构建开始
  async notifyBuildStarted(build) {
    const message = {
      text: `🚀 Build started for ${build.repository}`,
      attachments: [{
        color: 'good',
        fields: [
          {
            title: 'Branch',
            value: build.ref,
            short: true
          },
          {
            title: 'Priority',
            value: build.strategy.priority,
            short: true
          },
          {
            title: 'Steps',
            value: build.steps.length.toString(),
            short: true
          }
        ]
      }]
    };

    await this.slack.sendMessage(message);
  }

  // 通知构建状态
  async notifyBuildStatus(buildId, status) {
    const build = this.buildHistory.get(buildId);
    const color = status === 'success' ? 'good' : status === 'failure' ? 'danger' : 'warning';

    const message = {
      text: `Build ${buildId} ${status}`,
      attachments: [{
        color: color,
        fields: [
          {
            title: 'Repository',
            value: build.repository,
            short: true
          },
          {
            title: 'Step',
            value: build.step || 'Unknown',
            short: true
          },
          {
            title: 'Duration',
            value: this.calculateDuration(build),
            short: true
          }
        ]
      }]
    };

    await this.slack.sendMessage(message);
  }
}

// Express 路由示例
const buildTrigger = new SmartBuildTrigger(
  new GitHubClient(),
  new DockerHubClient(),
  new SlackClient()
);

app.post('/webhook/github', async (req, res) => {
  const event = req.body;

  switch (event.type) {
    case 'push':
      await buildTrigger.handleGitHubPush(event);
      break;
    case 'pull_request':
      await buildTrigger.handlePullRequest(event);
      break;
    case 'release':
      await buildTrigger.handleRelease(event);
      break;
  }

  res.status(200).json({ status: 'received' });
});
```

```javascript
// 智能部署管理器
export class IntelligentDeploymentManager {
  constructor(k8s, prometheus, alertManager) {
    this.k8s = k8s;
    this.prometheus = prometheus;
    this.alertManager = alertManager;
    this.deployments = new Map();
  }

  // 处理部署事件
  async handleDeploymentEvent(event) {
    const {
      application,
      version,
      environment,
      deployment_strategy,
      rollout_config
    } = event.data;

    console.log(`Deploying ${application} v${version} to ${environment}`);

    // 获取当前部署状态
    const currentDeployment = await this.getCurrentDeployment(application, environment);

    // 创建部署计划
    const deploymentPlan = await this.createDeploymentPlan({
      application,
      version,
      environment,
      currentDeployment,
      strategy: deployment_strategy,
      config: rollout_config
    });

    // 执行部署
    await this.executeDeployment(deploymentPlan);

    // 监控部署过程
    this.monitorDeployment(deploymentPlan.id);
  }

  // 创建部署计划
  async createDeploymentPlan(config) {
    const plan = {
      id: crypto.randomUUID(),
      application: config.application,
      version: config.version,
      environment: config.environment,
      strategy: config.strategy,
      phases: [],
      status: 'pending',
      createdAt: new Date().toISOString()
    };

    // 基于策略生成部署阶段
    switch (config.strategy) {
      case 'rolling':
        plan.phases = this.generateRollingPhases(config);
        break;
      case 'blue-green':
        plan.phases = this.generateBlueGreenPhases(config);
        break;
      case 'canary':
        plan.phases = this.generateCanaryPhases(config);
        break;
      case 'a-b-testing':
        plan.phases = this.generateABTestPhases(config);
        break;
    }

    return plan;
  }

  // 生成滚动更新阶段
  generateRollingPhases(config) {
    const phases = [];
    const replicas = config.config.replicas || 3;
    const batchSize = Math.max(1, Math.floor(replicas / 3));

    for (let i = 0; i < replicas; i += batchSize) {
      phases.push({
        name: `batch-${Math.floor(i / batchSize) + 1}`,
        type: 'rollout',
        replicas: Math.min(batchSize, replicas - i),
        timeout: 300000,
        healthChecks: ['readiness', 'liveness'],
        rollbackOnFailure: true
      });
    }

    return phases;
  }

  // 生成金丝雀发布阶段
  generateCanaryPhases(config) {
    return [
      {
        name: 'canary-5%',
        type: 'canary',
        traffic: 5,
        duration: 600000, // 10分钟
        metrics: {
          errorRate: { threshold: 0.01, operator: 'lt' },
          responseTime: { threshold: 500, operator: 'lt' },
          throughput: { threshold: 100, operator: 'gt' }
        }
      },
      {
        name: 'canary-20%',
        type: 'canary',
        traffic: 20,
        duration: 600000,
        metrics: {
          errorRate: { threshold: 0.01, operator: 'lt' },
          responseTime: { threshold: 500, operator: 'lt' }
        }
      },
      {
        name: 'full-rollout',
        type: 'rollout',
        traffic: 100,
        duration: 300000
      }
    ];
  }

  // 监控部署
  async monitorDeployment(deploymentId) {
    const deployment = this.deployments.get(deploymentId);

    // 设置监控指标
    const metrics = await this.setupDeploymentMetrics(deployment);

    // 定期检查部署状态
    const monitorInterval = setInterval(async () => {
      const status = await this.checkDeploymentStatus(deploymentId, metrics);

      if (status.completed) {
        clearInterval(monitorInterval);
        await this.handleDeploymentCompletion(deploymentId, status);
      } else if (status.needsRollback) {
        clearInterval(monitorInterval);
        await this.rollbackDeployment(deploymentId);
      }
    }, 10000); // 每10秒检查一次
  }

  // 检查部署状态
  async checkDeploymentStatus(deploymentId, metrics) {
    const deployment = this.deployments.get(deploymentId);
    const currentPhase = deployment.phases.find(p => p.status === 'in-progress');

    if (!currentPhase) {
      return { completed: true };
    }

    // 获取实时指标
    const currentMetrics = await this.getCurrentMetrics(deployment, metrics);

    // 健康检查
    const healthStatus = await this.performHealthChecks(currentPhase, currentMetrics);

    // 金丝雀指标检查
    if (currentPhase.type === 'canary') {
      const metricsCheck = await this.validateCanaryMetrics(
        currentPhase.metrics,
        currentMetrics
      );

      if (!metricsCheck.passed) {
        return {
          completed: false,
          needsRollback: true,
          reason: metricsCheck.reason
        };
      }
    }

    return {
      completed: false,
      needsRollback: false,
      healthStatus,
      metrics: currentMetrics
    };
  }

  // 验证金丝雀指标
  async validateCanaryMetrics(expectedMetrics, currentMetrics) {
    for (const [metric, config] of Object.entries(expectedMetrics)) {
      const value = currentMetrics[metric];

      if (!this.compareMetric(value, config.threshold, config.operator)) {
        return {
          passed: false,
          reason: `${metric} ${value} ${config.operator} ${config.threshold}`
        };
      }
    }

    return { passed: true };
  }

  // 比较指标
  compareMetric(value, threshold, operator) {
    switch (operator) {
      case 'lt':
        return value < threshold;
      case 'gt':
        return value > threshold;
      case 'eq':
        return value === threshold;
      case 'ne':
        return value !== threshold;
      default:
        return false;
    }
  }
}
```

```javascript
// 实时订单处理系统
export class RealtimeOrderProcessor {
  constructor(inventory, payment, shipping, analytics) {
    this.inventory = inventory;
    this.payment = payment;
    this.shipping = shipping;
    this.analytics = analytics;
    this.orderQueue = new Map();
  }

  // 处理订单创建事件
  async handleOrderCreated(event) {
    const order = event.data;

    console.log(`Processing order ${order.id}`);

    try {
      // 1. 验证库存
      const inventoryCheck = await this.checkInventory(order.items);

      if (!inventoryCheck.available) {
        await this.handleOutOfStock(order, inventoryCheck.unavailable);
        return;
      }

      // 2. 预留库存
      await this.reserveInventory(order.id, order.items);

      // 3. 处理支付
      const paymentResult = await this.processPayment(order.payment);

      if (!paymentResult.success) {
        await this.handlePaymentFailure(order, paymentResult);
        return;
      }

      // 4. 确认订单
      await this.confirmOrder(order);

      // 5. 触发履约流程
      await this.triggerFulfillment(order);

      // 6. 更新分析数据
      await this.updateAnalytics('order_completed', order);

    } catch (error) {
      console.error('Order processing error:', error);
      await this.handleOrderError(order, error);
    }
  }

  // 处理支付事件
  async handlePaymentEvent(event) {
    const {
      order_id,
      status,
      payment_method,
      amount,
      transaction_id
    } = event.data;

    console.log(`Payment ${status} for order ${order_id}`);

    const order = await this.getOrder(order_id);

    if (status === 'succeeded') {
      // 支付成功，触发发货
      await this.triggerShipping(order);
    } else if (status === 'failed') {
      // 支付失败，取消订单
      await this.cancelOrder(order_id, 'payment_failed');
    }

    // 更新支付分析
    await this.updateAnalytics('payment_status', {
      orderId: order_id,
      status,
      method: payment_method,
      amount
    });
  }

  // 处理库存更新
  async handleInventoryUpdate(event) {
    const {
      product_id,
      quantity,
      operation,
      location,
      timestamp
    } = event.data;

    // 更新库存缓存
    await this.updateInventoryCache(product_id, quantity, operation);

    // 检查是否触发补货
    await this.checkReorderPoint(product_id, quantity);

    // 更新库存分析
    await this.updateAnalytics('inventory_change', {
      productId: product_id,
      quantity,
      operation,
      location
    });
  }

  // 处理物流更新
  async handleShippingUpdate(event) {
    const {
      order_id,
      tracking_number,
      status,
      estimated_delivery,
      location,
      carrier
    } = event.data;

    // 更新订单状态
    await this.updateOrderStatus(order_id, {
      shippingStatus: status,
      trackingNumber,
      estimatedDelivery,
      lastLocation: location
    });

    // 发送客户通知
    await this.sendShippingNotification(order_id, {
      status,
      trackingNumber,
      estimatedDelivery
    });

    // 更新分析数据
    await this.updateAnalytics('shipping_update', {
      orderId: order_id,
      status,
      carrier
    });
  }

  // 智能库存检查
  async checkInventory(items) {
    const unavailable = [];
    let allAvailable = true;

    // 并行检查所有商品库存
    const checks = await Promise.allSettled(
      items.map(async item => {
        const availability = await this.inventory.checkAvailability({
          productId: item.product_id,
          quantity: item.quantity,
          location: this.getOptimalLocation(item.product_id, item.quantity)
        });

        return {
          ...item,
          available: availability.available,
          location: availability.location,
          alternativeLocations: availability.alternatives
        };
      })
    );

    for (const check of checks) {
      if (check.status === 'fulfilled') {
        if (!check.value.available) {
          unavailable.push(check.value);
          allAvailable = false;
        }
      }
    }

    return {
      available: allAvailable,
      unavailable
    };
  }

  // 获取最优库存位置
  getOptimalLocation(productId, quantity) {
    // 基于地理位置、运输成本等因素选择最优位置
    // 这里简化为固定逻辑
    return 'warehouse-1';
  }

  // 预留库存
  async reserveInventory(orderId, items) {
    for (const item of items) {
      await this.inventory.reserve({
        productId: item.product_id,
        quantity: item.quantity,
        orderId: orderId,
        location: this.getOptimalLocation(item.product_id, item.quantity),
        ttl: 1800000 // 30分钟
      });
    }

    // 设置定时器释放未支付的库存
    setTimeout(() => {
      this.releaseInventoryReservation(orderId);
    }, 1800000);
  }

  // 触发履约流程
  async triggerFulfillment(order) {
    // 根据订单类型选择履约策略
    const fulfillmentStrategy = this.determineFulfillmentStrategy(order);

    switch (fulfillmentStrategy.type) {
      case 'immediate':
        await this.immediateFulfillment(order);
        break;
      case 'batched':
        await this.addToFulfillmentBatch(order);
        break;
      case 'dropship':
        await this.initiateDropship(order);
        break;
    }
  }

  // 确定履约策略
  determineFulfillmentStrategy(order) {
    // 根据商品类型、数量、客户位置等因素确定策略
    const hasLargeItems = order.items.some(item => item.weight > 10);
    const hasInternationalShipping = order.shipping.country !== 'US';

    if (hasLargeItems || hasInternationalShipping) {
      return { type: 'dropship' };
    }

    if (order.items.length > 10) {
      return { type: 'batched' };
    }

    return { type: 'immediate' };
  }

  // 发送客户通知
  async sendShippingNotification(orderId, shippingInfo) {
    const order = await this.getOrder(orderId);
    const customer = await this.getCustomer(order.customer_id);

    const notification = {
      type: 'shipping_update',
      customer: {
        email: customer.email,
        phone: customer.phone
      },
      content: {
        orderNumber: orderId,
        trackingNumber: shippingInfo.trackingNumber,
        status: shippingInfo.status,
        estimatedDelivery: shippingInfo.estimatedDelivery,
        message: this.generateShippingMessage(shippingInfo.status)
      },
      channels: ['email', 'sms', 'push']
    };

    await this.sendNotification(notification);
  }

  // 生成运输消息
  generateShippingMessage(status) {
    const messages = {
      'shipped': 'Your order has been shipped and is on its way!',
      'in_transit': 'Your package is in transit to your location.',
      'out_for_delivery': 'Your package is out for delivery today!',
      'delivered': 'Your package has been delivered. Enjoy!',
      'delayed': 'Your package delivery has been delayed. We apologize for the inconvenience.'
    };

    return messages[status] || 'Your order status has been updated.';
  }

  // 实时分析更新
  async updateAnalytics(eventType, data) {
    const analyticsEvent = {
      eventType,
      timestamp: new Date().toISOString(),
      data: this.sanitizeData(data)
    };

    // 发送到多个分析平台
    await Promise.allSettled([
      this.analytics.track('google_analytics', analyticsEvent),
      this.analytics.track('mixpanel', analyticsEvent),
      this.analytics.track('segment', analyticsEvent),
      this.analytics.track('internal', analyticsEvent)
    ]);
  }

  // 数据脱敏
  sanitizeData(data) {
    const sensitiveFields = ['email', 'phone', 'creditCard', 'ssn'];
    const sanitized = { ...data };

    for (const field of sensitiveFields) {
      if (sanitized[field]) {
        sanitized[field] = this.maskSensitiveData(sanitized[field]);
      }
    }

    return sanitized;
  }

  // 敏感数据掩码
  maskSensitiveData(data) {
    if (typeof data === 'string') {
      return data.substring(0, 3) + '***' + data.substring(data.length - 4);
    }
    return '***MASKED***';
  }
}
```

```javascript
// IoT 设备监控器
export class IoTDeviceMonitor {
  constructor(influxDB, alertManager, deviceRegistry) {
    this.influxDB = influxDB;
    this.alertManager = alertManager;
    this.deviceRegistry = deviceRegistry;
    this.deviceStates = new Map();
    this.alertRules = new Map();
  }

  // 处理设备数据上报
  async handleDeviceData(event) {
    const {
      device_id,
      device_type,
      timestamp,
      metrics,
      location,
      status
    } = event.data;

    // 验证设备
    if (!await this.validateDevice(device_id)) {
      console.warn(`Unknown device: ${device_id}`);
      return;
    }

    // 存储时序数据
    await this.storeMetrics(device_id, metrics, timestamp);

    // 更新设备状态
    await this.updateDeviceState(device_id, {
      lastSeen: new Date(timestamp),
      status,
      location,
      metrics
    });

    // 检查告警规则
    await this.checkAlertRules(device_id, metrics);

    // 预测性维护分析
    await this.predictiveMaintenance(device_id, metrics);

    // 更新设备位置
    if (location) {
      await this.updateDeviceLocation(device_id, location);
    }
  }

  // 处理设备告警
  async handleDeviceAlert(event) {
    const {
      device_id,
      alert_type,
      severity,
      message,
      data
    } = event.data;

    console.log(`Device ${device_id} alert: ${alert_type}`);

    // 获取设备信息
    const device = await this.getDevice(device_id);

    // 创建告警
    const alert = {
      id: crypto.randomUUID(),
      deviceId: device_id,
      deviceType: device.type,
      type: alert_type,
      severity,
      message,
      data,
      timestamp: new Date().toISOString(),
      status: 'active'
    };

    // 发送告警通知
    await this.sendAlert(alert);

    // 触发自动化响应
    await this.triggerAutomatedResponse(alert);

    // 记录告警
    await this.recordAlert(alert);
  }

  // 检查告警规则
  async checkAlertRules(deviceId, metrics) {
    const device = await this.getDevice(deviceId);
    const rules = await this.getAlertRules(device.type);

    for (const rule of rules) {
      const violation = await this.evaluateRule(rule, metrics);

      if (violation.triggered) {
        // 创建告警
        const alert = {
          deviceId,
          deviceType: device.type,
          type: rule.name,
          severity: rule.severity,
          message: violation.message,
          data: violation.data,
          ruleId: rule.id
        };

        await this.sendAlert(alert);
        await this.recordAlert(alert);
      }
    }
  }

  // 评估告警规则
  async evaluateRule(rule, metrics) {
    const { conditions, operator } = rule;
    const results = [];

    for (const condition of conditions) {
      const value = metrics[condition.metric];
      const result = this.evaluateCondition(value, condition);
      results.push(result);
    }

    const triggered = operator === 'AND'
      ? results.every(r => r)
      : results.some(r => r);

    return {
      triggered,
      message: triggered ? rule.message : null,
      data: {
        ruleId: rule.id,
        results,
        metrics
      }
    };
  }

  // 预测性维护
  async predictiveMaintenance(deviceId, metrics) {
    // 获取历史数据
    const historicalData = await this.getHistoricalMetrics(
      deviceId,
      24 * 7 // 过去7天
    );

    // 运行预测模型
    const prediction = await this.runPredictionModel(
      historicalData,
      metrics
    );

    if (prediction.risk > 0.7) {
      // 高风险，创建维护告警
      const alert = {
        deviceId,
        type: 'predictive_maintenance',
        severity: 'warning',
        message: `设备需要维护，故障风险: ${(prediction.risk * 100).toFixed(1)}%`,
        data: {
          risk: prediction.risk,
          predictedFailure: prediction.predictedFailure,
          recommendations: prediction.recommendations
        }
      };

      await this.sendAlert(alert);
      await this.scheduleMaintenance(deviceId, prediction);
    }
  }

  // 处理设备状态变更
  async handleDeviceStatusChange(event) {
    const {
      device_id,
      old_status,
      new_status,
      reason,
      timestamp
    } = event.data;

    // 更新设备状态
    await this.updateDeviceStatus(device_id, new_status, reason);

    // 特殊状态处理
    switch (new_status) {
      case 'offline':
        await this.handleDeviceOffline(device_id);
        break;
      case 'error':
        await this.handleDeviceError(device_id, reason);
        break;
      case 'maintenance':
        await this.handleDeviceMaintenance(device_id);
        break;
    }

    // 通知相关系统
    await this.notifyStatusChange(device_id, old_status, new_status);
  }

  // 处理设备离线
  async handleDeviceOffline(deviceId) {
    const device = await this.getDevice(deviceId);

    // 检查是否是预期离线
    if (device.expectedOffline) {
      return;
    }

    // 创建离线告警
    const alert = {
      deviceId,
      type: 'device_offline',
      severity: 'critical',
      message: `设备 ${deviceId} 已离线`,
      data: {
        lastSeen: device.lastSeen,
        offlineDuration: Date.now() - new Date(device.lastSeen).getTime()
      }
    };

    await this.sendAlert(alert);

    // 尝试远程诊断
    await this.attemptRemoteDiagnosis(deviceId);
  }

  // 处理批量设备事件
  async handleBatchDeviceEvents(event) {
    const {
      device_ids,
      event_type,
      data,
      batch_id
    } = event.data;

    console.log(`Processing batch ${batch_id} for ${device_ids.length} devices`);

    // 并行处理
    const results = await Promise.allSettled(
      device_ids.map(async (deviceId) => {
        try {
          // 为每个设备创建单独事件
          const deviceEvent = {
            id: crypto.randomUUID(),
            type: event_type,
            timestamp: new Date().toISOString(),
            data: {
              ...data,
              device_id: deviceId
            }
          };

          await this.processDeviceEvent(deviceEvent);
          return { deviceId, success: true };
        } catch (error) {
          return { deviceId, success: false, error: error.message };
        }
      })
    );

    // 统计结果
    const successful = results.filter(r => r.status === 'fulfilled' && r.value.success).length;
    const failed = device_ids.length - successful;

    // 发送批处理报告
    await this.sendBatchReport(batch_id, {
      total: device_ids.length,
      successful,
      failed,
      results
    });
  }

  // 处理固件更新事件
  async handleFirmwareUpdate(event) {
    const {
      device_id,
      version,
      update_status,
      progress,
      error_message
    } = event.data;

    // 更新固件版本记录
    await this.updateFirmwareVersion(device_id, version, update_status);

    if (update_status === 'completed') {
      // 验证更新
      await this.verifyFirmwareUpdate(device_id, version);
    } else if (update_status === 'failed') {
      // 更新失败，尝试回滚
      await this.attemptFirmwareRollback(device_id, error_message);
    }

    // 更新进度
    if (progress !== undefined) {
      await this.updateUpdateProgress(device_id, progress);
    }
  }

  // 设备健康评分
  async calculateDeviceHealth(deviceId) {
    const device = await this.getDevice(deviceId);
    const recentMetrics = await this.getRecentMetrics(deviceId, 24); // 24小时

    let healthScore = 100;

    // 基于指标计算健康分
    for (const [metric, value] of Object.entries(recentMetrics)) {
      const metricRule = await this.getMetricRule(device.type, metric);
      if (metricRule) {
        const score = this.calculateMetricScore(value, metricRule);
        healthScore = Math.min(healthScore, score);
      }
    }

    // 基于告警历史调整
    const recentAlerts = await this.getRecentAlerts(deviceId, 24);
    healthScore -= recentAlerts.length * 5;

    // 基于在线时间调整
    const uptime = await this.calculateUptime(deviceId, 24);
    if (uptime < 0.95) {
      healthScore -= (1 - uptime) * 20;
    }

    return Math.max(0, Math.min(100, healthScore));
  }

  // 获取设备总览
  async getDeviceOverview() {
    const devices = await this.getAllDevices();
    const overview = {
      total: devices.length,
      online: 0,
      offline: 0,
      error: 0,
      maintenance: 0,
      healthDistribution: {
        excellent: 0,
        good: 0,
        fair: 0,
        poor: 0
      }
    };

    for (const device of devices) {
      overview[device.status]++;

      const health = await this.calculateDeviceHealth(device.id);
      if (health >= 90) overview.healthDistribution.excellent++;
      else if (health >= 75) overview.healthDistribution.good++;
      else if (health >= 60) overview.healthDistribution.fair++;
      else overview.healthDistribution.poor++;
    }

    return overview;
  }
}
```
