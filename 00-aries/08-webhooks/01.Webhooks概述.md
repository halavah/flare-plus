# Webhooks 概述

## 🎯 什么是 Webhooks？

Webhooks 是一种"反向 API"，允许应用程序在特定事件发生时自动向其他系统发送 HTTP 请求。与传统的轮询方式不同，Webhooks 采用事件驱动模式，当数据发生变化或事件触发时，系统会主动推送通知到预配置的 URL 端点。

### 工作原理
```
事件触发 → Webhook 触发 → HTTP POST 请求 → 目标系统处理
    ↓
[Cloudflare 服务] → [Webhook 配置] → [目标 URL]
```

### 与轮询的对比

| 特性 | Webhooks | 轮询 |
|------|----------|------|
| 延迟 | 近实时（秒级） | 延迟（分钟/小时级） |
| 资源消耗 | 低（事件驱动） | 高（持续请求） |
| 实时性 | 高 | 低 |
| 服务器负载 | 按需发送 | 持续负载 |
| 复杂性 | 需要接收端处理 | 实现简单 |

## 🔧 Cloudflare Webhooks 服务

### 核心功能
1. **事件订阅**：订阅 Cloudflare 各类服务事件
2. **灵活配置**：支持自定义过滤条件和重试策略
3. **安全认证**：提供签名验证和 IP 白名单
4. **监控管理**：实时监控 Webhook 调用状态和成功率

### 支持的事件类型

#### 1. Workers 事件
- Worker 脚本部署/更新
- Worker 执行失败/超时
- Cron 触发器执行
- Worker 仪表盘事件

#### 2. Pages 事件
- Pages 项目构建
- 部署成功/失败
- 预览环境发布
- 分支推送

#### 3. R2 存储事件
- 对象创建/删除
- 存储桶配置变更
- 生命周期策略执行
- 数据传输事件

#### 4. D1 数据库事件
- 数据库创建/删除
- 查询执行
- 数据变更
- 备份完成

#### 5. KV 存储事件
- 键值对写入/删除
- 命名空间创建
- 批量操作
- 缓存失效

#### 6. Durable Objects 事件
- 对象创建/销毁
- 连接建立/断开
- 状态变更
- 消息传递

## 🚀 快速开始

### 1. 创建 Webhook 端点

首先，你需要创建一个能够接收 Webhook 请求的服务器端点：

```javascript
// webhook-receiver.js - 使用 Node.js/Express
const express = require('express');
const crypto = require('crypto');

const app = express();
app.use(express.raw({ type: 'application/json' }));

// Cloudflare Webhook 签名验证中间件
function verifyCloudflareSignature(req, res, next) {
  const signature = req.headers['cf-webhook-signature'];
  const timestamp = req.headers['cf-webhook-timestamp'];
  const secret = process.env.CLOUDFLARE_WEBHOOK_SECRET;

  if (!signature || !timestamp) {
    return res.status(401).json({ error: 'Missing signature headers' });
  }

  // 构造待验证字符串
  const payload = `${timestamp}.${req.body}`;

  // 计算预期签名
  const expectedSignature = crypto
    .createHmac('sha256', secret)
    .update(payload)
    .digest('hex');

  // 验证签名
  if (signature !== expectedSignature) {
    return res.status(401).json({ error: 'Invalid signature' });
  }

  next();
}

// Webhook 接收端点
app.post('/webhook/cloudflare', verifyCloudflareSignature, (req, res) => {
  try {
    const event = JSON.parse(req.body.toString());

    console.log('Received Cloudflare event:', {
      type: event.type,
      timestamp: event.timestamp,
      data: event.data
    });

    // 处理不同类型的事件
    switch (event.type) {
      case 'worker.deploy':
        handleWorkerDeploy(event);
        break;
      case 'pages.build':
        handlePagesBuild(event);
        break;
      case 'r2.object_created':
        handleR2ObjectCreated(event);
        break;
      default:
        console.log('Unknown event type:', event.type);
    }

    res.status(200).json({ status: 'received' });
  } catch (error) {
    console.error('Webhook processing error:', error);
    res.status(500).json({ error: 'Processing failed' });
  }
});

// 处理 Worker 部署事件
function handleWorkerDeploy(event) {
  const { worker_name, script_id, deployment_id } = event.data;

  console.log(`Worker ${worker_name} deployed with ID ${deployment_id}`);

  // 更新部署记录
  updateDeploymentRecord(worker_name, deployment_id);

  // 发送通知
  sendNotification(`Worker ${worker_name} has been successfully deployed`);
}

// 处理 Pages 构建事件
function handlePagesBuild(event) {
  const { project_name, build_id, status, url } = event.data;

  if (status === 'success') {
    console.log(`Pages build ${build_id} completed successfully`);
    console.log(`Preview URL: ${url}`);

    // 自动部署到生产环境
    if (shouldDeployToProduction(project_name)) {
      promoteToProduction(project_name, build_id);
    }
  } else {
    console.error(`Pages build ${build_id} failed`);
    sendErrorNotification(event.data);
  }
}

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Webhook receiver listening on port ${PORT}`);
});
```

### 2. 配置 Cloudflare Webhook

通过 Cloudflare API 或 Dashboard 创建 Webhook：

```javascript
// 创建 Webhook 配置
async function createWebhook() {
  const response = await fetch('https://api.cloudflare.com/client/v4/accounts/{account_id}/webhooks', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${CLOUDFLARE_API_TOKEN}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      name: 'Production Webhook',
      url: 'https://your-server.com/webhook/cloudflare',
      secret: 'your-webhook-secret',
      event_filters: [
        {
          type: 'worker.deploy',
          filters: {
            script_names: ['my-worker', 'api-worker']
          }
        },
        {
          type: 'pages.build',
          filters: {
            project_names: ['my-website']
          }
        }
      ],
      retry_config: {
        max_retries: 3,
        retry_delay: 5000
      }
    })
  });

  const result = await response.json();
  console.log('Webhook created:', result);
  return result;
}
```

## 🔒 安全最佳实践

### 1. 签名验证

```javascript
// 验证 Cloudflare 签名
function verifyWebhookSignature(payload, signature, secret, timestamp) {
  // 检查时间戳是否在允许范围内（5分钟）
  const now = Math.floor(Date.now() / 1000);
  if (Math.abs(now - timestamp) > 300) {
    return false;
  }

  // 构造待验证字符串
  const message = `${timestamp}.${payload}`;

  // 计算 HMAC-SHA256
  const expectedSignature = crypto
    .createHmac('sha256', secret)
    .update(message)
    .digest('hex');

  // 使用安全比较防止时序攻击
  return crypto.timingSafeEqual(
    Buffer.from(signature, 'hex'),
    Buffer.from(expectedSignature, 'hex')
  );
}
```

### 2. IP 白名单

```javascript
// 验证请求来源 IP
const CLOUDFLARE_IPS = [
  '103.21.244.0/22',
  '103.22.200.0/22',
  '103.31.4.0/22',
  '104.16.0.0/13',
  '104.24.0.0/14',
  '108.162.192.0/18',
  '131.0.72.0/22',
  '141.101.64.0/18',
  '162.158.0.0/15',
  '172.64.0.0/13',
  '173.245.48.0/20',
  '188.114.96.0/20',
  '190.93.240.0/20',
  '197.234.240.0/22',
  '198.41.128.0/17'
];

function isCloudflareIP(ip) {
  const ipInt = ipToInt(ip);

  return CLOUDFLARE_IPS.some(range => {
    const [network, prefixLength] = range.split('/');
    const networkInt = ipToInt(network);
    const mask = (0xFFFFFFFF << (32 - parseInt(prefixLength))) >>> 0;

    return (ipInt & mask) === (networkInt & mask);
  });
}

// IP 验证中间件
function verifyCloudflareIP(req, res, next) {
  const clientIP = req.ip || req.connection.remoteAddress;

  if (!isCloudflareIP(clientIP)) {
    return res.status(403).json({ error: 'Unauthorized IP' });
  }

  next();
}
```

### 3. 重放攻击防护

```javascript
// 防止重放攻击的 nonce 存储
const processedNonces = new Set();

function checkReplayAttack(nonce, maxAge = 300000) { // 5分钟
  // 检查 nonce 是否已处理
  if (processedNonces.has(nonce)) {
    return false; // 重放攻击
  }

  // 解码 nonce 中的时间戳
  const timestamp = parseInt(nonce.split('_')[1]);
  const now = Date.now();

  // 检查是否过期
  if (now - timestamp > maxAge) {
    return false;
  }

  // 记录已处理的 nonce
  processedNonces.add(nonce);

  // 定期清理旧的 nonce
  setTimeout(() => {
    processedNonces.delete(nonce);
  }, maxAge);

  return true;
}
```

## 📈 监控和分析

### Webhook 性能指标

```javascript
// Webhook 监控类
class WebhookMonitor {
  constructor() {
    this.metrics = {
      totalReceived: 0,
      successful: 0,
      failed: 0,
      averageLatency: 0,
      eventTypeCounts: new Map()
    };
  }

  recordEvent(eventType, processingTime, success) {
    this.metrics.totalReceived++;

    if (success) {
      this.metrics.successful++;
    } else {
      this.metrics.failed++;
    }

    // 更新平均延迟
    this.metrics.averageLatency =
      (this.metrics.averageLatency + processingTime) / 2;

    // 更新事件类型计数
    const count = this.metrics.eventTypeCounts.get(eventType) || 0;
    this.metrics.eventTypeCounts.set(eventType, count + 1);
  }

  getMetrics() {
    return {
      ...this.metrics,
      successRate: this.metrics.successful / this.metrics.totalReceived,
      eventTypeCounts: Object.fromEntries(this.metrics.eventTypeCounts)
    };
  }
}
```
