# Webhooks 扩展模式

## 🚀 扩展架构概述

Webhooks 扩展模式允许你超越基础事件通知功能，构建复杂的、可扩展的事件驱动生态系统。通过结合各种服务、工具和技术，可以打造强大的分布式系统和实时数据处理平台。

### 扩展层次架构
```
┌─────────────────────────────────────┐
│           应用层扩展                    │
│  ┌─────────────────────────────────┐  │
│  │     业务流程自动化               │  │
│  └─────────────────────────────────┘  │
├─────────────────────────────────────┤
│           服务层扩展                    │
│  ┌─────────────────────────────────┐  │
│  │     微服务编排与协调             │  │
│  └─────────────────────────────────┘  │
├─────────────────────────────────────┤
│           数据层扩展                    │
│  ┌─────────────────────────────────┐  │
│  │     事件流处理与分析             │  │
│  └─────────────────────────────────┘  │
├─────────────────────────────────────┤
│           基础设施扩展                  │
│  ┌─────────────────────────────────┐  │
│  │     多云与边缘计算               │  │
│  └─────────────────────────────────┘  │
└─────────────────────────────────────┘
```

```javascript
// 高级事件流处理器
export class AdvancedEventStreamProcessor {
  constructor(options = {}) {
    this.kafka = options.kafka;
    this.redis = options.redis;
    this.postgres = options.postgres;
    this.elasticsearch = options.elasticsearch;

    this.streams = new Map();
    this.processors = new Map();
    this.transformers = new Map();
    this.aggregators = new Map();

    this.metrics = {
      processed: 0,
      failed: 0,
      throughput: 0,
      latency: 0
    };
  }

  // 创建事件流
  async createStream(streamConfig) {
    const stream = {
      id: streamConfig.id,
      name: streamConfig.name,
      source: streamConfig.source,
      processors: streamConfig.processors || [],
      transformers: streamConfig.transformers || [],
      aggregators: streamConfig.aggregators || [],
      outputs: streamConfig.outputs || [],
      bufferSize: streamConfig.bufferSize || 1000,
      batchTimeout: streamConfig.batchTimeout || 5000,
      retryPolicy: streamConfig.retryPolicy || {
        maxRetries: 3,
        backoff: 'exponential'
      }
    };

    // 创建 Kafka 主题
    await this.kafka.createTopic(stream.id, {
      partitions: streamConfig.partitions || 3,
      replicationFactor: 2
    });

    // 注册流处理器
    this.registerStreamProcessors(stream);

    this.streams.set(stream.id, stream);
    return stream;
  }

  // 注册流处理器
  registerStreamProcessors(stream) {
    for (const processorConfig of stream.processors) {
      const processor = new StreamProcessor(processorConfig, this);
      this.processors.set(processorConfig.id, processor);
    }
  }

  // 处理原始事件
  async processRawEvent(event) {
    const startTime = Date.now();

    try {
      // 确定目标流
      const targetStreams = await this.determineTargetStreams(event);

      // 并行处理到多个流
      const processingPromises = targetStreams.map(async (streamId) => {
        const stream = this.streams.get(streamId);
        return this.processEventToStream(event, stream);
      });

      await Promise.allSettled(processingPromises);

      // 更新指标
      this.updateMetrics(Date.now() - startTime, true);

    } catch (error) {
      this.updateMetrics(Date.now() - startTime, false);
      console.error('Stream processing error:', error);
      throw error;
    }
  }

  // 处理事件到特定流
  async processEventToStream(event, stream) {
    // 应用转换器
    let transformedEvent = event;
    for (const transformerConfig of stream.transformers) {
      const transformer = this.transformers.get(transformerConfig.id);
      if (transformer) {
        transformedEvent = await transformer.transform(transformedEvent);
      }
    }

    // 应用处理器
    for (const processorConfig of stream.processors) {
      const processor = this.processors.get(processorConfig.id);
      if (processor) {
        await processor.process(transformedEvent);
      }
    }

    // 应用聚合器
    for (const aggregatorConfig of stream.aggregators) {
      const aggregator = this.aggregators.get(aggregatorConfig.id);
      if (aggregator) {
        await aggregator.aggregate(transformedEvent);
      }
    }

    // 发送到输出
    for (const output of stream.outputs) {
      await this.sendToOutput(transformedEvent, output);
    }
  }

  // 确定目标流
  async determineTargetStreams(event) {
    const streams = [];

    // 基于事件类型
    const eventTypeStreams = this.getStreamsByType(event.type);
    streams.push(...eventTypeStreams);

    // 基于事件源
    const sourceStreams = this.getStreamsBySource(event.source);
    streams.push(...sourceStreams);

    // 基于标签
    const tagStreams = this.getStreamsByTags(event.tags || []);
    streams.push(...tagStreams);

    // 去重
    return [...new Set(streams)];
  }

  // 发送到输出
  async sendToOutput(event, output) {
    switch (output.type) {
      case 'kafka':
        await this.kafka.send(output.topic, event);
        break;
      case 'redis':
        await this.redis.lpush(output.key, JSON.stringify(event));
        break;
      case 'postgres':
        await this.postgres.insert(output.table, event);
        break;
      case 'elasticsearch':
        await this.elasticsearch.index(output.index, event);
        break;
      case 'webhook':
        await this.sendWebhook(output.url, event);
        break;
    }
  }
}

// 流处理器
class StreamProcessor {
  constructor(config, streamProcessor) {
    this.id = config.id;
    this.type = config.type;
    this.config = config;
    this.streamProcessor = streamProcessor;
    this.state = new Map();
  }

  async process(event) {
    switch (this.type) {
      case 'filter':
        return await this.filterEvent(event);
      case 'enrich':
        return await this.enrichEvent(event);
      case 'validate':
        return await this.validateEvent(event);
      case 'transform':
        return await this.transformEvent(event);
      case 'route':
        return await this.routeEvent(event);
      default:
        throw new Error(`Unknown processor type: ${this.type}`);
    }
  }

  async filterEvent(event) {
    const condition = this.config.condition;
    return this.evaluateCondition(condition, event);
  }

  async enrichEvent(event) {
    const enrichers = this.config.enrichers || [];
    for (const enricher of enrichers) {
      event = await this.applyEnricher(enricher, event);
    }
    return event;
  }

  async validateEvent(event) {
    const schema = this.config.schema;
    return this.validateAgainstSchema(event, schema);
  }
}

// 事件转换器
class EventTransformer {
  constructor(config) {
    this.id = config.id;
    this.type = config.type;
    this.config = config;
  }

  async transform(event) {
    switch (this.type) {
      case 'field_mapping':
        return this.mapFields(event);
      case 'data_type_conversion':
        return this.convertDataTypes(event);
      case 'format_change':
        return this.changeFormat(event);
      case 'content_extraction':
        return this.extractContent(event);
      default:
        return event;
    }
  }

  mapFields(event) {
    const mapping = this.config.mapping;
    const transformed = {};

    for (const [targetField, sourceField] of Object.entries(mapping)) {
      if (typeof sourceField === 'string') {
        transformed[targetField] = this.getNestedValue(event, sourceField);
      } else if (typeof sourceField === 'object') {
        transformed[targetField] = this.computeValue(event, sourceField);
      }
    }

    return transformed;
  }

  getNestedValue(obj, path) {
    return path.split('.').reduce((current, key) => current?.[key], obj);
  }

  computeValue(event, computation) {
    switch (computation.type) {
      case 'concat':
        const values = computation.fields.map(f => this.getNestedValue(event, f));
        return values.join(computation.separator || '');
      case 'sum':
        return computation.fields.reduce((sum, f) => sum + (this.getNestedValue(event, f) || 0), 0);
      case 'timestamp':
        return new Date().toISOString();
      default:
        return null;
    }
  }
}

// 事件聚合器
class EventAggregator {
  constructor(config) {
    this.id = config.id;
    this.type = config.type;
    this.config = config;
    this.buffer = [];
    this.lastFlush = Date.now();
  }

  async aggregate(event) {
    this.buffer.push(event);

    // 检查是否需要刷新
    const now = Date.now();
    const shouldFlush =
      this.buffer.length >= this.config.batchSize ||
      (now - this.lastFlush) >= this.config.flushInterval;

    if (shouldFlush) {
      await this.flush();
    }
  }

  async flush() {
    if (this.buffer.length === 0) return;

    const batch = [...this.buffer];
    this.buffer = [];
    this.lastFlush = Date.now();

    switch (this.type) {
      case 'count':
        await this.aggregateCount(batch);
        break;
      case 'sum':
        await this.aggregateSum(batch);
        break;
      case 'average':
        await this.aggregateAverage(batch);
        break;
      case 'window':
        await this.aggregateWindow(batch);
        break;
    }
  }

  async aggregateCount(batch) {
    const counts = {};
    const groupBy = this.config.groupBy;

    for (const event of batch) {
      const key = this.getGroupKey(event, groupBy);
      counts[key] = (counts[key] || 0) + 1;
    }

    // 发送聚合结果
    await this.sendAggregationResult('count', counts);
  }

  getGroupKey(event, groupBy) {
    if (!groupBy) return 'default';
    return groupBy.map(field => this.getNestedValue(event, field)).join('|');
  }
}
```

```javascript
// 复杂事件处理器 (CEP)
export class ComplexEventProcessor {
  constructor(options = {}) {
    this.patterns = new Map();
    this.matches = new Map();
    this.timers = new Map();
    this.store = options.store || new Map();
  }

  // 注册事件模式
  registerPattern(patternConfig) {
    const pattern = {
      id: patternConfig.id,
      name: patternConfig.name,
      type: patternConfig.type, // sequence, selection, negation, etc.
      events: patternConfig.events,
      conditions: patternConfig.conditions || [],
      timeWindow: patternConfig.timeWindow || 60000,
      actions: patternConfig.actions || [],
      priority: patternConfig.priority || 1
    };

    this.patterns.set(pattern.id, pattern);
    return pattern;
  }

  // 处理事件
  async processEvent(event) {
    // 检查所有相关模式
    const relevantPatterns = this.getRelevantPatterns(event);

    for (const pattern of relevantPatterns) {
      await this.processEventForPattern(event, pattern);
    }

    // 清理过期匹配
    await this.cleanupExpiredMatches();
  }

  // 处理特定模式的事件
  async processEventForPattern(event, pattern) {
    const match = this.getOrCreateMatch(pattern.id);

    switch (pattern.type) {
      case 'sequence':
        await this.processSequencePattern(event, pattern, match);
        break;
      case 'selection':
        await this.processSelectionPattern(event, pattern, match);
        break;
      case 'negation':
        await this.processNegationPattern(event, pattern, match);
        break;
      case 'repetition':
        await this.processRepetitionPattern(event, pattern, match);
        break;
    }
  }

  // 处理序列模式
  async processSequencePattern(event, pattern, match) {
    const expectedEventIndex = match.events.length;

    if (expectedEventIndex >= pattern.events.length) {
      return; // 模式已完成
    }

    const expectedEvent = pattern.events[expectedEventIndex];

    // 检查事件是否匹配
    if (this.eventMatches(event, expectedEvent)) {
      // 添加到匹配序列
      match.events.push({
        event,
        timestamp: Date.now()
      });

      // 设置定时器
      this.setPatternTimer(pattern.id);

      // 检查是否完成
      if (match.events.length === pattern.events.length) {
        if (this.evaluateConditions(pattern.conditions, match.events.map(e => e.event))) {
          await this.triggerPatternActions(pattern, match);
          this.completeMatch(pattern.id);
        }
      }
    }
  }

  // 处理选择模式
  async processSelectionPattern(event, pattern, match) {
    // 检查事件是否匹配任何选项
    for (const eventOption of pattern.events) {
      if (this.eventMatches(event, eventOption)) {
        match.events.push({
          event,
          timestamp: Date.now()
        });

        // 选择模式只需要一个匹配即可完成
        if (this.evaluateConditions(pattern.conditions, match.events.map(e => e.event))) {
          await this.triggerPatternActions(pattern, match);
          this.completeMatch(pattern.id);
        }
        break;
      }
    }
  }

  // 处理否定模式
  async processNegationPattern(event, pattern, match) {
    // 检查是否出现了不应该出现的事件
    for (const negatedEvent of pattern.events) {
      if (this.eventMatches(event, negatedEvent)) {
        // 否定模式被违反
        await this.triggerPatternActions({
          ...pattern,
          type: 'negation_violation'
        }, match);
        this.failMatch(pattern.id);
        return;
      }
    }

    // 正面条件检查
    if (pattern.positiveConditions) {
      if (this.evaluateConditions(pattern.positiveConditions, [event])) {
        match.positiveEvents.push({
          event,
          timestamp: Date.now()
        });

        // 检查是否满足正面条件
        if (this.evaluateConditions(pattern.conditions, match.positiveEvents.map(e => e.event))) {
          await this.triggerPatternActions(pattern, match);
          this.completeMatch(pattern.id);
        }
      }
    }
  }

  // 事件匹配检查
  eventMatches(event, eventTemplate) {
    // 检查事件类型
    if (eventTemplate.type && event.type !== eventTemplate.type) {
      return false;
    }

    // 检查属性
    if (eventTemplate.attributes) {
      for (const [key, value] of Object.entries(eventTemplate.attributes)) {
        if (event.data[key] !== value) {
          return false;
        }
      }
    }

    // 检查条件
    if (eventTemplate.conditions) {
      for (const condition of eventTemplate.conditions) {
        if (!this.evaluateCondition(condition, event)) {
          return false;
        }
      }
    }

    return true;
  }

  // 评估条件
  evaluateCondition(condition, context) {
    switch (condition.operator) {
      case 'equals':
        return this.getValue(condition.field, context) === condition.value;
      case 'greater_than':
        return this.getValue(condition.field, context) > condition.value;
      case 'less_than':
        return this.getValue(condition.field, context) < condition.value;
      case 'contains':
        return this.getValue(condition.field, context).includes(condition.value);
      case 'matches':
        const regex = new RegExp(condition.value);
        return regex.test(this.getValue(condition.field, context));
      default:
        return true;
    }
  }

  // 获取值
  getValue(path, context) {
    if (path.startsWith('event.')) {
      const fieldPath = path.substring(6);
      return this.getNestedValue(context[0], fieldPath);
    } else if (path.startsWith('match.')) {
      const matchIndex = parseInt(path.split('.')[1]);
      const fieldPath = path.split('.').slice(2).join('.');
      return this.getNestedValue(context[matchIndex], fieldPath);
    }
    return path;
  }

  getNestedValue(obj, path) {
    return path.split('.').reduce((current, key) => current?.[key], obj);
  }

  // 触发模式动作
  async triggerPatternActions(pattern, match) {
    for (const action of pattern.actions) {
      await this.executeAction(action, pattern, match);
    }
  }

  // 执行动作
  async executeAction(action, pattern, match) {
    switch (action.type) {
      case 'webhook':
        await this.sendActionWebhook(action, pattern, match);
        break;
      case 'event':
        await this.emitActionEvent(action, pattern, match);
        break;
      case 'command':
        await this.executeCommand(action, pattern, match);
        break;
      case 'notification':
        await this.sendNotification(action, pattern, match);
        break;
    }
  }
}
```

### 1. 工作流引擎

```javascript
// 微服务工作流引擎
export class MicroserviceWorkflowEngine {
  constructor(options = {}) {
    this.workflows = new Map();
    this.instances = new Map();
    this.services = new Map();
    this.taskQueue = new PriorityQueue();
    this.stateStore = options.stateStore || new Map();
    this.eventBus = options.eventBus;
  }

  // 注册工作流
  async registerWorkflow(workflowConfig) {
    const workflow = {
      id: workflowConfig.id,
      name: workflowConfig.name,
      version: workflowConfig.version || '1.0',
      description: workflowConfig.description,
      tasks: await this.parseTasks(workflowConfig.tasks),
      triggers: workflowConfig.triggers || [],
      variables: workflowConfig.variables || {},
      errorHandling: workflowConfig.errorHandling || {
        retryPolicy: 'exponential',
        maxRetries: 3,
        timeout: 300000
      },
      compensation: workflowConfig.compensation || []
    };

    this.workflows.set(workflow.id, workflow);
    return workflow;
  }

  // 注册服务
  registerService(serviceConfig) {
    const service = {
      id: serviceConfig.id,
      name: serviceConfig.name,
      endpoint: serviceConfig.endpoint,
      healthCheck: serviceConfig.healthCheck,
      timeout: serviceConfig.timeout || 30000,
      retryPolicy: serviceConfig.retryPolicy,
      circuitBreaker: serviceConfig.circuitBreaker,
      authentication: serviceConfig.authentication
    };

    this.services.set(service.id, service);
    return service;
  }

  // 启动工作流实例
  async startWorkflow(workflowId, input, options = {}) {
    const workflow = this.workflows.get(workflowId);
    if (!workflow) {
      throw new Error(`Workflow not found: ${workflowId}`);
    }

    const instance = {
      id: crypto.randomUUID(),
      workflowId,
      status: 'running',
      startTime: Date.now(),
      input,
      variables: { ...workflow.variables, ...input },
      currentTask: null,
      completedTasks: [],
      failedTasks: [],
      context: {
        instanceId: crypto.randomUUID(),
        correlationId: options.correlationId || crypto.randomUUID(),
        parentInstanceId: options.parentInstanceId,
        retryCount: 0
      },
      history: []
    };

    this.instances.set(instance.id, instance);

    // 保存状态
    await this.saveInstanceState(instance);

    // 开始执行第一个任务
    await this.executeNextTask(instance);

    return instance;
  }

  // 执行下一个任务
  async executeNextTask(instance) {
    const workflow = this.workflows.get(instance.workflowId);
    const nextTask = this.getNextTask(workflow, instance);

    if (!nextTask) {
      // 工作流完成
      await this.completeWorkflow(instance, 'success');
      return;
    }

    instance.currentTask = nextTask.id;
    instance.status = 'executing';

    // 记录历史
    instance.history.push({
      type: 'task_started',
      taskId: nextTask.id,
      timestamp: Date.now()
    });

    await this.saveInstanceState(instance);

    // 执行任务
    try {
      const result = await this.executeTask(nextTask, instance);
      await this.handleTaskSuccess(instance, nextTask, result);
    } catch (error) {
      await this.handleTaskFailure(instance, nextTask, error);
    }
  }

  // 执行任务
  async executeTask(task, instance) {
    const service = this.services.get(task.serviceId);
    if (!service) {
      throw new Error(`Service not found: ${task.serviceId}`);
    }

    // 准备任务输入
    const taskInput = await this.prepareTaskInput(task, instance);

    // 执行服务调用
    const result = await this.callService(service, {
      action: task.action,
      input: taskInput,
      context: instance.context
    });

    return result;
  }

  // 调用服务
  async callService(service, request) {
    // 检查熔断器
    if (this.isCircuitBreakerOpen(service)) {
      throw new Error(`Circuit breaker open for service: ${service.id}`);
    }

    const startTime = Date.now();

    try {
      const response = await fetch(service.endpoint, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          ...this.buildAuthHeaders(service.authentication),
          'X-Correlation-ID': request.context.correlationId
        },
        body: JSON.stringify(request),
        signal: AbortSignal.timeout(service.timeout)
      });

      if (!response.ok) {
        throw new Error(`Service call failed: ${response.status} ${response.statusText}`);
      }

      const result = await response.json();

      // 记录成功
      this.recordServiceCall(service.id, true, Date.now() - startTime);

      return result;

    } catch (error) {
      // 记录失败
      this.recordServiceCall(service.id, false, Date.now() - startTime);
      throw error;
    }
  }

  // 处理任务成功
  async handleTaskSuccess(instance, task, result) {
    // 更新变量
    if (task.outputVariable) {
      instance.variables[task.outputVariable] = result.data;
    }

    // 记录完成
    instance.completedTasks.push(task.id);
    instance.history.push({
      type: 'task_completed',
      taskId: task.id,
      result,
      timestamp: Date.now()
    });

    // 检查是否有分支条件
    const nextTaskId = this.evaluateBranchConditions(task, instance);
    if (nextTaskId) {
      const workflow = this.workflows.get(instance.workflowId);
      const branchTask = workflow.tasks.find(t => t.id === nextTaskId);
      if (branchTask) {
        await this.executeTask(branchTask, instance);
        return;
      }
    }

    // 执行下一个任务
    await this.saveInstanceState(instance);
    await this.executeNextTask(instance);
  }

  // 处理任务失败
  async handleTaskFailure(instance, task, error) {
    const workflow = this.workflows.get(instance.workflowId);
    const retryPolicy = workflow.errorHandling;

    instance.failedTasks.push({
      taskId: task.id,
      error: error.message,
      timestamp: Date.now()
    });

    instance.history.push({
      type: 'task_failed',
      taskId: task.id,
      error: error.message,
      timestamp: Date.now()
    });

    // 检查重试策略
    if (instance.context.retryCount < retryPolicy.maxRetries) {
      instance.context.retryCount++;
      instance.status = 'retrying';

      // 计算重试延迟
      const delay = this.calculateRetryDelay(
        retryPolicy.retryPolicy,
        instance.context.retryCount
      );

      // 延迟重试
      setTimeout(() => {
        this.executeTask(task, instance);
      }, delay);

    } else {
      // 执行补偿操作
      if (workflow.compensation && workflow.compensation.length > 0) {
        await this.executeCompensation(instance, task, error);
      }

      await this.completeWorkflow(instance, 'failed', error);
    }
  }

  // 执行补偿操作
  async executeCompensation(instance, failedTask, error) {
    const workflow = this.workflows.get(instance.workflowId);
    const compensationTasks = workflow.compensation
      .filter(c => c.forTaskId === failedTask.id || c.forAll === true);

    for (const compTask of compensationTasks) {
      try {
        const service = this.services.get(compTask.serviceId);
        if (service) {
          await this.callService(service, {
            action: compTask.action,
            input: {
              originalInput: instance.input,
              failedTaskId: failedTask.id,
              error: error.message,
              compensationData: compTask.data || {}
            },
            context: instance.context
          });
        }
      } catch (compError) {
        console.error('Compensation failed:', compError);
      }
    }
  }

  // 完成工作流
  async completeWorkflow(instance, status, error = null) {
    instance.status = status;
    instance.endTime = Date.now();
    instance.duration = instance.endTime - instance.startTime;

    if (error) {
      instance.error = error.message;
    }

    instance.history.push({
      type: 'workflow_completed',
      status,
      timestamp: instance.endTime
    });

    // 发送完成事件
    if (this.eventBus) {
      await this.eventBus.emit('workflow.completed', {
        instanceId: instance.id,
        workflowId: instance.workflowId,
        status,
        duration: instance.duration,
        input: instance.input,
        output: instance.variables,
        error
      });
    }

    // 保存最终状态
    await this.saveInstanceState(instance);

    // 清理实例（可选）
    if (status === 'success' || instance.context.retentionTime === 0) {
      setTimeout(() => {
        this.instances.delete(instance.id);
      }, 60000); // 1分钟后清理
    }
  }

  // 获取下一个任务
  getNextTask(workflow, instance) {
    // 如果有当前任务，找下一个
    if (instance.currentTask) {
      const currentIndex = workflow.tasks.findIndex(t => t.id === instance.currentTask);
      if (currentIndex >= 0 && currentIndex < workflow.tasks.length - 1) {
        return workflow.tasks[currentIndex + 1];
      }
    } else {
      // 返回第一个任务
      return workflow.tasks[0];
    }

    return null;
  }

  // 评估分支条件
  evaluateBranchConditions(task, instance) {
    if (!task.branches) return null;

    for (const branch of task.branches) {
      if (this.evaluateCondition(branch.condition, instance)) {
        return branch.toTaskId;
      }
    }

    return task.defaultNextTaskId;
  }
}
```

```javascript
    return job;
  }

  // 构建作业图
  async buildJobGraph(job) {
    let stream = await this.createSourceStream(job.source);

    // 应用转换
    for (const transformation of job.transformations) {
      stream = await this.applyTransformation(stream, transformation);
    }

    // 应用窗口
    if (job.windows.length > 0) {
      stream = await this.applyWindows(stream, job.windows);
    }

    // 应用聚合
    if (job.aggregations.length > 0) {
      stream = await this.applyAggregations(stream, job.aggregations);
    }

    // 添加输出
    for (const sink of job.sinks) {
      stream = await this.addSink(stream, sink);
    }

    return stream;
  }

  // 创建源流
  async createSourceStream(sourceConfig) {
    switch (sourceConfig.type) {
      case 'kafka':
        return await this.flink.fromKafka({
          topics: sourceConfig.topics,
          groupId: sourceConfig.groupId,
          bootstrapServers: sourceConfig.bootstrapServers,
          format: sourceConfig.format || 'json'
        });

      case 'socket':
        return await this.flink.fromSocket({
          host: sourceConfig.host,
          port: sourceConfig.port,
          delimiter: sourceConfig.delimiter || '\n'
        });

      case 'file':
        return await this.flink.fromFile({
          path: sourceConfig.path,
          format: sourceConfig.format || 'json'
        });

      default:
        throw new Error(`Unknown source type: ${sourceConfig.type}`);
    }
  }

  // 应用转换
  async applyTransformation(stream, transformation) {
    switch (transformation.type) {
      case 'filter':
        return stream.filter(event => this.evaluateFilterCondition(event, transformation.condition));

      case 'map':
        return stream.map(event => this.applyMapFunction(event, transformation.function));

      case 'flatMap':
        return stream.flatMap(event => this.applyFlatMapFunction(event, transformation.function));

      case 'enrich':
        return await this.enrichStream(stream, transformation);

      case 'validate':
        return stream.filter(event => this.validateEvent(event, transformation.schema));

      default:
        return stream;
    }
  }

  // 应用窗口
  async applyWindows(stream, windows) {
    let windowedStream = stream;

    for (const windowConfig of windows) {
      switch (windowConfig.type) {
        case 'tumbling':
          windowedStream = stream.windowAll(
            TumblingProcessingTimeWindows.of(Time.milliseconds(windowConfig.size))
          );
          break;

        case 'sliding':
          windowedStream = stream.windowAll(
            SlidingProcessingTimeWindows.of(
              Time.milliseconds(windowConfig.size),
              Time.milliseconds(windowConfig.slide)
            )
          );
          break;

        case 'session':
          windowedStream = stream.windowAll(
            EventTimeSessionWindows.withGap(Time.milliseconds(windowConfig.gap))
          );
          break;

        case 'global':
          windowedStream = stream.windowAll(GlobalWindows.create());
          break;
      }
    }

    return windowedStream;
  }

  // 应用聚合
  async applyAggregations(stream, aggregations) {
    let aggregatedStream = stream;

    for (const aggConfig of aggregations) {
      switch (aggConfig.type) {
        case 'count':
          aggregatedStream = stream.process((windowedStream) => {
            return windowedStream.aggregate(new Count());
          });
          break;

        case 'sum':
          aggregatedStream = stream.process((windowedStream) => {
            return windowedStream.aggregate(
              new Sum(aggConfig.field, aggConfig.alias || `sum_${aggConfig.field}`)
            );
          });
          break;

        case 'avg':
          aggregatedStream = stream.process((windowedStream) => {
            return windowedStream.aggregate(
              new Average(aggConfig.field, aggConfig.alias || `avg_${aggConfig.field}`)
            );
          });
          break;

        case 'min':
          aggregatedStream = stream.process((windowedStream) => {
            return windowedStream.aggregate(
              new Min(aggConfig.field, aggConfig.alias || `min_${aggConfig.field}`)
            );
          });
          break;

        case 'max':
          aggregatedStream = stream.process((windowedStream) => {
            return windowedStream.aggregate(
              new Max(aggConfig.field, aggConfig.alias || `max_${aggConfig.field}`)
            );
          });
          break;

        case 'distinct':
          aggregatedStream = stream.process((windowedStream) => {
            return windowedStream.aggregate(
              new Distinct(aggConfig.field, aggConfig.alias || `distinct_${aggConfig.field}`)
            );
          });
          break;

        case 'custom':
          aggregatedStream = stream.process((windowedStream) => {
            return windowedStream.aggregate(
              this.createCustomAggregator(aggConfig)
            );
          });
          break;
      }
    }

    return aggregatedStream;
  }

  // 添加输出

      case 'kafka':

          index: sinkConfig.index,
          format: sinkConfig.format || 'json'
        }));
        break;

      case 'websocket':

        break;
```

```javascript
      // 预处理
      const preprocessed = await this.preprocessFeatures(features, model);

      // 模型推理
      const rawPrediction = await this.runInference(model, preprocessed);

      // 后处理
      const prediction = await this.postprocessPrediction(rawPrediction, model);

      // 记录预测
      await this.recordPrediction(modelId, data, prediction);

      return prediction;
    } catch (error) {

      throw error;
    }
  }

  // 批量预测
  async batchPredict(modelId, dataList) {
    const predictions = [];

    // 分批处理以控制内存使用
    const batchSize = 100;
    for (let i = 0; i < dataList.length; i += batchSize) {
      const batch = dataList.slice(i, i + batchSize);
      const batchPredictions = await Promise.all(
        batch.map(data => this.predict(modelId, data))
      );
      predictions.push(...batchPredictions);
    }

    return predictions;
  }

  // 特征提取
  async extractFeatures(data, model) {
    const features = {};

    // 从模型配置获取特征列表
    const modelFeatures = await this.getModelFeatures(model);

    for (const featureConfig of modelFeatures) {
      const feature = this.features.get(featureConfig.id);
      if (!feature) continue;

      let value = this.extractFeatureValue(data, feature);

      // 应用转换
      if (feature.transformation) {
        value = this.applyTransformation(value, feature.transformation);
      }

      // 应用编码
      if (feature.encoding) {
        value = this.applyEncoding(value, feature.encoding);
      }

      features[featureConfig.id] = value;
    }

    return features;
  }

  extractFeatureValue(data, feature) {
    const extraction = feature.extraction;

    switch (extraction.type) {
      case 'field':

      case 'computed':
        return this.computeValue(data, extraction.formula);

      case 'aggregate':
        return this.aggregateValue(data, extraction.aggFunction, extraction.field);

      case 'time':
        return this.extractTimeFeature(data, extraction.timeFeature);

      case 'text':
        return this.extractTextFeature(data, extraction.method);

      default:
        return null;
    }
  }

  // 运行推理
  async runInference(model, features) {
    switch (model.framework) {
      case 'tensorflow':
        return await this.runTensorFlowInference(model, features);
      case 'pytorch':
        return await this.runPyTorchInference(model, features);
      case 'onnx':
        return await this.runONNXInference(model, features);
      case 'custom':
        return await this.runCustomInference(model, features);
      default:
        throw new Error(`Unsupported framework: ${model.framework}`);
    }
  }

  // TensorFlow 推理
  async runTensorFlowInference(model, features) {
    // 转换为 TensorFlow 张量格式
    const tensor = this.convertToTensor(features, model.inputSchema);

    // 运行模型
    const result = await this.tfEngine.execute(model.modelPath, tensor);

    // 解析结果
    return this.parseTensorFlowResult(result, model.outputSchema);
  }

  // 异常检测
  async detectAnomaly(modelId, data) {
    const prediction = await this.predict(modelId, data);

    if (prediction.anomalyScore > this.models.get(modelId).threshold) {
      return {
        isAnomaly: true,
        score: prediction.anomalyScore,
        reasons: this.explainAnomaly(data, prediction),
        confidence: prediction.confidence
      };
    }

    return {
      isAnomaly: false,
      score: prediction.anomalyScore,
      confidence: prediction.confidence
    };
  }

  // 在线学习
  async onlineLearning(modelId, newData, label) {
    const model = this.models.get(modelId);
    if (model.updateStrategy === 'none') {
      return;
    }

    // 获取当前模型性能
    const currentPerformance = await this.evaluateModelPerformance(modelId);

    // 决定是否需要更新
    if (this.shouldUpdateModel(currentPerformance, newData)) {
      await this.updateModel(modelId, newData, label);
    }
  }

  // 模型 A/B 测试
  async abTest(request, models, trafficSplit) {
    const total = Object.values(trafficSplit).reduce((a, b) => a + b, 0);
    const random = Math.random() * total;
    let accumulated = 0;
    let selectedModelId = null;

    for (const [modelId, split] of Object.entries(trafficSplit)) {
      accumulated += split;
      if (random <= accumulated) {
        selectedModelId = modelId;
        break;
      }
    }

    // 执行预测
    const prediction = await this.predict(selectedModelId, request);

    // 记录 A/B 测试数据
    await this.recordABTest({
      requestId: crypto.randomUUID(),
      timestamp: Date.now(),
      selectedModelId,
      trafficSplit,
      input: request,
      output: prediction
    });

    return {
      modelId: selectedModelId,
      prediction
    };
  }
}
```

```javascript
    // 检查延迟
    const latency = await this.getTenantLatency(tenantId);
    health.metrics.latency = latency;
    if (latency > 5000) { // 5秒
      health.status = 'degraded';
      health.issues.push(`High latency: ${latency}ms`);
      health.recommendations.push('Consider upgrading plan for better performance');
    }

    // 检查资源利用率
    const utilization = await this.getTenantResourceUtilization(tenantId);
    health.metrics.resourceUtilization = utilization;
    if (utilization > 0.8) { // 80%
      health.status = 'degraded';
      health.issues.push(`High resource utilization: ${(utilization * 100).toFixed(1)}%`);
      health.recommendations.push('Scale up resources or optimize webhook processing');
    }

    // 检查配额使用
    const quotaUsage = await this.getTenantQuotaUsage(tenantId);
    health.metrics.quotaUsage = quotaUsage.webhooks / tenant.webhookQuota.monthly;
    if (health.metrics.quotaUsage > 0.9) { // 90%
      health.issues.push(`Approaching quota limit: ${(health.metrics.quotaUsage * 100).toFixed(1)}%`);
      health.recommendations.push('Consider upgrading to higher tier plan');
    }

    return health;
  }
}

// 租户速率限制器
class TenantRateLimiter {
  constructor(quota) {
    this.quota = quota;
    this.counters = new Map();
    this.windows = {
      second: new Map(),
      minute: new Map(),
      hour: new Map(),
      day: new Map()
    };
  }

  async checkLimit(eventType) {
    const limits = this.quota[eventType] || this.quota.default;
    const now = Date.now();

    // 检查不同时间窗口
    for (const [windowName, windowConfig] of Object.entries(limits.windows)) {
      const windowStart = Math.floor(now / windowConfig.size) * windowConfig.size;
      const key = `${eventType}:${windowName}`;

      let counter = this.windows[windowName].get(key);
      if (!counter || counter.windowStart !== windowStart) {
        counter = { count: 0, windowStart };
        this.windows[windowName].set(key, counter);
      }

      if (counter.count >= windowConfig.limit) {
        return {
          allowed: false,
          reason: `${windowName} limit exceeded: ${counter.count}/${windowConfig.limit}`
        };
      }

      counter.count++;
    }

    return { allowed: true };
  }
}
```
