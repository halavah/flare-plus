# Workers 性能测试

## 测试概述

性能测试对于优化 Worker 代码、确保在免费额度内高效运行至关重要。本章介绍如何测试和优化 Worker 的 CPU 使用率、内存消耗和响应时间。

## CPU 性能测试

### 1. CPU 时间测量

```javascript
// cpu-test.js - CPU 时间基准测试
export default {
  async fetch(request) {
    const url = new URL(request.url);
    const testType = url.searchParams.get('test') || 'simple';

    const results = {
      test: testType,
      iterations: 1000,
      times: []
    };

    for (let i = 0; i < results.iterations; i++) {
      const start = performance.now();

      switch (testType) {
        case 'simple':
          await simpleCalculation();
          break;
        case 'json':
          await jsonProcessing();
          break;
        case 'crypto':
          await cryptoOperation();
          break;
        case 'string':
          await stringOperation();
          break;
        default:
          await simpleCalculation();
      }

      const end = performance.now();
      results.times.push(end - start);
    }

    const stats = calculateStats(results.times);

    return new Response(JSON.stringify({
      ...results,
      stats
    }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
};

async function simpleCalculation() {
  let sum = 0;
  for (let i = 0; i < 1000; i++) {
    sum += Math.sqrt(i);
  }
  return sum;
}

async function jsonProcessing() {
  const data = Array.from({ length: 100 }, (_, i) => ({
    id: i,
    name: `item-${i}`,
    tags: [`tag-${i % 10}`],
    value: Math.random()
  }));
  return JSON.stringify(data);
}

async function cryptoOperation() {
  const encoder = new TextEncoder();
  const data = encoder.encode('test string for hashing');
  const hashBuffer = await crypto.subtle.digest('SHA-256', data);
  return hashBuffer;
}

async function stringOperation() {
  let text = 'hello world';
  for (let i = 0; i < 100; i++) {
    text = text.toUpperCase().toLowerCase();
    text = text.replace('world', 'workers');
  }
  return text;
}

function calculateStats(times) {
  const sorted = times.sort((a, b) => a - b);
  return {
    avg: times.reduce((a, b) => a + b) / times.length,
    min: Math.min(...times),
    max: Math.max(...times),
    p50: sorted[Math.floor(sorted.length / 2)],
    p95: sorted[Math.floor(sorted.length * 0.95)],
    p99: sorted[Math.floor(sorted.length * 0.99)]
  };
}
```

### 2. CPU 优化测试

```javascript
// optimization-test.js - 优化效果对比
export default {
  async fetch(request) {
    const url = new URL(request.url);
    const optimization = url.searchParams.get('opt') || 'none';

    const results = await runOptimizationTest(optimization);

    return new Response(JSON.stringify(results), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
};

async function runOptimizationTest(type) {
  const iterations = 100;
  const times = [];

  for (let i = 0; i < iterations; i++) {
    const start = performance.now();

    switch (type) {
      case 'none':
        await naiveStringConcat();
        break;
      case 'array':
        await arrayJoin();
        break;
      case 'template':
        await templateLiteral();
        break;
      case 'cache':
        await cachedCalculation(i);
        break;
    }

    times.push(performance.now() - start);
  }

  return {
    type,
    iterations,
    avgTime: times.reduce((a, b) => a + b) / times.length,
    minTime: Math.min(...times),
    maxTime: Math.max(...times)
  };
}

// 未优化：字符串拼接
async function naiveStringConcat() {
  let result = '';
  for (let i = 0; i < 1000; i++) {
    result += 'chunk-' + i;
  }
  return result;
}

// 优化：数组 join
async function arrayJoin() {
  const chunks = [];
  for (let i = 0; i < 1000; i++) {
    chunks.push('chunk-' + i);
  }
  return chunks.join('');
}

// 优化：模板字符串
async function templateLiteral() {
  const chunks = Array.from({ length: 1000 }, (_, i) => `chunk-${i}`);
  return chunks.join('');
}

// 优化：缓存结果
const cache = new Map();
async function cachedCalculation(i) {
  if (cache.has(i)) {
    return cache.get(i);
  }
  const result = Math.sqrt(i) * Math.PI;
  cache.set(i, result);
  return result;
}
```

### 1. 内存使用监控

```javascript
// memory-test.js - 内存使用测试
export default {
  async fetch(request) {
    const url = new URL(request.url);
    const test = url.searchParams.get('test');

    switch (test) {
      case 'leak':
        return testMemoryLeak();
      case 'optimize':
        return testMemoryOptimization();
      case 'stream':
        return testStreamMemory();
      default:
        return new Response('Add ?test= parameter');
    }
  }
};

async function testMemoryLeak() {
  const results = [];
  const memoryMap = new Map();

  for (let i = 0; i < 1000; i++) {
    // 模拟内存泄漏 - 不断添加数据到 Map
    memoryMap.set(`key-${i}`, new Array(1000).fill(0));

    // 强制垃圾回收（仅在开发环境可用）
    if (global.gc) {
      global.gc();
    }

    const memory = performance.memory;
    if (memory) {
      results.push({
        iteration: i,
        used: memory.usedJSHeapSize,
        total: memory.totalJSHeapSize
      });
    }
  }

  return new Response(JSON.stringify(results), {
    headers: { 'Content-Type': 'application/json' }
  });
}

async function testMemoryOptimization() {
  const results = [];

  for (let i = 0; i < 100; i++) {
    const start = performance.now();

    // 使用循环的变量会被自动回收
    let tempData = null;

    {
      const scopeData = new Array(10000).fill(i);
      tempData = processData(scopeData);
    } // scopeData 在这里被销毁

    const end = performance.now();

    if (i % 10 === 0 && performance.memory) {
      results.push({
        iteration: i,
        time: end - start,
        memory: performance.memory.usedJSHeapSize
      });
    }
  }

  return new Response(JSON.stringify(results), {
    headers: { 'Content-Type': 'application/json' }
  });
}

function processData(data) {
  return data.map(x => x * 2).filter(x => x > 100);
}

async function testStreamMemory() {
  // 测试流式处理的内存效率
  const stream = new ReadableStream({
    start(controller) {
      for (let i = 0; i < 10000; i++) {
        controller.enqueue(new TextEncoder().encode(`chunk-${i}\n`));
      }
      controller.close();
    }
  });

  const response = new Response(stream);
  return response;
}
```

```javascript
// latency-test.js - 响应时间基准
export default {
  async fetch(request) {
    const url = new URL(request.url);
    const testType = url.searchParams.get('type') || 'simple';

    const start = Date.now();
    const startPerf = performance.now();

    let result;
    switch (testType) {
      case 'simple':
        result = await simpleResponse();
        break;
      case 'fetch':
        result = await fetchResponse();
        break;
      case 'kv':
        result = await kvResponse();
        break;
      case 'cpu':
        result = await cpuIntensive();
        break;
      case 'composite':
        result = await compositeOperations();
        break;
      default:
        result = await simpleResponse();
    }

    const endPerf = performance.now();
    const end = Date.now();

    // 记录 Cloudflare 特有信息
    const cf = request.cf;

    const metadata = {
      requestStart: start,
      requestEnd: end,
      durationMs: end - start,
      perfDuration: endPerf - startPerf,
      colo: cf.colo,
      country: cf.country,
      asn: cf.asn,
      httpProtocol: cf.httpProtocol
    };

    // 如果返回的是 Response，添加头部
    if (result instanceof Response) {
      result.headers.set('X-Duration', String(metadata.durationMs));
      result.headers.set('X-Colo', metadata.colo);
      return result;
    }

    return new Response(JSON.stringify({
      result,
      metadata
    }), {
      headers: {
        'Content-Type': 'application/json',
        'X-Duration': String(metadata.durationMs),
        'X-Colo': metadata.colo
      }
    });
  }
};

async function simpleResponse() {
  return { message: 'Hello World' };
}

async function fetchResponse() {
  const response = await fetch('https://httpbin.org/json');
  return await response.json();
}

async function kvResponse(env) {
  // 假设有 KV 绑定
  // return await env.MY_KV.get('test-key');
  return { kv: 'test-value' };
}

async function cpuIntensive() {
  let sum = 0;
  for (let i = 0; i < 100000; i++) {
    sum += Math.sqrt(i) * Math.sin(i);
  }
  return { result: sum };
}

async function compositeOperations() {
  const results = await Promise.all([
    simpleResponse(),
    cpuIntensive(),
    new Promise(r => setTimeout(() => r({ delayed: true }), 10))
  ]);
  return { composite: results };
}
```

## 并发测试

### 1. 并发请求处理

```javascript
// concurrent-test.js - 并发性能测试
export default {
  async fetch(request) {
    const url = new URL(request.url);
    const concurrent = parseInt(url.searchParams.get('concurrent')) || 10;
    const delay = parseInt(url.searchParams.get('delay')) || 0;

    // 创建多个并发任务
    const tasks = Array.from({ length: concurrent }, (_, i) =>
      simulateAsyncWork(i, delay)
    );

    const start = performance.now();
    const results = await Promise.all(tasks);
    const end = performance.now();

    return new Response(JSON.stringify({
      concurrent,
      delay,
      duration: end - start,
      results
    }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
};

async function simulateAsyncWork(id, delay) {
  const start = performance.now();

  // 模拟异步操作
  await new Promise(resolve => setTimeout(resolve, delay));

  // 模拟 CPU 工作
  let sum = 0;
  for (let i = 0; i < 1000; i++) {
    sum += Math.sqrt(id * i);
  }

  const end = performance.now();

  return {
    id,
    duration: end - start,
    result: sum
  };
}
```

```html
        let isRunning = false;
        let completed = 0;
        let errors = 0;
        let latencies = [];
        let startTime;

        async function runTest() {
            if (isRunning) return;

            isRunning = true;
            completed = 0;
            errors = 0;
            latencies = [];
            startTime = Date.now();

            const url = document.getElementById('workerUrl').value;
            const concurrent = parseInt(document.getElementById('concurrent').value);
            const iterations = parseInt(document.getElementById('iterations').value);

            const resultsDiv = document.getElementById('results');
            resultsDiv.innerHTML = '';

            // 分批发送请求
            for (let i = 0; i < iterations; i += concurrent) {
                const batch = Math.min(concurrent, iterations - i);
                const promises = [];

                for (let j = 0; j < batch; j++) {
                    promises.push(makeRequest(url, i + j));
                }

                await Promise.all(promises);

                if (!isRunning) break;
            }

            isRunning = false;
            updateStats();
        }

        async function makeRequest(url, id) {
            const start = performance.now();

            try {
                const response = await fetch(url + '?id=' + id + '&type=cpu');
                const end = performance.now();
                const latency = end - start;

                latencies.push(latency);
                completed++;

                if (!response.ok) {
                    errors++;
                }

                addResult(id, latency, response.ok);
            } catch (error) {
                errors++;
                addResult(id, -1, false, error.message);
            }

            updateStats();
        }

            const resultsDiv = document.getElementById('results');
            const div = document.createElement('div');
            div.className = `result ${success ? 'success' : 'error'}`;
            div.textContent = `Request ${id}: ${latency >= 0 ? latency + 'ms' : error}`;
            resultsDiv.appendChild(div);
        }

            const statsDiv = document.getElementById('stats');
            const duration = (Date.now() - startTime) / 1000;

                    <div>Avg Latency: ${(latencies.reduce((a, b) => a + b) / latencies.length).toFixed(2)}ms</div>
                    <div>Min Latency: ${Math.min(...latencies).toFixed(2)}ms</div>
                    <div>Max Latency: ${Math.max(...latencies).toFixed(2)}ms</div>
                    <div>Throughput: ${(completed / duration).toFixed(2)} req/s</div>
                `;
            }
        }

        function stopTest() {
            isRunning = false;
        }
    </script>
```

```javascript
// performance-analyzer.js - 性能分析器
export default {
  async fetch(request) {
    const url = new URL(request.url);
    const action = url.searchParams.get('action');

    switch (action) {
      case 'benchmark':
        return runFullBenchmark();
      case 'profile':
        return profileCode();
      case 'compare':
        return compareOptimizations();
      default:
        return new Response('Use ?action=benchmark|profile|compare');
    }
  }
};

async function runFullBenchmark() {
  const tests = [
    { name: 'String Operations', fn: stringBenchmark },
    { name: 'JSON Operations', fn: jsonBenchmark },
    { name: 'Crypto Operations', fn: cryptoBenchmark },
    { name: 'Array Operations', fn: arrayBenchmark }
  ];

  const results = {};

  for (const test of tests) {
    console.log(`Running ${test.name}...`);
    results[test.name] = await runBenchmark(test.fn, 100);
  }

  return new Response(JSON.stringify(results, null, 2), {
    headers: { 'Content-Type': 'application/json' }
  });
}

async function runBenchmark(testFn, iterations) {
  const times = [];

  for (let i = 0; i < iterations; i++) {
    const start = performance.now();
    await testFn();
    times.push(performance.now() - start);
  }

  return {
    iterations,
    mean: average(times),
    median: median(times),
    p95: percentile(times, 0.95),
    p99: percentile(times, 0.99),
    min: Math.min(...times),
    max: Math.max(...times)
  };
}

// 具体测试函数
async function stringBenchmark() {
  let str = 'hello';
  for (let i = 0; i < 1000; i++) {
    str += i.toString();
  }
  return str.length;
}

async function jsonBenchmark() {
  const data = Array.from({ length: 100 }, (_, i) => ({ id: i, value: Math.random() }));
  return JSON.stringify(data);
}

async function cryptoBenchmark() {
  const encoder = new TextEncoder();
  const data = encoder.encode('benchmark test');
  return await crypto.subtle.digest('SHA-256', data);
}

async function arrayBenchmark() {
  const arr = Array.from({ length: 1000 }, (_, i) => i);
  return arr.map(x => x * 2).filter(x => x > 100).reduce((a, b) => a + b, 0);
}

// 辅助函数
function average(arr) {
  return arr.reduce((a, b) => a + b) / arr.length;
}

function median(arr) {
  const sorted = arr.sort((a, b) => a - b);
  return sorted[Math.floor(sorted.length / 2)];
}

function percentile(arr, p) {
  const sorted = arr.sort((a, b) => a - b);
  return sorted[Math.floor(sorted.length * p)];
}

async function profileCode() {
  // 简单的代码性能分析
  const samples = [];

  for (let i = 0; i < 100; i++) {
    const ops = {
      calculations: 0,
      memory: 0,
      io: 0
    };

    // 模拟不同操作
    const start = performance.now();

    // 计算密集型
    for (let j = 0; j < 1000; j++) {
      Math.sqrt(j);
      ops.calculations++;
    }

    // 内存操作
    const arr = new Array(100).fill(0);
    ops.memory += arr.length;

    // IO 模拟
    await new Promise(r => setTimeout(r, 1));
    ops.io++;

    const end = performance.now();
    samples.push({
      duration: end - start,
      ops
    });
  }

  return new Response(JSON.stringify(samples), {
    headers: { 'Content-Type': 'application/json' }
  });
}
```

## 最佳实践

### 1. 性能优化技巧

```javascript
// 优化示例
export default {
  async fetch(request) {
    const url = new URL(request.url);

    // 1. 缓存频繁访问的数据
    const cache = caches.default;
    const cacheKey = new Request(request.url);

    let response = await cache.match(cacheKey);
    if (response) {
      return response;
    }

    // 2. 使用异步操作避免阻塞
    const backgroundFetch = fetch(request);

    // 3. 立即返回缓存或默认响应
    response = new Response('Loading...', { status: 200 });

    // 4. 在后台更新缓存
    event.waitUntil(
      backgroundFetch.then(fetchResponse => {
        if (fetchResponse.ok) {
          cache.put(cacheKey, fetchResponse.clone());
        }
      })
    );

    return response;
  }
};
```
