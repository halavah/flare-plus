# Workers API 参考

## 核心 API 概览

Cloudflare Workers 提供了丰富的 Web API 和专用 API，让开发者可以构建强大的边缘应用。

## Request 对象

### 基础属性

```javascript
export default {
  async fetch(request) {
    // URL 和路径信息
    console.log(request.url);          // 完整 URL
    console.log(request.method);       // HTTP 方法
    console.log(request.headers);      // Headers 对象
    console.log(request.body);         // ReadableStream
    console.log(request.bodyUsed);     // 是否已读取
    console.log(request.cache);        // 缓存模式
    console.log(request.credentials);  // 凭证模式
    console.log(request.destination);  // 目标
    console.log(request integrity);    // 完整性
    console.log(request.keepalive);    // Keep-alive
    console.log(request.mode);         // 请求模式
    console.log(request.redirect);     // 重定向模式
    console.log(request.referrer);     // 引用页面
    console.log(request.referrerPolicy); // 引用策略

    // Cloudflare 特有属性
    console.log(request.cf);           // Cloudflare 上下文
    return new Response('OK');
  }
};
```

## Response 对象

### 创建响应

```javascript
export default {
  async fetch(request) {
    // 简单文本响应
    const textResponse = new Response('Hello World');

    // JSON 响应
    const jsonResponse = new Response(
      JSON.stringify({ message: 'Hello' }),
      {
        status: 200,
        statusText: 'OK',
        headers: { 'Content-Type': 'application/json' }
      }
    );

    // HTML 响应
    const htmlResponse = new Response(
      '<h1>Hello HTML</h1>',
      { headers: { 'Content-Type': 'text/html' } }
    );

    // 流式响应
    const streamResponse = new Response(
      new ReadableStream({
        start(controller) {
          controller.enqueue('First chunk\n');
          setTimeout(() => {
            controller.enqueue('Second chunk\n');
            controller.close();
          }, 1000);
        }
      })
    );

    return jsonResponse;
  }
};
```

### 响应操作

```javascript
export default {
  async fetch(request) {
    const originalResponse = await fetch(request);

    // 克隆响应
    const clonedResponse = originalResponse.clone();

    // 创建新响应（基于现有）
    const modifiedResponse = new Response(
      originalResponse.body,
      {
        status: originalResponse.status,
        statusText: originalResponse.statusText,
        headers: {
          ...Object.fromEntries(originalResponse.headers),
          'X-Modified-By': 'Cloudflare Worker'
        }
      }
    );

    // 重定向响应
    const redirectResponse = Response.redirect(
      'https://example.com',
      301
    );

    // 错误响应
    const errorResponse = Response.error();

    return modifiedResponse;
  }
};
```

## Headers 对象

### 操作 Headers

```javascript
export default {
  async fetch(request) {
    // 读取请求头
    const userAgent = request.headers.get('User-Agent');
    const accept = request.headers.get('Accept');

    // 检查头部存在性
    const hasAuth = request.headers.has('Authorization');

    // 遍历所有头部
    for (const [key, value] of request.headers) {
      console.log(`${key}: ${value}`);
    }

    // 创建新的 Headers
    const newHeaders = new Headers();
    newHeaders.set('Content-Type', 'application/json');
    newHeaders.append('X-Custom-Header', 'value');
    newHeaders.append('X-Custom-Header', 'value2');

    // 创建响应
    const response = new Response('OK', {
      status: 200,
      headers: newHeaders
    });

    return response;
  }
};
```

## Fetch API

### 基础用法

```javascript
export default {
  async fetch(request) {
    // GET 请求
    const getResponse = await fetch('https://api.example.com/data');
    const getData = await getResponse.json();

    // POST 请求
    const postResponse = await fetch('https://api.example.com/submit', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer token'
      },
      body: JSON.stringify({
        key: 'value'
      })
    });

    // 转发原始请求
    const forwardResponse = await fetch(request, {
      cf: { cacheTtl: -1 } // 绕过缓存
    });

    // 处理响应
    if (forwardResponse.ok) {
      const responseText = await forwardResponse.text();
      return new Response(responseText);
    } else {
      return new Response('Error', { status: 500 });
    }
  }
};
```

### 高级 Fetch 配置

```javascript
export default {
  async fetch(request) {
    // 超时控制
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 5000);

    try {
      const response = await fetch('https://api.example.com/slow', {
        signal: controller.signal,
        // 自定义 Cloudflare 选项
        cf: {
          cacheEverything: true,
          cacheTtl: 86400,     // 缓存1天
          cacheKey: 'custom',  // 自定义缓存键
          scrapeShield: true,  // 启用 Scrape Shield
          minify: {            // 压缩
            html: true,
            css: true,
            js: true
          },
          image: {             // 图片优化
            format: 'webp',
            quality: 80
          }
        }
      });

      clearTimeout(timeout);
      return response;
    } catch (error) {
      clearTimeout(timeout);
      if (error.name === 'AbortError') {
        return new Response('Request timeout', { status: 504 });
      }
      return new Response('Fetch error', { status: 502 });
    }
  }
};
```

## URL 和 URLSearchParams

### 解析和操作 URL

```javascript
export default {
  async fetch(request) {
    const url = new URL(request.url);

    // URL 基础属性
    console.log(url.href);        // 完整 URL
    console.log(url.protocol);    // 协议
    console.log(url.hostname);    // 主机名
    console.log(url.port);        // 端口
    console.log(url.pathname);    // 路径
    console.log(url.search);      // 查询字符串
    console.log(url.hash);        // 锚点

    // URLSearchParams 操作
    const searchParams = url.searchParams;

    // 获取参数
    const id = searchParams.get('id');
    const tags = searchParams.getAll('tags');

    // 设置参数
    searchParams.set('page', '1');
    searchParams.append('filter', 'new');

    // 检查参数
    const hasFilter = searchParams.has('filter');

    // 删除参数
    searchParams.delete('old-param');

    // 构建新 URL
    const newUrl = new URL(url.origin + url.pathname);
    newUrl.search = searchParams.toString();

    return new Response(`Processed URL: ${newUrl.href}`);
  }
};
```

## Crypto API

### 加密功能

```javascript
export default {
  async fetch(request) {
    // 生成随机字符串
    const randomBytes = crypto.getRandomValues(new Uint8Array(16));
    const randomString = Array.from(randomBytes)
      .map(b => b.toString(16).padStart(2, '0'))
      .join('');

    // UUID 生成
    const uuid = crypto.randomUUID();

    // 字符串编码/解码
    const text = 'Hello World';
    const encoder = new TextEncoder();
    const decoder = new TextDecoder();
    const encoded = encoder.encode(text);
    const decoded = decoder.decode(encoded);

    // 哈希计算
    const encoder2 = new TextEncoder();
    const data = encoder2.encode('message');
    const hashBuffer = await crypto.subtle.digest('SHA-256', data);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');

    return new Response(JSON.stringify({
      randomString,
      uuid,
      encoded: Array.from(encoded),
      decoded,
      sha256: hashHex
    }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
};
```

## Stream API

### 处理流数据

```javascript
export default {
  async fetch(request) {
    const url = new URL(request.url);

    // 流式读取
    if (url.pathname === '/stream') {
      const stream = new ReadableStream({
        start(controller) {
          let count = 0;
          const interval = setInterval(() => {
            if (count < 10) {
              controller.enqueue(`data: ${count}\n\n`);
              count++;
            } else {
              controller.close();
              clearInterval(interval);
            }
          }, 1000);
        }
      });

      return new Response(stream, {
        headers: { 'Content-Type': 'text/event-stream' }
      });
    }

    // 流式转换
    if (url.pathname === '/transform') {
      const response = await fetch(request);
      const transformedStream = response.body
        .pipeThrough(new TextDecoderStream())
        .pipeThrough(new TransformStream({
          transform(chunk, controller) {
            controller.enqueue(chunk.toUpperCase());
          }
        }));

      return new Response(transformedStream);
    }

    return new Response('Use /stream or /transform');
  }
};
```

## 性能监控 API

### 性能指标

```javascript
export default {
  async fetch(request) {
    // 时间测量
    const start = performance.now();

    // 执行一些操作
    const result = await expensiveOperation();

    const end = performance.now();
    const duration = end - start;

    // 内存使用（有限）
    const memory = performance.memory || {};

    return new Response(JSON.stringify({
      duration: `${duration.toFixed(2)}ms`,
      memory: {
        used: memory.usedJSHeapSize,
        total: memory.totalJSHeapSize
      },
      result
    }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
};

async function expensiveOperation() {
  // 模拟耗时操作
  await new Promise(resolve => setTimeout(resolve, 10));
  return 'Operation complete';
}
```

## 注意事项

### API 限制
- 不支持 `eval()` 和 `new Function()`
- 不支持 Web Workers
- 部分 Node.js API 不可用
- 文件系统访问受限

### 最佳实践
- 使用异步操作避免阻塞
- 及时释放大对象
- 利用缓存减少计算
- 合理使用错误处理
