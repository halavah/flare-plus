## 与其他 Cloudflare 服务集成

### 1. Workers + KV 存储

```javascript
// src/kv-integration.js
export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const path = url.pathname;

    // API 缓存示例
    if (path === '/api/cache') {
      return handleApiCache(request, env);
    }

    // 用户会话管理
    if (path === '/api/session') {
      return handleSession(request, env);
    }

    // 配置管理
    if (path === '/api/config') {
      return handleConfig(request, env);
    }

    return new Response('Not Found', { status: 404 });
  }
};

// API 响应缓存
async function handleApiCache(request, env) {
  const cacheKey = `api:${request.url}`;
  const cached = await env.CACHE_KV.get(cacheKey);

  if (cached) {
    const response = new Response(cached, {
      headers: { 'Content-Type': 'application/json' }
    });
    response.headers.set('X-Cache', 'HIT');
    return response;
  }

  // 缓存未命中，从源获取
  const response = await fetch('https://api.example.com/data');
  const data = await response.text();

  // 存储到缓存（5分钟TTL）
  ctx.waitUntil(env.CACHE_KV.put(cacheKey, data, { expirationTtl: 300 }));

  const newResponse = new Response(data, {
    headers: { 'Content-Type': 'application/json' }
  });
  newResponse.headers.set('X-Cache', 'MISS');

  return newResponse;
}

// 用户会话管理
async function handleSession(request, env) {
  const sessionId = request.headers.get('Authorization')?.replace('Bearer ', '');

  if (!sessionId) {
    return new Response('Missing session ID', { status: 401 });
  }

  const sessionData = await env.SESSIONS_KV.get(`session:${sessionId}`, {
    type: 'json'
  });

  if (!sessionData) {
    return new Response('Invalid session', { status: 401 });
  }

  // 更新最后访问时间
  sessionData.lastAccess = new Date().toISOString();
  ctx.waitUntil(
    env.SESSIONS_KV.put(
      `session:${sessionId}`,
      JSON.stringify(sessionData),
      { expirationTtl: 3600 } // 1小时过期
    )
  );

  return new Response(JSON.stringify(sessionData), {
    headers: { 'Content-Type': 'application/json' }
  });
}

// 动态配置管理
async function handleConfig(request, env) {
  const url = new URL(request.url);
  const key = url.searchParams.get('key');

  if (request.method === 'GET') {
    const value = await env.CONFIG_KV.get(key);
    return value
      ? new Response(value)
      : new Response('Not Found', { status: 404 });
  }

  if (request.method === 'PUT') {
    const value = await request.text();
    await env.CONFIG_KV.put(key, value);
    return new Response('Configuration updated');
  }
}
```

### 2. Workers + D1 数据库

```javascript
// src/d1-integration.js
export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname;

    // CRUD 操作示例
    if (path.startsWith('/api/users')) {
      return handleUsers(request, env);
    }

    // 批量操作
    if (path === '/api/batch') {
      return handleBatch(request, env);
    }

    return new Response('Not Found', { status: 404 });
  }
};

// 用户管理 API
async function handleUsers(request, env) {
  const url = new URL(request.url);
  const userId = url.pathname.split('/')[3];

  switch (request.method) {
    case 'GET':
      if (userId) {
        return getUserById(userId, env);
      } else {
        return listUsers(url, env);
      }

    case 'POST':
      return createUser(request, env);

    case 'PUT':
      if (!userId) {
        return new Response('User ID required', { status: 400 });
      }
      return updateUser(userId, request, env);

    case 'DELETE':
      if (!userId) {
        return new Response('User ID required', { status: 400 });
      }
      return deleteUser(userId, env);

    default:
      return new Response('Method not allowed', { status: 405 });
  }
}

async function listUsers(url, env) {
  const limit = parseInt(url.searchParams.get('limit')) || 10;
  const offset = parseInt(url.searchParams.get('offset')) || 0;

  const stmt = env.DB.prepare('SELECT * FROM users ORDER BY created_at DESC LIMIT ? OFFSET ?');
  const result = await stmt.bind(limit, offset).all();

  // 获取总数
  const countStmt = env.DB.prepare('SELECT COUNT(*) as total FROM users');
  const countResult = await countStmt.first();

  return new Response(JSON.stringify({
    users: result.results,
    pagination: {
      limit,
      offset,
      total: countResult.total
    }
  }), {
    headers: { 'Content-Type': 'application/json' }
  });
}

async function createUser(request, env) {
  const user = await request.json();

  const stmt = env.DB.prepare(`
    INSERT INTO users (name, email, created_at)
    VALUES (?, ?, ?)
  `);

  try {
    const result = await stmt.bind(
      user.name,
      user.email,
      new Date().toISOString()
    ).run();

    const newUser = await env.DB.prepare(
      'SELECT * FROM users WHERE id = ?'
    ).bind(result.meta.last_row_id).first();

    return new Response(JSON.stringify(newUser), {
      status: 201,
      headers: { 'Content-Type': 'application/json' }
    });
  } catch (error) {
    return new Response(JSON.stringify({
      error: error.message
    }), {
      status: 400,
      headers: { 'Content-Type': 'application/json' }
    });
  }
}

// 批量操作示例
async function handleBatch(request, env) {
  const operations = await request.json();
  const results = [];

  // 使用事务确保数据一致性
  const batch = [
    env.DB.prepare('INSERT INTO logs (message, created_at) VALUES (?, ?)'),
    env.DB.prepare('UPDATE stats SET count = count + 1 WHERE name = ?'),
    env.DB.prepare('DELETE FROM temp_table WHERE expired_at < ?')
  ];

  // 批量执行
  for (const op of operations) {
    try {
      const result = await batch[op.type].bind(...op.params).run();
      results.push({ success: true, meta: result.meta });
    } catch (error) {
      results.push({ success: false, error: error.message });
    }
  }

  return new Response(JSON.stringify(results), {
    headers: { 'Content-Type': 'application/json' }
  });
}
```

### 3. Workers + R2 存储

```javascript
// src/r2-integration.js
export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname;

    // 文件上传
    if (path === '/api/upload') {
      return handleUpload(request, env);
    }

    // 文件下载
    if (path.startsWith('/api/download/')) {
      return handleDownload(path, env);
    }

    // 文件列表
    if (path === '/api/files') {
      return listFiles(request, env);
    }

    return new Response('Not Found', { status: 404 });
  }
};

// 文件上传（支持多部分上传）
async function handleUpload(request, env) {
  const formData = await request.formData();
  const file = formData.get('file');
  const filename = formData.get('filename') || file.name;

  try {
    // 小文件直接上传
    if (file.size < 100 * 1024 * 1024) { // 100MB
      await env.BUCKET.put(filename, file);
    } else {
      // 大文件使用多部分上传
      await multipartUpload(env.BUCKET, filename, file);
    }

    const fileUrl = `https://${env.BUCKET_DOMAIN}/${filename}`;

    return new Response(JSON.stringify({
      filename,
      url: fileUrl,
      size: file.size
    }), {
      status: 200,
      headers: { 'Content-Type': 'application/json' }
    });
  } catch (error) {
    return new Response(JSON.stringify({
      error: error.message
    }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' }
    });
  }
}

// 多部分上传
async function multipartUpload(bucket, filename, file) {
  const chunkSize = 5 * 1024 * 1024; // 5MB chunks
  const uploadId = await bucket.createMultipartUpload(filename);
  const parts = [];

  for (let offset = 0; offset < file.size; offset += chunkSize) {
    const end = Math.min(offset + chunkSize, file.size);
    const chunk = file.slice(offset, end);

    const part = await bucket.uploadPart(uploadId, offset + 1, chunk);
    parts.push({ partNumber: offset / chunkSize + 1, etag: part.etag });
  }

  await bucket.completeMultipartUpload(uploadId, parts);
}

// 文件下载（支持预签名URL）
async function handleDownload(path, env) {
  const filename = path.replace('/api/download/', '');

  try {
    const object = await env.BUCKET.head(filename);

    // 检查是否需要预签名URL
    const isPrivate = object.customMetadata?.private === 'true';

    if (isPrivate) {
      // 生成预签名URL
      const signedUrl = await env.BUCKET.getPresignedUrl({
        key: filename,
        expiresIn: 3600 // 1小时
      });

      return Response.redirect(signedUrl, 302);
    }

    // 直接返回文件
    const objectData = await env.BUCKET.get(filename);
    return new Response(objectData.body, {
      headers: {
        'Content-Type': objectData.httpMetadata.contentType || 'application/octet-stream',
        'Content-Length': objectData.size
      }
    });
  } catch (error) {
    return new Response('File not found', { status: 404 });
  }
}

// 文件列表
async function listFiles(request, env) {
  const url = new URL(request.url);
  const prefix = url.searchParams.get('prefix') || '';
  const limit = parseInt(url.searchParams.get('limit')) || 100;

  const options = {
    prefix,
    limit
  };

  try {
    const result = await env.BUCKET.list(options);
    const files = result.objects.map(obj => ({
      key: obj.key,
      size: obj.size,
      lastModified: obj.lastModified,
      etag: obj.etag
    }));

    return new Response(JSON.stringify({
      files,
      truncated: result.truncated
    }), {
      headers: { 'Content-Type': 'application/json' }
    });
  } catch (error) {
    return new Response(JSON.stringify({
      error: error.message
    }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' }
    });
  }
}
```

### 4. Workers + Durable Objects

```javascript
// src/durable-objects-integration.js

// Durable Object 类
export class ChatRoom {
  constructor(state, env) {
    this.state = state;
    this.env = env;
    this.sessions = new Set();
    this.messages = [];
  }

  async fetch(request) {
    const url = new URL(request.url);

    // WebSocket 连接处理
    if (url.pathname === '/ws') {
      return this.handleWebSocket(request);
    }

    // API 调用
    if (url.pathname === '/history') {
      return this.getHistory();
    }

    return new Response('Not Found', { status: 404 });
  }

  async handleWebSocket(request) {
    const pair = new WebSocketPair();
    const [client, server] = Object.values(pair);

    this.sessions.add(server);

    server.accept();
    server.addEventListener('message', event => {
      this.handleMessage(event.data, server);
    });

    server.addEventListener('close', () => {
      this.sessions.delete(server);
    });

    return new Response(null, {
      status: 101,
      webSocket: client
    });
  }

  async handleMessage(data, sender) {
    const message = JSON.parse(data);
    message.timestamp = new Date().toISOString();

    // 存储消息
    this.messages.push(message);
    if (this.messages.length > 100) {
      this.messages.shift(); // 保持最近100条消息
    }

    // 广播给所有连接
    const messageData = JSON.stringify(message);
    this.sessions.forEach(session => {
      if (session !== sender) {
        session.send(messageData);
      }
    });

    // 持久化到 D1
    this.state.storage.put(`message:${message.timestamp}`, message);
  }

  async getHistory() {
    return new Response(JSON.stringify({
      messages: this.messages
    }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
}

// Worker 主程序
export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname;

    // 创建或获取聊天室
    if (path.startsWith('/chat/')) {
      const roomId = path.split('/')[2];
      const id = env.CHAT_ROOM.idFromName(roomId);
      const stub = env.CHAT_ROOM.get(id);

      // 重写 URL 并转发
      const newUrl = new URL(url.pathname.replace(`/chat/${roomId}`, ''), url);
      const newRequest = new Request(newUrl, request);

      return stub.fetch(newRequest);
    }

    // 聊天室列表
    if (path === '/api/rooms') {
      return listRooms(env);
    }

    return new Response('Use /chat/{room-id}');
  }
};

async function listRooms(env) {
  // 从 D1 获取活跃聊天室
  const result = await env.DB.prepare(`
    SELECT DISTINCT room_id, COUNT(*) as active_users
    FROM active_sessions
    WHERE last_seen > datetime('now', '-5 minutes')
    GROUP BY room_id
  `).all();

  return new Response(JSON.stringify(result.results), {
    headers: { 'Content-Type': 'application/json' }
  });
}
```

### 1. API 网关模式

```javascript
// src/api-gateway.js
export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    // 认证
    const authResult = await authenticate(request, env);
    if (!authResult.success) {
      return new Response('Unauthorized', { status: 401 });
    }

    // 速率限制
    const rateLimitResult = await checkRateLimit(request, env);
    if (!rateLimitResult.allowed) {
      return new Response('Rate limit exceeded', { status: 429 });
    }

    // 路由到后端服务
    const response = await routeToService(request, env);

    // 添加响应头
    response.headers.set('X-Gateway-Version', '1.0');
    response.headers.set('X-Request-ID', request.id || crypto.randomUUID());

    return response;
  }
};

async function routeToService(request, env) {
  const url = new URL(request.url);
  const path = url.pathname;

  // 服务路由配置
  const routes = {
    '/api/users': { service: 'user-service', stripPrefix: false },
    '/api/orders': { service: 'order-service', stripPrefix: false },
    '/api/products': { service: 'product-service', stripPrefix: false },
    '/api/notifications': { service: 'notification-service', stripPrefix: false }
  };

  for (const [prefix, config] of Object.entries(routes)) {
    if (path.startsWith(prefix)) {
      const serviceUrl = env[config.service.toUpperCase().replace('-', '_')];

      let targetPath = path;
      if (config.stripPrefix) {
        targetPath = path.replace(prefix, '');
      }

      const targetUrl = new URL(targetPath + url.search, serviceUrl);

      const response = await fetch(targetUrl, {
        method: request.method,
        headers: request.headers,
        body: request.body
      });

      return response;
    }
  }

  return new Response('Service not found', { status: 404 });
}
```

### 2. 事件驱动架构

```javascript
// src/event-bus.js
export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    if (url.pathname === '/events/publish') {
      return publishEvent(request, env);
    }

    if (url.pathname === '/events/subscribe') {
      return subscribeToEvents(request, env);
    }

    return new Response('Not Found', { status: 404 });
  },

  async scheduled(event, env, ctx) {
    // 处理队列中的事件
    await processEventQueue(env);
  }
};

async function publishEvent(request, env) {
  const event = await request.json();

  // 验证事件
  if (!event.type || !event.data) {
    return new Response('Invalid event', { status: 400 });
  }

  // 添加元数据
  const enrichedEvent = {
    ...event,
    id: crypto.randomUUID(),
    timestamp: new Date().toISOString(),
    version: '1.0'
  };

  // 存储到事件流
  await env.EVENTS_KV.put(
    `event:${enrichedEvent.id}`,
    JSON.stringify(enrichedEvent)
  );

  // 添加到处理队列
  await env.QUEUE.send({ event: enrichedEvent });

  // 触发事件处理器
  await triggerHandlers(enrichedEvent, env);

  return new Response(JSON.stringify({
    eventId: enrichedEvent.id
  }), {
    status: 201,
    headers: { 'Content-Type': 'application/json' }
  });
}

async function triggerHandlers(event, env) {
  const handlers = await env.EVENT_HANDLERS_KV.list({
    prefix: `handler:${event.type}:`
  });

  for (const key of handlers.keys) {
    const handlerUrl = key.name.split(':')[2];

    // 异步调用处理器
    fetch(handlerUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(event)
    }).catch(error => {
      console.error(`Handler failed: ${handlerUrl}`, error);
    });
  }
}

async function processEventQueue(env) {
  const messages = await env.QUEUE.batch(100);

  for (const message of messages) {
    try {
      await processMessage(message, env);
      message.ack();
    } catch (error) {
      console.error('Failed to process message:', error);
      message.retry();
    }
  }
}
```

```javascript
// src/stripe-integration.js
export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname;

    if (path === '/api/create-payment-intent') {
      return createPaymentIntent(request, env);
    }

    if (path === '/api/confirm-payment') {
      return confirmPayment(request, env);
    }

    if (path === '/api/webhook/stripe') {
      return handleStripeWebhook(request, env);
    }

    return new Response('Not Found', { status: 404 });
  }
};

async function createPaymentIntent(request, env) {
  const { amount, currency, orderId } = await request.json();

  try {
    const response = await fetch('https://api.stripe.com/v1/payment_intents', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${env.STRIPE_SECRET_KEY}`,
        'Content-Type': 'application/x-www-form-urlencoded'
      },
      body: new URLSearchParams({
        amount: amount * 100, // 转换为分
        currency,
        metadata: { orderId }
      })
    });

    const paymentIntent = await response.json();

    return new Response(JSON.stringify({
      clientSecret: paymentIntent.client_secret
    }), {
      headers: { 'Content-Type': 'application/json' }
    });
  } catch (error) {
    return new Response(JSON.stringify({
      error: error.message
    }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' }
    });
  }
}

async function handleStripeWebhook(request, env) {
  const signature = request.headers.get('stripe-signature');
  const body = await request.text();

  // 验证签名
  const event = stripe.webhooks.constructEvent(
    body,
    signature,
    env.STRIPE_WEBHOOK_SECRET
  );

  // 处理事件
  switch (event.type) {
    case 'payment_intent.succeeded':
      await handlePaymentSuccess(event.data.object, env);
      break;

    case 'payment_intent.payment_failed':
      await handlePaymentFailure(event.data.object, env);
      break;
  }

  return new Response('OK');
}
```

```javascript
      return sendEmail(request, env);

async function sendEmail(request, env) {
  const { to, subject, template, data } = await request.json();

  try {
    const response = await fetch('https://api.sendgrid.com/v3/mail/send', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${env.SENDGRID_API_KEY}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        from: env.FROM_EMAIL,
        templateId: env.SENDGRID_TEMPLATES[template],
        personalizations: [{
          to: [{ email: to }],
          dynamicTemplateData: data
        }]
      })
    });

    if (response.ok) {
      // 记录邮件发送日志
      await env.EMAIL_LOGS_KV.put(
        `email:${Date.now()}`,
        JSON.stringify({
          to,
          subject,
          template,
          timestamp: new Date().toISOString()
        })
      );
    }

    return new Response('Email sent', { status: 200 });
  } catch (error) {
    return new Response('Failed to send email', { status: 500 });
  }
}
```
