# Workers 最佳实践

## 架构设计原则

### 1. 单一职责原则

每个 Worker 应该专注于单一功能，避免过于复杂的逻辑。

```javascript
// ❌ 避免将所有功能放在一个 Worker
export default {
  async fetch(request) {
    // 认证
    // 路由
    // 数据处理
    // 缓存
    // 日志
    // 返回响应
  }
};

// ✅ 分离关注点
// auth-worker.js - 专门处理认证
// router-worker.js - 专门处理路由
// api-worker.js - 专门处理 API 请求
// cache-worker.js - 专门处理缓存
```

### 2. 模块化设计

使用 ES6 模块组织代码，提高可维护性。

```javascript
// src/auth.js
export async function authenticate(request) {
  const authHeader = request.headers.get('Authorization');
  if (!authHeader) return null;

  const token = authHeader.replace('Bearer ', '');
  return await verifyToken(token);
}

// src/utils.js
export function createResponse(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: { 'Content-Type': 'application/json' }
  });
}

// src/index.js
import { authenticate } from './auth.js';
import { createResponse } from './utils.js';

export default {
  async fetch(request, env, ctx) {
    const user = await authenticate(request);
    if (!user) {
      return createResponse({ error: 'Unauthorized' }, 401);
    }

    // 处理请求...
  }
};
```

## 性能优化

### 1. 缓存策略

```javascript
// 多层缓存策略
export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const cacheKey = new Request(url.toString());

    // 1. 检查 Cloudflare CDN 缓存
    const cache = caches.default;
    let response = await cache.match(cacheKey);
    if (response) {
      // 添加缓存指示
      response.headers.set('X-Cache', 'HIT');
      return response;
    }

    // 2. 检查 KV 缓存（适用于动态内容）
    const kvKey = `cache:${url.pathname}`;
    const cached = await env.CACHE_KV.get(kvKey);
    if (cached) {
      response = new Response(cached, {
        headers: { 'Content-Type': 'application/json' }
      });
      response.headers.set('X-Cache', 'KV-HIT');

      // 更新 CDN 缓存
      ctx.waitUntil(cache.put(cacheKey, response.clone()));
      return response;
    }

    // 3. 获取新内容
    response = await fetch(request);

    // 4. 存储到 KV（设置 TTL）
    if (response.ok) {
      const body = await response.clone().text();
      ctx.waitUntil(
        env.CACHE_KV.put(kvKey, body, { expirationTtl: 60 })
      );

      // 5. 存储到 CDN 缓存
      ctx.waitUntil(cache.put(cacheKey, response.clone()));
    }

    response.headers.set('X-Cache', 'MISS');
    return response;
  }
};
```

### 2. 异步操作优化

```javascript
// 使用 Promise.all 并行处理
export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const id = url.searchParams.get('id');

    // ✅ 并行执行独立操作
    const [user, permissions, settings] = await Promise.all([
      env.USERS_KV.get(id),
      env.PERMISSIONS_KV.get(id),
      env.SETTINGS_KV.get('default')
    ]);

    return new Response(JSON.stringify({
      user: JSON.parse(user),
      permissions: JSON.parse(permissions),
      settings: JSON.parse(settings)
    }));
  }
};

// 使用 event.waitUntil 处理后台任务
export default {
  async fetch(request, env, ctx) {
    // 立即返回响应
    const response = new Response('Accepted', { status: 202 });

    // 在后台处理耗时操作
    ctx.waitUntil(
      fetch('https://api.example.com/process', {
        method: 'POST',
        body: request.body
      }).then(res => {
        if (!res.ok) {
          console.error('Background task failed:', res.statusText);
        }
      })
    );

    return response;
  }
};
```

### 3. 内存优化

```javascript
// 避免内存泄漏
export default {
  async fetch(request) {
    // ✅ 及时释放大对象
    let largeObject;
    try {
      largeObject = await fetchLargeData();
      const result = processLargeData(largeObject);
      // largeObject 自动被 GC 回收
      return result;
    } finally {
      largeObject = null;
    }
  }
};

// 使用流式处理大文件
export default {
  async fetch(request) {
    // ✅ 流式处理，避免内存占用
    const stream = new TransformStream({
      transform(chunk, controller) {
        // 逐块处理数据
        const transformed = processChunk(chunk);
        controller.enqueue(transformed);
      }
    });

    return fetch(request)
      .then(response => response.body)
      .then(body => body.pipeThrough(stream))
      .then(stream => new Response(stream));
  }
};
```

## 错误处理

### 1. 全局错误捕获

```javascript
// src/error-handler.js
export class WorkerError extends Error {
  constructor(message, code, status = 500) {
    super(message);
    this.name = 'WorkerError';
    this.code = code;
    this.status = status;
  }
}

export function handleError(error, env) {
  console.error('Worker Error:', error);

  // 记录错误到监控系统
  if (env.ERROR_WEBHOOK) {
    env.waitUntil(
      fetch(env.ERROR_WEBHOOK, {
        method: 'POST',
        body: JSON.stringify({
          message: error.message,
          stack: error.stack,
          timestamp: new Date().toISOString()
        })
      })
    );
  }

  // 返回用户友好的错误信息
  if (error instanceof WorkerError) {
    return new Response(
      JSON.stringify({
        error: error.message,
        code: error.code
      }),
      {
        status: error.status,
        headers: { 'Content-Type': 'application/json' }
      }
    );
  }

  // 默认错误响应
  return new Response(
    JSON.stringify({
      error: 'Internal Server Error',
      code: 'INTERNAL_ERROR'
    }),
    {
      status: 500,
      headers: { 'Content-Type': 'application/json' }
    }
  );
}

// src/index.js
import { handleError } from './error-handler.js';

export default {
  async fetch(request, env, ctx) {
    try {
      return await handleRequest(request, env, ctx);
    } catch (error) {
      return handleError(error, env);
    }
  }
};
```

      data: [],
      message: 'Service temporarily unavailable'
    }), {
      status: 503,
      headers: { 'Content-Type': 'application/json' }
    });
  }
};

## 安全实践

### 1. 输入验证

```javascript
// src/validator.js
export function validateInput(data, schema) {
  const errors = [];

  for (const [key, rules] of Object.entries(schema)) {
    const value = data[key];

    if (rules.required && (value === undefined || value === null)) {
      errors.push(`${key} is required`);
      continue;
    }

    if (value !== undefined) {
      if (rules.type && typeof value !== rules.type) {
        errors.push(`${key} must be of type ${rules.type}`);
      }

      if (rules.min && value.length < rules.min) {
        errors.push(`${key} must be at least ${rules.min} characters`);
      }

      if (rules.max && value.length > rules.max) {
        errors.push(`${key} must not exceed ${rules.max} characters`);
      }

      if (rules.pattern && !rules.pattern.test(value)) {
        errors.push(`${key} format is invalid`);
      }
    }
  }

  if (errors.length > 0) {
    throw new WorkerError('Validation failed', 'VALIDATION_ERROR', 400);
  }

  return true;
}

// 使用示例
export default {
  async fetch(request) {
    const data = await request.json();

    const schema = {
      username: {
        required: true,
        type: 'string',
        min: 3,
        max: 50,
        pattern: /^[a-zA-Z0-9_]+$/
      },
      email: {
        required: true,
        type: 'string',
        pattern: /^[^\s@]+@[^\s@]+\.[^\s@]+$/
      }
    };

    validateInput(data, schema);
    // 继续处理...
  }
};
```

### 2. CORS 处理

```javascript
// src/cors.js
export function handleCORS(request) {
  const corsHeaders = {
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type, Authorization',
    'Access-Control-Max-Age': '86400'
  };

  if (request.method === 'OPTIONS') {
    return new Response(null, {
      status: 204,
      headers: corsHeaders
    });
  }

  return { headers: corsHeaders };
}

// src/index.js
import { handleCORS } from './cors.js';

export default {
  async fetch(request) {
    const cors = handleCORS(request);
    if (request.method === 'OPTIONS') {
      return cors;
    }

    const response = await handleRequest(request);

    // 添加 CORS 头到响应
    Object.entries(cors.headers).forEach(([key, value]) => {
      response.headers.set(key, value);
    });

    return response;
  }
};
```

// src/logger.js
export function log(level, message, context = {}) {
  const logEntry = {
    level,
    message,
    timestamp: new Date().toISOString(),
    requestId: crypto.randomUUID(),
    ...context
  };

  console.log(JSON.stringify(logEntry));
}

export const logger = {
  info: (message, context) => log('INFO', message, context),
  warn: (message, context) => log('WARN', message, context),
  error: (message, context) => log('ERROR', message, context),
  debug: (message, context) => log('DEBUG', message, context)
};

// 使用示例
export default {
  async fetch(request, env) {
    const requestId = crypto.randomUUID();

    logger.info('Request started', {
      requestId,
      method: request.method,
      url: request.url
    });

    try {
      const response = await processRequest(request);

      logger.info('Request completed', {
        requestId,
        status: response.status
      });

      return response;
    } catch (error) {
      logger.error('Request failed', {
        requestId,
        error: error.message,
        stack: error.stack
      });

      throw error;
    }
  }
};

### 2. 性能监控

```javascript
// src/metrics.js
export class MetricsCollector {
  constructor(env) {
    this.env = env;
    this.counters = new Map();
    this.timings = [];
  }

  increment(name, value = 1, tags = {}) {
    const key = this.serializeKey(name, tags);
    const current = this.counters.get(key) || 0;
    this.counters.set(key, current + value);

    // 发送到外部监控系统
    if (this.env.METRICS_WEBHOOK) {
      this.env.waitUntil(
        fetch(this.env.METRICS_WEBHOOK, {
          method: 'POST',
          body: JSON.stringify({
            metric: name,
            value,
            tags,
            timestamp: Date.now()
          })
        })
      );
    }
  }

  timing(name, duration, tags = {}) {
    this.timings.push({ name, duration, tags });
    this.increment(`${name}_duration_ms`, duration, tags);
  }

  serializeKey(name, tags) {
    return `${name}:${JSON.stringify(tags)}`;
  }

  async flush() {
    const metrics = {
      counters: Object.fromEntries(this.counters),
      timings: this.timings
    };

    if (this.env.METRICS_WEBHOOK) {
      await fetch(this.env.METRICS_WEBHOOK, {
        method: 'POST',
        body: JSON.stringify(metrics)
      });
    }

    return metrics;
  }
}

// src/index.js
import { MetricsCollector } from './metrics.js';

export default {
  async fetch(request, env, ctx) {
    const metrics = new MetricsCollector(env);
    const start = performance.now();

    metrics.increment('requests_total', 1, {
      method: request.method,
      path: new URL(request.url).pathname
    });

    try {
      const response = await handleRequest(request);

      metrics.timing('request_duration_ms', performance.now() - start);
      metrics.increment('responses_total', 1, { status: response.status });

      return response;
    } finally {
      ctx.waitUntil(metrics.flush());
    }
  }
};
```
