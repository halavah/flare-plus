# Workers 环境变量

## 环境变量概述

Cloudflare Workers 支持两种类型的环境变量：
- **Vars**：公开的配置值，适用于非敏感信息
- **Secrets**：加密的敏感信息，如 API 密钥、密码等

## 配置方法

### 1. 使用 wrangler.toml

```toml
name = "my-worker"
main = "src/index.js"
compatibility_date = "2023-12-01"

# 普通环境变量
[vars]
API_BASE_URL = "https://api.example.com"
ENVIRONMENT = "production"
CACHE_TTL = "3600"
DEBUG = "false"

# 不同环境的配置
[env.development.vars]
API_BASE_URL = "https://dev-api.example.com"
DEBUG = "true"

[env.staging.vars]
API_BASE_URL = "https://staging-api.example.com"
DEBUG = "true"

[env.production.vars]
API_BASE_URL = "https://api.example.com"
DEBUG = "false"
```

### 2. 使用 CLI 命令

```bash
# 普通变量
wrangler secret put API_KEY
wrangler secret put DATABASE_URL

# 设置普通变量（不敏感）
wrangler var put ENVIRONMENT --value="production"
wrangler var put API_VERSION --value="v1"

# 列出所有变量
wrangler secret list
wrangler var list

# 删除变量
wrangler secret delete API_KEY
wrangler var delete ENVIRONMENT
```

## 在代码中使用环境变量

### 基础用法

```javascript
export default {
  async fetch(request, env, ctx) {
    // 访问普通变量
    const apiBaseUrl = env.API_BASE_URL;
    const environment = env.ENVIRONMENT;
    const cacheTtl = Number(env.CACHE_TTL);
    const debug = env.DEBUG === 'true';

    // 访问密钥（secrets）
    const apiKey = env.API_KEY;
    const dbUrl = env.DATABASE_URL;

    // 使用环境变量
    if (debug) {
      console.log(`Running in ${environment} mode`);
    }

    const response = await fetch(`${apiBaseUrl}/data`, {
      headers: {
        'Authorization': `Bearer ${apiKey}`,
        'X-Environment': environment
      }
    });

    return response;
  }
};
```

### 类型安全的环境变量

```javascript
// src/env.js
export class Env {
  constructor(env) {
    this.apiBaseUrl = env.API_BASE_URL;
    this.environment = env.ENVIRONMENT || 'development';
    this.debug = (env.DEBUG || 'false') === 'true';
    this.cacheTtl = Number(env.CACHE_TTL) || 3600;
    this.maxRequests = Number(env.MAX_REQUESTS) || 1000;

    // 密钥
    this.apiKey = env.API_KEY;
    this.databaseUrl = env.DATABASE_URL;
    this.encryptionKey = env.ENCRYPTION_KEY;

    // 验证必需的变量
    this.validate();
  }

  validate() {
    const required = ['API_BASE_URL', 'API_KEY'];
    const missing = required.filter(key => !this[key]);

    if (missing.length > 0) {
      throw new Error(`Missing required environment variables: ${missing.join(', ')}`);
    }
  }

  isProduction() {
    return this.environment === 'production';
  }

  isDevelopment() {
    return this.environment === 'development';
  }
}

// src/index.js
import { Env } from './env.js';

export default {
  async fetch(request, env, ctx) {
    const envConfig = new Env(env);

    // 使用类型安全的配置
    if (envConfig.debug) {
      console.log('Debug mode enabled');
    }

    const response = await fetch(`${envConfig.apiBaseUrl}/data`, {
      headers: {
        'Authorization': `Bearer ${envConfig.apiKey}`
      }
    });

    return response;
  }
};
```

```javascript
export class Config {

    // API 配置
    this.api = {

      key: env.API_KEY,

    // 数据库配置
    this.database = {
      url: env.DATABASE_URL,
      maxConnections: Number(env.DB_MAX_CONNECTIONS) || 10,
      connectionTimeout: Number(env.DB_TIMEOUT) || 5000

    this.cache = {
      defaultTtl: Number(env.CACHE_TTL) || 3600,
      maxAge: Number(env.CACHE_MAX_AGE) || 86400

    // 功能开关
    this.features = {
      analytics: (env.FEATURE_ANALYTICS || 'false') === 'true',
      logging: (env.FEATURE_LOGGING || 'true') === 'true',
      metrics: (env.FEATURE_METRICS || 'false') === 'true'
    };

    // 外部服务
    this.services = {
      webhookUrl: env.WEBHOOK_URL,
      emailService: env.EMAIL_SERVICE_URL,
      smsService: env.SMS_SERVICE_URL
    };
  }

  getFeature(featureName) {
    return this.features[featureName] || false;
  }
}

import { Config } from './config.js';

export default {
  async fetch(request, env, ctx) {
    const config = new Config(env);

    // 根据功能开关执行不同逻辑
    if (config.getFeature('analytics')) {
      // 执行分析逻辑
      trackRequest(request);
    }

  }
};
```

### 2. 动态配置加载

```javascript
// src/dynamic-config.js
export class DynamicConfig {
  constructor(env) {
    this.env = env;
    this.cache = new Map();
  }

  async get(key, fallback = null) {
    // 检查缓存
    if (this.cache.has(key)) {
      return this.cache.get(key);
    }

    // 尝试从环境变量获取
    let value = this.env[key];

    // 如果环境变量不存在，尝试从 KV 获取动态配置
    if (!value && this.env.CONFIG_KV) {
      value = await this.env.CONFIG_KV.get(`config:${key}`);
    }

    // 设置缓存（5分钟过期）
    if (value) {
      this.cache.set(key, value);
      setTimeout(() => this.cache.delete(key), 5 * 60 * 1000);
    }

    return value || fallback;
  }

  async set(key, value, ttl = 3600) {
    // 设置到 KV
    if (this.env.CONFIG_KV) {
      await this.env.CONFIG_KV.put(`config:${key}`, value, {
        expirationTtl: ttl
      });
    }

    // 更新缓存
    this.cache.set(key, value);
  }

  async reload() {
    this.cache.clear();
  }
}

// src/index.js
import { DynamicConfig } from './dynamic-config.js';

export default {
  async fetch(request, env, ctx) {
    const config = new DynamicConfig(env);

    // 动态获取配置
    const featureFlag = await config.get('FEATURE_NEW_API', 'false');
    const rateLimit = Number(await config.get('RATE_LIMIT', '100'));

    if (featureFlag === 'true') {
      return handleNewApi(request);
    }

    return handleLegacyApi(request);
  }
};
```

```javascript
export class SecretManager {
  constructor(env) {
    this.env = env;
    this.rotationPeriod = 30 * 24 * 60 * 60 * 1000; // 30天
  }

  getCurrentSecret(key) {
    // 支持密钥版本控制
    const current = this.env[`${key}_CURRENT`];
    const next = this.env[`${key}_NEXT`];
    const rotationTime = Number(this.env[`${key}_ROTATION_TIME`] || '0');

    if (next && Date.now() > rotationTime) {
      return { secret: next, version: 'next', needsUpdate: true };
    }

    return { secret: current, version: 'current', needsUpdate: false };
  }

  async rotateIfNeeded(key) {
    const { needsUpdate } = this.getCurrentSecret(key);

    if (needsUpdate) {
      console.log(`Rotating secret: ${key}`);
      // 触发密钥更新逻辑
      await this.triggerRotation(key);
    }
  }

  async triggerRotation(key) {
    // 实现密钥更新逻辑
    // 可能包括更新外部服务、通知管理员等
  }
}
```

### 3. 密钥验证

```javascript
// src/secret-validator.js
export class SecretValidator {
  constructor(env) {
    this.env = env;
  }

  validateRequired(secrets) {
    const missing = [];

    for (const secret of secrets) {
      if (!this.env[secret]) {
        missing.push(secret);
      }
    }

    if (missing.length > 0) {
      throw new Error(`Missing required secrets: ${missing.join(', ')}`);
    }
  }

  validateFormat(secret, pattern, name) {
    const value = this.env[secret];

    if (value && !pattern.test(value)) {
      throw new Error(`Invalid format for ${name}: ${secret}`);
    }
  }

  validateAll() {
    this.validateRequired(['API_KEY', 'DATABASE_URL']);
    this.validateFormat('API_KEY', /^sk_[a-zA-Z0-9]+/, 'API Key');
    this.validateFormat('DATABASE_URL', /^postgresql:\/\//, 'Database URL');
  }
}
```

## 配置热重载

### 1. 通过 KV 实现配置热重载

```javascript
// src/hot-reload.js
export class HotReloadConfig {
  constructor(env) {
    this.env = env;
    this.version = 0;
    this.config = new Map();
    this.subscribers = [];

    // 初始加载配置
    this.loadConfig();

    // 监听配置更新
    this.watchConfig();
  }

  async loadConfig() {
    const version = await this.env.CONFIG_KV.get('config_version') || '0';
    this.version = Number(version);

    // 加载所有配置项
    const keys = await this.env.CONFIG_KV.list({ prefix: 'config:' });
    for (const key of keys.keys) {
      const value = await this.env.CONFIG_KV.get(key.name);
      const configKey = key.name.replace('config:', '');
      this.config.set(configKey, value);
    }
  }

  async watchConfig() {
    // 定期检查配置更新
    setInterval(async () => {
      const newVersion = await this.env.CONFIG_KV.get('config_version') || '0';

      if (Number(newVersion) > this.version) {
        await this.loadConfig();
        this.notifySubscribers();
      }
    }, 5000); // 每5秒检查一次
  }

  get(key, fallback = null) {
    return this.config.get(key) || fallback;
  }

  subscribe(callback) {
    this.subscribers.push(callback);
  }

  notifySubscribers() {
    this.subscribers.forEach(callback => {
      try {
        callback(this.config);
      } catch (error) {
        console.error('Config update callback error:', error);
      }
    });
  }
}
```

## 安全注意事项

### 1. 避免在日志中暴露敏感信息

```javascript
// src/secure-logging.js
export class SecureLogger {
  constructor(env) {
    this.env = env;
    this.sensitiveKeys = ['API_KEY', 'DATABASE_URL', 'SECRET'];
  }

  sanitize(data) {
    const sanitized = { ...data };

    for (const key of this.sensitiveKeys) {
      if (sanitized[key]) {
        sanitized[key] = '***REDACTED***';
      }
    }

    return sanitized;
  }

  log(level, message, context = {}) {
    console.log(JSON.stringify({
      level,
      message,
      context: this.sanitize(context),
      timestamp: new Date().toISOString()
    }));
  }
}
```

### 2. 加密敏感配置

```javascript
// src/config-encryption.js
export class ConfigEncryption {
  constructor(secretKey) {
    this.secretKey = secretKey;
  }

  async encrypt(text) {
    const encoder = new TextEncoder();
    const data = encoder.encode(text);

    const key = await crypto.subtle.importKey(
      'raw',
      encoder.encode(this.secretKey),
      { name: 'AES-GCM' },
      false,
      ['encrypt']
    );

    const iv = crypto.getRandomValues(new Uint8Array(12));
    const encrypted = await crypto.subtle.encrypt(
      { name: 'AES-GCM', iv },
      key,
      data
    );

    // 组合 IV 和加密数据
    const combined = new Uint8Array(iv.length + encrypted.byteLength);
    combined.set(iv);
    combined.set(new Uint8Array(encrypted), iv.length);

    return btoa(String.fromCharCode(...combined));
  }

  async decrypt(encryptedText) {
    const combined = new Uint8Array(
      atob(encryptedText).split('').map(c => c.charCodeAt(0))
    );

    const iv = combined.slice(0, 12);
    const encrypted = combined.slice(12);

    const encoder = new TextEncoder();
    const key = await crypto.subtle.importKey(
      'raw',
      encoder.encode(this.secretKey),
      { name: 'AES-GCM' },
      false,
      ['decrypt']
    );

    const decrypted = await crypto.subtle.decrypt(
      { name: 'AES-GCM', iv },
      key,
      encrypted
    );

    return new TextDecoder().decode(decrypted);
  }
}
```
