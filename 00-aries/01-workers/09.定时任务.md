# Workers 定时任务

## 定时任务概述

Cloudflare Workers 支持通过 Cron Triggers 实现定时任务，可以在指定时间或按计划自动执行 Worker 代码。这对于后台数据处理、定期报告、缓存预热等场景非常有用。

## 基础配置

### 1. wrangler.toml 配置

```toml
name = "scheduled-worker"
main = "src/index.js"
compatibility_date = "2023-12-01"

# 添加 scheduled handler
[triggers]
crons = ["* * * * *"]  # 每分钟执行一次

# 多个定时任务
# crons = [
#   "0 * * * *",      # 每小时执行
#   "0 0 * * *",      # 每天午夜执行
#   "0 0 * * 0"       # 每周日午夜执行
# ]
```

### 2. Cron 表达式格式

Cron 表达式由 5 个字段组成，表示：分钟、小时、日期、月份、星期

```
*  *  *  *  *
│  │  │  │  │
│  │  │  │  └─── 星期几 (0-7, 0和7都表示周日)
│  │  │  └────── 月份 (1-12)
│  │  └───────── 日期 (1-31)
│  └──────────── 小时 (0-23)
└─────────────── 分钟 (0-59)
```

### 3. 常用 Cron 表达式

| 表达式 | 说明 |
|--------|------|
| `* * * * *` | 每分钟 |
| `0 * * * *` | 每小时（在第0分钟） |
| `0 0 * * *` | 每天午夜 |
| `0 12 * * *` | 每天中午12点 |
| `0 0 * * 1` | 每周一午夜 |
| `0 0 1 * *` | 每月1号午夜 |
| `0 0 1 * *` | 每月1号午夜 |
| `*/5 * * * *` | 每5分钟 |
| `*/10 * * * *` | 每10分钟 |
| `0 */2 * * *` | 每2小时 |
| `0 9,17 * * 1-5` | 工作日的上午9点和下午5点 |
| `0 0 * * MON-FRI` | 工作日午夜（使用名称） |
| `0 0 L * *` | 每月最后一天 |

## 实现定时任务

### 1. 基础定时任务

```javascript
// src/index.js
export default {
  // 处理定时任务
  async scheduled(event, env, ctx) {
    console.log('Scheduled event triggered at:', event.scheduledTime);
    console.log('Cron:', event.cron);

    try {
      // 执行定时任务
      await performScheduledTask(env);
    } catch (error) {
      console.error('Scheduled task failed:', error);
      throw error;
    }
  },

  // 处理普通请求
  async fetch(request, env, ctx) {
    return new Response('Worker is running');
  }
};

async function performScheduledTask(env) {
  // 清理过期数据
  await cleanupExpiredData(env);

  // 更新缓存
  await updateCache(env);

  // 发送报告
  await sendDailyReport(env);

  console.log('Scheduled task completed successfully');
}

async function cleanupExpiredData(env) {
  // 清理 KV 中的过期数据
  const list = await env.DATA_KV.list();
  const now = Date.now();

  for (const key of list.keys) {
    const item = await env.DATA_KV.get(key.name, { type: 'json' });

    if (item && item.expiresAt && item.expiresAt < now) {
      await env.DATA_KV.delete(key.name);
      console.log(`Deleted expired key: ${key.name}`);
    }
  }
}

async function updateCache(env) {
  // 预热缓存
  const popularItems = ['/api/popular', '/api/trending', '/api/latest'];

  for (const path of popularItems) {
    try {
      const response = await fetch(`https://api.example.com${path}`);
      if (response.ok) {
        const data = await response.json();
        await env.CACHE_KV.put(path, JSON.stringify(data), {
          expirationTtl: 3600 // 1小时
        });
      }
    } catch (error) {
      console.error(`Failed to update cache for ${path}:`, error);
    }
  }
}

async function sendDailyReport(env) {
  // 获取统计数据
  const stats = await getDailyStats(env);

  // 发送报告到 Slack 或邮件
  if (env.REPORT_WEBHOOK) {
    await fetch(env.REPORT_WEBHOOK, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        text: `Daily Report: ${JSON.stringify(stats)}`,
        timestamp: new Date().toISOString()
      })
    });
  }
}
```

### 2. 高级定时任务模式

```javascript
// src/scheduler.js
export class Scheduler {
  constructor(env) {
    this.env = env;
    this.tasks = new Map();
    this.loadTasks();
  }

  loadTasks() {
    // 注册所有定时任务
    this.tasks.set('cleanup', {
      handler: this.cleanupTask,
      schedule: '0 0 * * *', // 每天午夜
      lastRun: null
    });

    this.tasks.set('report', {
      handler: this.reportTask,
      schedule: '0 9 * * *', // 每天上午9点
      lastRun: null
    });

    this.tasks.set('backup', {
      handler: this.backupTask,
      schedule: '0 2 * * 0', // 每周日凌晨2点
      lastRun: null
    });
  }

  async execute(cron) {
    const tasksToRun = Array.from(this.tasks.entries())
      .filter(([name, task]) => task.schedule === cron);

    const results = [];

    for (const [name, task] of tasksToRun) {
      try {
        console.log(`Executing task: ${name}`);
        const start = Date.now();

        await task.handler.call(this);

        const duration = Date.now() - start;
        task.lastRun = new Date().toISOString();

        results.push({
          name,
          status: 'success',
          duration,
          timestamp: task.lastRun
        });

        console.log(`Task ${name} completed in ${duration}ms`);
      } catch (error) {
        console.error(`Task ${name} failed:`, error);
        results.push({
          name,
          status: 'error',
          error: error.message,
          timestamp: new Date().toISOString()
        });
      }
    }

    // 保存执行结果
    await this.saveResults(results);
    return results;
  }

  async cleanupTask() {
    // 清理超过30天的日志
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - 30);

    const logs = await this.env.LOGS_KV.list({ prefix: 'log:' });
    let deleted = 0;

    for (const key of logs.keys) {
      const log = await this.env.LOGS_KV.get(key.name, { type: 'json' });

      if (log && new Date(log.timestamp) < cutoffDate) {
        await this.env.LOGS_KV.delete(key.name);
        deleted++;
      }
    }

    console.log(`Cleaned up ${deleted} old log entries`);
  }

  async reportTask() {
    // 生成日报
    const stats = await this.generateStats();

    await this.env.REPORTS_KV.put(`report:${new Date().toISOString()}`, JSON.stringify(stats));

    // 发送通知
    if (this.env.NOTIFICATION_WEBHOOK) {
      await fetch(this.env.NOTIFICATION_WEBHOOK, {
        method: 'POST',
        body: JSON.stringify({
          text: `Daily Report Generated`,
          stats
        })
      });
    }
  }

  async backupTask() {
    // 备份重要数据到外部存储
    const data = await this.exportData();

    await fetch(this.env.BACKUP_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        date: new Date().toISOString(),
        data
      })
    });
  }

  async saveResults(results) {
    const key = `schedule_results:${new Date().toISOString()}`;
    await this.env.METRICS_KV.put(key, JSON.stringify(results), {
      expirationTtl: 7 * 24 * 60 * 60 // 保存7天
    });
  }
}

// src/index.js
import { Scheduler } from './scheduler.js';

export default {
  async scheduled(event, env, ctx) {
    const scheduler = new Scheduler(env);
    const results = await scheduler.execute(event.cron);

    console.log('Scheduled tasks results:', results);
  }
};
```

## 多任务管理

### 1. 任务队列系统

```javascript
// src/task-queue.js
export class TaskQueue {
  constructor(env) {
    this.env = env;
    this.queueName = 'scheduled_tasks';
  }

  async addTask(task, options = {}) {
    const taskData = {
      id: crypto.randomUUID(),
      type: task,
      scheduledAt: options.scheduledAt || Date.now(),
      retryCount: 0,
      maxRetries: options.maxRetries || 3,
      data: options.data || {}
    };

    await this.env.QUEUE_KV.put(
      `task:${taskData.id}`,
      JSON.stringify(taskData),
      { expirationTtl: 24 * 60 * 60 } // 24小时过期
    );

    // 添加到队列索引
    const queue = await this.getQueue();
    queue.push(taskData.id);
    await this.saveQueue(queue);

    console.log(`Task added to queue: ${taskData.id}`);
    return taskData.id;
  }

  async getQueue() {
    const queueData = await this.env.QUEUE_KV.get(this.queueName);
    return queueData ? JSON.parse(queueData) : [];
  }

  async saveQueue(queue) {
    await this.env.QUEUE_KV.put(this.queueName, JSON.stringify(queue));
  }

  async processTasks() {
    const queue = await this.getQueue();
    const now = Date.now();
    const processed = [];

    for (const taskId of queue) {
      const taskData = await this.env.QUEUE_KV.get(`task:${taskId}`, {
        type: 'json'
      });

      if (!taskData) {
        // 任务已过期或不存在
        processed.push(taskId);
        continue;
      }

      if (taskData.scheduledAt > now) {
        // 还未到执行时间
        continue;
      }

      try {
        await this.executeTask(taskData);
        processed.push(taskId);
      } catch (error) {
        taskData.retryCount++;

        if (taskData.retryCount < taskData.maxRetries) {
          // 延迟重试
          taskData.scheduledAt = now + (60 * 60 * 1000); // 1小时后重试
          await this.env.QUEUE_KV.put(
            `task:${taskId}`,
            JSON.stringify(taskData)
          );
          console.log(`Task ${taskId} scheduled for retry`);
        } else {
          // 超过最大重试次数
          console.error(`Task ${taskId} failed after ${taskData.maxRetries} retries`);
          processed.push(taskId);
        }
      }
    }

    // 从队列中移除已处理的任务
    const remainingQueue = queue.filter(id => !processed.includes(id));
    await this.saveQueue(remainingQueue);
  }

  async executeTask(taskData) {
    console.log(`Executing task: ${taskData.type}`, taskData.data);

    switch (taskData.type) {
      case 'send_email':
        await this.sendEmail(taskData.data);
        break;
      case 'process_data':
        await this.processData(taskData.data);
        break;
      case 'generate_report':
        await this.generateReport(taskData.data);
        break;
      default:
        throw new Error(`Unknown task type: ${taskData.type}`);
    }
  }

  async sendEmail(data) {
    // 实现邮件发送逻辑
    console.log('Sending email:', data);
  }

  async processData(data) {
    // 实现数据处理逻辑
    console.log('Processing data:', data);
  }

  async generateReport(data) {
    // 实现报告生成逻辑
    console.log('Generating report:', data);
  }
}

// src/index.js
import { TaskQueue } from './task-queue.js';

export default {
  async scheduled(event, env, ctx) {
    const queue = new TaskQueue(env);

    // 处理队列中的任务
    await queue.processTasks();
  }
};
```

## 监控和错误处理

### 1. 任务监控系统

```javascript
// src/monitor.js
export class TaskMonitor {
  constructor(env) {
    this.env = env;
  }

  async recordTaskExecution(taskName, duration, status, error = null) {
    const record = {
      taskName,
      duration,
      status,
      error: error ? error.message : null,
      timestamp: new Date().toISOString(),
      date: new Date().toISOString().split('T')[0]
    };

    // 保存到 KV
    const key = `task_metric:${Date.now()}`;
    await this.env.METRICS_KV.put(key, JSON.stringify(record), {
      expirationTtl: 30 * 24 * 60 * 60 // 30天
    });

    // 更新每日统计
    await this.updateDailyStats(record);
  }

  async updateDailyStats(record) {
    const statsKey = `daily_stats:${record.date}`;
    const existing = await this.env.METRICS_KV.get(statsKey, {
      type: 'json'
    });

    const stats = existing || {
      date: record.date,
      tasks: {},
      totalTasks: 0,
      totalErrors: 0,
      totalDuration: 0
    };

    // 更新统计
    if (!stats.tasks[record.taskName]) {
      stats.tasks[record.taskName] = {
        count: 0,
        errors: 0,
        totalDuration: 0,
        avgDuration: 0
      };
    }

    const taskStats = stats.tasks[record.taskName];
    taskStats.count++;
    taskStats.totalDuration += record.duration;
    taskStats.avgDuration = taskStats.totalDuration / taskStats.count;

    if (record.status === 'error') {
      taskStats.errors++;
      stats.totalErrors++;
    }

    stats.totalTasks++;
    stats.totalDuration += record.duration;

    await this.env.METRICS_KV.put(statsKey, JSON.stringify(stats), {
      expirationTtl: 90 * 24 * 60 * 60 // 90天
    });
  }

  async getRecentStats(days = 7) {
    const stats = [];
    const cutoff = new Date();
    cutoff.setDate(cutoff.getDate() - days);

    const list = await this.env.METRICS_KV.list({
      prefix: 'daily_stats:'
    });

    for (const key of list.keys) {
      const date = key.name.split(':')[1];
      if (new Date(date) >= cutoff) {
        const stat = await this.env.METRICS_KV.get(key.name, {
          type: 'json'
        });
        stats.push(stat);
      }
    }

    return stats.sort((a, b) => new Date(b.date) - new Date(a.date));
  }
}

// src/index.js
import { TaskMonitor } from './monitor.js';

export default {
  async scheduled(event, env, ctx) {
    const monitor = new TaskMonitor(env);
    const start = Date.now();

    try {
      // 执行任务
      await runScheduledTasks(env);

      const duration = Date.now() - start;
      await monitor.recordTaskExecution('main_scheduler', duration, 'success');
    } catch (error) {
      const duration = Date.now() - start;
      await monitor.recordTaskExecution('main_scheduler', duration, 'error', error);
      throw error;
    }
  }
};
```

## 最佳实践

### 1. 任务设计原则

```javascript
// ✅ 好的实践
export default {
  async scheduled(event, env, ctx) {
    // 1. 快速失败
    const start = Date.now();

    // 2. 幂等性 - 相同输入产生相同结果
    const taskRunId = `run_${event.scheduledTime}`;

    // 3. 检查是否已执行
    const alreadyRun = await env.METRICS_KV.get(taskRunId);
    if (alreadyRun) {
      console.log('Task already executed for:', event.scheduledTime);
      return;
    }

    // 4. 分批处理大量数据
    await processBatch(env);

    // 5. 标记完成
    await env.METRICS_KV.put(taskRunId, 'completed', {
      expirationTtl: 24 * 60 * 60 // 24小时
    });

    // 6. 记录指标
    console.log(`Task completed in ${Date.now() - start}ms`);
  }
};

async function processBatch(env) {
  const batchSize = 100;
  let offset = 0;
  let hasMore = true;

  while (hasMore) {
    const items = await getBatchItems(env, offset, batchSize);

    if (items.length < batchSize) {
      hasMore = false;
    }

    await processItems(items);
    offset += batchSize;

    // 每10批次让出控制权
    if (offset % (batchSize * 10) === 0) {
      await new Promise(resolve => setTimeout(resolve, 0));
    }
  }
}
```

### 2. 错误处理和重试

```javascript
// src/resilient-scheduler.js
export class ResilientScheduler {
  constructor(env) {
    this.env = env;
  }

  async executeWithRetry(task, maxRetries = 3) {
    let lastError;

    for (let attempt = 0; attempt < maxRetries; attempt++) {
      try {
        console.log(`Attempt ${attempt + 1} for task: ${task.name}`);
        const result = await task.execute();
        console.log(`Task ${task.name} completed successfully`);
        return result;
      } catch (error) {
        lastError = error;
        console.error(`Task ${task.name} failed on attempt ${attempt + 1}:`, error);

        if (attempt < maxRetries - 1) {
          // 指数退避
          const delay = Math.pow(2, attempt) * 1000;
          await new Promise(resolve => setTimeout(resolve, delay));
        }
      }
    }

    // 最终失败，记录错误
    await this.logFailure(task, lastError);
    throw lastError;
  }

  async logFailure(task, error) {
    const failure = {
      taskName: task.name,
      error: error.message,
      stack: error.stack,
      timestamp: new Date().toISOString()
    };

    await this.env.FAILURES_KV.put(
      `failure:${Date.now()}`,
      JSON.stringify(failure)
    );

    // 发送警报
    if (this.env.ALERT_WEBHOOK) {
      await fetch(this.env.ALERT_WEBHOOK, {
        method: 'POST',
        body: JSON.stringify({
          text: `Scheduled task failed: ${task.name}`,
          error: error.message
        })
      });
    }
  }
}
```
