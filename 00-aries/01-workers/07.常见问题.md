## 性能问题

### 1. CPU 时间超限

**问题**: Error 1102: Worker exceeded CPU time limit

**原因**:
- 循环过多或深度嵌套
- 同步阻塞操作
- 大量数据处理

**解决方案**:

```javascript
// ❌ 同步处理大量数据
function processSync(data) {
  return data.map(x => expensiveOperation(x));
}

// ✅ 分批异步处理
async function processAsync(data) {
  const batchSize = 100;
  const results = [];

  for (let i = 0; i < data.length; i += batchSize) {
    const batch = data.slice(i, i + batchSize);
    const batchResults = await Promise.all(
      batch.map(item => expensiveOperation(item))
    );
    results.push(...batchResults);

    // 每10批次让出控制权
    if (i % (batchSize * 10) === 0) {
      await new Promise(resolve => setTimeout(resolve, 0));
    }
  }

  return results;
}

// 使用 Web Workers 处理密集计算（如果支持）
// 使用 event.waitUntil 处理后台任务
export default {
  async fetch(request, env, ctx) {
    // 立即返回
    const response = new Response('Processing started', { status: 202 });

    // 后台处理
    ctx.waitUntil(
      heavyComputation()
    );

    return response;
  }
};
```

### 2. 内存不足

**问题**: Out of memory 错误

**解决方案**:

```javascript
// 使用流处理大文件
export default {
  async fetch(request) {
    return fetch('https://example.com/large-file.json')
      .then(response => response.body)
      .then(body => {
        const transform = new TransformStream({
          transform(chunk, controller) {
            // 逐块处理，避免内存占用
            controller.enqueue(chunk);
          }
        });
        return new Response(body.pipeThrough(transform));
      });
  }
};

// 避免内存泄漏
export default {
  async fetch(request, env) {
    const data = await fetchData();

    try {
      // 处理数据
      const result = processData(data);
      return new Response(JSON.stringify(result));
    } finally {
      // 清理数据
      data = null;
    }
  }
};
```

## 兼容性问题

### 1. 浏览器 API 不可用

```javascript
// ❌ Workers 不支持
function unsupportedCode() {
  window.alert('Hello');      // No window object
  document.getElementById();   // No DOM
  localStorage.setItem();     // No localStorage
}

// ✅ Workers 支持
function supportedCode() {
  console.log('Hello');       // Yes
  fetch('/api/data');         // Yes
  crypto.getRandomValues();   // Yes
  new URL('/path', 'https://example.com'); // Yes
}
```

### 2. Node.js 模块限制

```javascript
// ❌ 原生 Node.js 模块
import fs from 'fs';              // No file system
import path from 'path';          // No path module
import http from 'http';          // Node.js HTTP not available

// ✅ 替代方案
// 使用 URL API 替代 path
const url = new URL('/file.txt', 'https://example.com');

// 使用 fetch 替代 http
const response = await fetch('https://api.example.com');

// 使用 Web Crypto API 替代 crypto Node.js 模块
const array = new Uint32Array(10);
crypto.getRandomValues(array);
```

## 请求处理问题

### 1. CORS 错误

```javascript
// 完整的 CORS 解决方案
export default {
  async fetch(request) {
    // 预检请求处理
    if (request.method === 'OPTIONS') {
      return handleOptions();
    }

    // 处理实际请求
    const response = await handleRequest(request);
    return addCorsHeaders(response);
  }
};

function handleOptions() {
  return new Response(null, {
    status: 204,
    headers: {
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
      'Access-Control-Allow-Headers': 'Content-Type, Authorization',
      'Access-Control-Max-Age': '86400'
    }
  });
}

function addCorsHeaders(response) {
  response.headers.set('Access-Control-Allow-Origin', '*');
  return response;
}
```

### 2. 请求体读取错误

```javascript
export default {
  async fetch(request) {
    // ❌ 多次读取请求体
    // const body1 = await request.text();
    // const body2 = await request.json(); // 错误：流已关闭

    // ✅ 正确的请求体处理
    const bodyText = await request.text();
    let jsonData;

    try {
      jsonData = JSON.parse(bodyText);
    } catch (error) {
      return new Response('Invalid JSON', { status: 400 });
    }

    // ✅ 或者使用 clone()
    const clonedRequest = request.clone();
    const body = await clonedRequest.json();
    const text = await request.text(); // 原始请求仍可使用

    return new Response('OK');
  }
};
```

### 3. 响应超时

```javascript
export default {
  async fetch(request) {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), 5000);

    try {
      const response = await fetch('https://api.example.com/data', {
        signal: controller.signal
      });

      clearTimeout(timeoutId);
      return response;
    } catch (error) {
      clearTimeout(timeoutId);

      if (error.name === 'AbortError') {
        return new Response('Request timeout', { status: 504 });
      }

      return new Response('Request failed', { status: 502 });
    }
  }
};
```

## 常见错误码

| 错误码 | 描述 | 解决方案 |
|--------|------|----------|
| 1102 | CPU 时间超限 | 优化代码，减少计算量 |
| 1105 | Worker 已被停用 | 检查费用，重新启用 |
| 1106 | 脚本大小超限 | 优化代码，删除依赖 |
| 1107 | 语法错误 | 检查代码语法 |
| 1108 | 运行时错误 | 修复代码中的错误 |
| 1109 | 请求体过大 | 减少请求体大小或使用流 |
| 1110 | 响应头过大 | 减少响应头大小 |
| 1111 | 方法未实现 | 实现对应的方法处理 |

### 2. 调试工具

```javascript
// 简单的调试工具
export class Debugger {
  constructor(enabled = false) {
    this.enabled = enabled;
    this.logs = [];
  }

  log(...args) {
    if (this.enabled) {
      console.log(...args);
      this.logs.push(args);
    }
  }

  dump() {
    return this.logs;
  }

  clear() {
    this.logs = [];
  }
}

// 使用
const debug = new Debugger(true);
debug.log('Debug info');
```
