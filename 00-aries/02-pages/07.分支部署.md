# Pages 分支部署

## 分支部署概述

Cloudflare Pages 支持基于 Git 分支的自动化部署，让你可以为不同环境（开发、测试、预览、生产）创建独立的部署。

## 分支部署类型

### 1. 生产分支（Production Branch）

```javascript
// 主要特点
✅ 绑定自定义域名
✅ 生产环境配置
✅ SEO 优化
✅ 性能监控
✅ 严格的构建要求

// 默认设置
- 生产分支: main
- 部署到: example.com
- 环境: production
```

### 2. 预览分支（Preview Branches）

```javascript
// 功能特点
✅ 自动生成预览 URL
✅ 每个 PR 独立部署
✅ 共享配置
✅ 开发友好功能

// URL 格式
https://branch-name.project-name.pages.dev
https://pr-123.project-name.pages.dev

// 默认预览分支
- main
- dev
- develop
- staging
- 所有 feature/* 分支
```

### 3. 环境特定配置

```javascript
// 构建配置
// vite.config.js
export default defineConfig(({ mode }) => {
  const isProduction = mode === 'production';
  const isPreview = mode === 'preview';

  return {
    base: isProduction ? '/' : '/preview/',
    build: {
      sourcemap: !isProduction,
      minify: isProduction ? 'terser' : 'esbuild',
      // 生产环境优化
      ...(isProduction && {
        rollupOptions: {
          output: {
            manualChunks: {
              vendor: ['vue', 'vue-router'],
              ui: ['element-plus']
            }
          }
        }
      })
    },
    define: {
      __VUE_ENV__: JSON.stringify(isProduction ? 'production' : 'development'),
      __API_URL__: JSON.stringify(
        process.env.VITE_API_URL || 'https://api.example.com'
      )
    }
  };
});
```

```bash
# 1. 创建功能分支
git checkout -b feature/new-feature

# 2. 开发功能
# ... 编写代码 ...

# 3. 提交代码

# 4. 推送分支

```

### 3. 自动化部署配置

```yaml
# .github/workflows/deploy.yml
name: Deploy to Cloudflare Pages

on:
  push:
    branches:
      - main
      - develop
      - 'feature/*'
  pull_request:
    branches:
      - main

jobs:
  deploy:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      deployments: write

    steps:
      - name: Checkout
        uses: actions/checkout@v3

      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '18'

      - name: Install dependencies
        run: npm ci

      - name: Run tests
        run: npm test

      - name: Build
        run: npm run build

      - name: Determine environment
        id: env
        run: |
          if [[ "${{ github.ref }}" == "refs/heads/main" ]]; then
            echo "environment=production" >> $GITHUB_OUTPUT
          elif [[ "${{ github.ref }}" == "refs/heads/develop" ]]; then
            echo "environment=staging" >> $GITHUB_OUTPUT
          else
            echo "environment=preview" >> $GITHUB_OUTPUT
          fi

      - name: Deploy to Cloudflare Pages
        uses: cloudflare/pages-action@v1
        with:
          apiToken: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          accountId: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
          projectName: my-project
          directory: dist
          gitHubToken: ${{ secrets.GITHUB_TOKEN }}
```

## 测试和验证

### 1. 预览部署测试

```javascript
// 自动化测试脚本
// scripts/test-preview.js
async function testPreviewDeployment(url) {
  const results = [];

  // 基础功能测试
  const tests = [
    { path: '/', expect: 200 },
    { path: '/api/health', expect: 200 },
    { path: '/nonexistent', expect: 404 }
  ];

  for (const test of tests) {
    try {
      const response = await fetch(`${url}${test.path}`);
      const passed = response.status === test.expect;

      results.push({
        path: test.path,
        expected: test.expect,
        actual: response.status,
        passed
      });
    } catch (error) {
      results.push({
        path: test.path,
        expected: test.expect,
        actual: 'ERROR',
        passed: false,
        error: error.message
      });
    }
  }

  return results;
}

// 在 CI/CD 中使用
async function main() {
  const previewUrl = process.env.PREVIEW_URL;
  const results = await testPreviewDeployment(previewUrl);

  const failed = results.filter(r => !r.passed);
  if (failed.length > 0) {
    console.error('Tests failed:', failed);
    process.exit(1);
  }

  console.log('All tests passed! ✅');
}
```

## 监控和日志

### 1. 分支监控

```javascript
// 构建状态监控
async function monitorBranchBuilds() {
  const accountId = process.env.CLOUDFLARE_ACCOUNT_ID;
  const apiToken = process.env.CLOUDFLARE_API_TOKEN;

  const response = await fetch(
    `https://api.cloudflare.com/client/v4/accounts/${accountId}/pages/projects`,
    {
      headers: {
        'Authorization': `Bearer ${apiToken}`
      }
    }
  );

  const projects = await response.json();

  // 检查构建状态
  for (const project of projects.result) {
    const builds = await getRecentBuilds(project.name);

    builds.forEach(build => {
      if (build.status === 'failed') {
        // 发送告警
        sendAlert(`Build failed for ${project.name}: ${build.id}`);
      }
    });
  }
}

function sendAlert(message) {
  console.error(message);
  // 发送到 Slack、邮件或其他通知系统
}
```

### 2. 环境特定分析

```javascript
// Google Analytics 配置
// analytics.js
export function initAnalytics() {
  const isProduction = import.meta.env.PROD;
  const trackingId = import.meta.env.VITE_GA_TRACKING_ID;

  if (!trackingId) return;

  // 生产环境启用完整跟踪
  if (isProduction) {
    gtag('config', trackingId, {
      send_page_view: true,
      debug_mode: false
    });
  } else {
    // 开发环境启用调试模式
    gtag('config', trackingId, {
      send_page_view: true,
      debug_mode: true
    });
  }
}

// 事件跟踪
export function trackEvent(eventName, parameters) {
  if (import.meta.env.PROD) {
    gtag('event', eventName, parameters);
  } else {
    console.log('Analytics Event:', eventName, parameters);
  }
}
```
