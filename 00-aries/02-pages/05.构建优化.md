# Pages 构建优化

## 构建优化概述

构建优化直接影响网站的性能、SEO 评分和用户体验。本章介绍如何优化 Cloudflare Pages 的构建过程和输出。

## 构建性能优化

### 1. 构建时间优化

#### 减少构建依赖

```javascript
// package.json 优化
{
  "dependencies": {
    // ✅ 只包含运行时依赖
    "vue": "^3.3.0",
    "vue-router": "^4.2.0"
  },
  "devDependencies": {
    // ✅ 开发依赖分类
    "@vitejs/plugin-vue": "^4.3.0",
    "vite": "^4.4.0",
    "tailwindcss": "^3.3.0"
  },
  "scripts": {
    // ✅ 优化的构建命令
    "build": "vite build --minify",
    "build:analyze": "vite build --mode analyze"
  }
}
```

#### 并行构建

```javascript
// Vite 配置 - 启用并行构建
// vite.config.js
export default {
  build: {
    parallel: true,  // 启用并行构建
    rollupOptions: {
      output: {
        // 分包策略
        manualChunks: {
          vendor: ['vue', 'vue-router'],
          utils: ['lodash', 'axios']
        }
      }
    }
  }
}
```

#### 缓存策略

```javascript
// wrangler.toml - 构建缓存配置
[build]
command = "npm run build"
publish = "dist"

# 启用构建缓存
[build.upload]
format = "modules"
```

### 2. CSS 优化

#### CSS 压缩和合并

```javascript
// PostCSS 配置
// postcss.config.js
export default {
  plugins: [
    // 压缩 CSS
    'cssnano',
    // 自动添加浏览器前缀
    'autoprefixer',
    // 移除未使用的 CSS
    'postcss-purgecss',
    // 媒体查询合并
    'postcss-merge-media-queries'
  ]
};
```

#### Critical CSS

```javascript
// 内联关键 CSS
// functions/_middleware.js
import { getAssetFromKV } from '@cloudflare/kv-asset-handler';

export async function onRequest(event) {
  const response = await getAssetFromKV(event);

  // 注入 Critical CSS
  const criticalCSS = `
    body { margin: 0; font-family: system-ui; }
    h1 { font-size: 2rem; }
  `;

  const html = await response.text();
  const modifiedHtml = html.replace(
    '</head>',
    `<style>${criticalCSS}</style></head>`
  );

  return new Response(modifiedHtml, response);
}
```

### 3. JavaScript 优化

        reduce_funcs: true

        // 变量名混淆
        toplevel: true,
        // 类名混淆

  }

### 2. 构建条件

```javascript
// 条件构建
// vite.config.js
export default defineConfig(({ command, mode }) => {
  const isProduction = mode === 'production';

  return {
    build: {
      sourcemap: !isProduction,
      minify: isProduction,
      // 生产环境优化
      ...(isProduction && {
        rollupOptions: {
          output: {
            manualChunks: {
              vendor: ['vue', 'vue-router'],
              ui: ['element-plus']
            }
          }
        }
      })
    }
  };
});
```

```javascript

command = "npm run build"
publish = "dist"

# 缓存控制
[env.production]
[env.production.vars]
CACHE_CONTROL = "public, max-age=31536000"

[env.preview]
[env.preview.vars]
CACHE_CONTROL = "public, max-age=3600"
```

### 2. 压缩配置

```javascript
// 自动压缩
// wrangler.toml
[build]
command = "npm run build"
publish = "dist"

# 启用压缩
[build.upload]
format = "modules"
compression = true
```

### 3. 边缘优化

```javascript
// Functions 优化
// functions/_middleware.js
export async function onRequest(event) {
  const url = new URL(event.request.url);

  // 设置缓存头
  const response = await getAssetFromKV(event, {
    cacheControl: {
      browserTTL: 60 * 60 * 24 * 365, // 1年
      edgeTTL: 60 * 60 * 24 * 30,      // 30天
      bypassCache: false
    }
  });

  // 添加性能头
  response.headers.set('X-Frame-Options', 'DENY');
  response.headers.set('X-Content-Type-Options', 'nosniff');

  return response;
}
```

### 部署前检查

```javascript
// 部署前脚本
// scripts/pre-deploy.js
import { execSync } from 'child_process';
import { readFileSync } from 'fs';

// 检查构建大小
const stats = JSON.parse(
  readFileSync('dist/stats.json', 'utf8')
);

const mainBundleSize = stats.assets.find(
  asset => asset.name.includes('main')
).size;

if (mainBundleSize > 200 * 1024) { // 200KB
  console.error('Main bundle too large:', mainBundleSize);
  process.exit(1);
}

// 检查 Lighthouse 分数
execSync('lighthouse dist/index.html --quiet --output=json');

console.log('Pre-deployment checks passed ✅');
```
