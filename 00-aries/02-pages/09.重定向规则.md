# Pages 重定向规则

## 重定向概述

Cloudflare Pages 提供多种重定向配置方式，包括 Page Rules、Functions、配置文件等，帮助你管理 URL 转发、域名迁移、SEO 优化等。

## 重定向类型

### 1. 永久重定向 (301)
```javascript
// 用途
- 域名迁移
- URL 结构变更
- SEO 优化
- 内容合并

// 特点
- 浏览器缓存
- 搜索引擎更新索引
- 传递 SEO 权重
```

### 2. 临时重定向 (302/307)
```javascript
// 用途
- A/B 测试
- 维护页面
- 临时内容
- 地理定位

// 特点
- 不缓存
- 搜索引擎保留原 URL
- 灵活切换
```

### 2. Functions 实现重定向

```javascript
// functions/redirects/[[path]].js
export async function onRequest(context) {
  const { request } = context;
  const url = new URL(request.url);
  const path = url.pathname;

  // 重定向规则映射
  const redirects = [
    { from: '/old-page', to: '/new-page', code: 301 },
    { from: '/blog/old-post', to: '/blog/new-post', code: 301 },
    { from: '/downloads/*', to: '/assets/$1', code: 301 },
    { from: '/temp/*', to: '/temporary/$1', code: 302 }
  ];

  // 检查重定向规则
  for (const rule of redirects) {
    if (matchesPattern(path, rule.from)) {
      const toPath = replacePattern(path, rule.from, rule.to);
      const redirectUrl = new URL(toPath, url.origin);

      return Response.redirect(redirectUrl, rule.code);
    }
  }

  // 如果没有匹配规则，继续正常处理
  return context.next();
}

// 模式匹配函数
function matchesPattern(path, pattern) {
  if (pattern.includes('*')) {
    const regex = new RegExp('^' + pattern.replace(/\*/g, '.*') + '$');
    return regex.test(path);
  }
  return path === pattern;
}

// 模式替换函数
function replacePattern(path, pattern, replacement) {
  if (pattern.includes('*')) {
    const regex = new RegExp('^' + pattern.replace(/\*/g, '(.*)') + '$');
    const match = path.match(regex);
    if (match) {
      return replacement.replace('*', match[1]);
    }
  }
  return path === pattern ? replacement : path;
}
```

### 4. JavaScript 配置

```javascript
// functions/_config.js - 集中管理重定向规则
export const redirectRules = [
  {
    // 简单重定向
    from: '/about',
    to: '/about-us',
    code: 301
  },
  {
    // 通配符重定向
    from: '/products/*',
    to: '/shop/:splat',
    code: 301
  },
  {
    // 正则表达式重定向
    from: /^\/category\/(\d+)-(.*)$/,
    to: '/category/$2',
    code: 301
  },
  {
    // 条件重定向
    from: '/special',
    to: () => {
      const now = new Date();
      return now.getMonth() === 11 ? '/holiday' : '/regular';
    },
    code: 302
  },
  {
    // 外部重定向
    from: '/external',
    to: 'https://external-site.com/page',
    code: 301
  }
];

// functions/_middleware.js
import { redirectRules } from './_config.js';

export async function onRequest(context) {
  const { request } = context;
  const url = new URL(request.url);
  const path = url.pathname;

  for (const rule of redirectRules) {
    if (typeof rule.from === 'string') {
      // 字符串匹配
      if (rule.from.includes('*')) {
        const pattern = rule.from.replace(/\*/g, '(.*)');
        const regex = new RegExp(`^${pattern}$`);
        const match = path.match(regex);

        if (match) {
          const to = typeof rule.to === 'function' ? rule.to() : rule.to;
          const finalTo = to.replace(':splat', match[1]);
          const redirectUrl = rule.to.startsWith('http')
            ? new URL(finalTo)
            : new URL(finalTo, url.origin);

          return Response.redirect(redirectUrl, rule.code);
        }
      } else if (path === rule.from) {
        const to = typeof rule.to === 'function' ? rule.to() : rule.to;
        const redirectUrl = rule.to.startsWith('http')
          ? new URL(to)
          : new URL(to, url.origin);

        return Response.redirect(redirectUrl, rule.code);
      }
    } else if (rule.from instanceof RegExp) {
      // 正则表达式匹配
      const match = path.match(rule.from);
      if (match) {
        const to = typeof rule.to === 'function' ? rule.to() : rule.to;
        let finalTo = to;

        // 替换捕获组
        for (let i = 1; i < match.length; i++) {
          finalTo = finalTo.replace(`$${i}`, match[i]);
        }

        const redirectUrl = rule.to.toString().startsWith('http')
          ? new URL(finalTo)
          : new URL(finalTo, url.origin);

        return Response.redirect(redirectUrl, rule.code);
      }
    }
  }

  return context.next();
}
```

## 高级重定向场景

### 1. 域名迁移

```javascript
// functions/middleware/domain-migration.js
export async function domainMigration(context) {
  const { request } = context;
  const url = new URL(request.url);

  // 旧域名列表
  const oldDomains = ['old-domain.com', 'legacy-site.org'];
  const newDomain = 'new-domain.com';

  if (oldDomains.includes(url.hostname)) {
    // 保持路径和查询参数
    const newUrl = new URL(url.pathname + url.search, `https://${newDomain}`);

    return Response.redirect(newUrl, 301);
  }

  return context.next();
}
```

### 2. 语言和地区重定向

```javascript
// functions/middleware/geo-redirect.js
export async function geoRedirect(context) {
  const { request } = context;
  const url = new URL(request.url);
  const country = request.cf.country;

  // 地区域名映射
  const domainMap = {
    'US': 'example.com',
    'GB': 'example.co.uk',
    'DE': 'example.de',
    'FR': 'example.fr',
    'JP': 'example.jp'
  };

  const localizedDomain = domainMap[country];

  if (localizedDomain && url.hostname !== localizedDomain) {
    const newUrl = new URL(url.pathname + url.search, `https://${localizedDomain}`);

    // 首次访问时重定向
    const hasVisited = request.headers.get('Cookie')?.includes('geo_redirected=true');

    if (!hasVisited) {
      const response = Response.redirect(newUrl, 302);
      response.headers.set('Set-Cookie', 'geo_redirected=true; Max-Age=86400');
      return response;
    }
  }

  return context.next();
}
```

### 3. 设备类型重定向

```javascript
// functions/middleware/device-redirect.js
export async function deviceRedirect(context) {
  const { request } = context;
  const url = new URL(request.url);

  // 从 User-Agent 检测设备
  const userAgent = request.headers.get('User-Agent') || '';
  const isMobile = /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(userAgent);

  // 移动端重定向
  if (isMobile && !url.pathname.startsWith('/mobile')) {
    // 检查是否已经在移动端页面
    if (!url.pathname.startsWith('/m/') && !url.searchParams.has('mobile')) {
      const mobileUrl = new URL(`/mobile${url.pathname}`, url.origin);
      return Response.redirect(mobileUrl, 302);
    }
  }

  return context.next();
}
```

### 4. A/B 测试重定向

```javascript
// functions/middleware/ab-test.js
export async function abTestRedirect(context) {
  const { request } = context;
  const url = new URL(request.url);

  // 获取 A/B 测试配置
  const testConfig = await getABTestConfig(context.env);
  const testKey = url.searchParams.get('test');

  if (testKey && testConfig[testKey]) {
    const test = testConfig[testKey];
    const userId = getUserId(request);
    const testGroup = getTestGroup(userId, test);

    // 重定向到测试版本
    if (testGroup === 'B' && !url.pathname.includes('/variant-b')) {
      const testUrl = new URL(
        url.pathname.replace('/variant-a', '/variant-b') + url.search,
        url.origin
      );

      return Response.redirect(testUrl, 302);
    }
  }

  return context.next();
}

function getUserId(request) {
  // 从 Cookie 或生成新 ID
  const cookie = request.headers.get('Cookie') || '';
  const match = cookie.match(/user_id=([^;]+)/);
  return match ? match[1] : crypto.randomUUID();
}

function getTestGroup(userId, test) {
  // 简单哈希分组
  const hash = userId.split('').reduce((a, b) => {
    a = ((a << 5) - a) + b.charCodeAt(0);
    return a & a;
  }, 0);

  return Math.abs(hash) % 100 < test.splitPercentage ? 'B' : 'A';
}
```

### 5. 临时维护重定向

```javascript
// functions/middleware/maintenance-redirect.js
export async function maintenanceRedirect(context) {
  const { request, env } = context;

  // 检查维护模式
  const maintenanceStatus = await env.CONFIG_KV.get('maintenance_mode');

  if (maintenanceStatus === 'true') {
    const url = new URL(request.url);

    // 管理员 IP 绕过维护
    const adminIPs = (env.ADMIN_IPS || '').split(',');
    const clientIP = request.headers.get('CF-Connecting-IP');

    if (!adminIPs.includes(clientIP)) {
      // 重定向到维护页面
      const maintenanceUrl = new URL('/maintenance.html', url.origin);

      return Response.redirect(maintenanceUrl, 302);
    }
  }

  return context.next();
}
```

## 性能优化

### 1. 缓存重定向

```javascript
// functions/middleware/cached-redirect.js
export async function cachedRedirect(context) {
  const { request, env } = context;
  const url = new URL(request.url);

  // 检查缓存
  const cacheKey = `redirect:${url.pathname}`;
  const cachedRedirect = await env.REDIRECT_CACHE_KV.get(cacheKey);

  if (cachedRedirect) {
    const { to, code } = JSON.parse(cachedRedirect);
    return Response.redirect(to, code);
  }

  // 执行重定向逻辑
  const result = await performRedirectLogic(url);

  // 缓存结果（5分钟）
  if (result) {
    await env.REDIRECT_CACHE_KV.put(
      cacheKey,
      JSON.stringify(result),
      { expirationTtl: 300 }
    );
  }

  return result
    ? Response.redirect(result.to, result.code)
    : context.next();
}
```

### 2. 批量重定向导入

```javascript
// functions/api/admin/import-redirects.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const formData = await request.formData();
  const file = formData.get('file');

  if (!file || !file.name.endsWith('.csv')) {
    return new Response('Invalid file format', { status: 400 });
  }

  const text = await file.text();
  const lines = text.split('\n');

  const redirects = [];

  for (const line of lines) {
    if (line.trim()) {
      const [from, to, code = '301'] = line.split(',').map(s => s.trim());
      redirects.push({ from, to, code: parseInt(code) });
    }
  }

  // 批量插入到数据库
  for (const redirect of redirects) {
    await env.DB.prepare(`
      INSERT OR REPLACE INTO redirects (from_path, to_url, status_code)
      VALUES (?, ?, ?)
    `).bind(redirect.from, redirect.to, redirect.code).run();
  }

  return Response.json({
    message: `Imported ${redirects.length} redirects`
  });
}
```

### 1. 重定向跟踪

```javascript
// functions/middleware/redirect-tracking.js
export async function trackRedirect(context) {
  const { request, env } = context;
  const url = new URL(request.url);

  // 记录重定向
  context.waitUntil(
    env.REDIRECT_LOGS_KV.put(
      `redirect:${Date.now()}:${crypto.randomUUID()}`,
      JSON.stringify({
        from: url.pathname,
        to: context.redirectUrl,
        code: context.statusCode,
        ip: request.headers.get('CF-Connecting-IP'),
        userAgent: request.headers.get('User-Agent'),
        referer: request.headers.get('Referer'),
        timestamp: new Date().toISOString()
      }),
      { expirationTtl: 86400 * 30 } // 30天
    )
  );

  return context.next();
}
```

### 2. 重定向分析

```javascript
// functions/api/admin/redirect-analytics.js
export async function onRequestGet(context) {
  const { env } = context;
  const url = new URL(context.request.url);
  const days = parseInt(url.searchParams.get('days')) || 7;

  // 获取重定向统计
  const logs = await env.REDIRECT_LOGS_KV.list({
    prefix: 'redirect:',
    limit: 1000
  });

  const analytics = {
    totalRedirects: 0,
    topPaths: {},
    statusCodeDistribution: {}
  };

  for (const key of logs.keys) {
    const log = JSON.parse(await env.REDIRECT_LOGS_KV.get(key.name));

    analytics.totalRedirects++;

    // 统计路径
    analytics.topPaths[log.from] = (analytics.topPaths[log.from] || 0) + 1;

    // 统计状态码
    analytics.statusCodeDistribution[log.code] =
      (analytics.statusCodeDistribution[log.code] || 0) + 1;
  }

  // 排序热门路径
  analytics.topPaths = Object.entries(analytics.topPaths)
    .sort(([, a], [, b]) => b - a)
    .slice(0, 10)
    .reduce((obj, [path, count]) => ({ ...obj, [path]: count }), {});

  return Response.json(analytics);
}
```

### 2. 避免重定向链

```javascript
// ❌ 错误：重定向链
/page-old → /page-temp → /page-new

// ✅ 正确：直接重定向
/page-old → /page-new

// 检测重定向链
function detectRedirectChain(redirects) {
  const graph = {};

  // 构建图
  redirects.forEach(r => {
    graph[r.from] = r.to;
  });

  // 检测循环
  function hasCycle(node, visited) {
    if (visited.has(node)) return true;
    if (!graph[node]) return false;

    visited.add(node);
    return hasCycle(graph[node], visited);
  }

  return Object.keys(graph).some(key => hasCycle(key, new Set()));
}
```

### 3. 更新内部链接

```javascript
// 构建时更新链接
// scripts/update-internal-links.js
const fs = require('fs');
const path = require('path');

function updateInternalLinks(dir, redirectMap) {
  const files = fs.readdirSync(dir, { withFileTypes: true });

  files.forEach(file => {
    const filePath = path.join(dir, file.name);

    if (file.isDirectory()) {
      updateInternalLinks(filePath, redirectMap);
    } else if (file.name.endsWith('.html')) {
      let content = fs.readFileSync(filePath, 'utf8');

      // 更新内部链接
      Object.entries(redirectMap).forEach(([from, to]) => {
        const regex = new RegExp(`href="${from}"`, 'g');
        content = content.replace(regex, `href="${to}"`);
      });

      fs.writeFileSync(filePath, content);
    }
  });
}
```
