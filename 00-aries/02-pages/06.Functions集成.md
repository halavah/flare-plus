# Pages Functions 集成

## Functions 概述

Cloudflare Pages Functions 让你能够在静态网站中添加服务器端功能，实现动态内容、API 路由、表单处理等功能，无需单独的服务器。

## Functions 基础

### 1. Functions 目录结构

```
my-pages-project/
├── functions/
│   ├── api/
│   │   ├── [[path]].js     # 捕获所有路由
│   │   ├── users/
│   │   │   ├── [id].js     # 动态路由 /api/users/:id
│   │   │   └── index.js    # /api/users
│   │   ├── auth.js         # /api/auth
│   │   └── search.js       # /api/search
│   ├── middleware/
│   │   ├── auth.js         # 认证中间件
│   │   └── cors.js         # CORS 处理
│   └── _middleware.js      # 全局中间件
├── src/
└── dist/
```

### 2. 路由规则

```javascript
// 路由优先级（从高到低）
1. /functions/_middleware.js       - 全局中间件
2. /functions/文件名.js             - 精确匹配
3. /functions/文件夹/index.js       - 目录默认
4. /functions/[param].js           - 单参数匹配
5. /functions/[[path]].js          - 捕获所有路由
```

## 中间件模式

### 1. 全局中间件

```javascript
// functions/_middleware.js
export async function onRequest(context) {
  const { request, next, env } = context;

  // 添加 CORS 头
  const corsHeaders = {
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE',
    'Access-Control-Allow-Headers': 'Content-Type, Authorization'
  };

  // 处理 OPTIONS 请求
  if (request.method === 'OPTIONS') {
    return new Response(null, {
      status: 204,
      headers: corsHeaders
    });
  }

  // 请求日志
  console.log(`[${request.method}] ${request.url}`);

  // 执行下一个中间件或路由
  const response = await next();

  // 添加响应头
  Object.entries(corsHeaders).forEach(([key, value]) => {
    response.headers.set(key, value);
  });

  response.headers.set('X-Powered-By', 'Cloudflare Pages');
  response.headers.set('X-Response-Time', Date.now());

  return response;
}
```

### 2. 认证中间件

```javascript
// functions/middleware/auth.js
export async function requireAuth(context) {
  const { request, next } = context;

  const authHeader = request.headers.get('Authorization');

  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return new Response('Unauthorized', { status: 401 });
  }

  const token = authHeader.replace('Bearer ', '');
  const user = await verifyToken(token);

  if (!user) {
    return new Response('Invalid token', { status: 401 });
  }

  // 将用户信息添加到上下文
  context.user = user;

  return next();
}

// 使用认证中间件
export async function onRequest(context) {
  // 公共路由直接处理
  if (context.request.url.includes('/api/public')) {
    return next();
  }

  // 私有路由需要认证
  return requireAuth(context);
}
```

### 3. 错误处理中间件

```javascript
// functions/middleware/error-handler.js
export async function withErrorHandling(context) {
  try {
    return await context.next();
  } catch (error) {
    console.error('Function error:', error);

    // 开发环境返回详细错误
    if (context.env.ENVIRONMENT === 'development') {
      return new Response(JSON.stringify({
        error: error.message,
        stack: error.stack
      }), {
        status: 500,
        headers: { 'Content-Type': 'application/json' }
      });
    }

    // 生产环境返回通用错误
    return new Response(JSON.stringify({
      error: 'Internal Server Error'
    }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' }
    });
  }
}
```

## 数据库集成

### 1. D1 数据库

```javascript
// functions/api/posts.js - 集成 D1
export async function onRequestGet(context) {
  const { env } = context;

  // 查询所有文章
  const { results } = await env.DB.prepare(`
    SELECT id, title, created_at
    FROM posts
    ORDER BY created_at DESC
  `).all();

  return Response.json(results);
}

export async function onRequestPost(context) {
  const { env, request } = context;
  const { title, content } = await request.json();

  // 插入新文章
  const result = await env.DB.prepare(`
    INSERT INTO posts (title, content, created_at)
    VALUES (?, ?, ?)
  `).bind(title, content, new Date().toISOString()).run();

  // 获取插入的文章
  const post = await env.DB.prepare(`
    SELECT * FROM posts WHERE id = ?
  `).bind(result.meta.last_row_id).first();

  return Response.json(post, { status: 201 });
}
```

### 2. KV 存储

```javascript
// functions/api/cache.js - 缓存实现
export async function onRequestGet(context) {
  const { env, request } = context;
  const url = new URL(request.url);
  const cacheKey = `data:${url.pathname}`;

  // 尝试从缓存获取
  let data = await env.CACHE_KV.get(cacheKey);

  if (!data) {
    // 缓存未命中，从数据库获取
    data = await fetchData();

    // 存储到缓存（5分钟）
    await env.CACHE_KV.put(cacheKey, JSON.stringify(data), {
      expirationTtl: 300
    });
  } else {
    data = JSON.parse(data);
  }

  return Response.json(data);
}
```

## 文件处理

### 1. 表单处理

```javascript
// functions/api/contact.js - 表单提交
export async function onRequestPost(context) {
  const { request, env } = context;
  const formData = await request.formData();

  // 提取表单数据
  const name = formData.get('name');
  const email = formData.get('email');
  const message = formData.get('message');

  // 验证数据
  if (!name || !email || !message) {
    return new Response('Missing required fields', { status: 400 });
  }

  // 存储到 D1
  await env.DB.prepare(`
    INSERT INTO contacts (name, email, message, created_at)
    VALUES (?, ?, ?, ?)
  `).bind(name, email, message, new Date().toISOString()).run();

  // 发送邮件通知
  if (env.EMAIL_WEBHOOK) {
    await fetch(env.EMAIL_WEBHOOK, {
      method: 'POST',
      body: JSON.stringify({ name, email, message })
    });
  }

  return Response.json({ success: true });
}
```

### 2. 文件上传

```javascript
// functions/api/upload.js - 文件上传
export async function onRequestPost(context) {
  const { request, env } = context;
  const formData = await request.formData();
  const file = formData.get('file');

  if (!file) {
    return new Response('No file uploaded', { status: 400 });
  }

  // 检查文件大小（10MB 限制）
  if (file.size > 10 * 1024 * 1024) {
    return new Response('File too large', { status: 400 });
  }

  // 生成唯一文件名
  const fileName = `${Date.now()}-${file.name}`;

  // 上传到 R2
  await env.BUCKET.put(fileName, file);

  // 返回文件信息
  return Response.json({
    fileName,
    size: file.size,
    type: file.type
  });
}
```

## 认证和授权

### 1. JWT 认证

```javascript
// functions/api/auth/login.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const { email, password } = await request.json();

  // 验证用户
  const user = await validateUser(email, password, env);

  if (!user) {
    return new Response('Invalid credentials', { status: 401 });
  }

  // 生成 JWT
  const token = await generateJWT(user, env.JWT_SECRET);

  return Response.json({
    user: { id: user.id, email: user.email },
    token
  });
}

// functions/middleware/jwt.js
export async function verifyJWT(context) {
  const { request, next, env } = context;

  const authHeader = request.headers.get('Authorization');
  const token = authHeader?.replace('Bearer ', '');

  if (!token) {
    return new Response('Missing token', { status: 401 });
  }

  try {
    const payload = await decodeJWT(token, env.JWT_SECRET);
    context.user = payload;
    return next();
  } catch (error) {
    return new Response('Invalid token', { status: 401 });
  }
}
```

### 2. OAuth 集成

```javascript
// functions/api/auth/oauth.js
export async function onRequestGet(context) {
  const { request, env } = context;
  const url = new URL(request.url);
  const provider = url.searchParams.get('provider');

  switch (provider) {
    case 'github':
      return handleGitHubOAuth(env);
    case 'google':
      return handleGoogleOAuth(env);
    default:
      return new Response('Unsupported provider', { status: 400 });
  }
}

async function handleGitHubOAuth(env) {
  const authUrl = 'https://github.com/login/oauth/authorize';
  const clientId = env.GITHUB_CLIENT_ID;
  const redirectUri = `${env.BASE_URL}/api/auth/github/callback`;

  return Response.redirect(
    `${authUrl}?client_id=${clientId}&redirect_uri=${redirectUri}`,
    302
  );
}
```

### 1. 响应格式标准化

```javascript
// functions/utils/response.js
export function createResponse(data, status = 200, meta = {}) {
  return Response.json({
    success: status < 400,
    data,
    meta: {
      timestamp: new Date().toISOString(),
      ...meta
    }
  }, { status });
}

export function createError(message, status = 500, code = null) {
  return Response.json({
    success: false,
    error: {
      message,
      code,
      timestamp: new Date().toISOString()
    }
  }, { status });
}

// 使用示例
export async function onRequestGet(context) {
  try {
    const data = await fetchData();
    return createResponse(data);
  } catch (error) {
    return createError('Failed to fetch data', 500, 'FETCH_ERROR');
  }
}
```

### 2. 请求验证

```javascript
// functions/utils/validation.js
export function validateBody(schema) {
  return async (context) => {
    const body = await context.request.json();

    for (const [key, rules] of Object.entries(schema)) {
      const value = body[key];

      if (rules.required && !value) {
        return Response.json({
          error: `${key} is required`
        }, { status: 400 });
      }

      if (value && rules.type && typeof value !== rules.type) {
        return Response.json({
          error: `${key} must be of type ${rules.type}`
        }, { status: 400 });
      }
    }

    context.validatedBody = body;
    return context.next();
  };
}

// 使用示例
export async function onRequestPost(context) {
  return validateBody({
    email: { required: true, type: 'string' },
    password: { required: true, type: 'string', minLength: 8 }
  })(context);
}
```

## 性能优化

### 1. 缓存策略

```javascript
// functions/middleware/cache.js
export function withCache(ttl = 300) {
  return async (context) => {
    const { request, env, next } = context;
    const cacheKey = new Request(request.url);

    // 尝试从缓存获取
    let response = await caches.default.match(cacheKey);

    if (response) {
      response.headers.set('X-Cache', 'HIT');
      return response;
    }

    // 执行原始请求
    response = await next();

    // 添加到缓存
    if (response.ok) {
      const cachedResponse = response.clone();
      cachedResponse.headers.set('Cache-Control', `public, max-age=${ttl}`);
      context.waitUntil(
        caches.default.put(cacheKey, cachedResponse)
      );
    }

    response.headers.set('X-Cache', 'MISS');
    return response;
  };
}

// 使用示例
export async function onRequest(context) {
  return withCache(600)(context);
}
```

### 2. 异步处理

```javascript
// functions/api/async-process.js
export async function onRequestPost(context) {
  const { request, env } = context;
  const data = await request.json();

  // 立即返回
  const response = Response.json({
    message: 'Processing started',
    taskId: crypto.randomUUID()
  });

  // 后台处理
  context.waitUntil(
    processData(data, env)
  );

  return response;
}

async function processData(data, env) {
  // 耗时操作在后台执行
  await performHeavyComputation(data);

  // 存储结果
  await env.RESULTS_KV.put(data.taskId, JSON.stringify({
    status: 'completed',
    timestamp: new Date().toISOString()
  }));
}
```
