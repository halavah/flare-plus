# Turnstile 验证码 - 快速开始

## 📋 项目简介

**10-turnstile** 是 Cloudflare Turnstile 验证码的后端验证服务，提供隐私优先的 CAPTCHA 替代方案。

### 核心特性

- ✅ Token 验证
- ✅ 会话管理
- ✅ 速率限制
- ✅ 统计分析
- ✅ 多种验证模式
- ✅ 隐私保护

## 环境准备

### 1. 安装依赖

```bash
cd 10-turnstile
npm install
```

### 2. 获取 Turnstile 密钥

1. 访问 [Cloudflare Dashboard](https://dash.cloudflare.com/)
2. 转到 **Turnstile** 部分
3. 创建新站点，获取：
   - **Site Key**（公钥 - 用于前端）
   - **Secret Key**（私钥 - 用于后端）

### 3. 配置环境变量

在 `wrangler.toml` 中配置：

```toml
[vars]
TURNSTILE_SECRET_KEY = "your_secret_key"
TURNSTILE_SITE_KEY = "your_site_key"
```

### 4. 初始化数据库

```bash
# 本地环境
npm run db:local

# 生产环境
npm run db:migrate
```

## 本地开发

```bash
npm run dev
```

服务在 `http://localhost:8787` 启动。

## API 使用示例

### 1. 验证 Turnstile Token

```bash
curl -X POST http://localhost:8787/api/verify \
  -H "Content-Type: application/json" \
  -d '{
    "token": "turnstile_token_from_frontend",
    "remoteip": "user_ip_address"
  }'
```

响应（成功）：

```json
{
  "success": true,
  "verified": true,
  "challenge_ts": "2024-01-15T10:00:00Z",
  "hostname": "example.com",
  "sessionId": "sess_abc123..."
}
```

响应（失败）：

```json
{
  "success": false,
  "verified": false,
  "error": "Invalid token"
}
```

### 2. 创建会话

验证成功后自动创建会话，也可手动创建：

```bash
curl -X POST http://localhost:8787/api/session \
  -H "Content-Type": "application/json" \
  -d '{
    "userId": "123",
    "metadata": {
      "action": "login"
    }
  }'
```

### 3. 验证会话

```bash
curl http://localhost:8787/api/session/sess_abc123...
```

### 4. 获取统计信息

```bash
curl http://localhost:8787/api/stats?days=7
```

响应：

```json
{
  "success": true,
  "stats": {
    "total_verifications": 1500,
    "successful": 1450,
    "failed": 50,
    "success_rate": "96.67%",
    "by_date": [...]
  }
}
```

```html
      turnstileToken = token;
      document.getElementById('submitBtn').disabled = false;

    document.getElementById('myForm').addEventListener('submit', async (e) => {

      if (!turnstileToken) {
        showMessage('请完成人机验证', 'error');
        return;
      }

      const email = document.getElementById('email').value;
      const password = document.getElementById('password').value;

      try {
        // 1. 验证 Turnstile Token
        const verifyResponse = await fetch(`${API_URL}/api/verify`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json'
          },
          body: JSON.stringify({
            token: turnstileToken,
            remoteip: ''  // 可选
          })
        });

        const verifyResult = await verifyResponse.json();

        if (!verifyResult.success || !verifyResult.verified) {
          showMessage('验证失败：' + (verifyResult.error || '无效的 token'), 'error');
          return;
        }

        // 2. 执行登录逻辑
        // 这里可以调用你的登录 API
        console.log('验证通过，会话 ID:', verifyResult.sessionId);

        showMessage('登录成功！', 'success');

        // 重置表单
        document.getElementById('myForm').reset();
        document.getElementById('submitBtn').disabled = true;

        // 重置 Turnstile
        turnstile.reset();

      } catch (error) {
        console.error('请求失败:', error);
        showMessage('请求失败：' + error.message, 'error');
      }
    });

    function showMessage(text, type) {
      const messageDiv = document.getElementById('message');
      messageDiv.textContent = text;
      messageDiv.className = `message ${type}`;

      setTimeout(() => {
        messageDiv.textContent = '';
        messageDiv.className = '';
      }, 5000);
    }
  </script>
</body>
</html>
```

### React 集成

```jsx
import { useState } from 'react';
import Turnstile from 'react-turnstile';

function LoginForm() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [turnstileToken, setTurnstileToken] = useState(null);
  const [message, setMessage] = useState(null);
  const [loading, setLoading] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();

    if (!turnstileToken) {
      setMessage({ type: 'error', text: '请完成人机验证' });
      return;
    }

    setLoading(true);

    try {
      // 验证 Turnstile Token
      const verifyResponse = await fetch('http://localhost:8787/api/verify', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          token: turnstileToken
        })
      });

      const verifyResult = await verifyResponse.json();

      if (!verifyResult.success || !verifyResult.verified) {
        setMessage({
          type: 'error',
          text: '验证失败：' + (verifyResult.error || '无效的 token')
        });
        return;
      }

      // 执行登录逻辑
      console.log('验证通过，会话 ID:', verifyResult.sessionId);

      setMessage({ type: 'success', text: '登录成功！' });

      // 重置表单
      setEmail('');
      setPassword('');
      setTurnstileToken(null);

    } catch (error) {
      setMessage({ type: 'error', text: '请求失败：' + error.message });
    } finally {
      setLoading(false);
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      <div>
        <label>邮箱：</label>
        <input
          type="email"
          value={email}
          onChange={(e) => setEmail(e.target.value)}
          required
        />
      </div>

      <div>
        <label>密码：</label>
        <input
          type="password"
          value={password}
          onChange={(e) => setPassword(e.target.value)}
          required
        />
      </div>

      <Turnstile
        sitekey="your_site_key"
        onVerify={(token) => setTurnstileToken(token)}
        onError={() => setMessage({ type: 'error', text: '验证失败' })}
      />

      <button type="submit" disabled={!turnstileToken || loading}>
        {loading ? '处理中...' : '登录'}
      </button>

      {message && (
        <div className={`message ${message.type}`}>
          {message.text}
        </div>
      )}
    </form>
  );
}

export default LoginForm;
```

### Vue 集成

```vue
<template>
  <form @submit.prevent="handleSubmit">
    <div>
      <label>邮箱：</label>
      <input v-model="email" type="email" required />
    </div>

    <div>
      <label>密码：</label>
      <input v-model="password" type="password" required />
    </div>

    <div ref="turnstileWidget"></div>

    <button type="submit" :disabled="!turnstileToken || loading">
      {{ loading ? '处理中...' : '登录' }}
    </button>

    <div v-if="message" :class="['message', message.type]">
      {{ message.text }}
    </div>
  </form>
</template>

<script setup>
import { ref, onMounted } from 'vue';

const email = ref('');
const password = ref('');
const turnstileToken = ref(null);
const message = ref(null);
const loading = ref(false);
const turnstileWidget = ref(null);

onMounted(() => {
  // 加载 Turnstile
  const script = document.createElement('script');
  script.src = 'https://challenges.cloudflare.com/turnstile/v0/api.js';
  script.async = true;
  script.onload = () => {
    window.turnstile.render(turnstileWidget.value, {
      sitekey: 'your_site_key',
      callback: (token) => {
        turnstileToken.value = token;
      },
      'error-callback': () => {
        message.value = { type: 'error', text: '验证失败' };
      }
    });
  };
  document.head.appendChild(script);
});

const handleSubmit = async () => {
  if (!turnstileToken.value) {
    message.value = { type: 'error', text: '请完成人机验证' };
    return;
  }

  loading.value = true;

  try {
    const response = await fetch('http://localhost:8787/api/verify', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ token: turnstileToken.value })
    });

    const result = await response.json();

    if (result.success && result.verified) {
      message.value = { type: 'success', text: '登录成功！' };
      email.value = '';
      password.value = '';
      window.turnstile.reset();
    } else {
      message.value = { type: 'error', text: result.error || '验证失败' };
    }
  } catch (error) {
    message.value = { type: 'error', text: '请求失败：' + error.message };
  } finally {
    loading.value = false;
  }
};
</script>
```

## Turnstile 模式

### 1. 托管模式（Managed）

自动选择挑战难度，推荐使用：

```html
<div class="cf-turnstile" data-sitekey="your_key"></div>
```

### 2. 非交互模式（Non-Interactive）

后台验证，对用户无感：

```html
<div class="cf-turnstile"
     data-sitekey="your_key"
     data-theme="light"
     data-size="invisible"></div>
```

### 3. 交互模式（Interactive）

总是显示挑战：

```html
<div class="cf-turnstile"
     data-sitekey="your_key"
     data-appearance="always"></div>
```

## 常见问题

### 1. Turnstile 与 reCAPTCHA 有什么不同？

- **隐私优先**：不追踪用户
- **更好的UX**：更少的挑战
- **免费无限制**：无使用限制
- **更快**：加载更快

### 2. 如何测试？

Cloudflare 提供测试密钥：
- **测试 Site Key**: `1x00000000000000000000AA`
- **测试 Secret Key**: `1x0000000000000000000000000000000AA`

### 3. Token 有效期多长？

Token 有效期约 5 分钟，验证后立即失效。
